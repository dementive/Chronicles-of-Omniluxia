namespace = omni_elemental_colossus

# Only scheduled for an active summon. Cohort scopes survive army merges and splits.
omni_elemental_colossus.1 = {
	type = country_event
	hidden = yes

	immediate = {
		# Preserve the surviving cohort and its strength cap in older saves.
		if = {
			limit = {
				has_variable = omni_colossus_strength
				exists = var:omni_colossus_cohort
			}
			var:omni_colossus_cohort = {
				set_variable = { name = omni_colossus_strength value = root.var:omni_colossus_strength }
			}
			add_to_variable_list = { name = omni_colossus_cohorts target = var:omni_colossus_cohort }
			set_variable = omni_colossus_active
			remove_variable = omni_colossus_strength
		}

		every_in_list = {
			variable = omni_colossus_cohorts
			limit = { exists = owning_unit }
			if = {
				limit = { owning_unit = { unit_owner = root } }
				# The engine reinforces the whole army. Undo reinforcement per summoned cohort.
				if = {
					limit = { subunit_strength_percentage > var:omni_colossus_strength }
					add_subunit_strength = { value = var:omni_colossus_strength subtract = subunit_strength_percentage }
				}
				else = {
					set_variable = { name = omni_colossus_strength value = subunit_strength_percentage }
				}
			}
			else_if = {
				limit = { owning_unit = { unit_size = 1 } }
				owning_unit = { destroy_unit = yes }
			}
			else = { destroy_subunit = yes }
		}

		# Only the army with the tracked cohort gets the siege bonus, even if the summon is split.
		if = {
			limit = {
				NOT = {
					var:omni_colossus_cohort = { owning_unit = { unit_owner = root } }
				}
			}
			remove_variable = omni_colossus_cohort
			random_in_list = {
				variable = omni_colossus_cohorts
				limit = { owning_unit = { unit_owner = root } }
				root = { set_variable = { name = omni_colossus_cohort value = prev } }
			}
		}
		if = {
			limit = { exists = var:omni_colossus_cohort }
			omni_colossus_refresh_army_effect = yes
			trigger_event = { id = omni_elemental_colossus.1 days = 1 }
		}
		else = {
			omni_colossus_clear_army_effect = yes
			clear_variable_list = omni_colossus_cohorts
			remove_variable = omni_colossus_active
			remove_variable = omni_colossus_strength
		}
	}
}
