country_decisions = {
	daimyo_to_shogun = {
	
		potential = {
			num_of_cities >= 1
			government = daimyo
		}
		allow = {
			num_of_cities >= 1
			government = daimyo
			is_subject = no
			legitimacy >= 65
			political_influence >= 25		
		}
		
		effect = {
			custom_tooltip = orc_levelup1_tt
			trigger_event = {
				id = orc_flavor.3
			}
		}
	}

	shogun_to_shogunate = {
	
		potential = {
			num_of_cities >= 1
			government = independent_shogun
		}
		allow = {
			num_of_cities >= 1
			government = independent_shogun
			legitimacy >= 80
			political_influence >= 50
			any_subject = {
				count >= 2
			}
		}
		
		effect = {
			custom_tooltip = orc_levelup2_tt
			trigger_event = {
				id = orc_flavor.4
			}
		}
	}
}
