country_decisions = {
	# A junior crown whose request for a peaceful release was refused
	# (omni_dynastic_union.16.b) holds a claim to its independence for ten
	# years. The subject type has subject_can_cancel = no, so this is the only
	# way to press it outside the succession events.
	omni_dynastic_union_declare_independence = {
		potential = {
			is_subject_type = dynastic_union_subject
			has_variable = omni_du_independence_claim
		}
		allow = {
			omni_dynastic_union_can_fight_overlord_trigger = yes
		}
		effect = {
			remove_variable = omni_du_independence_claim
			omni_dynastic_union_independence_war_effect = yes
		}
		ai_will_do = {
			base = 0
			modifier = {
				add = 50
				has_subject_loyalty < 50
				num_of_cohorts >= overlord.num_of_cohorts
			}
		}
	}
}
