omni_cast_detect_minerals_ii = {
	sound = "event:/SFX/UI/Actions/sfx_ui_action_invoke_omen"
	on_other_nation = no
	on_own_nation = yes

	potential_trigger = {
		scope:actor = { is_ai = no }
		scope:target = { omni_can_select_territory_detect_minerals_ii = yes }
	}

	allowed_trigger = {
		scope:actor = { any_owned_province = { omni_spell_target_mineral = yes } }
		scope:target = { omni_can_afford_spell = { BASE = 50 } }
	}

	province_actor_trigger = { omni_spell_target_mineral = yes }

	effect = {
		if = {
			limit = {
				exists = scope:recipient
				scope:recipient = {
					owner = scope:actor
					omni_spell_target_mineral = yes
				}
				scope:target = {
					OR = { employer = scope:actor country = scope:actor }
					omni_can_select_territory_detect_minerals_ii = yes
					omni_can_afford_spell = { BASE = 50 }
				}
			}
			scope:target = {
				omni_spell_mana_cost_effect = { BASE = 50 }
			}
			scope:recipient = { add_province_modifier = { name = amten_elementalist_spell_t2_mod duration = 180 } }
		}
	}
}

omni_cast_summon_dryads = {
	sound = "event:/SFX/UI/Actions/sfx_ui_action_invoke_omen"
	on_other_nation = no
	on_own_nation = yes

	potential_trigger = {
		scope:actor = { is_ai = no }
		scope:target = { omni_can_select_territory_summon_dryads = yes }
	}

	allowed_trigger = {
		scope:actor = { NOT = { has_variable = omni_summon_dryads_cooldown } any_owned_province = { omni_spell_target_forest = yes } }
		scope:target = { omni_can_afford_spell = { BASE = 100 } }
	}

	province_actor_trigger = { omni_spell_target_forest = yes }

	effect = {
		if = {
			limit = {
				exists = scope:recipient
				scope:actor = { NOT = { has_variable = omni_summon_dryads_cooldown } }
				scope:recipient = {
					owner = scope:actor
					omni_spell_target_forest = yes
				}
				scope:target = {
					OR = { employer = scope:actor country = scope:actor }
					omni_can_select_territory_summon_dryads = yes
					omni_can_afford_spell = { BASE = 100 }
				}
			}
			scope:target = {
				omni_spell_mana_cost_effect = { BASE = 100 }
				save_scope_as = dryad_caster
			}
			scope:actor = {
				set_variable = { name = omni_summon_dryads_cooldown days = 365 }
				create_unit = {
					navy = no
					name = "Thornwood Dryads"
					location = scope:recipient
					hidden_effect = {
						while = { count = omni_dryad_cohort_count add_subunit = omni_dryad }
						add_food = 800
						add_morale = 5
					}
				}
			}
			scope:recipient = { add_province_modifier = { name = summon_dryads_mod duration = 365 } }
		}
	}
}
