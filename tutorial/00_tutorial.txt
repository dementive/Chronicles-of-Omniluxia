# Omniluxia tutorial: "A Royal Progress". Started by the main menu Tutorial button,
# which the engine hard-codes to tag ROM (Darinfur, setup/main/02_countries.txt).
# Shown in the vanilla tutorial panel (gui/tutorial_view.gui). Each objective is
# completed by its Complete button once `allow` is met. Completion variables
# keep their original IDs for existing saves. Milestones come from omni_tutorial_mark
# (common/scripted_effects/omni_tutorial_effects.txt). Start: omni_tutorial.1.

omni_tut_market = {
	step = 1
	potential = {
		tag = ROM
		is_tutorial_active = yes
	}
	allow = {
		has_variable = omni_tut_on
		NOT = { has_variable = omni_tut_s1 }
		has_variable = omni_tut_market_visited
	}
	completed = {
		has_variable = omni_tut_s1
	}
	highlight = {
		scope:province = {
			province_id = 1409
		}
	}
	effect = {
		set_variable = omni_tut_s1
		add_treasury = 100
	}
}

	}
}

omni_tut_set_out = {
	step = 7
	potential = {
		tag = ROM
		is_tutorial_active = yes
	}
	allow = {
		has_variable = omni_tut_on
		has_variable = omni_tut_s2
		has_variable = omni_tut_s5
		NOT = { has_variable = omni_tut_s7 }
		# Away now, or already back (cooldown only exists after a finished trip). The ward
		# itself is proven by step 5; requiring it live here stranded the step on return.
		current_ruler = {
			OR = {
				has_variable = on_omni_adventure
				has_variable = omni_adventure_cooldown
			}
		}
	}
	completed = {
		has_variable = omni_tut_s7
	}
	highlight = {
		scope:province = {
			owner = root num_of_adventurer_guild_building >= 1
		}
	}
	effect = {
		omni_tut_complete_set_out_effect = yes # also auto-run on departure (adventure.3)
	}
}

omni_tut_retake = {
	step = 12
	potential = {
		tag = ROM
		is_tutorial_active = yes
	}
	allow = {
		has_variable = omni_tut_on
		has_variable = omni_tut_s12
		NOT = { has_variable = omni_tut_s13 }
		exists = var:omni_tut_raid_province
		OR = {
			var:omni_tut_raid_province = { owner = root }
			# Raid war is over and the raiders no longer hold it (annexed or taken by a
			# third party): nothing left to retake, so let the tutorial finish.
			AND = {
				omni_tutorial_raid_war_over = yes
				exists = var:omni_tut_raiders
				var:omni_tut_raid_province = { NOT = { owner = root.var:omni_tut_raiders } }
			}
		}
	}
	completed = {
		has_variable = omni_tut_s13
	}
	highlight = {
		scope:province = {
			exists = root.var:omni_tut_raid_province this = root.var:omni_tut_raid_province
		}
	}
	effect = {
		set_variable = omni_tut_s13
		add_treasury = 250
		add_political_influence = 25
		hidden_effect = {
			omni_tutorial_cleanup_effect = yes
			remove_country_modifier = omni_tutorial_boost
			set_variable = omni_tut_done
			omni_unlock_achievement = { ACH = omni_ach_royal_progress EVENT = omni_ach.17 }
		}
		trigger_event = { id = omni_tutorial.90 }
	}
}
