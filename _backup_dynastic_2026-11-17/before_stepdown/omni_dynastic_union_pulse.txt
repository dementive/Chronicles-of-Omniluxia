# Shared heirs. When a child of the union's two spouses takes either throne, .6
# (junior) or .7 (senior) decides what that means. The *_pending variable stops
# the spouse-death events (.4 / .5) from firing a second, conflicting set of
# options for the same succession.
on_ruler_change = { on_actions = { omni_dynastic_union_pulse_on_ruler_change } }
omni_dynastic_union_pulse_on_ruler_change = {
	effect = {
		if = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				has_variable = dynastic_union_junior_spouse
				exists = current_ruler
				current_ruler = {
					is_child_of = var:dynastic_union_senior_spouse
					is_child_of = var:dynastic_union_junior_spouse
				}
			}
			set_variable = { name = dynastic_union_shared_heir_pending days = 60 }
			current_ruler = { save_scope_as = dynastic_shared_heir_ruler }
			save_scope_as = dynastic_union_subject
			trigger_event = {
				id = omni_dynastic_union.6
				days = 30
			}
		}
		if = {
			limit = {
				is_monarchy = yes
				has_land = yes
				exists = current_ruler
				any_subject = {
					is_subject_type = dynastic_union_subject
					has_land = yes
					has_variable = dynastic_union_senior_spouse
					has_variable = dynastic_union_junior_spouse
					root.current_ruler = {
						is_child_of = prev.var:dynastic_union_senior_spouse
						is_child_of = prev.var:dynastic_union_junior_spouse
					}
				}
			}
			random_subject = {
				limit = {
					is_subject_type = dynastic_union_subject
					has_land = yes
					has_variable = dynastic_union_senior_spouse
					has_variable = dynastic_union_junior_spouse
					root.current_ruler = {
						is_child_of = prev.var:dynastic_union_senior_spouse
						is_child_of = prev.var:dynastic_union_junior_spouse
					}
				}
				save_scope_as = dynastic_union_subject
			}
			set_variable = { name = dynastic_union_shared_heir_pending days = 60 }
			current_ruler = { save_scope_as = dynastic_shared_heir_ruler }
			save_scope_as = dynastic_senior_country
			trigger_event = {
				id = omni_dynastic_union.7
				days = 30
			}
		}
	}
}

# The union follows the two SPOUSES, not whoever sits on each throne.
# FIX: .4 / .5 used to fire on any ruler change whose former ruler matched the
# spouse variables. If the engine moves a married ruler to the other court, the
# junior throne changes hands on the wedding day and .4 announced the spouse's
# "death" at once. Firing on the spouse's actual death works either way.
on_character_death = { on_actions = { omni_dynastic_union_on_character_death } }
omni_dynastic_union_on_character_death = {
	effect = {
		if = {
			limit = {
				has_variable = omni_du_junior_crown
				exists = var:omni_du_junior_crown
			}
			save_scope_as = former_ruler
			var:omni_du_junior_crown = {
				if = {
					limit = {
						has_land = yes
						is_subject_type = dynastic_union_subject
						has_variable = dynastic_union_junior_spouse
						var:dynastic_union_junior_spouse = scope:former_ruler
					}
					trigger_event = { id = omni_dynastic_union.4 days = 3 }
				}
			}
		}
		if = {
			limit = {
				has_variable = omni_du_senior_crown
				exists = var:omni_du_senior_crown
			}
			save_scope_as = former_ruler
			var:omni_du_senior_crown = {
				if = {
					limit = {
						has_land = yes
						any_subject = {
							is_subject_type = dynastic_union_subject
							has_land = yes
							has_variable = dynastic_union_senior_spouse
							var:dynastic_union_senior_spouse = scope:former_ruler
						}
					}
					trigger_event = { id = omni_dynastic_union.5 days = 3 }
				}
			}
		}
	}
}

monthly_country_pulse = { on_actions = { omni_dynastic_union_pulse_monthly_country_pulse } }
omni_dynastic_union_pulse_monthly_country_pulse = {
	effect = {
		if = {
			limit = {
				omni_dynastic_marriage_available_alert_trigger = yes
			}
			set_variable = {
				name = omni_dynastic_marriage_available_alert
				days = 40
			}
		}
		else = {
			remove_variable = omni_dynastic_marriage_available_alert
		}
		# Save migration: unions formed before the spouse-death hooks existed
		# get their spouse links once. New unions bind at formation.
		if = {
			limit = {
				is_subject_type = dynastic_union_subject
				NOT = { has_variable = omni_du_spouses_bound }
			}
			omni_dynastic_union_bind_spouses_effect = yes
		}
		# The compact is a bond between two independent monarchies. If either
		# side stops being one, it lapses without blame. Cheap: the first check
		# is false for every country that is not a junior crown.
		if = {
			limit = {
				is_subject_type = dynastic_union_subject
				OR = {
					is_monarchy = no
					overlord = {
						OR = {
							is_monarchy = no
							is_subject = yes
						}
					}
				}
			}
			omni_dynastic_union_lapse_effect = yes
		}
	}
}
