omni_friendly_magic_unit_button = {
	scope = unit
	saved_scopes = {
		player
	}

	# Shown when the selected unit is one of the player's OWN armies/fleets and the
	# player has a character capable of casting magic. Mirror of
	# omni_offensive_magic_unit_button, but for friendly (buff) spells.
	is_shown = {
		OR = {
			is_army = yes
			is_navy = yes
		}
		unit_owner = {
			this = scope:player
		}
		exists = commander
		commander = {
			OR = {
				is_general = yes
				is_admiral = yes
			}
			is_alive = yes
			NOT = { has_variable = is_the_revant }
		}
		scope:player = {
			is_ai = no
			any_character = {
				OR = {
					is_ruler = yes
					is_general = yes
					is_admiral = yes
					has_office = office_physician
					has_office = office_republic_physician
					has_office = office_apothecary
				}
				OR = {
					has_trait = mage_trait
					omni_any_discipline_mastered = yes
					has_trait = warlock_trait
					has_trait = cleric_trait
					has_trait = necromancer_trait
					has_trait = lich_trait
					has_trait = druid_trait
					has_trait = elementalist_trait
					has_trait = diviner_trait
					has_trait = hierophant_trait
					has_trait = illusionist_trait
					has_trait = vampire_trait
					has_trait = blood_angel_curse_trait
					has_variable = folk_unlocked_var
				}
				NOR = {
					has_trait = brain_damaged
					has_trait = goblin_trait
					has_variable = on_orc_adventure
				}
			}
		}
	}

	ai_is_valid = {
		always = no
	}

	# is_valid drives the ENABLED state. In combat this button is only for friendly
	# heat-of-battle buffs, not travel or hostile spellcraft.
	is_valid = {
		custom_tooltip = {
			text = omni_friendly_magic_unit_button_tooltip
		}
		trigger_if = {
			limit = {
				scope:player = {
					has_variable = omni_battle_spell_cooldown
				}
			}
			custom_tooltip = {
				text = omni_battle_spell_cooldown_tt
				scope:player = {
					NOT = { has_variable = omni_battle_spell_cooldown }
				}
			}
		}
		custom_tooltip = {
			text = omni_battle_friendly_need_spell_tt
			commander = {
				OR = {
					AND = {
						omni_can_afford_spell = { BASE = 15 }
						omni_caster_tier_1_any_school = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 25 }
						omni_caster_any_school_t2 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 50 }
						omni_caster_any_school_t2 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 75 }
						omni_any_discipline_mastered = yes
					}
				}
			}
		}
	}

	effect = {
		commander = {
			save_scope_as = target
		}
		scope:player = {
			save_scope_as = battle_spell_context
			trigger_event = { id = me_zorgo_magic_focus.130 }
		}
	}
}
