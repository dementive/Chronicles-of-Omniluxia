omni_friendly_magic_unit_button = {
	scope = unit
	saved_scopes = {
		player
	}

	# Shown when the selected unit is one of the player's OWN armies/fleets and the
	# player has a character capable of casting magic. Mirror of
	# omni_offensive_magic_unit_button, but for friendly (buff) spells.
	is_shown = {
		OR = {
			is_army = yes
			is_navy = yes
		}
		unit_owner = {
			this = scope:player
		}
		exists = commander
		commander = {
			OR = {
				is_general = yes
				is_admiral = yes
			}
			is_alive = yes
			NOT = { has_variable = is_the_revant }
		}
		scope:player = {
			is_ai = no
			omni_player_has_eligible_mage = yes
		}
	}

	ai_is_valid = {
		always = no
	}

	# is_valid drives the enabled state. Hearthward is a local ward available
	# to a commander at home between battles.
	is_valid = {
		custom_tooltip = {
			text = omni_friendly_magic_unit_button_tooltip
		}

		custom_tooltip = {
			text = omni_battle_friendly_need_spell_tt
			OR = {
				AND = {
					is_army = yes
					in_combat = no
					unit_location = { owner = scope:player }
					commander = { omni_can_cast_hearthward = yes }
				}
				commander = {
					OR = {
						AND = {
							omni_can_afford_spell = { BASE = 15 }
							omni_caster_any_school_t1 = yes
						}
						AND = {
							omni_can_afford_spell = { BASE = 25 }
							omni_caster_any_school_t2 = yes
						}
						AND = {
							omni_can_afford_spell = { BASE = 50 }
							omni_caster_any_school_t2 = yes
						}
						AND = {
							omni_can_afford_spell = { BASE = 75 }
							omni_any_discipline_mastered = yes
						}
					}
				}
			}
		}
	}

	effect = {
		commander = {
			save_scope_as = target
		}
		if = {
			limit = { in_combat = yes }
			scope:player = { save_scope_as = battle_spell_context }
		}
		scope:player = {
			trigger_event = { id = me_zorgo_magic_focus.130 }
		}
	}
}
