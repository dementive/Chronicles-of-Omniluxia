	saved_scopes = {

	}

	is_shown = {
		is_army = yes

			this = scope:player
		}

		commander = {
			is_general = yes
			is_alive = yes

			any_character_unit = {
				is_army = yes
				in_combat = no
			}
		}
		scope:player = {
			is_ai = no
		}
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		custom_tooltip = {
			text = omni_teleport_unit_button_tooltip
		}

			text = omni_unit_magic_need_mana_tt

		}
	}

	effect = {
		commander = {
			save_scope_as = target
		}

			trigger_event = { id = me_zorgo_magic_focus.36 }
		}
	}
}
