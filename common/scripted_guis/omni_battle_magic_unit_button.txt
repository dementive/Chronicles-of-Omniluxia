omni_battle_magic_unit_button = {
	scope = unit
	saved_scopes = {
		player
		battle_enemy_unit
	}


	# Root is the leading unit of the battle side. An attached unit can supply
	# the caster when the leading commander has no available battle spell.
	is_shown = {
		in_combat = yes
		scope:player = { is_ai = no }
		OR = {
			omni_battle_mage_unit = yes
			unit_location = {
				any_unit_in_province = { omni_battle_mage_unit = yes }
			}
		}
	}

	ai_is_valid = { always = no }

	is_valid = {
		in_combat = yes
		custom_tooltip = { text = omni_battle_magic_unit_button_tooltip }
		custom_tooltip = {
			text = omni_battle_magic_need_spell_tt
			OR = {
				omni_battle_mage_unit_ready = yes
				unit_location = {
					any_unit_in_province = { omni_battle_mage_unit_ready = yes }
				}
			}
		}
	}

	effect = {
		if = {
			limit = { omni_battle_mage_unit_ready = yes }
			commander = { save_scope_as = target }
		}
		else = {
			unit_location = {
				every_unit_in_province = {
					limit = {
						NOT = { exists = scope:target }
						omni_battle_mage_unit_ready = yes
					}
					commander = { save_scope_as = target }
				}
			}
		}
		if = {
			limit = { exists = scope:battle_enemy_unit }
			scope:battle_enemy_unit = {
				save_scope_as = battle_enemy_force
				if = {
					limit = { exists = commander }
					commander = {
						save_scope_as = battle_enemy_target
					}
				}
			}
		}
		if = {
			limit = { exists = scope:target }
			scope:player = {
				save_scope_as = battle_spell_context
				trigger_event = { id = me_zorgo_magic_focus.29 }
			}
		}
	}
}
