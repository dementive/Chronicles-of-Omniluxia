omni_battle_magic_unit_button = {
	scope = unit
	saved_scopes = {
		player
		battle_enemy_unit
	}

	# Root is the force in battle. Using the unit as root avoids combat-window
	# refreshes passing an invalid placeholder character into ScriptedGui.
	is_shown = {
		OR = {
			is_army = yes
			is_navy = yes
		}
		unit_owner = {
			this = scope:player
		}
		exists = commander
		commander = {
			is_alive = yes
			OR = {
				is_general = yes
				is_admiral = yes
			}
			NOT = { has_variable = is_the_revant }
			omni_character_is_eligible_caster = yes
		}
		scope:player = {
			is_ai = no
		}
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		custom_tooltip = {
			text = omni_battle_magic_unit_button_tooltip
		}
		trigger_if = {
			limit = {
				scope:player = {
					has_variable = omni_battle_spell_cooldown
				}
			}
			custom_tooltip = {
				text = omni_battle_spell_cooldown_tt
				scope:player = {
					NOT = { has_variable = omni_battle_spell_cooldown }
				}
			}
		}
		custom_tooltip = {
			text = omni_battle_magic_need_spell_tt
			commander = {
				OR = {
					AND = {
						omni_can_afford_spell = { BASE = 10 }
						omni_caster_melodian_t1 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 25 }
						OR = {
							omni_caster_aldic_t1 = yes
							has_variable = folk_unlocked_var
						}
					}
					AND = {
						omni_can_afford_spell = { BASE = 30 }
						omni_caster_melodian_t2 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 50 }
						omni_caster_any_school_t2 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 60 }
						omni_caster_aldic_t2 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 75 }
						omni_caster_melodian_t3 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 90 }
						omni_caster_melodian_t3 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 100 }
						omni_caster_any_school_t3 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 165 }
						has_trait = archmage_trait
						omni_caster_melodian_t3 = yes
					}
					AND = {
						omni_can_afford_spell = { BASE = 175 }
						has_trait = archmage_trait
					}
				}
			}
		}
	}

	effect = {
		commander = {
			save_scope_as = target
		}
		if = {
			limit = { exists = scope:battle_enemy_unit }
			scope:battle_enemy_unit = {
				save_scope_as = battle_enemy_force
				if = {
					limit = { exists = commander }
					commander = {
						save_scope_as = battle_enemy_target
					}
				}
			}
		}
		scope:player = {
			save_scope_as = battle_spell_context
			trigger_event = { id = me_zorgo_magic_focus.29 }
		}
	}
}
