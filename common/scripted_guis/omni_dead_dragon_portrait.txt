# Portrait checks for DEAD dragons only; living dragons are matched by their
# variables directly in the GUI. Lists are filled by
# common/on_action/omni_dead_dragons.txt at the moment of death.

    scope = character
    ai_is_valid = { always = no }
    is_shown = {

    }
}
