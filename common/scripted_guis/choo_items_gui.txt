omni_character_items_button = {
	scope = character
	saved_scopes = {
		player
	}

	# Shown on the player's own characters (court members, rulers, generals,
	# governors). Always visible even with an empty inventory so you can hover
	# non-rulers and inspect their gear, but only the ruler can click it.
	is_shown = {
		exists = root
		is_alive = yes
		OR = {
			country = scope:player
			employer = scope:player
		}
	}

	is_valid = {
		is_ruler = yes
	}

	effect = {
		hidden_effect = {
		# Opens the event-based item menu on the shown character.
		trigger_event = { id = omni_items.1 }
		}
			# --- Currently Equipped: army loadout + character items ---
			if = {
				limit = {
					OR = {
						has_variable_list = equiped_item_list
						country = { OR = { has_variable = omni_army_slot_weapon has_variable = omni_army_slot_ranged has_variable = omni_army_slot_armour has_variable = omni_army_slot_shield } }
					}
				}
				custom_tooltip = omni_equipped_header
				# National army loadout (generated items)
				if = { limit = { country = { has_variable = omni_army_slot_weapon } } custom_tooltip = omni_equipped_slot_weapon show_as_tooltip = { country = { omni_show_army_item = { SLOTVAR = omni_army_slot_weapon } } } }
				if = { limit = { country = { has_variable = omni_army_slot_ranged } } custom_tooltip = omni_equipped_slot_ranged show_as_tooltip = { country = { omni_show_army_item = { SLOTVAR = omni_army_slot_ranged } } } }
				if = { limit = { country = { has_variable = omni_army_slot_armour } } custom_tooltip = omni_equipped_slot_armour show_as_tooltip = { country = { omni_show_army_item = { SLOTVAR = omni_army_slot_armour } } } }

				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_chainmail } } custom_tooltip = omni_equipped_item_chainmail }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_chainmail_coif } } custom_tooltip = omni_equipped_item_chainmail_coif }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_plated_coat } } custom_tooltip = omni_equipped_item_plated_coat }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_grand_helmet } } custom_tooltip = omni_equipped_item_grand_helmet }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_plate_armour } } custom_tooltip = omni_equipped_item_plate_armour }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_armet_helmet } } custom_tooltip = omni_equipped_item_armet_helmet }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_light_armour } } custom_tooltip = omni_equipped_item_light_armour }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_bow } } custom_tooltip = omni_equipped_item_bow }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_elven_bow } } custom_tooltip = omni_equipped_item_elven_bow }

				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_artifact_treasure_wolf_tooth } } custom_tooltip = omni_equipped_item_artifact_treasure_wolf_tooth }
				if = { limit = { is_target_in_variable_list = { name = equiped_item_list target = flag:item_artifact_treasure_zehia_horse_saddle } } custom_tooltip = omni_equipped_item_artifact_treasure_zehia_horse_saddle }
			}
			else = { custom_tooltip = omni_equipped_none }
	}
}

omni_character_city_button = {
	scope = character
	saved_scopes = {
		player
	}

	# Shows a city interaction button when one of the player's characters is standing
	# in a city with at least one supported interaction building they can access.
	is_shown = {
		exists = root
		is_alive = yes
		OR = {
			country = scope:player
			employer = scope:player
		}
		OR = {
			omni_character_has_local_market = yes
			omni_character_has_local_adventurer_guild = yes
			omni_character_has_local_library = yes
			omni_character_has_local_brewery = yes
			omni_character_has_local_circus = yes
			omni_character_has_local_magic_academy = yes
			# A hostile/occupied city has no traditional buildings open to a
			# visiting ruler or general, but the Black Market and Requisition
			# Food options should always be reachable there.
			omni_character_can_visit_hostile_city = yes
		}
	}

	is_valid = {
		custom_tooltip = {
			text = omni_character_city_disloyal_tt
			loyalty > 33
		}
	}

	effect = {
		hidden_effect = {
			if = {
				limit = { is_ruler = yes }
				save_scope_as = omni_city_visitor
				trigger_event = { id = omni_items.7 }
			}
			else = {
				# Non-ruler character events are auto-resolved by the engine. Keep the
				# commander as the city actor, but display the event on the player ruler.
				save_scope_as = omni_city_visitor
				scope:player.current_ruler = { trigger_event = { id = omni_items.7 } }
			}
		}
	}
}
