omni_offensive_magic_unit_button = {
	scope = unit
	saved_scopes = {
		player
	}

	is_shown = {
		OR = {
			is_army = yes
			is_navy = yes
		}
		unit_owner = {
			NOT = { tag = BAR }
			NOT = { tag = REB }
			NOT = { tag = PIR }
			NOT = { this = scope:player }
			scope:player = {
				war_with = PREV
			}
		}
		exists = commander
		commander = {
			OR = {
				is_general = yes
				is_admiral = yes
			}
			is_alive = yes
			NOT = { has_variable = is_the_revant }
		}
		scope:player = {
			is_ai = no
			omni_player_has_eligible_mage = yes
		}
	}

	ai_is_valid = {
		always = no
	}

	# is_valid drives the ENABLED state. Greyed out when no eligible mage can afford
	# even the cheapest offensive spell (5 MP scaled).
	is_valid = {
		custom_tooltip = {
			text = omni_offensive_magic_unit_button_tooltip
		}
		custom_tooltip = {
			text = omni_unit_magic_need_mana_tt
			scope:player = { omni_player_has_affordable_mage = { BASE = 5 } }
		}
	}

	effect = {
		commander = {
			save_scope_as = target
		}
		scope:player = {
			trigger_event = { id = me_zorgo_magic_focus.8 }
		}
	}
}
