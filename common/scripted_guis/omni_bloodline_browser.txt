# Explicit human button actions only. UI lists are owned by the player's country.
omni_bl_open = {
    scope = country
    ai_is_valid = { always = no }
    effect = {
        if = {
            limit = { NOT = { has_variable = omni_bl_selected } }
            set_variable = { name = omni_bl_selected value = 1 }
        }
        set_variable = { name = omni_bl_filter value = 0 }
        remove_variable = omni_bl_men_first
        omni_bl_refresh_effect = yes
    }
}
omni_bl_refresh = {
    scope = country
    ai_is_valid = { always = no }
    effect = { omni_bl_refresh_effect = yes }
}
omni_bl_close = {
    scope = country
    ai_is_valid = { always = no }
    effect = {
        omni_bl_clear_effect = yes
        remove_variable = omni_bl_filter
        remove_variable = omni_bl_men_first
    }
}
