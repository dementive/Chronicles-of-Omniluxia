### OMNILUXIA CUSTOM ACHIEVEMENTS - SCRIPTED GUIS ###
#
# The achievements window (gui/achievements/jomini_achievements_window.gui) reads
# these through ScriptedGui.IsShown / ScriptedGui.IsValid:
#
#   is_shown  -> the achievement is UNLOCKED (drives the gold ornament + "Unlocked" tab)
#   is_valid  -> the achievement is still POSSIBLE this campaign (drives "Available"
#                vs. "Not possible" tabs; race-locked achievements gate on culture group)
#
# There are deliberately NO effect blocks / re-check buttons here: unlocking is fully
# automatic (see common/scripted_effects/00_achievement_effects.txt). Conditions live
# in common/scripted_triggers/omni_achievement_triggers.txt so they never duplicate.

# --- GAME RULE TOGGLE -------------------------------------------------------
# Toggles the global variable that enables/disables the whole achievement system.
# is_shown is TRUE while achievements are ENABLED (variable absent = default ON),
# which the window uses to pick the correct button/tooltip. Executing it flips
# the rule. Everything is gated on this rule inside omni_unlock_achievement, so
# turning it off stops all future unlocks (already-earned ones are kept).
omni_achievements_toggle = {
    scope = country
    ai_is_valid = { always = no }
    is_shown = { NOT = { has_global_variable = game_rule_disable_achievements } }
    effect = {
        if = {
            limit = { has_global_variable = game_rule_disable_achievements }
            remove_global_variable = game_rule_disable_achievements
        }
        else = {
            set_global_variable = game_rule_disable_achievements
        }
    }
}
