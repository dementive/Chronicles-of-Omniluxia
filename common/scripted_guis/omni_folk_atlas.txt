omni_folk_atlas_hut = {
 scope = province
 is_shown = { omni_folk_hut_catalog = yes }
}

omni_folk_atlas_refresh_directory = {
 scope = country
 ai_is_valid = { always = no }

 effect = {
  clear_variable_list = omni_folk_atlas_local_huts
  every_in_global_list = {
   variable = omni_folk_atlas_huts
   limit = {
    exists = owner
    owner = {
     OR = {
      this = root
      root = { in_diplomatic_range = prev }
     }
    }
   }
   root = {
    add_to_variable_list = { name = omni_folk_atlas_local_huts target = prev }
   }
  }
 }
}
