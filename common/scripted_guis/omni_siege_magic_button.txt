omni_siege_magic_button = {
	scope = character
	saved_scopes = {
		player
	}

	# Shown on the Siege window. The root scope is the besieging commander
	# (SiegeView.GetSiege.GetCommander). The COMMANDER LEADING THE ARMY must
	# personally be a living, capable mage - the caster has to be present at
	# the siege. A mage sitting back home cannot reach out and level a fort.
	is_shown = {
		is_general = yes
		is_alive = yes
		NOT = { has_variable = is_the_revant }
		# This commander must be able to cast magic themselves.
		omni_character_is_eligible_caster = yes
		# ...and must be leading one of the player's own armies that is
		# currently conducting a siege.
		any_character_unit = {
			is_army = yes
			unit_owner = {
				this = scope:player
			}
			unit_location = {
				has_siege = yes
			}
		}
		scope:player = {
			is_ai = no
		}
	}

	ai_is_valid = {
		always = no
	}

	# is_valid drives the button's ENABLED state. is_shown (mage commander at a
	# siege) keeps it VISIBLE; this greys it out - exactly like Assume Control -
	# when the commander cannot afford even the cheapest siege spell.
	is_valid = {
		custom_tooltip = {
			text = omni_siege_magic_unit_button_tooltip
		}
		custom_tooltip = {
			text = omni_siege_need_mana_tt
			omni_can_afford_spell = { BASE = 25 }
		}
	}

	effect = {
		save_scope_as = target
		scope:player = {
			trigger_event = { id = me_zorgo_magic_focus.30 }
		}
	}
}
