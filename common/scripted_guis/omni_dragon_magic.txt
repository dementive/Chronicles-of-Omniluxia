omni_sb_dragon_magic_visible = {
    scope = country
    saved_scopes = { selected_mage }
    is_shown = {
        omni_has_casting_dragon_sovereign = yes
        exists = scope:selected_mage
        current_ruler = { this = scope:selected_mage }
    }
}

omni_sb_cast_dragon_cinderstorm = {
    scope = country
    saved_scopes = { selected_mage }
    is_shown = { omni_can_cast_dragon_magic = { SPELL = cinderstorm } }
    is_valid = {
        omni_can_cast_dragon_cinderstorm = yes
        exists = scope:selected_mage
        current_ruler = { this = scope:selected_mage }
    }
    ai_is_valid = { always = no }
    effect = { hidden_effect = { omni_cast_dragon_cinderstorm = yes } }
}
