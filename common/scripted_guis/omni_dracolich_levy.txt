omni_dracolich_raise_with_levy = {
	scope = country
	is_valid = {
		OR = {
			omni_can_deploy_dracolich = yes
			omni_can_deploy_wyrm = yes
		}
	}
	effect = {
		if = {
			limit = { omni_can_deploy_dracolich = yes }
			omni_deploy_dracolich = yes
		}
		else_if = {
			limit = { omni_can_deploy_wyrm = yes }
			omni_deploy_wyrm = yes
		}
	}
}
