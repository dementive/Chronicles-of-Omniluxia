# Lightweight menu-only indicator. It is evaluated only for visible mercenary
# list entries and deliberately checks the explicit per-captain trait. Generic
# caster eligibility is broader than this feature and must not show this icon.
omni_mercenary_magic_indicator = {
	scope = character
	saved_scopes = { player }

	is_shown = {
		is_mercenary = yes
		is_alive = yes
		has_trait = omni_magical_mercenary_trait
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		always = yes
	}
}
