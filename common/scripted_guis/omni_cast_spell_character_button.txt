# Character-window magic button.
# Main magic entry point for the viewed character: self-casting, spells cast on
# the character, empowering their commanded force, and siege magic when relevant.
omni_cast_spell_character_button = {
	scope = character
	saved_scopes = {
		player
	}

	is_shown = {
		is_alive = yes
		employer = scope:player
		employer = {
			NOT = { tag = MER }
			NOT = { tag = BAR }
			NOT = { tag = REB }
			NOT = { tag = PIR }
		}
		scope:player = {
			is_ai = no
			omni_player_has_eligible_mage = yes
		}
		OR = {
			AND = {
				omni_character_is_eligible_caster = yes
				OR = {
					omni_can_afford_spell = { BASE = 15 }
					AND = {
						has_trait = vampire_trait
						NOT = { has_variable = omni_vampiric_feeding_cooldown }
					}
				}
			}
			AND = {
				is_adult = yes
				is_mercenary = no
				scope:player = {
					omni_player_has_character_magic_ready = yes
				}
			}
			AND = {
				OR = {
					is_general = yes
					is_admiral = yes
				}
				any_character_unit = {
					OR = {
						is_army = yes
						is_navy = yes
					}
					unit_owner = {
						this = scope:player
					}
				}
			}
		}
	}

	ai_is_valid = {
		always = no
	}

	is_valid = {
		trigger_if = {
			limit = {
				NOT = {
					OR = {
						AND = {
							omni_character_is_eligible_caster = yes
							OR = {
								omni_can_afford_spell = { BASE = 15 }
								AND = {
									has_trait = vampire_trait
									NOT = { has_variable = omni_vampiric_feeding_cooldown }
								}
							}
						}
						AND = {
							is_adult = yes
							is_mercenary = no
							scope:player = {
								omni_player_has_character_magic_ready = yes
							}
						}
						AND = {
							OR = {
								is_general = yes
								is_admiral = yes
							}
							any_character_unit = {
								OR = {
									is_army = yes
									is_navy = yes
								}
								unit_owner = { this = scope:player }
							}
							scope:player = {
								omni_player_has_character_magic_ready = yes
							}
						}
						AND = {
							is_general = yes
							omni_character_is_eligible_caster = yes
							omni_can_afford_spell = { BASE = 25 }
							any_character_unit = {
								is_army = yes
								unit_owner = { this = scope:player }
								unit_location = { has_siege = yes }
							}
						}
					}
				}
			}
			custom_tooltip = {
				text = omni_cast_spell_no_affordable_mage_tooltip
				always = yes
			}
		}
	}

	effect = {
		save_scope_as = target
		scope:player = {
			trigger_event = { id = me_zorgo_magic_focus.40 }
		}
	}
}
