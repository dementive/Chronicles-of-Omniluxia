# Early tribal coalition, before the Drow escalation. Roll once per world per
# year from 1010; from 1020, form at the first opportunity with a reachable enemy.
yearly_country_pulse = {
    effect = {
        if = {
            limit = {
                is_ai = no
                current_date >= 1010.1.1
                NOT = { has_global_variable = zerywani_horde_formed }
                NOT = { has_global_variable = zerywani_first_annual_roll }
                any_country = { zerywani_horde_can_launch_trigger = yes }
            }
            set_global_variable = { name = zerywani_first_annual_roll days = 300 }
            random = {
                chance = 20
                modifier = {
                    factor = 5
                    current_date >= 1020.1.1
                }
                ordered_country = {
                    order_by = zerywani_horde_power_score_svalue
                    position = 0
                    limit = { zerywani_horde_can_launch_trigger = yes }
                    zerywani_horde_formation_effect = yes
                }
            }
        }
    }
}
