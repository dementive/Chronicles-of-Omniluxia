monthly_country_pulse = {
    on_actions = { omni_wild_dragon_pulse }
}

omni_wild_dragon_pulse = {
    trigger = { NOT = { has_global_variable = omni_dragon_month_lock } }
    effect = {
        set_global_variable = { name = omni_dragon_month_lock days = 27 }
        # Saves from before the scorched-land flag may already hold marked land.
        if = {
            limit = { NOT = { has_global_variable = omni_dragon_scorch_migrated } }
            set_global_variable = omni_dragon_scorch_migrated
            set_global_variable = omni_dragon_has_scorched
        }
        if = {
            limit = {
                exists = global_var:omni_dragon_horde
                global_var:omni_dragon_horde = {
                    unit_owner = { omni_is_wild_dragon_owner = yes }
                    in_combat = no
                    any_sub_unit = { sub_unit_type = omni_dragon }
                }
            }
            global_var:omni_dragon_horde = {
                # Wild hordes can pick up ordinary troops after a successful raid.
                every_sub_unit = { limit = { NOT = { sub_unit_type = omni_dragon } } destroy_subunit = yes }
                # Remember land a rebel dragon takes so it is handed back when it falls.
                if = {
                    limit = { exists = c:REB unit_owner = c:REB }
                    unit_location = {
                        if = {
                            limit = { has_owner = yes controller = c:REB }
                            set_variable = omni_dragon_scorched
                            set_global_variable = omni_dragon_has_scorched
                        }
                    }
                }
            }
        }
        # A dragon that has roosted on one continent long enough flies on.
        if = {
            limit = {
                exists = global_var:omni_dragon_horde
                global_var:omni_dragon_horde = {
                    unit_owner = { omni_is_wild_dragon_owner = yes }
                    in_combat = no
                    has_variable = omni_wild_morrigon
                    NOT = { has_variable = omni_dragon_roosting }
                    any_sub_unit = { sub_unit_type = omni_dragon }
                }
            }
            omni_wild_dragon_take_wing = yes
        }
        if = {
            limit = { NOT = { omni_wild_dragon_alive = yes } }
            omni_release_dragon_scorched_land = yes
            if = {
                limit = { has_global_variable = omni_dragon_in_field }
                remove_global_variable = omni_dragon_in_field
            }
            if = {
                limit = {
                    NOR = { has_global_variable = omni_dragon_hunt has_global_variable = omni_dragon_roll_lock }
                }
                set_global_variable = { name = omni_dragon_roll_lock days = 85 }
                random = { chance = 30 set_global_variable = omni_dragon_hunt }
            }
            if = {
                limit = { has_global_variable = omni_dragon_hunt }
                # Each new dragon rises on the next continent in the rotation.
                omni_spawn_rotating_wild_dragon = yes
                # Fallback: if no rebel dragon could be raised, wake a barbarian
                # horde or stir one of the barbarian strongholds as before.
                if = {
                    limit = { NOT = { exists = scope:omni_new_wild_dragon } exists = c:BAR }
                    c:BAR = {
                        random_unit = {
                            limit = { omni_dragon_horde_candidate = yes }
                            omni_awaken_wild_dragon = yes
                        }
                    }
                    # A den that has since been settled can never spawn a horde.
                    if = {
                        limit = {
                            exists = global_var:omni_dragon_den
                            global_var:omni_dragon_den = { has_owner = yes }
                        }
                        global_var:omni_dragon_den = { remove_province_modifier = omni_dragon_stirring }
                        remove_global_variable = omni_dragon_den
                    }
                    if = {
                        limit = {
                            NOT = { exists = global_var:omni_dragon_den }
                            NOT = { omni_wild_dragon_alive = yes }
                        }
                        random_province = {
                            limit = {
                                omni_dragon_den_candidate = yes
                            }
                            set_global_variable = { name = omni_dragon_den value = this }
                            add_province_modifier = { name = omni_dragon_stirring duration = -1 }
                        }
                    }
                }
            }
        }
    }
}

# Native diplomacy can dismiss a horde. The hunt survives that and resumes on
# another raid. Defeating its marked army in battle actually slays the dragon.
on_battle_lost_country = { on_actions = { omni_dragon_slain_in_battle } }

omni_dragon_slain_in_battle = {
    trigger = {
        omni_is_wild_dragon_owner = yes
        has_global_variable = omni_dragon_hunt
        OR = {
            AND = { exists = scope:actor scope:actor = { has_variable = omni_wild_morrigon } }
            # The engine may already have deleted a completely wiped stack.
            AND = {
                NOT = { exists = scope:actor }
                has_global_variable = omni_dragon_in_field
                NOT = { exists = global_var:omni_dragon_horde }
            }
        }
    }
    effect = {
        remove_global_variable = omni_dragon_hunt
        remove_global_variable = omni_dragon_in_field
        remove_global_variable = omni_dragon_horde
        set_global_variable = { name = omni_dragon_roll_lock days = 365 }
        if = {
            limit = { exists = scope:actor }
            scope:actor = { save_scope_as = omni_fallen_dragon }
        }
        # Finish surviving cohorts after the battle engine releases them.
        scope:target.unit_location = { save_scope_as = omni_dragon_battlefield }
        scope:target.unit_owner = { trigger_event = { id = omni_dragon.4 days = 1 } }
    }
}
