# Ported from Invictus (00_deficit_pulse.txt, "invictus_ai_rehydration").
# Scripted invasions spawn armies with create_unit, which the AI cannot disband. Without this,
# the Drow and Zerywani invaders sink into endless debt and their campaigns collapse.
# on_deficit_pulse only fires for countries already in deficit, and the trigger is two cheap
# checks, so this costs nothing for everyone else. Imperator stacks on_actions with the same key,
# so the vanilla/Omniluxia deficit random_events in 00_specific_from_code.txt still fire.
# Flag: is_not_allowed_to_go_broke - set in drow_invasion_formation_effect /
# zerywani_horde_formation_effect (10-year expiry as a safety net), removed in their cleanup effects.
on_deficit_pulse = {
	on_actions = {
		omni_invader_rehydration
	}
}

omni_invader_rehydration = {
	trigger = {
		is_ai = yes
		has_variable = is_not_allowed_to_go_broke
	}
	effect = {
		if = {
			limit = { treasury < 0 }
			add_treasury = {
				value = treasury
				multiply = -1
			}
			add_treasury = {
				value = has_monthly_income
				multiply = 9
				max = 300
			}
		}
		if = {
			limit = {
				exists = current_ruler
				current_ruler = { wealth < 50 }
			}
			current_ruler = { add_gold = 250 }
		}
		if = {
			limit = { political_influence < 0 }
			add_political_influence = {
				value = political_influence
				multiply = -1
			}
			add_political_influence = {
				value = { integer_range = { min = 20 max = 40 } }
			}
		}
	}
}
