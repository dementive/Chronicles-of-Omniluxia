				set_global_variable = omni_warborn_summoned
			}
		}
	}

# PERF: the five graduation events used to sit in the flavor list above, so every country ran five
# separate court scans every month. One combined scan now gates all five; the events (and their own
# triggers) only run for a country that actually has a grown-up pupil. Same events, same result.
omni_monthly_graduation_pulse = {

	trigger = {
		any_character = {
			OR = {
				has_variable = ec_land_var
				has_variable = ec_sea_var
				has_variable = ec_sophistry_var
				has_variable = ec_law_var
				has_variable = ec_admin_var
				has_variable = ec_trade_var
				has_variable = ec_secular_var
				has_variable = ec_religious_var
				has_variable = ec_omnic_var
				has_variable = ec_aldic_var
				has_variable = ec_amten_var
				has_variable = ec_melodian_var
			}
			is_adult = yes
		}
	}

	events = {

		omni_educate_child.7 #Graduation

		omni_educate_child.8 #Graduation

		omni_educate_child.9 #Graduation

		omni_educate_child.10 #Graduation

		omni_educate_child.11 #Graduation

	}

}
