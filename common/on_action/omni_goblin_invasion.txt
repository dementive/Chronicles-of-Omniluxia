# Goblin Invasion of the World's Lungs. Chained with on_actions = { } so other mods' hooks on the
# same pulses survive (see tools/fix_onaction_collisions.py).
yearly_country_pulse = { on_actions = { omni_goblin_invasion_yearly } }
monthly_country_pulse = { on_actions = { omni_goblin_root_ashes_monthly } }

# An eligible goblin realm can march once it controls both Skyreach passages.
omni_goblin_invasion_yearly = {
	trigger = {
		is_ai = yes
		NOT = { has_global_variable = goblin_invasion_fired }
		goblin_invasion_eligible_trigger = yes
		goblin_invasion_root_stands_trigger = yes
		goblin_invasion_controls_skyreach_trigger = yes
	}
	effect = {
		goblin_invasion_formation_effect = yes
	}
}

omni_goblin_root_ashes_monthly = {
	trigger = {
		has_global_variable = root_of_first_spring_felled
		OR = {
			owns = 2356
			has_country_modifier = root_of_first_spring_ashes_country
		}
	}
	effect = {
		root_of_first_spring_ashes_upkeep_effect = yes
	}
}
