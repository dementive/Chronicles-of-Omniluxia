monthly_country_pulse = {
	on_actions = {
		delay = { days = { 1 27 } }
		omni_engage_in_piracy_monthly_pulse
	}
}

omni_engage_in_piracy_monthly_pulse = {
	effect = {
		# every_unit is a GLOBAL scan and this on_action runs per country, so two
		# guards keep it cheap:
		#  - omni_active_piracy_count (maintained by the engage_in_piracy ability
		#    toggle) skips the scan entirely whenever no fleet is actively pirating.
		#  - omni_piracy_pulse_done is a 27-day global lock so only the FIRST
		#    country each month runs the world-wide scan, not every country.
		if = {
		limit = {
			has_global_variable = omni_active_piracy_count
			global_var:omni_active_piracy_count >= 1
			NOT = { has_global_variable = omni_piracy_pulse_done }
		}
		set_global_variable = { name = omni_piracy_pulse_done days = 27 }
		every_unit = {
			limit = {
				has_variable = engage_in_piracy_active
				is_navy = yes
				in_combat = no
				is_exiled = no
				unit_location = {
					any_neighbor_province = {
						is_sea = no
						has_owner = yes
						NOT = { owner = root }
						owner = {
							NOT = { alliance_with = root }
							NOT = { is_subject_of = root }
						}
					}
				}
			}
			hidden_effect = {
				save_scope_as = omni_piracy_fleet
				scope:omni_piracy_fleet.unit_owner = {
					save_scope_as = omni_piracy_raider_country
				}
				unit_location = {
					random_neighbor_province = {
						limit = {
							is_sea = no
							has_owner = yes
							NOT = { owner = scope:omni_piracy_raider_country }
							owner = {
								NOT = { alliance_with = scope:omni_piracy_raider_country }
								NOT = { is_subject_of = scope:omni_piracy_raider_country }
							}
						}
						save_scope_as = omni_piracy_raided_province
						add_province_modifier = {
							name = engage_in_piracy_raided_port_modifier
							duration = 35
						}
						owner = {
							add_opinion = {
								modifier = opinion_engage_in_piracy_raided_us
								target = scope:omni_piracy_raider_country
							}
							if = {
								limit = {
									NOT = { has_variable = engage_in_piracy_raid_event_cooldown }
									NOT = { war_with = scope:omni_piracy_raider_country }
								}
								set_variable = {
									name = engage_in_piracy_raid_event_cooldown
									days = 180
								}
								trigger_event = {
									id = omni_piracy.1
								}
							}
						}
					}
				}
				scope:omni_piracy_raider_country = {
					random_list = {
						300 = { set_variable = { name = omni_piracy_num_raids value = 1 } }
						300 = { set_variable = { name = omni_piracy_num_raids value = 3 } }
						200 = { set_variable = { name = omni_piracy_num_raids value = 5 } }
						200 = { set_variable = { name = omni_piracy_num_raids value = 8 } }
						150 = { set_variable = { name = omni_piracy_num_raids value = 12 } }
						100 = { set_variable = { name = omni_piracy_num_raids value = 15 } }
						 75 = { set_variable = { name = omni_piracy_num_raids value = 20 } }
						 50 = { set_variable = { name = omni_piracy_num_raids value = 25 } }
						 25 = { set_variable = { name = omni_piracy_num_raids value = 30 } }
					}
					random_list = {
						40 = { trigger_event = { id = omni_piracy.10 } }
						25 = { trigger_event = { id = omni_piracy.11 } }
						25 = { trigger_event = { id = omni_piracy.12 } }
						10 = { trigger_event = { id = omni_piracy.13 } }
					}
				}
			}
		}
		} # closes piracy-active + once-per-month gate
	}
}
