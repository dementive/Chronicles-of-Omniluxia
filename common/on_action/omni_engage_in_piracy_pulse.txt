monthly_country_pulse = {
	on_actions = {
		omni_engage_in_piracy_monthly_pulse
	}
}

omni_engage_in_piracy_monthly_pulse = {
	effect = {
		engage_in_piracy_refresh_country_income_effect = yes
		every_unit = {
			limit = {
				is_unit_ability_used = engage_in_piracy
				NOT = { has_variable = engage_in_piracy_recently_disabled }
				is_navy = yes
				in_combat = no
				is_exiled = no
				unit_location = {
					any_neighbor_province = {
						engage_in_piracy_target_province_trigger = yes
						NOT = { owner = root }
						owner = {
							NOT = { alliance_with = root }
							NOT = { is_subject_of = root }
						}
					}
				}
			}
			engage_in_piracy_apply_privateering_effect = yes
			engage_in_piracy_apply_current_fleet_commerce_fallback_effect = yes
		}
	}
}
