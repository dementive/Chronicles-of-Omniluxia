# New-campaign fleet; initialization runs in country scope.
# Spawn is delayed one day: at on_game_initialized the port building in
# Srerescy (360) does not exist yet, so create_unit rejects the naval unit.
on_game_initialized = { on_actions = { omni_cymeva_starting_navy } }

omni_cymeva_starting_navy = {
    trigger = {
        tag = ACY
        owns = 360
        NOT = { has_variable = omni_cymeva_starting_navy_created }
    }
    effect = {
        set_variable = omni_cymeva_starting_navy_created
        trigger_event = { id = omni_cymeva_navy.1 days = 1 }
    }
}
