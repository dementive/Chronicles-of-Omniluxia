# OMNILUXIA - DEDICATED BUILDING EVENT PULSE
# ----------------------------------------------------------------------------------------------------------------------
# These events used to live inside main_event_pulse_country, competing against a ~26500 weight pool. That gave the whole
# set of 45 building events a ~3.4% chance per country-year of even being rolled, so the AI almost never saw them.
# They now live here with their own pool, and the pulse trigger short-circuits before any province scan happens.
#
# PERF: every one of these events opens with NOT = { has_variable = <category>_building_cooldown } and only then runs an
# any_owned_province scan. The pulse-level trigger below checks all three cooldowns first, so a country that is on
# cooldown for every category never enters the random_events block at all - no province iteration, no event triggers.
# ----------------------------------------------------------------------------------------------------------------------

omni_building_event_pulse = {
	trigger = {
		num_of_cities >= 1
		NOR = {
			tag = REB
			tag = BAR
		}
		# Cheap gate: if all three category cooldowns are running, every event below would fail anyway.
		NOT = {
			AND = {
				has_variable = military_building_cooldown
				has_variable = economy_building_cooldown
				has_variable = population_building_cooldown
			}
		}
	}

	random_events = {
		# ~33% chance per year to attempt a building event. Category cooldowns are 25 years, so this is deliberately
