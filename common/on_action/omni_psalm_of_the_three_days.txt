# Every new master of the Balthazarian Order learns the Psalm of the Three Days.
# on_ruler_change covers a new ruler; the monthly check also catches a new co-ruler.
on_ruler_change = { on_actions = { omni_psalm_teach_new_masters } }
monthly_country_pulse = { on_actions = { omni_psalm_teach_new_masters } }
omni_psalm_teach_new_masters = {
	trigger = { omni_psalm_master_needs_teaching = yes }
	effect = { omni_psalm_teach_order_masters = yes }
}
