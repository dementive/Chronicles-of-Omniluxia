on_ownership_change = {
	on_actions = { omni_mage_conquest_demotion }
}

omni_mage_conquest_demotion = {
	# Check the war while the ownership change is being processed.
	trigger = {
		has_city_status = yes
		num_of_mages >= 1
		exists = owner
		owner_at_war_with_previous_owner = yes
	}

	effect = {
		every_pops_in_province = {
			limit = { pop_type = mages }
			random_list = {
				25 = { set_pop_type = citizen }
				75 = {}
			}
		}
	}
}
