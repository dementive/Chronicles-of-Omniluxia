on_ruler_change = {
	on_actions = { omni_seedwatcher_succession omni_wenelru_succession_rite }
}

monthly_country_pulse = {
	on_actions = { omni_seedwatcher_ensure_office }
}

omni_seedwatcher_succession = {
	trigger = { tag = LF6 }
	effect = {
		if = {
			limit = {
				exists = scope:former_ruler
				scope:former_ruler = {
					is_alive = yes
					has_trait = seedwatcher_trait
				}
			}
			scope:former_ruler = {
				remove_trait = seedwatcher_trait
				recalculate_character_magic_vars = yes
			}
		}
		omni_seedwatcher_install = yes
	}
}

omni_seedwatcher_ensure_office = {
	trigger = { tag = LF6 }
	effect = { omni_seedwatcher_install = yes }
}

omni_wenelru_succession_rite = {
	effect = {
		if = {
			limit = {
				government = wenelru_settled_tribe
				exists = scope:former_ruler
				exists = current_ruler
				omni_wenelru_covenant = yes
				any_owned_province = { has_province_modifier = omni_wenelru_covenant_hearth }
				any_character = { omni_wenelru_delegate = yes }
			}
			random_character = {
				limit = { omni_wenelru_delegate = yes }
				save_scope_as = wenelru_delegate
			}
			current_ruler = { save_scope_as = wenelru_successor }
			trigger_event = { id = omni_wenelru.4 }
		}
	}
}
