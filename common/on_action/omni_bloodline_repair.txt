# One global catch-up for existing saves, on the first monthly country pulse.
# Do not re-seed current ruling families: repair only genuine parent inheritance.
monthly_country_pulse = {
    effect = {
        if = {
            limit = { NOT = { has_global_variable = omni_bloodline_initialization_v2 } }
            every_living_character = { set_racial_trait_effect = yes }
            bloodline_setup_effect = yes
            set_global_variable = { name = omni_bloodline_initialization_v2 value = 1 }
        }
    }
}

yearly_country_pulse = {
    effect = {
        if = {
            # Global sweep: guard globally, or every country repeats it once.
            limit = { NOT = { has_global_variable = omni_bloodline_race_repair_v1 } }
            set_global_variable = omni_bloodline_race_repair_v1
            every_living_character = { omni_cleanup_invalid_bloodlines_effect = yes }
        }
    }
}
