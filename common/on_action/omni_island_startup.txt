on_game_initialized = {
	on_actions = { omni_island_dragon_levies omni_island_starting_navies }
}

omni_island_starting_navies = {
	trigger = { OR = { tag = AZQ tag = QTB tag = QTR tag = ESQ tag = SVI tag = ZBR tag = RXD tag = TXD tag = VQR tag = THQ } }
	effect = {
		# Starting ports are built after on_game_initialized.
		trigger_event = { id = omni_island_startup.1 days = 1 }
	}
}

omni_island_dragon_levies = {
	trigger = {
		OR = { tag = RXD tag = TXD }
		government = omni_dragon_dominion
	}
	effect = { omni_set_dragon_levies = yes }
}
