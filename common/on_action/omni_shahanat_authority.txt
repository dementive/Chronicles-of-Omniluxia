monthly_country_pulse = {
    effect = {
        if = {
            limit = {
                OR = {
                    omni_hc_shannah_trigger = yes
                    has_country_modifier = omni_hc_authority_weak
                    has_country_modifier = omni_hc_authority_fragile
                    has_country_modifier = omni_hc_authority_established
                    has_country_modifier = omni_hc_authority_respected
                    has_country_modifier = omni_hc_authority_supreme
                }
            }
            omni_hc_sync_authority_effect = yes
        }
    }
}

# Registered in yearly_country_pulse. Institutional inertia: high Authority must be
# maintained through councils, recognition and pastoral service rather than banked.
omni_hc_authority_yearly = {
    trigger = { omni_hc_shannah_trigger = yes has_variable = omni_hc_authority_base }
    effect = {
        if = { limit = { var:omni_hc_authority_base > 75 } omni_hc_change_authority_effect = { AMOUNT = -2 } }
        else_if = { limit = { var:omni_hc_authority_base > 50 } omni_hc_change_authority_effect = { AMOUNT = -1 } }
    }
}
