# ----------------------------------------------------------------------------------------------------------------------
# Omniluxia AI pulse - anbeeld's Advanced AI systems, always enabled
# Adds to existing monthly/yearly pulses without replacing them
# ----------------------------------------------------------------------------------------------------------------------

monthly_country_pulse = {
	on_actions = {
		omniluxia_ai_monthly_pulse
	}
}

omniluxia_ai_monthly_pulse = {
	effect = {
		# Monthly country pulses are frequent and run once per country. Keep this cheap.
		# The expensive AI scans and construction decisions run from the yearly pulse.
		ai_avoid_to_invoke_devotio = yes
		# Global effect guarded by a 28-day global variable: the first country to tick runs it for
		# every AI at war, every other country only pays one has_global_variable check. Restored to
		# the monthly pulse (as in Invictus) so AI hires mercenaries all year, not just Jan-Oct.
		ai_recruit_mercenaries = yes
	}
}

yearly_country_pulse = {
	on_actions = {
		omniluxia_ai_yearly_pulse
		delay = { days = 181 }
		omniluxia_ai_midyear_pulse
	}
}

# Each country keeps the SAME slot in the year, every year. The old days = { 0 300 } re-rolled
# annually, so a country that drew day 250 one year and day 40 the next ran its "yearly" AI pass
# only 155 days apart. Every effect in the chain guards itself with a 345-day variable, so that
# second pass was silently skipped: roads, pop moving, economic policy and the research cache
# were lost for the whole year (roughly 4 years in 10 per country). A fixed slot keeps passes
# 360-370 days apart, which always clears the 345-day guards, and still spreads the world's
# countries over 12 slots so there is no Jan 1 spike.
omniluxia_ai_yearly_pulse = {
	effect = {
		if = {
			limit = { NOT = { has_variable = omni_ai_year_slot } }
			random_list = {
				1 = { set_variable = { name = omni_ai_year_slot value = 0 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 1 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 2 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 3 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 4 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 5 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 6 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 7 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 8 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 9 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 10 } }
				1 = { set_variable = { name = omni_ai_year_slot value = 11 } }
			}
		}
		if = { limit = { var:omni_ai_year_slot = 0 } trigger_event = { id = omni_perf.3 days = { 0 5 } } }
		else_if = { limit = { var:omni_ai_year_slot = 1 } trigger_event = { id = omni_perf.3 days = { 25 30 } } }
		else_if = { limit = { var:omni_ai_year_slot = 2 } trigger_event = { id = omni_perf.3 days = { 50 55 } } }
		else_if = { limit = { var:omni_ai_year_slot = 3 } trigger_event = { id = omni_perf.3 days = { 75 80 } } }
		else_if = { limit = { var:omni_ai_year_slot = 4 } trigger_event = { id = omni_perf.3 days = { 100 105 } } }
		else_if = { limit = { var:omni_ai_year_slot = 5 } trigger_event = { id = omni_perf.3 days = { 125 130 } } }
		else_if = { limit = { var:omni_ai_year_slot = 6 } trigger_event = { id = omni_perf.3 days = { 150 155 } } }
		else_if = { limit = { var:omni_ai_year_slot = 7 } trigger_event = { id = omni_perf.3 days = { 175 180 } } }
		else_if = { limit = { var:omni_ai_year_slot = 8 } trigger_event = { id = omni_perf.3 days = { 200 205 } } }
		else_if = { limit = { var:omni_ai_year_slot = 9 } trigger_event = { id = omni_perf.3 days = { 225 230 } } }
		else_if = { limit = { var:omni_ai_year_slot = 10 } trigger_event = { id = omni_perf.3 days = { 250 255 } } }
		else_if = { limit = { var:omni_ai_year_slot = 11 } trigger_event = { id = omni_perf.3 days = { 275 280 } } }
	}
}

# Colonization kept on its own schedule (the old midyear offset existed so colonization funds
# don't compete with the winter food-import window) - same per-country spreading treatment.
omniluxia_ai_midyear_pulse = {
	effect = {
		trigger_event = { id = omni_perf.4 days = { 0 60 } }
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after ending war
# ----------------------------------------------------------------------------------------------------------------------

on_ending_war = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_ending_war
	}
}

omniluxia_ai_on_ending_war = {
	effect = {
		ai_destroy_leftover_navies = yes
		optimize_state_capitals_and_forts = {
			periodic = 0
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after integrating subjects
# ----------------------------------------------------------------------------------------------------------------------

on_diplomatic_annex = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_diplomatic_annex
	}
}

omniluxia_ai_on_diplomatic_annex = {
	effect = {
		ai_destroy_leftover_navies = yes
		scope:actor = {
			optimize_state_capitals_and_forts = {
				periodic = 0
			}
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after annexing territories
# ----------------------------------------------------------------------------------------------------------------------

on_ownership_change = {
	on_actions = {
		delay = { days = 1 }
		omniluxia_ai_on_ownership_change
	}
}

omniluxia_ai_on_ownership_change = {
	effect = {
		ai_flag_acquired_fort_territory = yes
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after state revolting
# ----------------------------------------------------------------------------------------------------------------------

on_state_secession = {
	on_actions = {
		omniluxia_ai_on_state_secession
	}
}

omniluxia_ai_on_state_secession = {
	effect = {
		ordered_state_province = {
			order_by = {
				value = total_population
				if = {
					limit = {
						is_state_capital = yes
					}
					add = 100000
				}
			}
			position = 0
			check_range_bounds = no

			add_to_global_variable_list = {
				name = territories_that_just_revolted
				target = this
			}
			trigger_event = {
				id = ai_war_events.1
				days = 1
			}
		}
	}
}
