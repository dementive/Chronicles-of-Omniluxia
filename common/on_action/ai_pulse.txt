# ----------------------------------------------------------------------------------------------------------------------
# Omniluxia AI pulse - anbeeld's Advanced AI systems, always enabled
# Adds to existing monthly/yearly pulses without replacing them
# ----------------------------------------------------------------------------------------------------------------------

monthly_country_pulse = {
	on_actions = {
		omniluxia_ai_monthly_pulse
	}
}

omniluxia_ai_monthly_pulse = {
	effect = {
		create_list_of_ownable_territories = yes  # Once per game      Global, for old saves and edge cases
		ai_avoid_to_invoke_devotio = yes          # Every month
		ai_recruit_mercenaries = yes              # Every month        Global, spends gold
		optimize_state_capitals_and_forts = {     # Every 1.5 years    Spends gold, potentially a lot
			periodic = 1
		}
		cache_adjusted_research_efficiency = yes  # Every year         For players too, prevents lag if switching
		ai_determine_economic_policies = yes      # Every year         Depends on cache_adjusted_research_efficiency
		ai_revoke_free_hands = yes                # Every half year    Before loyalty fix to compensate if needed
		ai_fix_character_loyalty = yes            # Every quarter      Spends a bit of political influence
		ai_choose_and_enact_law = yes             # Every half year    Spends stability and political influence
		ai_perform_divine_sacrifice = yes         # Every quarter      Spends political influence
		ai_manually_move_pops_to_cities = yes     # Every year         Spends gold
		ai_try_upgrading_city_to_metropolis = yes # Every half year    Spends political influence and gold
		ai_choose_territory_and_found_city = yes  # Every half year    Spends political influence and gold
		ai_perform_custom_road_building = yes     # Every year         Spends gold, potentially a lot
		ai_press_hold_games_button = yes          # Every quarter      Tribal development, spends gold
		ai_choose_and_make_investment = yes       # Every half year    Spends political influence and maybe gold
	}
}

yearly_country_pulse = {
	on_actions = {
		omniluxia_ai_yearly_pulse
		delay = { days = 181 }
		omniluxia_ai_midyear_pulse
	}
}

omniluxia_ai_yearly_pulse = {
	effect = {
		ai_import_food_into_states_in_need = yes  # Every year         Global, food is lowest during winter
		ai_enshrine_social_equity = yes           # Every year         Only if Mixed Gender Rules are enabled
		ai_destroy_leftover_navies = yes          # Every year
	}
}

omniluxia_ai_midyear_pulse = {
	effect = {
		ai_colonize_unowned_territories = yes     # Every year         Global, spends a bit of gold
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after ending war
# ----------------------------------------------------------------------------------------------------------------------

on_ending_war = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_ending_war
	}
}

omniluxia_ai_on_ending_war = {
	effect = {
		ai_destroy_leftover_navies = yes
		optimize_state_capitals_and_forts = {
			periodic = 0
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after integrating subjects
# ----------------------------------------------------------------------------------------------------------------------

on_diplomatic_annex = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_diplomatic_annex
	}
}

omniluxia_ai_on_diplomatic_annex = {
	effect = {
		ai_destroy_leftover_navies = yes
		scope:actor = {
			optimize_state_capitals_and_forts = {
				periodic = 0
			}
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after annexing territories
# ----------------------------------------------------------------------------------------------------------------------

on_ownership_change = {
	on_actions = {
		delay = { days = 1 }
		omniluxia_ai_on_ownership_change
	}
}

omniluxia_ai_on_ownership_change = {
	effect = {
		ai_flag_acquired_fort_territory = yes
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after state revolting
# ----------------------------------------------------------------------------------------------------------------------

on_state_secession = {
	on_actions = {
		omniluxia_ai_on_state_secession
	}
}

omniluxia_ai_on_state_secession = {
	effect = {
		ordered_state_province = {
			order_by = {
				value = total_population
				if = {
					limit = {
						is_state_capital = yes
					}
					add = 100000
				}
			}
			position = 0
			check_range_bounds = no

			add_to_global_variable_list = {
				name = territories_that_just_revolted
				target = this
			}
			trigger_event = {
				id = ai_war_events.1
				days = 1
			}
		}
	}
}
