# ----------------------------------------------------------------------------------------------------------------------
# Omniluxia AI pulse - anbeeld's Advanced AI systems, always enabled
# Adds to existing monthly/yearly pulses without replacing them
# ----------------------------------------------------------------------------------------------------------------------

monthly_country_pulse = {
	on_actions = {
		omniluxia_ai_monthly_pulse
	}
}

omniluxia_ai_monthly_pulse = {
	effect = {
		# Monthly country pulses are frequent and run once per country. Keep this cheap.
		# The expensive AI scans and construction decisions run from the yearly pulse.
		ai_avoid_to_invoke_devotio = yes
	}
}

yearly_country_pulse = {
	on_actions = {
		omniluxia_ai_yearly_pulse
		delay = { days = 181 }
		omniluxia_ai_midyear_pulse
	}
}

omniluxia_ai_yearly_pulse = {
	effect = {
		omni_court_arcana_yearly = yes
		create_list_of_ownable_territories = yes
		ai_recruit_mercenaries = yes
		optimize_state_capitals_and_forts = { periodic = 1 }
		cache_adjusted_research_efficiency = yes
		ai_determine_economic_policies = yes
		ai_revoke_free_hands = yes
		ai_fix_character_loyalty = yes
		ai_choose_and_enact_law = yes
		ai_perform_divine_sacrifice = yes
		ai_manually_move_pops_to_cities = yes
		ai_try_upgrading_city_to_metropolis = yes
		ai_choose_territory_and_found_city = yes
		ai_perform_custom_road_building = yes
		ai_press_hold_games_button = yes
		ai_choose_and_make_investment = yes
		ai_import_food_into_states_in_need = yes  # Every year         Global, food is lowest during winter
		ai_enshrine_social_equity = yes           # Every year         Only if Mixed Gender Rules are enabled
		ai_destroy_leftover_navies = yes          # Every year
		omni_ai_study_magic_effect = yes          # Yearly chance      Uses Magic Academies to train useful mages
		omni_ai_seduce_character_effect = yes     # Yearly chance      Allows lustful AI characters and spouses to seduce
	}
}

omniluxia_ai_midyear_pulse = {
	effect = {
		ai_colonize_unowned_territories = yes     # Every year         Global, spends a bit of gold
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after ending war
# ----------------------------------------------------------------------------------------------------------------------

on_ending_war = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_ending_war
	}
}

omniluxia_ai_on_ending_war = {
	effect = {
		ai_destroy_leftover_navies = yes
		optimize_state_capitals_and_forts = {
			periodic = 0
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after integrating subjects
# ----------------------------------------------------------------------------------------------------------------------

on_diplomatic_annex = {
	on_actions = {
		delay = { days = 2 }
		omniluxia_ai_on_diplomatic_annex
	}
}

omniluxia_ai_on_diplomatic_annex = {
	effect = {
		ai_destroy_leftover_navies = yes
		scope:actor = {
			optimize_state_capitals_and_forts = {
				periodic = 0
			}
		}
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after annexing territories
# ----------------------------------------------------------------------------------------------------------------------

on_ownership_change = {
	on_actions = {
		delay = { days = 1 }
		omniluxia_ai_on_ownership_change
	}
}

omniluxia_ai_on_ownership_change = {
	effect = {
		ai_flag_acquired_fort_territory = yes
	}
}

# ----------------------------------------------------------------------------------------------------------------------
# AI actions after state revolting
# ----------------------------------------------------------------------------------------------------------------------

on_state_secession = {
	on_actions = {
		omniluxia_ai_on_state_secession
	}
}

omniluxia_ai_on_state_secession = {
	effect = {
		ordered_state_province = {
			order_by = {
				value = total_population
				if = {
					limit = {
						is_state_capital = yes
					}
					add = 100000
				}
			}
			position = 0
			check_range_bounds = no

			add_to_global_variable_list = {
				name = territories_that_just_revolted
				target = this
			}
			trigger_event = {
				id = ai_war_events.1
				days = 1
			}
		}
	}
}
