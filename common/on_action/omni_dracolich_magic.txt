monthly_country_pulse = { on_actions = { omni_ai_dracolich_magic } }

omni_ai_dracolich_magic = {
    trigger = {
        is_ai = yes
        NOT = { has_variable = omni_dracolich_magic_recovery }
        exists = current_ruler
        current_ruler = { has_trait = omni_dracolich_trait is_alive = yes prisoner = no }
    }
    effect = {
        current_ruler = { save_scope_as = selected_mage }
        if = {
            limit = { omni_can_cast_dracolich_eclipse = yes }
            omni_cast_dracolich_eclipse = yes
        }
        else_if = {
            limit = { omni_can_cast_dracolich_bone_levy = yes }
            omni_cast_dracolich_bone_levy = yes
        }
        else_if = {
            limit = { omni_can_cast_dracolich_silent_throne = yes }
            omni_cast_dracolich_silent_throne = yes
        }
    }
}
