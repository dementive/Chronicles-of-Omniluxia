scheme_purge_court_start = {
	effect = {
		set_variable = { name = ongoing_scheme_target value = this days = 400 }
		add_character_modifier = { name = ongoing_scheme duration = 400 }
		employer = { add_tyranny = 5 }
		add_corruption = 5
	}
}

scheme_purge_court_monthly = {
	random_events = {
		2 = 0
		1 = scheme_purge_court.1
	}
}

scheme_purge_court_finish = {
	effect = {
		custom_tooltip = purge_court_finish_tt
		employer = { add_tyranny = 3 }
		add_corruption = 3
		remove_scheme_variables_effect = yes
		set_variable = { name = purge_court_cooldown days = 1095 }
	}
}

scheme_purge_court_abort = {
	effect = {
		# Used by the character-window "Persuade to Drop Scheme" button.
		# Stopping early removes the purge's target marker and ongoing cost modifier,
		# but does not apply the completion costs or cooldown.
		remove_scheme_variables_effect = yes
	}
}
