scheme_purge_court_start = {
	effect = {
		set_variable = { name = ongoing_scheme_target value = this days = 400 }
		add_character_modifier = { name = ongoing_scheme duration = 400 }
		employer = { add_tyranny = 5 }
		add_corruption = 5
	}
}

scheme_purge_court_monthly = {
	effect = {
		# A queued hearing can outlive its accused when the game continues.
		if = {
			limit = {
				has_variable = purge_court_pending_target
				trigger_if = {
					limit = { exists = var:purge_court_pending_target }
					var:purge_court_pending_target = {
						NOT = { purge_court_candidate_trigger = { RULER = root } }
					}
				}
				trigger_else = { always = yes }
			}
			remove_variable = purge_court_pending_target
		}
		if = {
			limit = {
				purge_court_active_trigger = yes
				NOT = { has_variable = purge_court_pending_target }
				employer = {
					any_character = { purge_court_candidate_trigger = { RULER = root } }
				}
			}
			if = {
				limit = { NOT = { has_variable = purge_court_opened_hearing } }
				purge_court_open_hearing_effect = yes
			}
			else = {
				random_list = {
					2 = { }
					1 = { purge_court_open_hearing_effect = yes }
				}
			}
		}
	}
}

scheme_purge_court_finish = {
	effect = {
		if = {
			limit = { has_variable = purge_court_resolved_hearing }
			custom_tooltip = purge_court_finish_tt
			employer = { add_tyranny = 3 }
			add_corruption = 3
			set_variable = { name = purge_court_cooldown days = 1095 }
			if = {
				limit = { has_variable = purge_court_removed_courtier }
				trigger_event = scheme_purge_court.3
			}
			else = {
				trigger_event = scheme_purge_court.4
			}
		}
		else = {
			trigger_event = scheme_purge_court.2
		}
		purge_court_cleanup_effect = yes
	}
}

scheme_purge_court_abort = {
	effect = {
		purge_court_abort_effect = yes
	}
}
