# Named children keep equipment work out of the shared pulse's effect block.
omni_item_registry_monthly_pulse = {
    effect = { omni_ensure_item_registry = yes omni_ai_ruler_item_catchup = yes }
}
omni_ai_ruler_equipment_yearly_pulse = {
    trigger = { is_ai = yes exists = current_ruler }
    effect = {
        omni_ensure_item_registry = yes
        # Let the registry repair event finish before purchasing or equipping.
        current_ruler = {
            trigger_event = { id = omni_annual_purchase.1 days = 2 }
        }
    }
}
