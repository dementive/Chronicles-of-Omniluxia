# Resume the Oath choice only if a war interrupted its open event.
monthly_country_pulse = {
    effect = {
        if = {
            limit = {
                has_variable = omni_eptelon_oath_reconvene
                war = no
                has_civil_war = no
                OR = {
                    is_subject = no
                    NOT = { has_truce_with = overlord }
                }
            }
            remove_variable = omni_eptelon_oath_reconvene
            trigger_event = me_eptelon_gen.61
        }
    }
}
