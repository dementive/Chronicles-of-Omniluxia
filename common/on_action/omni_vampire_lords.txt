# The northern Arteon vampire lords must start cursed even when all three
# countries are AI-controlled and their player-only lore events do not fire.
on_game_initialized = {
	on_actions = { omni_vampire_lords_startup }
}

omni_vampire_lords_startup = {
	trigger = {
		OR = { tag = MH3 tag = MH4 tag = AR7 }
	}
	effect = {
		current_ruler = {
			if = {
				limit = { NOT = { has_trait = vampire_trait } }
				add_trait = vampire_trait
			}
		}
	}
}
