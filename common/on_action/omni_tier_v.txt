# Tier V capstones: clear a realm's Oracle / Mindlord flags when its last one dies.
# The trigger is two trait checks on the dying character, so it costs nothing
# for every other death.
on_character_death = { on_actions = { omni_tier_v_on_death } }

omni_tier_v_on_death = {
	trigger = {
		OR = {
			has_trait = oracle_trait
			has_trait = mindlord_trait
		}
	}
	effect = { omni_tier_v_on_character_death = yes }
}
