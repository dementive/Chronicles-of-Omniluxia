# Explicitly registered before any random delays in the main pulse dispatchers.
omni_shannah_papal_yearly = {
    trigger = {
        is_ai = yes
        has_land = yes
        religion = helluvian
        exists = current_ruler
        current_ruler = { has_trait = shannah_trait }
    }
    effect = {
        # Existing saves without the counter begin a fresh three-year interval.
        if = {
            limit = { NOT = { has_variable = omni_shannah_council_years_remaining } }
            set_variable = { name = omni_shannah_council_years_remaining value = 3 }
        }
        change_variable = { name = omni_shannah_council_years_remaining subtract = 1 }
        if = {
            limit = { var:omni_shannah_council_years_remaining <= 0 }
            omni_shannah_ai_council_effect = yes
        }
    }
}

# Existing/new saves get their first council on the next monthly pulse.
# The permanent country flag prevents further bootstrap scans or succession resets.
omni_shannah_papal_bootstrap = {
    trigger = {
        is_ai = yes
        has_land = yes
        religion = helluvian
        exists = current_ruler
        current_ruler = { has_trait = shannah_trait }
        NOT = { has_variable = omni_shannah_ai_council_initialized }
    }
    effect = { omni_shannah_ai_council_effect = yes }
}

on_ruler_change = { on_actions = { omni_shannah_papal_actions_on_ruler_change } }
omni_shannah_papal_actions_on_ruler_change = {
    effect = {
        if = {
            limit = {
                OR = {
                    has_variable = omni_shannah_crown_recognized
                    AND = {
                        exists = scope:former_ruler
                        scope:former_ruler = { has_variable = omni_shannah_crown_recognized }
                    }
                }
            }
            omni_shannah_remove_recognized_crown_effect = yes
        }
    }
}

on_character_death = { on_actions = { omni_shannah_papal_actions_on_character_death } }
omni_shannah_papal_actions_on_character_death = {
    effect = {
        if = {
            limit = {
                has_variable = omni_shannah_crown_recognized
                exists = employer
            }
            employer = {
                omni_shannah_remove_recognized_crown_effect = yes
            }
        }
    }
}

monthly_country_pulse = { on_actions = { omni_shannah_papal_actions_monthly_country_pulse } }
omni_shannah_papal_actions_monthly_country_pulse = {
    effect = {
        if = {
            limit = {
                has_variable = omni_shannah_crown_recognized
                OR = {
                    NOT = { religion = helluvian }
                    is_monarchy = no
                }
            }
            omni_shannah_remove_recognized_crown_effect = yes
        }
        # AUDIT FIX: a realm that leaves the faith stops being excommunicated. Otherwise the
        # permanent -10% same-religion happiness penalty landed on its new religion's pops.
        if = {
            limit = {
                has_country_modifier = omni_shannah_excommunicated
                NOT = { religion = helluvian }
            }
            save_scope_as = omni_shannah_lapsed_realm
            remove_country_modifier = omni_shannah_excommunicated
            if = { limit = { has_variable = omni_shannah_penance_period } remove_variable = omni_shannah_penance_period }
            every_country = {
                remove_opinion = { target = scope:omni_shannah_lapsed_realm modifier = omni_shannah_excommunication_opinion }
            }
        }
    }
}
