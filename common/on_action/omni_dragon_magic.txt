    trigger = {
        is_ai = yes
        omni_has_casting_dragon_sovereign = yes
        NOT = { has_variable = omni_dragon_magic_recovery }
    }
    effect = {
        if = {
            limit = { omni_can_cast_dragon_cinderstorm = yes }
            omni_cast_dragon_cinderstorm = yes
        }
        else_if = {
            limit = { omni_can_cast_dragon_aegis_of_scales = yes }
            omni_cast_dragon_aegis_of_scales = yes
        }
        else_if = {
            limit = { omni_can_cast_dragon_hearthfire_blessing = yes }
            omni_cast_dragon_hearthfire_blessing = yes
        }
    }
}
