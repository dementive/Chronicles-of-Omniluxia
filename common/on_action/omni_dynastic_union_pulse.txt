yearly_country_pulse = {
	events = {
		omni_dynastic_union.3
	}
}

on_ruler_change = {
	events = {
		omni_dynastic_union.4
		omni_dynastic_union.5
	}
	effect = {
		if = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				has_variable = dynastic_union_junior_spouse
				current_ruler = {
					is_child_of = var:dynastic_union_senior_spouse
					is_child_of = var:dynastic_union_junior_spouse
				}
			}
			trigger_event = {
				id = omni_dynastic_union.6
				days = 30
			}
		}
		if = {
			limit = {
				is_monarchy = yes
				has_land = yes
				any_subject = {
					is_subject_type = dynastic_union_subject
					has_land = yes
					has_variable = dynastic_union_senior_spouse
					has_variable = dynastic_union_junior_spouse
					root.current_ruler = {
						is_child_of = prev.var:dynastic_union_senior_spouse
						is_child_of = prev.var:dynastic_union_junior_spouse
					}
				}
			}
			trigger_event = {
				id = omni_dynastic_union.7
				days = 30
			}
		}
	}
}

monthly_country_pulse = {
	effect = {
		if = {
			limit = {
				omni_dynastic_marriage_available_alert_trigger = yes
			}
			set_variable = {
				name = omni_dynastic_marriage_available_alert
				days = 40
			}
		}
		else = {
			remove_variable = omni_dynastic_marriage_available_alert
		}
	}
}
