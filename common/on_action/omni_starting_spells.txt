# Game-start spell kits. on_game_initialized fires once per country; the kit runs a
# day later so mages created by on_game_initialized lore events are included.
on_game_initialized = { on_actions = { omni_sb_starting_spells_startup } }

omni_sb_starting_spells_startup = {
	trigger = { NOT = { has_variable = omni_sb_start_spells_done } }
	effect = {
		omni_sb_starting_spells_pass = yes
		trigger_event = { id = omni_starting_spells.1 days = 1 }
	}
}
