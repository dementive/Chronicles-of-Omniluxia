biyearly_country_pulse = {
	random_events = {
		125 = 0
		10 = state_improvement.9
		5 = family_events.10
	}
	effect = {
		every_character = {
			# Perf: same NOR gate as the one inside set_racial_trait_effect, hoisted here
			# so the effect isn't invoked at all for the (vast majority of) characters
			# that already have a racial trait. Behavior identical.
			limit = {
				NOR = {
					has_trait = forest_elf_trait
					has_trait = high_elf_trait
					has_trait = orc_trait
					has_trait = gold_dwarf_trait
					has_trait = copper_dwarf_trait
					has_trait = silver_dwarf_trait
					has_trait = snakefolk_trait
					has_trait = human_trait
					has_trait = goblin_trait
					has_trait = hobgoblin_trait
					has_trait = drow_trait
					has_trait = wraith_trait
					has_trait = zorg_trait
					has_trait = half_elf_trait
					has_trait = silver_halfling_trait
					has_trait = halfling_trait
					has_trait = gnome_trait
					has_trait = deep_gnome_trait
					has_trait = sea_elf_trait
					has_trait = kinones_trait
				}
			}
			set_racial_trait_effect = yes
		}
		# Hacky fix so the revant character can't have kids cause if he does their graphics are broken
		if = {
			limit = {
				exists = current_ruler
			}
			current_ruler = {
				if = {
					limit = {
						has_variable = is_the_revant
					}
					if = {
						limit = {
							exists = spouse
						}
						spouse = {
							death = {
								death_reason = death_decapitated
							}
						}
					}
					every_child = {
						death = {
							death_reason = death_being_born
						}
					}
				}
			}
		}
	}
}
