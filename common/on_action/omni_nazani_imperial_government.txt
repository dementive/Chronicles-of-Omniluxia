# Keep the title repair on the monthly pulse; do not redefine the civil-war
# victory hook, which sets civil_war_target for the base victory events.
# Also repair already completed civil wars in existing saves. This leaves
# religious governments and republics unchanged.
monthly_country_pulse = {
    effect = {
        if = {
            limit = {
                tag = WGA
                has_civil_war = no
                OR = {
                    government = despotic_monarchy
                    government = aristocratic_monarchy
                    government = stratocratic_monarchy
                    government = plutocratic_monarchy
                    government = feudal_monarchy
                }
            }
            change_government = imperium
        }
    }
}
