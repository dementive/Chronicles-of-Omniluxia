scheme_mana = {
	can_be_picked = {
		is_alive = yes
		is_adult = yes
		prisoner = no
		NOT = { has_trait = brain_damaged }
		NOT = { has_trait = goblin_trait }
		NOT = { has_variable = ongoing_scheme_target }
		NOT = { has_variable = on_omni_adventure }
		OR = {
			has_trait = mage_trait
			has_trait = archmage_trait
			has_trait = warlock_trait
			has_trait = cleric_trait
			has_trait = necromancer_trait
			has_trait = lich_trait
			has_trait = druid_trait
			has_trait = elementalist_trait
			has_trait = diviner_trait
			has_trait = hierophant_trait
			has_trait = illusionist_trait
			has_trait = vampire_trait
			has_trait = blood_angel_curse_trait
			has_variable = folk_unlocked_var
		}
		# Don't pick if already near max mana
		NOT = {
			char_has_magic = { value = 90 }
		}
	}

	finished_when = {
		custom_tooltip = {
			text = "scheme_ambition_fulfillment_always_yes"
		}
	}

	duration = 365

	skip_initial_abort = yes

	abort = {
		OR = {
			is_alive = no
			prisoner = yes
			has_character_modifier = indisposed_charmod
			has_variable = on_omni_adventure
			NOR = {
				has_trait = mage_trait
				has_trait = archmage_trait
				has_trait = warlock_trait
				has_trait = cleric_trait
				has_trait = necromancer_trait
				has_trait = lich_trait
				has_trait = druid_trait
				has_trait = elementalist_trait
				has_trait = diviner_trait
				has_trait = hierophant_trait
				has_trait = illusionist_trait
				has_trait = vampire_trait
				has_trait = blood_angel_curse_trait
				has_variable = folk_unlocked_var
			}
		}
	}

	chance = {
		base = 3
		modifier = {
			add = 3
			has_trait = archmage_trait
		}
		modifier = {
			add = 2
			has_trait = lich_trait
		}
		modifier = {
			add = 2
			has_trait = hierophant_trait
		}
		modifier = {
			add = -3
			char_has_magic = { value = 75 }
		}
		modifier = {
			add = 2
			has_trait = ambitious
		}
		modifier = {
			add = -5
			has_trait = content
		}
	}

	on_monthly = scheme_mana_monthly
	on_start = scheme_mana_start
	on_finish = scheme_mana_finish
	on_abort = scheme_abort
}
