# Imperial Mages - academy-trained battle casters. Glass cannon: the hardest-hitting
# cohort in the game, and the most fragile. Built to sit in a mixed army behind a
# screen of infantry; on their own they break fast and get overrun.
# Counters to them live in the other unit files (imperial_mages = X): light cavalry,
# horse archers and camels close the distance before a working is finished.
# Allow checks are ordered cheapest-first; any_owned_province exits on the first hit.
imperial_mages = {
	army = yes
	assault = yes
	enable = yes

	levy_tier = advanced	# required on every army unit; mages pop type is also advanced

			NOT = { has_law = magic_banned }
			invention = organized_mage_guilds_inv
