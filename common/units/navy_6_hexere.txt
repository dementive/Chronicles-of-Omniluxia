	# Keep in step with omni_state_has_ship_timber in omni_naval_triggers.txt.
	# Timber must be PRESENT in the state (a producing territory or an import),
	# not in surplus: surplus counts copies beyond the first and is exported away.

				is_importing_trade_good = wood
				is_importing_trade_good = cedar
				is_importing_trade_good = hardwood
