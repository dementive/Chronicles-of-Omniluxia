# Folk Mages - village hedge-casters. Cheaper, quicker to raise and hardier on the
# march than Imperial Mages, but weaker blasts and they shatter under a cavalry charge.
# No academy needed: any province with mage pops and a soulgem or elixir surplus.
folk_mages = {
	army = yes
	enable = yes

	levy_tier = advanced	# required on every army unit; mages pop type is also advanced

	allow = {	# state scope (recruiting state); country checks go through owner
		OR = {
			trade_good_surplus = { target = soulgems value > 0 }
			trade_good_surplus = { target = elixir value > 0 }
		}
		owner = {
			NOT = { has_law = magic_banned }
			# num_of_mages is generated from the pop-type database, which is not loaded yet when
			# units are parsed - suspected cause of the startup crash. Check pops directly.
			any_owned_province = {
				any_pops_in_province = { count >= 3 pop_type = mages }
			}
		}
	}

	maneuver = 2.5
	movement_speed = 2.5
	build_time = 90

	light_infantry = 1.9
	heavy_infantry = 1.6
	heavy_cavalry = 1.0
	warelephant = 1.7
	horse_archers = 0.85
	archers = 1.4
	chariots = 1.6
	light_cavalry = 1.1
	camels = 1.1
	folk_mages = 1.2
	supply_train = 2.0

	attrition_weight = 0.7
	attrition_loss = 0.03

	morale_damage_taken = 1.4
	strength_damage_taken = 1.5

	ai_max_percentage = 15
	build_cost = {
		gold = 20
		manpower = 1
	}
	food_consumption = 0.12
	food_storage = 3.0
}
