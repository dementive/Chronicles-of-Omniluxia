# Balthazarian Order task 32 (the night landing). Titles for the ruler and for the commander of
# the landing (scope ob_isle_sword), so the text stays right without a co-ruler.
# ob_isle_sword_is_ruler is set by ob_isle_save_scopes when the ruler leads the landing in person.
ob_isle_ruler_title = { type = country random_valid = no
    text = { localization_key = ob_isle_title_hearth trigger = { government = ob_order } }
    text = { localization_key = ob_isle_title_protector trigger = { government = ob_protectorate } }
    text = { localization_key = ob_isle_title_master trigger = { always = yes } }
}
ob_isle_ruler_title_cap = { type = country random_valid = no
    text = { localization_key = ob_isle_title_hearth_cap trigger = { government = ob_order } }
    text = { localization_key = ob_isle_title_protector_cap trigger = { government = ob_protectorate } }
    text = { localization_key = ob_isle_title_master_cap trigger = { always = yes } }
}
ob_isle_sword_title = { type = country random_valid = no
    text = { localization_key = ob_isle_title_sword trigger = { exists = current_co_ruler } }
    text = { localization_key = ob_isle_title_hearth trigger = { has_variable = ob_isle_sword_is_ruler government = ob_order } }
    text = { localization_key = ob_isle_title_protector trigger = { has_variable = ob_isle_sword_is_ruler government = ob_protectorate } }
    text = { localization_key = ob_isle_title_master trigger = { has_variable = ob_isle_sword_is_ruler } }
    text = { localization_key = ob_isle_title_captain trigger = { always = yes } }
}
ob_isle_sword_title_cap = { type = country random_valid = no
    text = { localization_key = ob_isle_title_sword_cap trigger = { exists = current_co_ruler } }
    text = { localization_key = ob_isle_title_hearth_cap trigger = { has_variable = ob_isle_sword_is_ruler government = ob_order } }
    text = { localization_key = ob_isle_title_protector_cap trigger = { has_variable = ob_isle_sword_is_ruler government = ob_protectorate } }
    text = { localization_key = ob_isle_title_master_cap trigger = { has_variable = ob_isle_sword_is_ruler } }
    text = { localization_key = ob_isle_title_captain_cap trigger = { always = yes } }
}
