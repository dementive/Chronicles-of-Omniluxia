# Province scope. The local name for wild dragons, by region. Arteon is the
# isle of dragons, so Artheousan is the default everywhere else.
omni_dragon_breed = {
    type = province
    random_valid = no

    text = {
        localization_key = omni_dragon_breed_kainaxar
        trigger = {
            OR = {
                is_in_region = dragons_point_region
                is_in_region = dragon_steppes_west_region
                is_in_region = dragon_steppes_east_region
                is_in_region = dragon_pass
                is_in_region = dragon_lands_west
                is_in_region = dragon_lands_east
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_hrimwyrm
        trigger = {
            OR = {
                is_in_region = polaris_glaciers
                is_in_region = polaris_landhold
                is_in_region = west_wild_northland
                is_in_region = east_wild_northland
                is_in_region = norrfalian_peninsula
                is_in_region = great_herue
                is_in_region = norrfalian_borderland
                is_in_region = errnorfalie_north
                is_in_region = errnorfalie_shout
                is_in_region = errnorfalie_center
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_steppe
        trigger = {
            OR = {
                is_in_region = zainddian_sttepes
                is_in_region = westland_sttepes
                is_in_region = coastland_sttepes
                is_in_region = central_sttepes
                is_in_region = north_sttepes
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_skyreach
        trigger = {
            OR = {
                is_in_region = the_iron_foothils_region
                is_in_region = bronze_foothils
                is_in_region = western_shout_skyreach_foothils
                is_in_region = eastern_shout_skyreach_foothils
                is_in_region = western_skyreach_foothils
                is_in_region = the_skyreach_passages
                is_in_region = the_dwarven_grandlands
                is_in_region = golden_foothills_eteradica
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_tidewyrm
        trigger = {
            OR = {
                is_in_region = dissolved_island_region
                is_in_region = great_mushroom_island
                is_in_region = minor_mushroom_isles
                is_in_region = jade_island_north_region
                is_in_region = jade_island_shout_region
                is_in_region = ant_island_west_region
                is_in_region = ant_island_central_region
                is_in_region = ant_island_east_region
                is_in_region = arame_island_north_region
                is_in_region = arame_island_west_region
                is_in_region = arame_island_east_region
                is_in_region = bloody_islands_remun_region
                is_in_region = bloody_islands_rendun_region
                is_in_region = austropetolian_archipelago
                is_in_region = northern_austropetolian_isle
                is_in_region = southern_austropetolian_isle
                is_in_region = central_omniluxian_isle
                is_in_region = middle_intermare_isles_region
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_fenwyrm
        trigger = {
            OR = {
                is_in_region = marshland_region
                is_in_region = wherhenland_intermarenican_coast
                is_in_region = central_wherhenland
                is_in_region = east_wherhenland
                is_in_region = north_wherhenland
                is_in_region = werhenssia
                is_in_region = worlds_lungs_forest_region
                is_in_region = peaceful_valeys_region
                is_in_region = deep_rift_valeys_region
                is_in_region = north_flusenland
                is_in_region = central_flusenland
                is_in_region = shout_flusenland
            }
        }
    }

    text = {
        localization_key = omni_dragon_breed_artheousan
    }
}
