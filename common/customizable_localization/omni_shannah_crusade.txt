# Country-scoped response text for the roster event's saved country rows.
omni_crusade_roster_status = {
    type = country
    text = {
        localization_key = omni_crusade_status_joined
        trigger = { has_variable = omni_crusade_joined }
    }
    text = {
        localization_key = omni_crusade_status_unfulfilled
        trigger = {
            has_variable = omni_crusade_pledged
            NOT = { has_variable = omni_crusade_joined }
            exists = var:omni_crusade_host
            var:omni_crusade_host = { has_variable = omni_crusade_active }
        }
    }
    text = {
        localization_key = omni_crusade_status_pledged
        trigger = {
            has_variable = omni_crusade_pledged
            NOT = { has_variable = omni_crusade_joined }
            exists = var:omni_crusade_host
            var:omni_crusade_host = { NOT = { has_variable = omni_crusade_active } }
        }
    }
    text = {
        localization_key = omni_crusade_status_donated
        trigger = { has_variable = omni_crusade_donated }
    }
    text = {
        localization_key = omni_crusade_status_unanswered
        trigger = {
            has_variable = omni_crusade_invited
            exists = var:omni_crusade_host
            var:omni_crusade_host = { has_variable = omni_crusade_active }
        }
    }
    text = {
        localization_key = omni_crusade_status_pending
        trigger = {
            has_variable = omni_crusade_invited
            exists = var:omni_crusade_host
            var:omni_crusade_host = { NOT = { has_variable = omni_crusade_active } }
        }
    }
    text = {
        localization_key = omni_crusade_status_refused
        trigger = {
            NOR = {
                has_variable = omni_crusade_joined
                has_variable = omni_crusade_pledged
                has_variable = omni_crusade_donated
                has_variable = omni_crusade_invited
            }
        }
    }
}
