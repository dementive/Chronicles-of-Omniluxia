zaraken_ensure_elder_keeper_trait_effect = {
    if = {
        limit = {
            exists = current_ruler
            current_ruler = { NOT = { has_trait = elder_keeper_trait } }
        }
        current_ruler = { force_add_trait = elder_keeper_trait }
    }
}

zaraken_establish_keeper_government_effect = {
    if = {
        limit = { NOT = { government = zaraken_custodial_monarchy } }
        change_government = zaraken_custodial_monarchy
    }
    zaraken_ensure_elder_keeper_trait_effect = yes
}

zaraken_apply_keeper_flag_effect = {
    if = {
        limit = { tag = ZAE has_variable = zaraken_keeper_victory }
        change_country_flag = "OMNI_KEEPER_NEO_ZARAKEN_FLAG"
    }
    else = {
        change_country_flag = "OMNI_KEEPERS_OF_THE_TOMB_FLAG"
    }
    set_variable = zaraken_keeper_flag_applied
}
