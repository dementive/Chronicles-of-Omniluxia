### OMNILUXIA CUSTOM ACHIEVEMENTS - UNLOCK & CHECK EFFECTS ###
#
# omni_unlock_achievement : the ONE way an achievement is ever granted.
#   - Sets the tracking variable (used by the window's "unlocked" state).
#   - Queues the minor unlock event one day later, so the player still gets the
#     side-window notice without firing a popup inline during the check pulse.
#   - Idempotent: safe to call every tick, only acts the first time.
#
# omni_check_achievements : evaluates every "state based" achievement for the
#   current country and unlocks any whose condition is now met. Called from the
#   monthly player pulse (common/on_action/omni_achievements.txt), so unlocks feel
#   immediate - like real Steam achievements - with no manual "re-check" button and
#   no per-frame cost (it only ever runs for the human player).
#
# Event-driven achievements (e.g. winning a battle led by a child) are unlocked
# straight from their on_action instead of being polled here.

omni_unlock_achievement = {
    if = {
        limit = {
            is_ai = no
            NOT = { has_variable = $ACH$ }
        }
        set_variable = $ACH$
        trigger_event = { id = $EVENT$ days = 1 }
    }
}

omni_check_achievements = {

        limit = { NOT = { has_variable = omni_ach_court_mage } omni_ach_court_mage_trigger = yes }
