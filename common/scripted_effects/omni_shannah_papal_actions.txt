# Actor country scope; called by the interaction after all gameplay effects.
# Saved characters preserve the actual parties, rather than a recipient's ruler.
omni_shannah_announce_effect = {
    set_variable = omni_shannah_council_action_performed
    save_scope_as = omni_papal_actor_country
    current_ruler = { save_scope_as = omni_papal_shannah }
    scope:target = {
        save_scope_as = omni_papal_target_ruler
        employer = { save_scope_as = omni_papal_target_country }
    }
    # Public decrees alter relations or war rights across the faith.
    if = {
        limit = { always = $PUBLIC$ }
        every_country = {
            limit = {
                has_land = yes
                OR = {
                    religion = helluvian
                    this = scope:omni_papal_actor_country
                    this = scope:omni_papal_target_country
                }
            }
            trigger_event = { id = $EVENT$ }
        }
    }
    else = {
        trigger_event = { id = $EVENT$ }
        scope:omni_papal_target_country = {
            if = {
                limit = { NOT = { this = scope:omni_papal_actor_country } }
                trigger_event = { id = $EVENT$ }
            }
        }
    }
}

# Favorable powers earn the recipient's gratitude toward the issuing realm.
omni_shannah_grace_effect = {
    scope:target.employer = {
        remove_opinion = { target = scope:actor modifier = omni_shannah_grace_opinion }
        add_opinion = { target = scope:actor modifier = omni_shannah_grace_opinion }
    }
}

omni_shannah_excommunicate_effect = {
        scope:actor = {
            # BALANCE: was 25 influence and no Authority cost.
            add_political_influence = -75
            omni_hc_change_authority_effect = { AMOUNT = -5 }
            set_variable = { name = omni_shannah_excommunication_cooldown days = 365 }
        }
        scope:target.employer = {
            add_country_modifier = { name = omni_shannah_excommunicated duration = -1 }
            # AUDIT FIX: after five years the AI council may reconcile the realm (see omni_shannah_papal_ai.txt).
            set_variable = { name = omni_shannah_penance_period days = 1825 }
            if = {
                limit = { has_variable = omni_shannah_crown_recognized }
                omni_shannah_remove_recognized_crown_effect = yes
            }
        }
        hidden_effect = {
            every_country = {
                limit = {
                    has_land = yes
                    religion = helluvian
                    NOT = { this = scope:target.employer }
                }
                remove_opinion = { target = scope:target.employer modifier = omni_shannah_excommunication_opinion }
                add_opinion = { target = scope:target.employer modifier = omni_shannah_excommunication_opinion }
            }
        }
        hidden_effect = {
            scope:actor = { omni_shannah_announce_effect = { EVENT = omni_shannah_powers.10 PUBLIC = yes } }
        }
    
}


omni_shannah_lift_excommunication_effect = {
        scope:target.employer = { remove_country_modifier = omni_shannah_excommunicated }
        hidden_effect = {
            every_country = {
                remove_opinion = { target = scope:target.employer modifier = omni_shannah_excommunication_opinion }
            }
        }
        hidden_effect = {
            omni_shannah_grace_effect = yes scope:actor = { omni_hc_pastoral_authority_effect = yes omni_shannah_announce_effect = { EVENT = omni_shannah_powers.11 PUBLIC = yes } }
        }
    
}

# Issue an offer from the Shannah. The recipient pays only after accepting.
omni_shannah_offer_crown_effect = {
    if = {
        limit = {
            omni_shannah_recognize_crown_potential_trigger = { TARGET = scope:target }
            omni_shannah_recognize_crown_allowed_trigger = yes
            scope:actor = { NOT = { has_variable = omni_shannah_crown_offer_pending } }
        }
        scope:actor = {
            save_scope_as = omni_crown_offer_see
            current_ruler = { save_scope_as = omni_crown_offer_shannah }
            if = {
                limit = { NOT = { has_variable = omni_shannah_crown_offer_serial } }
                set_variable = { name = omni_shannah_crown_offer_serial value = 0 }
            }
            change_variable = { name = omni_shannah_crown_offer_serial add = 1 }
            var:omni_shannah_crown_offer_serial = { save_scope_as = omni_crown_offer_serial }
            set_variable = { name = omni_shannah_crown_offer_pending days = 30 }
        }
        scope:target = { save_scope_as = omni_crown_offer_ruler }
        scope:target.employer = { trigger_event = { id = omni_shannah_powers.16 } }
    }
}

# Both the approved petition and the accepted offer use this transaction.
omni_shannah_recognize_crown_effect = {
        scope:actor = {
            add_political_influence = -10
            set_variable = { name = omni_shannah_recognize_crown_cooldown days = 365 }
        }
        scope:target = { set_variable = omni_shannah_crown_recognized }
        scope:target.employer = {
            add_political_influence = -25
            add_legitimacy = 10
            set_variable = omni_shannah_crown_recognized
            save_scope_as = omni_shannah_recognized_scope
        }
        hidden_effect = {
            every_country = {
                limit = {
                    has_land = yes
                    religion = helluvian
                    NOT = { this = scope:omni_shannah_recognized_scope }
                }
                save_scope_as = omni_helluvian_country
                remove_opinion = { target = scope:omni_shannah_recognized_scope modifier = omni_shannah_recognized_crown_opinion }
                add_opinion = { target = scope:omni_shannah_recognized_scope modifier = omni_shannah_recognized_crown_opinion }
                scope:omni_shannah_recognized_scope = {
                    remove_opinion = { target = scope:omni_helluvian_country modifier = omni_shannah_recognized_crown_opinion }
                    add_opinion = { target = scope:omni_helluvian_country modifier = omni_shannah_recognized_crown_opinion }
                }
            }
        }
        hidden_effect = {
            scope:actor = { omni_hc_pastoral_authority_effect = yes omni_shannah_announce_effect = { EVENT = omni_shannah_powers.13 PUBLIC = yes } }
        }
    
}


omni_shannah_remove_recognized_crown_effect = {
        save_scope_as = omni_shannah_recognized_scope
        remove_variable = omni_shannah_crown_recognized
        if = {
            limit = { exists = current_ruler }
            current_ruler = { remove_variable = omni_shannah_crown_recognized }
        }
        if = {
            limit = { exists = scope:former_ruler }
            scope:former_ruler = { remove_variable = omni_shannah_crown_recognized }
        }
        every_country = {
            save_scope_as = omni_other_country
            remove_opinion = { target = scope:omni_shannah_recognized_scope modifier = omni_shannah_recognized_crown_opinion }
            scope:omni_shannah_recognized_scope = {
                remove_opinion = { target = scope:omni_other_country modifier = omni_shannah_recognized_crown_opinion }
            }
        }
}

# Pure blessing outcome shared by paid interactions and the free annual duty.
# Replacement refreshes a previous blessing without stacking it.
omni_shannah_apply_blessing_effect = {
    scope:target.employer = {
        remove_country_modifier = omni_shannah_blessing
        add_country_modifier = { name = omni_shannah_blessing duration = 1825 }
    }
    hidden_effect = {
        omni_shannah_grace_effect = yes scope:actor = { omni_hc_pastoral_authority_effect = yes omni_shannah_announce_effect = { EVENT = omni_shannah_powers.12 PUBLIC = no } }
    }
}
