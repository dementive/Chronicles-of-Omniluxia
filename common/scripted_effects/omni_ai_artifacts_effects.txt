omni_ai_manage_artifacts_effect = {
	if = {
		limit = {
			is_ai = yes
			war = yes
			NOT = { has_variable = ai_equipped_artifacts }
		}
		omni_ai_equip_artifacts_for_war = yes
		set_variable = ai_equipped_artifacts
	}
	if = {
		limit = {
			is_ai = yes
			war = no
			has_variable = ai_equipped_artifacts
		}
		omni_ai_unequip_artifacts_for_peace = yes
		remove_variable = ai_equipped_artifacts
	}
}

# Country scope. $CATEGORY$ must be a category registered by
# omni_assign_artifact_item_categories (choo_items_effects.txt).
omni_ai_place_artifact = {
	if = {
		limit = {
			exists = treasure:$TREASURE$
			OR = {
				any_country_treasure = { THIS = treasure:$TREASURE$ }
				any_owned_holy_site = { any_province_treasure = { THIS = treasure:$TREASURE$ } }
			}
		}
		ordered_character = {
			limit = {
				omni_ai_artifact_commander_trigger = yes
				NOT = { has_item = { item_ID = $ITEM$ } }
				has_any_other_same_category_item_equipped = { item_category = $CATEGORY$ item_ID = $ITEM$ amount_per_category = 1 }
			}
			order_by = omni_ai_artifact_holder_score
			save_scope_as = omni_art_recipient_$TREASURE$
		}
		if = {
			limit = {
				NOT = { exists = scope:omni_art_recipient_$TREASURE$ }
				exists = current_ruler
				current_ruler = {
					NOT = { has_item = { item_ID = $ITEM$ } }
					has_any_other_same_category_item_equipped = { item_category = $CATEGORY$ item_ID = $ITEM$ amount_per_category = 1 }
				}
			}
			current_ruler = { save_scope_as = omni_art_recipient_$TREASURE$ }
		}
		# If every candidate already fills that slot, the artifact stays in the reliquary.
		if = {
			limit = { exists = scope:omni_art_recipient_$TREASURE$ }
			treasure:$TREASURE$ = { destroy_treasure = yes }
			scope:omni_art_recipient_$TREASURE$ = {
				give_character_an_artifact = { item_ID = $ITEM$ }
				if = {
					limit = { NOT = { has_variable = omni_ai_artifacts_held } }
					set_variable = { name = omni_ai_artifacts_held value = 0 }
				}
				change_variable = { name = omni_ai_artifacts_held add = 1 }
			}
		}
	}
}

# Country scope. For artifacts with no registered item category: no slot rule,
# but still spread across commanders by omni_ai_artifact_holder_score.

		limit = {
			exists = treasure:$TREASURE$
			OR = {
				any_country_treasure = { THIS = treasure:$TREASURE$ }
				any_owned_holy_site = { any_province_treasure = { THIS = treasure:$TREASURE$ } }
			}
		}
		ordered_character = {

			}
			order_by = omni_ai_artifact_holder_score
			save_scope_as = omni_art_recipient_$TREASURE$
		}
		if = {
			limit = {
				NOT = { exists = scope:omni_art_recipient_$TREASURE$ }
				exists = current_ruler
			}
			current_ruler = { save_scope_as = omni_art_recipient_$TREASURE$ }

			limit = { exists = scope:omni_art_recipient_$TREASURE$ }
			treasure:$TREASURE$ = { destroy_treasure = yes }
			scope:omni_art_recipient_$TREASURE$ = {
				give_character_an_artifact = { item_ID = $ITEM$ }

					limit = { NOT = { has_variable = omni_ai_artifacts_held } }

				}
				change_variable = { name = omni_ai_artifacts_held add = 1 }
			}
		}
	}
}
