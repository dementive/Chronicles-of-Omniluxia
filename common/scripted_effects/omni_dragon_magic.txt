# Country scope. The country keeps its cooldowns through a change of ruler.
omni_apply_dragon_spell = {
    add_country_modifier = { name = omni_dragon_$SPELL$ duration = $DURATION$ }
    set_variable = { name = omni_dragon_cd_$SPELL$ days = 720 }
    set_variable = { name = omni_dragon_magic_recovery days = 360 }
}

omni_cast_dragon_cinderstorm = {
    omni_apply_dragon_spell = { SPELL = cinderstorm DURATION = 180 }
}

omni_cast_dragon_aegis_of_scales = {
    omni_apply_dragon_spell = { SPELL = aegis_of_scales DURATION = 180 }
}

omni_cast_dragon_hearthfire_blessing = {
    omni_apply_dragon_spell = { SPELL = hearthfire_blessing DURATION = 360 }
}
