# Signature mechanic: Ride the Bounds
zaraken_ride_the_bounds_effect = {
    add_treasury = -100
    if = {
        limit = { has_variable = 03_zaraken_bounds_renewed }
        custom_tooltip = 03_zaraken_bounds_effect_renewed_tt
        hidden_effect = {
            every_owned_province = {
                limit = {
                    is_in_region = golden_foothills_eteradica
                    is_state_capital = yes
                }
                add_state_loyalty = 15
            }
        }
    }
    else = {
        custom_tooltip = 03_zaraken_bounds_effect_tt
        hidden_effect = {
            every_owned_province = {
                limit = {
                    is_in_region = golden_foothills_eteradica
                    is_state_capital = yes
                }
                add_state_loyalty = 10
            }
        }
    }
    current_ruler = { add_popularity = 5 }
    custom_tooltip = 03_zaraken_bounds_risk_tt
    hidden_effect = {
        random = {
            chance = 50
            random_character = {
                limit = {
                    is_head_of_family = yes
                    is_ruler = no
                }
                add_loyalty = zaraken_ruler_abroad_loyalty
            }
        }
        if = {
            limit = { has_variable = zaraken_bounds_ridden }
            change_variable = { name = zaraken_bounds_ridden add = 1 }
        }
        else = {
            set_variable = { name = zaraken_bounds_ridden value = 1 }
        }
        if = {
            limit = { has_variable = 03_zaraken_bounds_renewed }
            set_variable = { name = 03_zaraken_bounds_cooldown days = 1095 }
        }
        else = {
            set_variable = { name = 03_zaraken_bounds_cooldown days = 1825 }
        }
    }
}
