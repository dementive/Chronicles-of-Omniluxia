# Country scope: Pharon is the office of the reigning Baetepian dynast.
omni_pharon_ensure_ruler_trait_effect = {
    if = {
        limit = {
            government = egyptian_dynasty
            exists = current_ruler
            current_ruler = { NOT = { has_trait = Pharon } }
        }
        current_ruler = {
            force_add_trait = Pharon
        }
    }
}
