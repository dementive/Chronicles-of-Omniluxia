# One bounded court selection per AI country per yearly pulse. Characters pay
# the same 50 personal gold as a hut lesson; distance is deliberately ignored.
# Only spells with implemented AI casting are offered.

        limit = { is_ai = yes }

                employer = root

                omni_character_can_learn_folk = yes

                    AND = {
                        wealth >= 50
                        OR = {
                            NOT = { omni_knows_spell = { SPELL = golden_hands } }

                            AND = { omni_can_afford_spell = { BASE = 50 } OR = { omni_knows_spell = { SPELL = golden_hands } omni_knows_spell = { SPELL = channeled_stars } } }
                            AND = { omni_can_afford_spell = { BASE = 15 } omni_knows_spell = { SPELL = serpentine_scrying } }

                }
            }
            if = {
                limit = { wealth >= 50 OR = { NOT = { omni_knows_spell = { SPELL = golden_hands } } NOT = { omni_knows_spell = { SPELL = serpentine_scrying } } NOT = { omni_knows_spell = { SPELL = channeled_stars } } AND = { OR = { is_general = yes is_admiral = yes } OR = { NOT = { omni_knows_spell = { SPELL = unified_blades } } NOT = { omni_knows_spell = { SPELL = raging_dragon } } NOT = { omni_knows_spell = { SPELL = victorys_blessing } } NOT = { omni_knows_spell = { SPELL = bloodlust } } NOT = { omni_knows_spell = { SPELL = scramble } } NOT = { omni_knows_spell = { SPELL = water_walker } } NOT = { omni_knows_spell = { SPELL = berserker_rage } } NOT = { omni_knows_spell = { SPELL = great_flare } } NOT = { omni_knows_spell = { SPELL = blade_storm } } NOT = { omni_knows_spell = { SPELL = steal } } NOT = { omni_knows_spell = { SPELL = stinging_nettles } } NOT = { omni_knows_spell = { SPELL = serpentine_swords } } NOT = { omni_knows_spell = { SPELL = imperial_dragons_claw } } NOT = { omni_knows_spell = { SPELL = sword_of_jaoz } } NOT = { omni_knows_spell = { SPELL = hellas_light } } NOT = { omni_knows_spell = { SPELL = explosion } } } } } }
                random_list = {
                    1 = { trigger = { NOT = { omni_knows_spell = { SPELL = golden_hands } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = golden_hands } }
                    1 = { trigger = { NOT = { omni_knows_spell = { SPELL = serpentine_scrying } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = serpentine_scrying } }

                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = unified_blades } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = unified_blades } }
                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = raging_dragon } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = raging_dragon } }

                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = steal } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = steal } }
                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = stinging_nettles } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = stinging_nettles } }

                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = hellas_light } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = hellas_light } }
                    1 = { trigger = { OR = { is_general = yes is_admiral = yes } NOT = { omni_knows_spell = { SPELL = explosion } } } add_gold = -50 omni_folk_initiate_practitioner = yes omni_learn_spell = { SPELL = explosion } }
                }
            }
            if = {
                limit = { omni_character_is_eligible_caster = yes omni_can_afford_spell = { BASE = 15 } }

                        trigger = { omni_knows_spell = { SPELL = golden_hands } omni_can_afford_spell = { BASE = 50 } }

                            chance = 10

# Player-equivalent temporary folk buffs, only while actually fighting. Each
# schedules the same guarded expiry event as the player spell.
omni_ai_folk_battlecast = {
    clear_saved_scope = omni_folk_enemy
    clear_saved_scope = omni_folk_enemy_unit
    random_character_unit = {
        limit = { in_combat = yes OR = { is_army = yes is_navy = yes } }
        unit_location = {
            random_unit_in_province = {
                limit = { in_combat = yes exists = commander unit_owner = { war_with = scope:caster_country } commander = { is_alive = yes } }
                save_scope_as = omni_folk_enemy_unit
                commander = { save_scope_as = omni_folk_enemy }
            }
        }
    }
    if = { limit = { exists = scope:omni_folk_enemy } omni_ai_folk_offensive_cast = yes }
    if = {
        limit = {
            NOT = { has_variable = omni_ai_battle_cast_succeeded }
            omni_knows_spell = { SPELL = explosion }
            omni_can_afford_spell = { BASE = 75 }
            any_character_unit = { is_army = yes in_combat = yes unit_location = { owner = { war_with = scope:caster_country } } }
        }
        random_character_unit = {
            limit = { is_army = yes in_combat = yes unit_location = { owner = { war_with = scope:caster_country } } }
            unit_location = {
                add_city_smoke_effect = yes
                add_city_fire_effect = yes
                ordered_pops_in_province = {
                    order_by = pop_hapiness
                    max = { value = total_population divide = 5 }
                    min = { value = 0 }
                    kill_pop = yes
                }
                if = { limit = { has_province_modifier = shoishoni_folk_mod } remove_province_modifier = shoishoni_folk_mod }
                add_province_modifier = { name = shoishoni_folk_mod duration = 64 }
            }
        }
        omni_spell_mana_cost_effect = { BASE = 75 }
        set_variable = omni_ai_battle_cast_succeeded
    }
    if = {
        limit = { NOT = { has_variable = omni_ai_battle_cast_succeeded } }
    if = {
        limit = { omni_knows_spell = { SPELL = victorys_blessing } omni_can_afford_spell = { BASE = 25 } NOT = { has_trait = viktriec_folk_trait } }
        force_add_trait = viktriec_folk_trait
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = { name = omni_spell_active_viktriec_folk_trait days = 180 }
        set_variable = omni_ai_battle_cast_succeeded
        save_scope_as = remove_magic_target
        employer = { trigger_event = { id = me_zorgo_magic_focus.22 days = 180 } }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = unified_blades } omni_can_afford_spell = { BASE = 25 } NOT = { has_trait = milon_folk_trait } }
        force_add_trait = milon_folk_trait
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = { name = omni_spell_active_milon_folk_trait days = 180 }
        set_variable = omni_ai_battle_cast_succeeded
        save_scope_as = remove_magic_target
        employer = { trigger_event = { id = me_zorgo_magic_focus.22 days = 180 } }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = raging_dragon } omni_can_afford_spell = { BASE = 50 } NOT = { has_trait = dragkhan_folk_trait } }
        force_add_trait = dragkhan_folk_trait
        omni_spell_mana_cost_effect = { BASE = 50 }
        set_variable = { name = omni_spell_active_dragkhan_folk_trait days = 180 }
        set_variable = omni_ai_battle_cast_succeeded
        save_scope_as = remove_magic_target
        employer = { trigger_event = { id = me_zorgo_magic_focus.22 days = 180 } }

        force_add_trait = drow_folk_trait

omni_ai_folk_offensive_cast = {
    if = {
        limit = { has_trait = blood_angel_curse_trait omni_can_afford_spell = { BASE = 25 } }
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy_unit = { damage_unit_percent = 0.25 }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = great_flare } omni_can_afford_spell = { BASE = 200 }  }
        omni_spell_mana_cost_effect = { BASE = 200 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy_unit = { destroy_unit = yes }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = sword_of_jaoz } omni_can_afford_spell = { BASE = 50 }  }
        omni_spell_mana_cost_effect = { BASE = 50 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy = { if = { limit = { has_trait = maimed } death = { death_reason = death_magic } } else = { force_add_trait = maimed } }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = hellas_light } omni_can_afford_spell = { BASE = 50 }  }
        omni_spell_mana_cost_effect = { BASE = 50 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy = { if = { limit = { has_trait = blind } death = { death_reason = death_magic } } else = { force_add_trait = blind } }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = blade_storm } omni_can_afford_spell = { BASE = 50 }  }
        omni_spell_mana_cost_effect = { BASE = 50 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy = { add_health = -2 }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = imperial_dragons_claw } omni_can_afford_spell = { BASE = 25 }  }
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy_unit = { damage_unit_percent = 0.08 }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = stinging_nettles } omni_can_afford_spell = { BASE = 25 }  }
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy_unit = { damage_unit_percent = 0.05 }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = serpentine_swords } omni_can_afford_spell = { BASE = 25 }  }
        omni_spell_mana_cost_effect = { BASE = 25 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy_unit = { damage_unit_percent = 0.05 }
    }
    else_if = {
        limit = { omni_knows_spell = { SPELL = steal } omni_can_afford_spell = { BASE = 15 } scope:omni_folk_enemy = { wealth >= 25 } }
        omni_spell_mana_cost_effect = { BASE = 15 }
        set_variable = omni_ai_battle_cast_succeeded
        scope:omni_folk_enemy = { add_gold = -25 }
        scope:caster_country = { add_treasury = 25 }
    }
}
