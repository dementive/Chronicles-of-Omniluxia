save_adventurer_var = {
	scope:$SCOPE$ = {
		if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_1 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_1
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
		else_if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_2 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_2
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
		else_if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_3 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_3
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
	}
}

# Shared bookkeeping for the new destination scenes. The existing legacy
# encounters keep their individual seen flags, while all arrival scenes use
# this one state machine to avoid duplicating reward and return plumbing.
omni_adventure_finish_contract_effect = {
	if = {
		limit = { var:adventure_score >= 105 }
		set_variable = adventure_outcome_legendary
	}
	else_if = {
		limit = { var:adventure_score >= 82 }
		set_variable = adventure_outcome_success
	}
	else_if = {
		limit = { var:adventure_score >= 62 }
		set_variable = adventure_outcome_mixed
	}
	else = { set_variable = adventure_outcome_failure }
}

# Contract choice is intentionally modest: it makes the selected job matter
# before the location scene and also nudges its final score tier.
omni_adventure_apply_contract_bonus_effect = {
	if = {
		limit = { has_variable = adventure_contract_rescue }
		omni_adventure_add_score_effect = { VALUE = 8 }
		add_popularity = 3
	}
	else_if = {
		limit = { has_variable = adventure_contract_ruin }
		omni_adventure_add_score_effect = { VALUE = 10 }
		add_finesse = 1
	}
	else_if = {
		limit = { has_variable = adventure_contract_monster }
		omni_adventure_add_score_effect = { VALUE = 8 }
		add_martial = 1
	}
	else_if = {
		limit = { has_variable = adventure_contract_relic }
		omni_adventure_add_score_effect = { VALUE = 10 }
		add_zeal = 1
	}
	else_if = {
		limit = { has_variable = adventure_contract_escort }
		omni_adventure_add_score_effect = { VALUE = 6 }
		add_popularity = 5
	}
	else_if = {
		limit = { has_variable = adventure_contract_intrigue }
		omni_adventure_add_score_effect = { VALUE = 8 }
		add_finesse = 1
	}
}

# Grants an adventure item, but ONLY if this adventure rolled the 1-in-3 item
# reward (adventure_grants_item) AND no item has been handed out yet this
# adventure (adventure_item_given).
#
# Generated items are awarded through one hidden event instead of expanding the
# 8k-item random list inside every encounter option.
omni_adventure_give_item_effect = {
	if = {
		limit = {
			has_variable = adventure_grants_item
			NOT = { has_variable = adventure_item_given }
		}
		set_variable = adventure_item_given
		random_list = {
			70 = {
				custom_tooltip = adventure_found_generated_item_tt
				hidden_effect = { trigger_event = adventure.6999 }
			}
			30 = {
				custom_tooltip = adventure_found_$item_ID$_tt
				hidden_effect = {
					give_character_an_item = { item_ID = $item_ID$ }
				}
			}
		}
	}
}
