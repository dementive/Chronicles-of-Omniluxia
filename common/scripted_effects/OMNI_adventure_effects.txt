				AND = { exists = var:adventure_party_1 var:adventure_party_1 = { is_alive = yes omni_adventure_role_warrior_trigger = yes } }

omni_adventure_risk_companion_wound_effect = {
	# One living companion takes the hit. A fresh wound is only a wound; a second
	# wound on top of an old one kills 18% of the time.
	if = {
		limit = { omni_adventure_party_has_any_trigger = yes }
		random_list = {
			1 = { trigger = { exists = var:adventure_party_1 var:adventure_party_1 = { is_alive = yes } } var:adventure_party_1 = { save_scope_as = adv_hit } }
			1 = { trigger = { exists = var:adventure_party_2 var:adventure_party_2 = { is_alive = yes } } var:adventure_party_2 = { save_scope_as = adv_hit } }
			1 = { trigger = { exists = var:adventure_party_3 var:adventure_party_3 = { is_alive = yes } } var:adventure_party_3 = { save_scope_as = adv_hit } }
		}
		if = {
			limit = { scope:adv_hit = { NOT = { has_trait = wounded } } }
			scope:adv_hit = { add_trait = wounded }
		}
		else = {
			random_list = {
				18 = { omni_adventure_companion_dies_effect = { WHO = adv_hit } }
				82 = {}
			}
		}
	}
	# Alone, every wound is the ruler's. A wound never kills the ruler outright.
	else_if = {
		limit = { NOT = { exists = var:adventure_party_1 } NOT = { has_trait = wounded } }
		add_trait = wounded
	}
}

# Shared bookkeeping for the new destination scenes. The existing legacy
# encounters keep their individual seen flags, while all arrival scenes use
# this one state machine to avoid duplicating reward and return plumbing.
# Tier thresholds are script values in common/script_values/OMNI_adventure_values.txt.
omni_adventure_finish_contract_effect = {
	if = {
		limit = { has_variable = adventure_called_home }
		set_variable = adventure_outcome_mixed
	}
	else_if = {
		limit = { var:adventure_score >= omni_adv_tier_legendary }
		set_variable = adventure_outcome_legendary
	}
	else_if = {
		limit = { var:adventure_score >= omni_adv_tier_success }
		set_variable = adventure_outcome_success
	}
	else_if = {
		limit = { var:adventure_score >= omni_adv_tier_mixed }
		set_variable = adventure_outcome_mixed
	}
	else = { set_variable = adventure_outcome_failure }
}

# Contract choice is intentionally modest: it makes the selected job matter
# before the location scene and also nudges its final score tier.
# Applied once on arrival. Every contract gets the same small base; the party
# gets the specialist bonus only when it brought the specialist the contract
# board asked for (see omni_adventure_has_contract_specialist_trigger).
omni_adventure_apply_contract_bonus_effect = {
	omni_adventure_add_score_effect = { VALUE = omni_adv_contract_base }
	if = {
		limit = { omni_adventure_has_contract_specialist_trigger = yes }
		custom_tooltip = adventure_contract_specialist_bonus_tt
		omni_adventure_add_score_effect = { VALUE = omni_adv_specialist_bonus }
	}
	if = { limit = { has_variable = adventure_contract_rescue } add_popularity = 3 }
	else_if = { limit = { has_variable = adventure_contract_escort } add_popularity = 3 }
}

# and if even that is empty the expedition simply ends (adventure.7). Pools are
# 6+ deep and at most 4 encounters are drawn, so the fallbacks are insurance.

	# Stall-free: a used-up theme pool is refreshed from the country flags first;
	# if it is still empty (all seen this trip) the generic pool takes over,
	# and if even that is empty the expedition ends (adventure.7). Generated.
	if = { limit = { has_variable = mountain_adventure } omni_adventure_refresh_mountain_pool_effect = yes }

# Loot is drawn from the destination owner's culture (normally the realm
# itself). A legendary-grade score or a "fine loot" hedge upgrades the odds.
omni_adventure_roll_generated_item_effect = {
	if = {
		limit = { exists = var:adventure_province.owner }
		var:adventure_province.owner = { save_scope_as = omni_adventure_loot_realm }
	}
	else = {
		employer = { save_scope_as = omni_adventure_loot_realm }
	}
	if = {
		limit = {
			OR = {
				var:adventure_score >= omni_adv_tier_legendary
				has_variable = adventure_fine_loot
			}
		}
		random_list = {
			40 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_common_item_effect = yes } }
			35 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_uncommon_item_effect = yes } }
			25 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_rare_item_effect = yes } }
		}
	}
	else = {
		random_list = {
			65 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_common_item_effect = yes } }
			25 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_uncommon_item_effect = yes } }
			10 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_rare_item_effect = yes } }
		}
	}
}

# Grants an adventure item, but ONLY if this adventure rolled the 1-in-3 item
# reward (adventure_grants_item) AND no item has been handed out yet this
# adventure (adventure_item_given).
#
# Generated items are awarded through one hidden event instead of expanding the
# 8k-item random list inside every encounter option.
omni_adventure_give_item_effect = {
	if = {
		limit = {
			has_variable = adventure_grants_item
			NOT = { has_variable = adventure_item_given }
		}
		set_variable = adventure_item_given
		random_list = {
			70 = {
				custom_tooltip = adventure_found_generated_item_tt
				hidden_effect = { trigger_event = adventure.6999 }
			}
			30 = {
				custom_tooltip = adventure_found_$item_ID$_tt
				hidden_effect = {
					give_character_an_item = { item_ID = $item_ID$ }
				}
			}
		}
	}
}


# ===========================================================================
# PRESS / HEDGE
# Every scene and encounter offers a "press" option (a gamble) and a "hedge"
# option (little or no readiness, but something tangible).
#
# Press odds: 50 win / 50 loss. The win weight is multiplied by 1.6 with the
# matching preparation and by 1.4 with the matching specialist (both: ~69%).
# Wound chance on a loss is 30%, halved by supplies. Scores come from
# omni_adv_press_win / omni_adv_press_loss / omni_adv_hedge. The Monte Carlo in
# tools/adventure/mc_adventure.py reads the numbers from the effect itself.
# Kinds: physical = supplies + warrior, wits = scouts + scout,
#        lore = rituals + scholar, social = supplies + diplomat.
# ===========================================================================
omni_adventure_press_effect = {
	custom_tooltip = adventure_press_$KIND$_tt
	if = { limit = { has_variable = adventure_$PREP$ } custom_tooltip = adventure_press_has_$PREP$_tt }
	if = { limit = { has_variable = adventure_party_$SPEC$ } custom_tooltip = adventure_press_has_$SPEC$_tt }
	random_list = {
		50 = {
			modifier = { factor = 1.6 has_variable = adventure_$PREP$ }
			modifier = { factor = 1.4 has_variable = adventure_party_$SPEC$ }
			custom_tooltip = adventure_press_win_tt
			hidden_effect = { omni_adventure_add_score_effect = { VALUE = omni_adv_press_win } }
		}
		50 = {
			custom_tooltip = adventure_press_loss_tt
			hidden_effect = { omni_adventure_add_score_effect = { VALUE = omni_adv_press_loss } }
			omni_adventure_press_wound_effect = yes
		}
	}
}

		30 = {
			modifier = { factor = 0.5 has_variable = adventure_supplies }

		}
		70 = {}
	}
}

omni_adventure_remove_from_party_effect = {
	# scope:$WHO$ leaves the party: the seats are packed again and the roles
	# recalculated (the ruler covers what is now missing).
	scope:$WHO$ = { remove_variable = on_omni_adventure }
	if = {
		limit = { exists = var:adventure_party_1 var:adventure_party_1 = { this = scope:$WHO$ } }
		remove_variable = adventure_party_1
		if = { limit = { exists = var:adventure_party_2 } set_variable = { name = adventure_party_1 value = var:adventure_party_2 } remove_variable = adventure_party_2 }
		if = { limit = { exists = var:adventure_party_3 } set_variable = { name = adventure_party_2 value = var:adventure_party_3 } remove_variable = adventure_party_3 }
	}
	else_if = {
		limit = { exists = var:adventure_party_2 var:adventure_party_2 = { this = scope:$WHO$ } }
		remove_variable = adventure_party_2
		if = { limit = { exists = var:adventure_party_3 } set_variable = { name = adventure_party_2 value = var:adventure_party_3 } remove_variable = adventure_party_3 }
	}
	else_if = {
		limit = { exists = var:adventure_party_3 var:adventure_party_3 = { this = scope:$WHO$ } }
		remove_variable = adventure_party_3
	}
	omni_adventure_calculate_party_roles_effect = yes
}
