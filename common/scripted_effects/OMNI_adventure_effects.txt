save_adventurer_var = {
	scope:$SCOPE$ = {
		if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_1 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_1
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
		else_if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_2 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_2
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
		else_if = {
			limit = {
				ROOT = {
					NOT = { has_variable = adventure_party_3 }
				}
			}
			ROOT = {
				set_variable = {
					name = adventure_party_3
					value = scope:$SCOPE$
					days = -1
				}
			}
		}
	}
}

omni_adventure_mark_party_busy_effect = {
	set_variable = {
		name = on_omni_adventure
		days = 1200
	}
	if = {
		limit = { exists = var:adventure_party_1 }
		var:adventure_party_1 = {
			set_variable = {
				name = on_omni_adventure
				value = ROOT
				days = 1200
			}
		}
	}
	if = {
		limit = { exists = var:adventure_party_2 }
		var:adventure_party_2 = {
			set_variable = {
				name = on_omni_adventure
				value = ROOT
				days = 1200
			}
		}
	}
	if = {
		limit = { exists = var:adventure_party_3 }
		var:adventure_party_3 = {
			set_variable = {
				name = on_omni_adventure
				value = ROOT
				days = 1200
			}
		}
	}
}

omni_adventure_add_score_effect = {
	add_to_variable = {
		name = adventure_score
		value = $VALUE$
	}
}

omni_adventure_release_party_effect = {
	remove_variable = on_omni_adventure
	if = {
		limit = { exists = var:adventure_party_1 }
		var:adventure_party_1 = {
			remove_variable = on_omni_adventure
		}
	}
}
