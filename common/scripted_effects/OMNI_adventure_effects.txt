	set_variable = {
		name = on_omni_adventure
		days = 1200
	}
	# Visible "away" status on the ruler. Timed to the same 1200 days as the
	# busy flag so it can never outlive a broken chain.

		limit = { NOT = { has_character_modifier = omni_adventure_on_the_road } }

			name = omni_adventure_on_the_road

# Imperator character attributes are permanent, so an expedition can grant at
# most one point across Martial, Finesse, Charisma, and Zeal.

omni_adventure_candidate_role_tooltips_effect = {
	if = { limit = { scope:$SCOPE$ = { OR = { martial >= 9 has_trait = brave } } } custom_tooltip = adventure_candidate_warrior_tt }
	if = { limit = { scope:$SCOPE$ = { OR = { finesse >= 9 has_trait = cautious has_trait = suspicious } } } custom_tooltip = adventure_candidate_scout_tt }
	if = { limit = { scope:$SCOPE$ = { OR = { zeal >= 9 has_trait = scholar has_trait = pious has_trait = mage_trait has_trait = cleric_trait } } } custom_tooltip = adventure_candidate_scholar_tt }
	if = { limit = { scope:$SCOPE$ = { charisma >= 9 } } custom_tooltip = adventure_candidate_diplomat_tt }
	# The contract is chosen before the party, so say plainly who fits it.
	if = {
		limit = {
			OR = {
				AND = {
					OR = { has_variable = adventure_contract_monster has_variable = adventure_contract_escort }
					scope:$SCOPE$ = { OR = { martial >= 9 has_trait = brave } }
				}
				AND = {
					has_variable = adventure_contract_ruin
					scope:$SCOPE$ = { OR = { finesse >= 9 has_trait = cautious has_trait = suspicious } }
				}
				AND = {
					has_variable = adventure_contract_relic
					scope:$SCOPE$ = { OR = { zeal >= 9 has_trait = scholar has_trait = pious has_trait = mage_trait has_trait = cleric_trait } }
				}
				AND = {
					OR = { has_variable = adventure_contract_rescue has_variable = adventure_contract_intrigue }
					scope:$SCOPE$ = { charisma >= 9 }
				}
			}
		}
		custom_tooltip = adventure_candidate_matches_contract_tt
	}
}

# Recalculate specialties once all three companions have been chosen.
omni_adventure_calculate_party_roles_effect = {
	remove_variable = adventure_party_warrior
	remove_variable = adventure_party_scout
	remove_variable = adventure_party_scholar
	remove_variable = adventure_party_diplomat
	if = {
		limit = { OR = {
			var:adventure_party_1 = { OR = { martial >= 9 has_trait = brave } }
			var:adventure_party_2 = { OR = { martial >= 9 has_trait = brave } }
			var:adventure_party_3 = { OR = { martial >= 9 has_trait = brave } }
		} }
		set_variable = adventure_party_warrior
	}
	if = {
		limit = { OR = {
			var:adventure_party_1 = { OR = { finesse >= 9 has_trait = cautious has_trait = suspicious } }
			var:adventure_party_2 = { OR = { finesse >= 9 has_trait = cautious has_trait = suspicious } }
			var:adventure_party_3 = { OR = { finesse >= 9 has_trait = cautious has_trait = suspicious } }
		} }
		set_variable = adventure_party_scout
	}
	if = {
		limit = { OR = {
			var:adventure_party_1 = { OR = { zeal >= 9 has_trait = scholar has_trait = pious has_trait = mage_trait has_trait = cleric_trait } }
			var:adventure_party_2 = { OR = { zeal >= 9 has_trait = scholar has_trait = pious has_trait = mage_trait has_trait = cleric_trait } }
			var:adventure_party_3 = { OR = { zeal >= 9 has_trait = scholar has_trait = pious has_trait = mage_trait has_trait = cleric_trait } }
		} }
		set_variable = adventure_party_scholar
	}
	if = {
		limit = { OR = {
			var:adventure_party_1 = { charisma >= 9 }
			var:adventure_party_2 = { charisma >= 9 }
			var:adventure_party_3 = { charisma >= 9 }
		} }
		set_variable = adventure_party_diplomat
	}
}

omni_adventure_risk_companion_wound_effect = {
	if = {
		limit = {
			OR = {
				var:adventure_party_1 = { is_alive = yes NOT = { has_trait = wounded } }
				var:adventure_party_2 = { is_alive = yes NOT = { has_trait = wounded } }
				var:adventure_party_3 = { is_alive = yes NOT = { has_trait = wounded } }
			}
		}
		random_list = {
			1 = { trigger = { var:adventure_party_1 = { is_alive = yes NOT = { has_trait = wounded } } } var:adventure_party_1 = { add_trait = wounded } }
			1 = { trigger = { var:adventure_party_2 = { is_alive = yes NOT = { has_trait = wounded } } } var:adventure_party_2 = { add_trait = wounded } }
			1 = { trigger = { var:adventure_party_3 = { is_alive = yes NOT = { has_trait = wounded } } } var:adventure_party_3 = { add_trait = wounded } }
		}
	}
}

# Emergency cleanup for death, succession, or other hard interruption. Unlike a
# completed expedition this does not impose a cooldown on a dead/former ruler.
omni_adventure_abort_effect = {
	remove_variable = on_omni_adventure
	omni_adventure_remove_road_modifier_effect = yes
	if = { limit = { exists = var:adventure_party_1 } var:adventure_party_1 = { remove_variable = on_omni_adventure } }
	if = { limit = { exists = var:adventure_party_2 } var:adventure_party_2 = { remove_variable = on_omni_adventure } }
	if = { limit = { exists = var:adventure_party_3 } var:adventure_party_3 = { remove_variable = on_omni_adventure } }
	omni_adventure_clear_variables_effect = yes
}

# Shared bookkeeping for the new destination scenes. The existing legacy
# encounters keep their individual seen flags, while all arrival scenes use
# this one state machine to avoid duplicating reward and return plumbing.
# Tier thresholds are script values in common/script_values/OMNI_adventure_values.txt.
omni_adventure_finish_contract_effect = {
	if = {
		limit = { var:adventure_score >= omni_adv_tier_legendary }
		set_variable = adventure_outcome_legendary
	}
	else_if = {
		limit = { var:adventure_score >= omni_adv_tier_success }
		set_variable = adventure_outcome_success
	}
	else_if = {
		limit = { var:adventure_score >= omni_adv_tier_mixed }
		set_variable = adventure_outcome_mixed
	}
	else = { set_variable = adventure_outcome_failure }
}

# Contract choice is intentionally modest: it makes the selected job matter
# before the location scene and also nudges its final score tier.
# Applied once on arrival. Every contract gets the same small base; the party
# gets the specialist bonus only when it brought the specialist the contract
# board asked for (see omni_adventure_has_contract_specialist_trigger).
omni_adventure_apply_contract_bonus_effect = {
	omni_adventure_add_score_effect = { VALUE = omni_adv_contract_base }
	if = {
		limit = { omni_adventure_has_contract_specialist_trigger = yes }
		custom_tooltip = adventure_contract_specialist_bonus_tt
		omni_adventure_add_score_effect = { VALUE = omni_adv_specialist_bonus }
	}
	if = { limit = { has_variable = adventure_contract_rescue } add_popularity = 3 }
	else_if = { limit = { has_variable = adventure_contract_escort } add_popularity = 3 }
}

# Loot is drawn from the destination owner's culture (normally the realm
# itself). A legendary-grade score or a "fine loot" hedge upgrades the odds.
omni_adventure_roll_generated_item_effect = {
	if = {
		limit = { exists = var:adventure_province.owner }
		var:adventure_province.owner = { save_scope_as = omni_adventure_loot_realm }
	}
	else = {
		employer = { save_scope_as = omni_adventure_loot_realm }
	}
	if = {
		limit = {
			OR = {
				var:adventure_score >= omni_adv_tier_legendary
				has_variable = adventure_fine_loot
			}
		}
		random_list = {
			40 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_common_item_effect = yes } }
			35 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_uncommon_item_effect = yes } }
			25 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_rare_item_effect = yes } }
		}
	}
	else = {
		random_list = {
			65 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_common_item_effect = yes } }
			25 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_uncommon_item_effect = yes } }
			10 = { scope:omni_adventure_loot_realm = { omni_adventure_roll_local_rare_item_effect = yes } }
		}
	}
}

# Grants an adventure item, but ONLY if this adventure rolled the 1-in-3 item
# reward (adventure_grants_item) AND no item has been handed out yet this
# adventure (adventure_item_given).
#
# Generated items are awarded through one hidden event instead of expanding the
# 8k-item random list inside every encounter option.
omni_adventure_give_item_effect = {
	if = {
		limit = {
			has_variable = adventure_grants_item
			NOT = { has_variable = adventure_item_given }
		}
		set_variable = adventure_item_given
		random_list = {
			70 = {
				custom_tooltip = adventure_found_generated_item_tt
				hidden_effect = { trigger_event = adventure.6999 }
			}
			30 = {
				custom_tooltip = adventure_found_$item_ID$_tt
				hidden_effect = {
					give_character_an_item = { item_ID = $item_ID$ }
				}
			}
		}
	}
}


# ===========================================================================
# PRESS / HEDGE
# Every scene and encounter offers a "press" option (a gamble) and a "hedge"
# option (little or no readiness, but something tangible).
#
# Press odds: 50 win / 50 loss. The win weight is multiplied by 1.6 with the
# matching preparation and by 1.4 with the matching specialist (both: ~69%).
# Wound chance on a loss is 30%, halved by supplies. Scores come from
# omni_adv_press_win / omni_adv_press_loss / omni_adv_hedge. The Monte Carlo in
# tools/adventure/mc_adventure.py reads the numbers from the effect itself.
# Kinds: physical = supplies + warrior, wits = scouts + scout,
#        lore = rituals + scholar, social = supplies + diplomat.
# ===========================================================================
omni_adventure_press_effect = {
	custom_tooltip = adventure_press_$KIND$_tt
	if = { limit = { has_variable = adventure_$PREP$ } custom_tooltip = adventure_press_has_$PREP$_tt }
	if = { limit = { has_variable = adventure_party_$SPEC$ } custom_tooltip = adventure_press_has_$SPEC$_tt }
	random_list = {
		50 = {
			modifier = { factor = 1.6 has_variable = adventure_$PREP$ }
			modifier = { factor = 1.4 has_variable = adventure_party_$SPEC$ }
			custom_tooltip = adventure_press_win_tt
			hidden_effect = { omni_adventure_add_score_effect = { VALUE = omni_adv_press_win } }
		}
		50 = {
			custom_tooltip = adventure_press_loss_tt
			hidden_effect = { omni_adventure_add_score_effect = { VALUE = omni_adv_press_loss } }
			omni_adventure_press_wound_effect = yes
		}
	}
}

omni_adventure_press_wound_effect = {
	random_list = {
		30 = {
			modifier = { factor = 0.5 has_variable = adventure_supplies }
			omni_adventure_risk_companion_wound_effect = yes
		}
		70 = {}
	}
}

omni_adventure_press_physical_effect = { omni_adventure_press_effect = { KIND = physical PREP = supplies SPEC = warrior } }

omni_adventure_press_wits_effect = { omni_adventure_press_effect = { KIND = wits PREP = scouts SPEC = scout } }

omni_adventure_press_lore_effect = { omni_adventure_press_effect = { KIND = lore PREP = rituals SPEC = scholar } }

omni_adventure_press_social_effect = { omni_adventure_press_effect = { KIND = social PREP = supplies SPEC = diplomat } }

	}
	else_if = {
		limit = { has_variable = adventure_contract_ruin }

	}
	else_if = {
		limit = { has_variable = adventure_contract_relic }
		omni_adventure_gain_zeal_effect = yes
	}
	else = {
		omni_adventure_gain_charisma_effect = yes
