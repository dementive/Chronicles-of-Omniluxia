# Picks a small pool of late-game rivals from the new world only.
omni_nw_antagonist_pick_effect = {
	random_country = {
		limit = {
			country_is_in_new_world = yes
			NOT = { THIS = ROOT }
			NOT = { is_antagonist = yes }
			NOT = { has_variable = omni_nw_antagonist_selected }
		}
		set_variable = omni_nw_antagonist_selected
		set_antagonist = yes
	}
}
