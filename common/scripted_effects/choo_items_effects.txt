give_character_an_item = {

if = {
    limit = { 
        NOT = { 
        
            is_target_in_variable_list = {  
                name = item_list   
                target = flag:$item_ID$
                                        }  

                
            }
    }

    add_to_variable_list = {
        name = item_list
        target = flag:$item_ID$
    }
    
    show_as_tooltip = { add_character_modifier = { name = $item_ID$_modifier } }





}

else = {
    
    custom_tooltip = character_already_have_this_item

}




    # AI-run characters auto-equip whatever they can whenever they gain an item,
    # so the AI actually uses the item system. Human players equip via the menu.
    if = {
        limit = { employer = { is_ai = yes } }
        omni_ai_auto_equip_items = yes
    }
}

# Dynamic item IDs are already variable targets; flag:var:... is invalid.
give_character_an_item_from_variable = {
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = item_list target = var:$ITEM_VAR$ } } }
		add_to_variable_list = { name = item_list target = var:$ITEM_VAR$ }
	}
}
# removes item of a character 
remove_character_item = {
    if = {
        limit = {
            has_item = {
                item_ID = $item_ID$
            }
        }
        remove_list_variable = {
            name = item_list
            target = flag:$item_ID$
        }

        # If the item was equipped, drop it from the equipped list so the
        # category slot frees up and it no longer counts as worn.
        if = {
            limit = {
                has_variable_list = equiped_item_list
                is_target_in_variable_list = { name = equiped_item_list target = flag:$item_ID$ }
            }
            remove_list_variable = {
                name = equiped_item_list
                target = flag:$item_ID$
            }
        }

        # Strip the character modifier the item applied on equip (if any).
        if = {
            limit = { has_character_modifier = $item_ID$_modifier }
            remove_character_modifier = $item_ID$_modifier
        }
    }
    else = {
        custom_tooltip = character_dont_have_such_item
    }
}

omni_add_item_to_shop_stock = {
    scope:omni_shop_owner = {
        if = {
            limit = {
                NOT = {
                    is_target_in_variable_list = {
                        name = omni_item_shop_stock
                        target = flag:$item_ID$
                    }
                }
                OR = {
                    AND = {
                        var:omni_shop_max_items = 4
                        var:omni_shop_item_count < 4
                    }
                    AND = {
                        var:omni_shop_max_items = 6
                        var:omni_shop_item_count < 6
                    }
                    AND = {
                        var:omni_shop_max_items = 8
                        var:omni_shop_item_count < 8
                    }
                    AND = {
                        var:omni_shop_max_items = 10
                        var:omni_shop_item_count < 10
                    }
                }
            }
            add_to_variable_list = {
                name = omni_item_shop_stock
                target = flag:$item_ID$
            }
            change_variable = {
                name = omni_shop_item_count
                add = 1
            }
        }
    }
}

omni_build_item_shop_stock = {
    save_scope_as = omni_shop_owner
    clear_variable_list = omni_item_shop_stock
    clear_variable_list = omni_item_shop_stock_gen
    set_variable = {
        name = omni_market_quality
        value = 0
    }
    set_variable = {
        name = omni_shop_item_count
        value = 0
    }
    set_variable = {
        name = omni_shop_max_items
        value = 4
    }
    root.location = {
        save_scope_as = omni_item_market_province
        if = { limit = { num_of_commerce_building >= 1 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        if = { limit = { num_of_commerce_building >= 2 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        if = { limit = { num_of_commerce_building >= 3 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        if = { limit = { num_of_port_building >= 1 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        if = { limit = { num_of_port_building >= 3 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        if = { limit = { num_of_workshop_building >= 1 } root = { change_variable = { name = omni_market_quality add = 8 } } }
        if = { limit = { num_of_foundry_building >= 1 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        if = { limit = { num_of_library_building >= 1 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        if = { limit = { num_of_academy_building >= 1 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        
        # Used Slots (Non-cumulative else_if)
        if = { limit = { num_of_used_building_slots >= 30 } root = { change_variable = { name = omni_market_quality add = 15 } } }
        else_if = { limit = { num_of_used_building_slots >= 16 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        else_if = { limit = { num_of_used_building_slots >= 8 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        
        # Population (Non-cumulative else_if)
        if = { limit = { total_population >= 100 } root = { change_variable = { name = omni_market_quality add = 20 } } }
        else_if = { limit = { total_population >= 50 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        else_if = { limit = { total_population >= 20 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        
        # Civilization Value (Non-cumulative else_if)
        if = { limit = { civilization_value >= 95 } root = { change_variable = { name = omni_market_quality add = 15 } } }
        else_if = { limit = { civilization_value >= 75 } root = { change_variable = { name = omni_market_quality add = 10 } } }
        else_if = { limit = { civilization_value >= 50 } root = { change_variable = { name = omni_market_quality add = 5 } } }
        
        if = { limit = { is_capital = yes } root = { change_variable = { name = omni_market_quality add = 5 } } }

# Deducts the price from the character's wealth, gives them the item, and clears it
# from that country's shop stock.
omni_buy_character_item = {
	save_scope_as = omni_purchase_host
	remove_variable = omni_purchase_succeeded
	if = {
		limit = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } }
		scope:omni_city_visitor = {
			if = {
				limit = { wealth >= $price$ }
				add_gold = { value = $price$ multiply = -1 }
				give_character_an_item = { item_ID = $item_ID$ }
				scope:omni_purchase_host = { set_variable = omni_purchase_succeeded }
			}
		}
	}
	else = {
		add_gold = { value = $price$ multiply = -1 }
		give_character_an_item = { item_ID = $item_ID$ }
		set_variable = omni_purchase_succeeded
	}
    if = {
		limit = { has_variable = omni_purchase_succeeded is_target_in_variable_list = { name = omni_item_shop_stock target = flag:$item_ID$ } }
        remove_list_variable = { name = omni_item_shop_stock target = flag:$item_ID$ }
    }
	remove_variable = omni_purchase_succeeded
}

equip_item = {
    if = {
        limit = {
            has_variable_list = equiped_item_list
            is_target_in_variable_list = { name = equiped_item_list target = flag:$item_ID$ }
        }
        remove_list_variable = {
            name = equiped_item_list
            target = flag:$item_ID$
        }
        custom_tooltip = unequiping_$item_ID$_tt
    }
    else = {
        custom_tooltip = equiping_$item_ID$_tt
        add_to_variable_list = {
            name = equiped_item_list
            target = flag:$item_ID$
        }
    }
}

equip_item_with_modifier = {
    if = {
        limit = {
            has_variable_list = equiped_item_list
            is_target_in_variable_list = { name = equiped_item_list target = flag:$item_ID$ }
        }
        remove_list_variable = {
            name = equiped_item_list
            target = flag:$item_ID$
        }
        custom_tooltip = unequiping_$item_ID$_tt
        remove_character_modifier = $item_ID$_modifier
    }
    else = {
        add_to_variable_list = {
            name = equiped_item_list
            target = flag:$item_ID$
        }
        custom_tooltip = equiping_$item_ID$_tt
        add_character_modifier = { name = $item_ID$_modifier  duration = -1 }
    }
}

give_character_an_artifact = {
    hidden_effect = {
        give_character_an_item = { item_ID = $item_ID$ }
    }
    add_character_modifier = { name = $item_ID$_modifier duration = -1 }
    country = { add_country_modifier = { name = $item_ID$_army duration = -1 } }
}
