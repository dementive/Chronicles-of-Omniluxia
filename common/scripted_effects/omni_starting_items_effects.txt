assign_starting_common_item = {
	current_ruler = {
		country = {
			omni_pick_random_generated_item_by_culture = { RARITY = common }
		}
	}
}

assign_starting_uncommon_item = {
	current_ruler = {
		country = {
			omni_pick_random_generated_item_by_culture = { RARITY = uncommon }
		}
	}
}

assign_starting_rare_item = {
	current_ruler = {
		country = {
			omni_pick_random_generated_item_by_culture = { RARITY = rare }
		}
	}
}

# Pick a random generated item by rarity from the appropriate cultural pool.
# Call with RARITY = common/uncommon/rare
# Scopes: country, with current_ruler available.
omni_pick_random_generated_item_by_culture = {
	# Pick from the global rarity pool (all generated items of that rarity across all cultures).
	# If per-culture differentiation is needed later, this can be expanded to use
	# omni_gen_pool_[culture_name]_[rarity] instead.
	random_in_global_list = {
		variable = "item_$RARITY$_rarity"
		save_scope_as = omni_start_item_pick
	}
	# Give the picked item to the ruler
	if = {
		limit = { exists = scope:omni_start_item_pick }
		current_ruler = {
			if = {
				limit = { NOT = { is_target_in_variable_list = { name = item_list target = scope:omni_start_item_pick } } }
				add_to_variable_list = { name = item_list target = scope:omni_start_item_pick }
			}
		}
	}
}

omni_distribute_starting_items = {
	if = {
		limit = {
			exists = current_ruler
			NOT = { has_variable = omni_starting_items_done }
		}
		set_variable = { name = omni_starting_pop_count value = country_population }
		if = {
			limit = { var:omni_starting_pop_count >= 1000 }
			random_list = {
				1 = { assign_starting_rare_item = yes }
				1 = { }
			}
			assign_starting_uncommon_item = yes
			assign_starting_common_item = yes
		}
		else_if = {
			limit = { var:omni_starting_pop_count >= 300 }
			random_list = {
				2 = { assign_starting_uncommon_item = yes }
				1 = { assign_starting_common_item = yes }
			}
			assign_starting_common_item = yes
		}
		else_if = {
			limit = { var:omni_starting_pop_count >= 100 }
			assign_starting_common_item = yes
			random_list = {
				1 = { assign_starting_uncommon_item = yes }
				3 = { assign_starting_common_item = yes }
			}
		}
		else = {
			assign_starting_common_item = yes
		}
		set_variable = { name = omni_starting_pop_count value = 0 }
		set_variable = { name = omni_starting_items_done value = 1 }
	}
}


# Called from choo_on_game_init_effects (on_game_initialized) at game start.
# Builds the global item category/rarity lists once, then hands starting items
# to every ruler. Both steps self-guard, so this is safe if it ever runs twice.
omni_items_game_start_effects = {
	if = {
		limit = { NOT = { has_global_variable = omni_item_lists_built } }
		set_global_variable = { name = omni_item_lists_built value = 1 }
		random_country = {
			limit = { always = yes }
			assign_all_items_to_categories = yes
			assign_all_items_to_rarities = yes
			omni_assign_generated_items_categories = yes
			omni_assign_generated_items_rarities = yes
			omni_assign_generated_items_units = yes
			omni_assign_generated_items_prices = yes
		}
	}
	every_country = {
		limit = { exists = current_ruler }
		if = { limit = { NOT = { has_variable = military_favor } } set_variable = { name = military_favor value = 0 } }
		if = { limit = { NOT = { has_variable = civic_favor } } set_variable = { name = civic_favor value = 0 } }
		if = { limit = { NOT = { has_variable = mercantile_favor } } set_variable = { name = mercantile_favor value = 0 } }
		if = { limit = { NOT = { has_variable = religious_favor } } set_variable = { name = religious_favor value = 0 } }
		if = { limit = { NOT = { has_variable = populist_favor } } set_variable = { name = populist_favor value = 0 } }
		omni_distribute_starting_items = yes
	}
}
