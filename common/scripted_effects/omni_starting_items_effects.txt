omni_distribute_starting_items = {
	if = {
		limit = {
			exists = current_ruler
			NOT = { has_variable = omni_starting_items_done }
		}
		set_variable = { name = omni_starting_pop_count value = 0 }
		every_owned_province = {
			limit = { has_province_rank = city }
			every_pops_in_province = {
				country = {
					change_variable = { name = omni_starting_pop_count add = 1 }
				}
			}
		}
		if = {
			limit = { var:omni_starting_pop_count >= 1000 }
			random_list = {
				1 = { assign_starting_rare_item = yes }
				1 = { }
			}
			assign_starting_uncommon_item = yes
			assign_starting_common_item = yes
		}
		else_if = {
			limit = { var:omni_starting_pop_count >= 300 }
			random_list = {
				2 = { assign_starting_uncommon_item = yes }
				1 = { assign_starting_common_item = yes }
			}
			assign_starting_common_item = yes
		}
		else_if = {
			limit = { var:omni_starting_pop_count >= 100 }
			assign_starting_common_item = yes
			random_list = {
				1 = { assign_starting_uncommon_item = yes }
				3 = { assign_starting_common_item = yes }
			}
		}
		else = {
			assign_starting_common_item = yes
		}
		set_variable = { name = omni_starting_pop_count value = 0 }
		set_variable = { name = omni_starting_items_done value = 1 }
	}
}


# Called from choo_on_game_init_effects (on_game_initialized) at game start.
# Builds the global item category/rarity lists once, then hands starting items
# to every ruler. Both steps self-guard, so this is safe if it ever runs twice.
omni_items_game_start_effects = {
	if = {
		limit = { NOT = { has_global_variable = omni_item_lists_built } }
		set_global_variable = { name = omni_item_lists_built value = 1 }
		random_country = {
			limit = { always = yes }
			assign_all_items_to_categories = yes
			assign_all_items_to_rarities = yes
		}
	}
	every_country = {
		limit = { exists = current_ruler }
		omni_distribute_starting_items = yes
	}
}
