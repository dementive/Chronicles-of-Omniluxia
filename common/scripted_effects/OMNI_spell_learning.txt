# Shared spell-learning primitives. Keep these effects O(1) at cast time.
omni_learn_spell = {
	set_variable = { name = $SPELL$_known_var value = 1 }
}

omni_grant_random_spell = {
	# School/tier-specific pools are emitted in OMNI_spell_pools_generated.txt.
	# This compatibility entry intentionally resolves to the beginner pool.
	omni_grant_random_beginner_spell = yes
}

omni_grant_random_beginner_spell = {
	random_list = {
		10 = { trigger = { NOT = { omni_knows_spell = { SPELL = fireball } } } omni_learn_spell = { SPELL = fireball } }
		10 = { trigger = { NOT = { omni_knows_spell = { SPELL = light_shields } } } omni_learn_spell = { SPELL = light_shields } }
		10 = { trigger = { NOT = { omni_knows_spell = { SPELL = restoration } } } omni_learn_spell = { SPELL = restoration } }
		10 = { trigger = { NOT = { omni_knows_spell = { SPELL = plant_growth } } } omni_learn_spell = { SPELL = plant_growth } }
		10 = { trigger = { NOT = { omni_knows_spell = { SPELL = summon_warborn } } } omni_learn_spell = { SPELL = summon_warborn } }
		1 = { }
	}
}

omni_grant_two_beginner_spells = {
	omni_grant_random_beginner_spell = yes
	omni_grant_random_beginner_spell = yes
}

omni_buy_catchup_beginner_spell = {
	omni_grant_random_beginner_spell = yes
	omni_apply_cramming_fatigue = yes
}

omni_apply_cramming_fatigue = {
	add_trait = cramming_fatigue_trait
	add_character_modifier = { name = cramming_fatigue_timer_modifier duration = 180 }
	recalculate_character_magic_vars = yes
}

# Compatibility bridge for existing saves. Legacy unlocks remain untouched;
# this only adds the equivalent spell flags, so loading a save cannot remove
# previously available magic.
omni_migrate_legacy_magic = {
	omni_check_magic_prestige = yes
	omni_migrate_legacy_magic_spells = yes
}

# PERF: the one-shot spell migration on its own, without the prestige pass.
# The yearly character maintenance calls this directly and runs
# omni_check_magic_prestige once afterwards (gated), instead of twice per
# character per year. Other callers keep using omni_migrate_legacy_magic.
omni_migrate_legacy_magic_spells = {
	# One-shot per character. This is a save-compatibility bridge; once a
	# character's legacy unlocks have been copied to the new spell flags there
	# is nothing left to migrate, so the guard skips the ~15 has_variable checks
	# on every subsequent yearly maintenance tick.
	if = {
	limit = { NOT = { has_variable = omni_magic_migrated } }
	set_variable = omni_magic_migrated
	if = { limit = { has_variable = diviner_t2_unlocked_var } omni_learn_spell = { SPELL = light_shields } }
	if = { limit = { has_variable = diviner_t3_unlocked_var } omni_learn_spell = { SPELL = light_rays } }
	if = { limit = { has_variable = diviner_t4_unlocked_var } omni_learn_spell = { SPELL = gravity_well } }
	if = { limit = { has_variable = hierophant_t2_unlocked_var } omni_learn_spell = { SPELL = exorcism } }
	if = { limit = { has_variable = cleric_t2_unlocked_var } omni_learn_spell = { SPELL = restoration } }
	if = { limit = { has_variable = illusionist_t2_unlocked_var } omni_learn_spell = { SPELL = altered_inspiration } }
	if = { limit = { has_variable = elementalist_t2_unlocked_var } omni_learn_spell = { SPELL = detect_minerals } }
	if = { limit = { has_variable = druid_t2_unlocked_var } omni_learn_spell = { SPELL = plant_growth } }
	if = { limit = { has_variable = necromancer_t2_unlocked_var } omni_learn_spell = { SPELL = summon_warborn } }
	if = { limit = { has_variable = warlock_t2_unlocked_var } omni_learn_spell = { SPELL = rage } }
	} # closes one-shot migration guard
	# V2 adds individual knowledge flags for tactical/civic spells that formerly
	# relied only on school tiers. Kept under its own guard because existing saves
	# may already have completed omni_magic_migrated.
	if = {
		limit = { NOT = { has_variable = omni_magic_casting_knowledge_migrated } }
		set_variable = omni_magic_casting_knowledge_migrated
		if = { limit = { omni_caster_omnic_t1 = yes } omni_learn_spell = { SPELL = battle_clarity } omni_learn_spell = { SPELL = sword_of_zanis } }
		if = { limit = { omni_caster_omnic_t2 = yes } omni_learn_spell = { SPELL = battle_hymn } omni_learn_spell = { SPELL = light_of_true_faith } }
		if = { limit = { omni_caster_omnic_t3 = yes } omni_learn_spell = { SPELL = blessing_of_war } omni_learn_spell = { SPELL = harbor_ward } omni_learn_spell = { SPELL = divine_ceremonies } }
		if = { limit = { omni_caster_aldic_t1 = yes } omni_learn_spell = { SPELL = arcane_aegis } omni_learn_spell = { SPELL = hand_of_adaralu } }
		if = { limit = { omni_caster_aldic_t2 = yes } omni_learn_spell = { SPELL = unrest_cleansing } omni_learn_spell = { SPELL = counterward } omni_learn_spell = { SPELL = illusion_walls } omni_learn_spell = { SPELL = golden_fingers_of_the_pharon } }
		if = { limit = { omni_caster_amten_t1 = yes } omni_learn_spell = { SPELL = blighted_ground } omni_learn_spell = { SPELL = mudbind } omni_learn_spell = { SPELL = burning_pitch } omni_learn_spell = { SPELL = conjured_supply } omni_learn_spell = { SPELL = arrows_of_lavas } }
		if = { limit = { omni_caster_amten_t2 = yes } omni_learn_spell = { SPELL = breaching_tremor } omni_learn_spell = { SPELL = weathercall_growth } omni_learn_spell = { SPELL = migration_beacon } omni_learn_spell = { SPELL = feast_of_ugone } }
		if = { limit = { omni_caster_amten_t3 = yes } omni_learn_spell = { SPELL = magical_construction } }
		if = { limit = { omni_caster_melodian_t1 = yes } omni_learn_spell = { SPELL = fear_hex } omni_learn_spell = { SPELL = cursed_rigging } omni_learn_spell = { SPELL = spectral_boarders } omni_learn_spell = { SPELL = wasting_rot } omni_learn_spell = { SPELL = battle_surge } }
		if = { limit = { omni_caster_melodian_t2 = yes } omni_learn_spell = { SPELL = hammer_of_melodias } }
		if = { limit = { omni_caster_melodian_t3 = yes } omni_learn_spell = { SPELL = storm_wrack } omni_learn_spell = { SPELL = heroic_resonance } omni_learn_spell = { SPELL = storm_of_the_siege } }
	}
}

# Preserve access to higher-tier spells after repairing their individual gates.
# Character scope; called before a player spellbook opens.
omni_migrate_spell_audit_knowledge = {
    # Repair existing casters immediately when their spellbook opens.
    omni_check_magic_prestige = yes
    if = {
        limit = { NOT = { has_variable = omni_spell_audit_knowledge_migrated } }
        set_variable = omni_spell_audit_knowledge_migrated
        if = { limit = { has_variable = elementalist_t3_unlocked_var omni_knows_spell = { SPELL = detect_minerals } } omni_learn_spell = { SPELL = detect_minerals_ii } }
        if = { limit = { has_variable = druid_t3_unlocked_var omni_knows_spell = { SPELL = plant_growth } } omni_learn_spell = { SPELL = plant_growth_ii } }
        if = { limit = { has_variable = druid_t4_unlocked_var omni_knows_spell = { SPELL = nature_bounty } } omni_learn_spell = { SPELL = nature_bounty_ii } }
        if = { limit = { has_variable = warlock_t4_unlocked_var omni_knows_spell = { SPELL = dark_offering } } omni_learn_spell = { SPELL = dark_offering_ii } }
        if = { limit = { has_variable = diviner_t3_unlocked_var omni_knows_spell = { SPELL = light_rays } } omni_learn_spell = { SPELL = light_rays_ii } }
        if = { limit = { has_variable = illusionist_t4_unlocked_var omni_knows_spell = { SPELL = warped_flesh } } omni_learn_spell = { SPELL = warped_flesh_ii } }
        if = { limit = { has_variable = cleric_t3_unlocked_var omni_knows_spell = { SPELL = healing_light } } omni_learn_spell = { SPELL = healing_light_ii } }
        if = { limit = { has_variable = necromancer_t2_unlocked_var omni_knows_spell = { SPELL = curse_ii } } omni_learn_spell = { SPELL = curse } }
    }
}

# Character scope. Keep at most one recurring study tick queued per mage.
omni_queue_spell_study = {
	if = {
		limit = { NOT = { has_variable = omni_spell_study_queued } }
		set_variable = omni_spell_study_queued
		trigger_event = { id = omni_items.90 days = 365 }
	}
}
