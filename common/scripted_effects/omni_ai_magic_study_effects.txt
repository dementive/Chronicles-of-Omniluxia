omni_ai_study_magic_effect = {
	if = {
		limit = {
			is_ai = yes
			NOT = { has_variable = omni_ai_magic_study_cooldown }
			treasury >= 50
			any_owned_province = {
				has_building = magic_academy_building
			}
			any_character = {
				is_alive = yes
				is_adult = yes
				NOT = { has_variable = omni_studying_magic_cooldown }
				NOT = { has_variable = is_the_revant }
				NOR = {
					has_trait = brain_damaged
					has_trait = incapable
					has_trait = goblin_trait
					has_variable = on_orc_adventure
				}
				OR = {
					is_ruler = yes
					is_general = yes
					is_admiral = yes
					is_governor = yes
					has_office = office_marshal
					has_office = office_high_priest_monarchy
					has_office = office_high_priest
					has_office = office_augur
					has_office = office_pontifex
					has_trait = mage_trait
					has_trait = archmage_trait
					prominence >= 60
					martial >= 8
					zeal >= 8
					finesse >= 8
				}
			}
		}
		random_list = {
				70 = {
				set_variable = {
					name = omni_ai_magic_study_cooldown
					days = 365
				}
				add_treasury = -50
				random_character = {
					limit = {
						is_alive = yes
						is_adult = yes
						NOT = { has_variable = omni_studying_magic_cooldown }
						NOT = { has_variable = is_the_revant }
						NOR = {
							has_trait = brain_damaged
							has_trait = incapable
							has_trait = goblin_trait
							has_variable = on_orc_adventure
						}
						OR = {
							is_ruler = yes
							is_general = yes
							is_admiral = yes
							is_governor = yes
							has_office = office_marshal
							has_office = office_high_priest_monarchy
							has_office = office_high_priest
							has_office = office_augur
							has_office = office_pontifex
							has_trait = mage_trait
							has_trait = archmage_trait
							prominence >= 60
							martial >= 8
							zeal >= 8
							finesse >= 8
						}
					}
					set_variable = omni_studying_magic_cooldown
					if = {
						limit = {
							NOT = { has_trait = mage_trait }
							NOT = { has_trait = archmage_trait }
						}
						set_variable = first_time_mage_var
					}
					random_list = {
						35 = { set_variable = omni_studying_diviner_var }
						25 = { set_variable = omni_studying_hierophant_var }
						25 = { set_variable = omni_studying_elementalist_var }
						15 = { set_variable = omni_studying_cleric_var }
					}
				}
				trigger_event = {
					id = me_magic_studying.1
					days = { 15 45 }
				}
			}
			65 = {}
		}
	}
}
