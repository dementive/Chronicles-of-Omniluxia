# ---------------------------------------------------------------------------
# AI magic study driver. Called yearly from omni_perf.7 (events/omni_perf_pulses.txt).
#
# Eligibility lives in omni_ai_study_*_candidate_trigger in
# common/scripted_triggers/OMNI_ai_study_triggers.txt rather than being inlined
# here three times, so the copies cannot drift apart.
#
# ENROLMENT MODEL: a country funds up to OMNI_STUDY_MAX_SLOTS students per cycle -
# the ruler first, then the highest-finesse governors/commanders it can still
# afford. Each slot costs OMNI_STUDY_SLOT_COST, and every slot after the ruler
# additionally requires the treasury to stay above OMNI_STUDY_RESERVE, so a rich
# empire trains a magical court while a poor one still gets its ruler schooled and
# does not bankrupt itself doing it. The budget check reads the treasury *after*
# the previous slot was paid for, so the loop self-limits with no counter.
#
# PERF: this is cheaper than the single-student version it replaces. Per country
# per cycle the scans are:
#   1. any_owned_province (short-circuits on the first academy)
#   2. any_character      (short-circuits on the first candidate)
#   3. current_ruler       - a direct scope link, not a scan
#   4. ordered_character   - ONE court pass, capped at max = 2
# The old version did an any_character priority pre-check AND a random_character
# roll on top of the outer any_character, i.e. three full court passes. This does
# one. The order_by sort is over an already-filtered court list of a few dozen at
# most, once per country per year.
# ---------------------------------------------------------------------------
omni_ai_study_magic_effect = {
	if = {
		limit = {
			is_ai = yes
			NOT = { has_variable = omni_ai_magic_study_cooldown }
			treasury >= OMNI_STUDY_SLOT_COST
			any_owned_province = {
				has_building = magic_academy_building
			}
			any_character = {
				employer = root
				omni_ai_study_candidate_trigger = yes
			}
		}
		random_list = {
			90 = {
				# Magic-invested countries (2+ academies) study twice as often.
				if = {
					limit = {
						any_owned_province = {
							count >= 2
							has_building = magic_academy_building
						}
					}
					set_variable = {
						name = omni_ai_magic_study_cooldown
						days = 180
					}
				}
				else = {
					set_variable = {
						name = omni_ai_magic_study_cooldown
						days = 365
					}
				}

				# --- Slot 1: the ruler. Always first when eligible. A magically
				# capable ruler is the visible, playable outcome of this whole
				# system, and current_ruler is a direct link so this costs nothing.
				if = {
					limit = { current_ruler = { omni_ai_study_ruler_candidate_trigger = yes } }
					current_ruler = { omni_ai_enrol_magic_student = yes }
				}

				# --- Slots 2-3: the best governors and commanders the treasury can
				# still cover. Ordered by finesse because finesse is what drives
				# omni_study_progress_gain_value, so these are the characters who
				# actually rank up rather than idling at T1 forever.
				#
				# The ruler may still appear in this list (the list is built before
				# the body runs), but omni_ai_enrol_magic_student re-checks
				# omni_ai_study_blocked_trigger and the ruler now carries a fresh
				# omni_studying_magic_cooldown, so they are skipped rather than
				# double-charged.
				ordered_character = {
					limit = {
						employer = root
						omni_ai_study_priority_candidate_trigger = yes
					}
					order_by = finesse
					# Literal, not a script value: no vanilla ordered_ list passes a
					# script value to max, and a silent parse failure here would
					# quietly enrol the whole court.
					max = 2
					if = {
						limit = {
							employer = { treasury >= OMNI_STUDY_RESERVE }
						}
						omni_ai_enrol_magic_student = yes
					}
				}

				# --- Fallback: if neither the ruler nor any priority candidate was
				# enrolled, spend the one slot on the wider pool so the cycle is not
				# wasted. The outer limit already guaranteed a candidate exists.
				if = {
					limit = { NOT = { has_variable = omni_ai_study_enrolled_this_cycle } }
					random_character = {
						limit = {
							employer = root
							omni_ai_study_candidate_trigger = yes
						}
						omni_ai_enrol_magic_student = yes
					}
				}
				if = {
					limit = { has_variable = omni_ai_study_enrolled_this_cycle }
					remove_variable = omni_ai_study_enrolled_this_cycle
				}
			}
			10 = {}
		}
	}
}

# Enrols ONE character. Evaluated in CHARACTER scope; the employer pays.
#
# The event is fired with no delay on purpose. Saved scopes are per event-chain,
# not per loop iteration, so queueing several delayed me_magic_studying.1 events
# from inside a loop would leave every one of them resolving against whichever
# student happened to be saved last - which is a fresh instance of exactly the
# scattering bug that made this system produce no mages at all. Firing inline
# means check_mage_effect reads scope:omni_study_target while it still points at
# this student. The 15-45 day cosmetic delay is not worth reintroducing that.
omni_ai_enrol_magic_student = {
	if = {
		limit = { NOT = { omni_ai_study_blocked_trigger = yes } }
		employer = {
			add_treasury = {
				value = 0
				subtract = OMNI_STUDY_SLOT_COST
			}
			set_variable = omni_ai_study_enrolled_this_cycle
		}
		omni_ai_prepare_magic_study_character = yes
		employer = {
			trigger_event = { id = me_magic_studying.1 }
		}
		clear_saved_scope = omni_study_target
	}
}

omni_ai_prepare_magic_study_character = {
	# CRITICAL: hand the enrolled character to the study event chain explicitly.
	# me_magic_studying.1 resolves its subject through check_mage_effect, which
	# used to re-roll a random flagged character on EVERY event in the chain. The
	# enrolled character therefore almost never received their own study progress,
	# and progress was smeared one point at a time across every character who had
	# ever been flagged - so nobody reached RANK_THRESHOLD and nobody got
	# mage_trait. check_mage_effect now prefers this scope.
	save_scope_as = omni_study_target
	# The cooldown MUST have a duration. It used to be a bare set_variable, and
	# nothing in the mod ever removed it, so a character enrolled in study once
	# became permanently ineligible for any further AI-driven study - AI mages
	# could never be re-enrolled to climb past their first school rank. Two years
	# is long enough that the AI spreads study across its court rather than
	# drilling one prodigy, and short enough that a long-lived mage can plausibly
	# reach T4 within a reign.
	set_variable = {
		name = omni_studying_magic_cooldown
		days = 730
	}
	if = {
		limit = {
			NOT = { has_trait = mage_trait }
			NOT = { has_trait = archmage_trait }
		}
		set_variable = first_time_mage_var
	}
	omni_ai_pick_study_school = yes
}

# School selection for AI study. Previously a flat 35/25/25/15 random roll that
# ignored the character entirely, so AI mages scattered across schools and rarely
# deepened one. Now it prefers continuity first (keep advancing a school the
# character already has affinity in), then aptitude, and only falls back to a
# random roll for blank-slate characters.
#
# PERF: all triggers here are has_variable / stat reads on a single already-scoped
# character - no scope iteration. Cost is a handful of comparisons, once per
# enrolment, at most a few times per country per decade.
omni_ai_pick_study_school = {
	# Refresh omni_best_affinity_var first: characters who accumulated affinity
	# before this bookkeeping existed (or in an older save) have the four raw
	# affinity vars but no resolved code yet.
	omni_update_best_affinity = yes
	if = {
		# --- 1. Continuity: deepen the school this character already leans into.
		limit = { omni_ai_has_any_magic_affinity = yes }
		random_list = {
			10 = {
				trigger = { omni_ai_best_affinity_is_omnic = yes }
				random_list = {
					1 = { set_variable = omni_studying_diviner_var }
					1 = { set_variable = omni_studying_hierophant_var }
				}
			}
			10 = {
				trigger = { omni_ai_best_affinity_is_aldic = yes }
				random_list = {
					1 = { set_variable = omni_studying_cleric_var }
					1 = { set_variable = omni_studying_illusionist_var }
				}
			}
			10 = {
				trigger = { omni_ai_best_affinity_is_amten = yes }
				random_list = {
					1 = { set_variable = omni_studying_elementalist_var }
					1 = { set_variable = omni_studying_druid_var }
				}
			}
			10 = {
				trigger = { omni_ai_best_affinity_is_melodian = yes }
				random_list = {
					1 = { set_variable = omni_studying_necromancer_var }
					1 = { set_variable = omni_studying_warlock_var }
				}
			}
		}
	}
	else = {
		# --- 2. Aptitude: match the school to what the character is good at.
		# Weights stay soft so the AI still produces some variety.
		random_list = {
			30 = {
				# Zealous / priestly -> healing and blessing schools.
				trigger = {
					OR = {
						zeal >= 8
						has_office = office_high_priest
						has_office = office_high_priest_monarchy
						has_office = office_augur
						has_office = office_pontifex
					}
				}
				random_list = {
					6 = { set_variable = omni_studying_cleric_var }
					4 = { set_variable = omni_studying_hierophant_var }
				}
			}
			30 = {
				# Martial commanders -> battlefield elementalism.
				trigger = {
					OR = {
						martial >= 8
						is_general = yes
						is_admiral = yes
						has_office = office_marshal
					}
				}
				set_variable = omni_studying_elementalist_var
			}
			30 = {
				# Finesse / administrators -> divination and illusion.
				trigger = { finesse >= 8 }
				random_list = {
					6 = { set_variable = omni_studying_diviner_var }
					4 = { set_variable = omni_studying_illusionist_var }
				}
			}
			20 = {
				# --- 3. Blank slate fallback: the old flat roll.
				random_list = {
					35 = { set_variable = omni_studying_diviner_var }
					25 = { set_variable = omni_studying_hierophant_var }
					25 = { set_variable = omni_studying_elementalist_var }
					15 = { set_variable = omni_studying_cleric_var }
				}
			}
		}
	}
}
