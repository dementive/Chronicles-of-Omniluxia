# Adventurer's Guild relics: equippable artifacts that exist nowhere at game start.
# A successful or legendary contract may turn one up; each can be found once per game.
# Character scope (the adventurer). Called from omni_adventure_bonus_rewards_effect.

omni_guild_relic_roll_effect = {
	if = {
		limit = {
			OR = { has_variable = adventure_outcome_legendary has_variable = adventure_outcome_success }
			OR = {
				NOT = { has_global_variable = omni_guild_relic_gilded_axe_dimnas }
				NOT = { has_global_variable = omni_guild_relic_spearhead_spear_wind }
				NOT = { has_global_variable = omni_guild_relic_copper_aegis_usreva }
				NOT = { has_global_variable = omni_guild_relic_peaked_helm_aydar }
				NOT = { has_global_variable = omni_guild_relic_mantle_istrilneth }
				NOT = { has_global_variable = omni_guild_relic_bronze_helm_lanieonguth }
				NOT = { has_global_variable = omni_guild_relic_spore_staff_tzagrin }
				NOT = { has_global_variable = omni_guild_relic_boar_horn_killdrukaath }
				NOT = { has_global_variable = omni_guild_relic_amulet_life }
				NOT = { has_global_variable = omni_guild_relic_blade_night }
				NOT = { has_global_variable = omni_guild_relic_chalice_blood }
				NOT = { has_global_variable = omni_guild_relic_cloak_shadows }
				NOT = { has_global_variable = omni_guild_relic_eye_storm }
				NOT = { has_global_variable = omni_guild_relic_helm_ancients }
				NOT = { has_global_variable = omni_guild_relic_horn_beasts }
				NOT = { has_global_variable = omni_guild_relic_kings_heart }
				NOT = { has_global_variable = omni_guild_relic_orb_knowledge }
				NOT = { has_global_variable = omni_guild_relic_ring_power }
				NOT = { has_global_variable = omni_guild_relic_staff_elements }
			}
		}
		random_list = {
			4 = {
				modifier = { factor = 2.5 has_variable = adventure_outcome_legendary }
				modifier = { factor = 2 OR = { has_variable = adventure_contract_relic has_variable = adventure_contract_ruin } }
				modifier = { factor = 1.5 has_variable = adventure_fine_loot }
				omni_guild_relic_grant_effect = yes
			}
			96 = { }
		}
	}
}
