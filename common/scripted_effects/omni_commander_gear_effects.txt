# COUNTRY scope. Monthly (from omni_perf.1). Only walks characters who carry an
# equipped-item list and currently lead a unit, so it is a handful per country.
omni_sync_country_commander_gear = {
	every_character = {
		limit = {
			has_variable_list = equiped_item_list
			is_ruler = no
			any_character_unit = { OR = { is_army = yes is_navy = yes } }
		}
		omni_sync_commander_gear_layers = yes
	}
}
