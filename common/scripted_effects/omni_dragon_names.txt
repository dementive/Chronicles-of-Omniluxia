omni_create_named_bound_dragon = {
    create_unit = {
        name = $NAME$
        location = scope:omni_dragon_capital
        sub_unit = omni_dragon
        save_scope_as = omni_bound_dragon
    }
    scope:omni_bound_dragon = { set_variable = { name = omni_dragon_name_index value = $INDEX$ } }
}

omni_create_named_wild_dragon = {
    omni_create_wild_dragon_self = { NAME = $NAME$ LAIR = scope:omni_dragon_lair }
    create_unit = {

        sub_unit = omni_dragon

# Country scope (the rebels). The wild dragon leads its own army: a dragon
# character sharing the army's name. Saved as scope:omni_dragon_rider.
omni_create_wild_dragon_self = {
    omni_retire_wild_dragon_self = yes
    create_character = {
        first_name = $NAME$
        culture = $LAIR$.dominant_province_culture
        religion = dragkhan_pantheon
        age = 40
        save_scope_as = omni_dragon_rider
    }
    scope:omni_dragon_rider = {
        set_variable = omni_wild_dragon_self
        omni_apply_dragon_sovereign_trait = yes
        random_list = {
            1 = { omni_set_dragon_rider_martial = { M = 10 } }
            1 = { omni_set_dragon_rider_martial = { M = 11 } }
            1 = { omni_set_dragon_rider_martial = { M = 12 } }
            1 = { omni_set_dragon_rider_martial = { M = 13 } }
            1 = { omni_set_dragon_rider_martial = { M = 14 } }
            1 = { omni_set_dragon_rider_martial = { M = 15 } }
        }
    }
    set_global_variable = { name = omni_wild_dragon_self value = scope:omni_dragon_rider }
}

# Any scope. The wild dragon's own character dies with its army.
omni_retire_wild_dragon_self = {
    if = {
        limit = { exists = global_var:omni_wild_dragon_self }
        global_var:omni_wild_dragon_self = {
            if = {
                limit = { is_alive = yes }
                death = { death_reason = death_magic }
            }
        }
        remove_global_variable = omni_wild_dragon_self
    }
}

omni_crown_named_dragon = {
    change_country_name = $NAME$_realm
    create_character = {
        first_name = $NAME$
        culture = scope:omni_dragon_capital.dominant_province_culture
        religion = dragkhan_pantheon
        age = 35
        female = no
        save_scope_as = omni_new_sovereign
    }
    scope:omni_new_sovereign = {
        set_variable = { name = omni_dragon_sovereign value = $INDEX$ }
        omni_apply_dragon_sovereign_trait = yes
    }
    set_as_ruler = scope:omni_new_sovereign
    omni_found_dragon_royal_family = yes
}
