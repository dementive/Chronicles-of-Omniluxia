omni_create_dragon_royal_family = {
    create_character = {
        first_name = $NAME$
        age = 32
        female = yes
        culture = scope:omni_new_sovereign.culture
        religion = dragkhan_pantheon
        set_variable = { name = omni_dragon_sovereign }
        save_scope_as = omni_dragon_bride
    }
    # Flag outside create_character too, matching the sovereign; the portrait GUI keys off this.
    scope:omni_dragon_bride = {
        set_variable = { name = omni_dragon_sovereign }
        omni_apply_dragon_sovereign_trait = yes
    }
    scope:omni_new_sovereign = { marry_character = scope:omni_dragon_bride }
    set_as_coruler = scope:omni_dragon_bride
    omni_create_dragon_hatchling = yes
}

# Country scope, with both parents saved by the coronation.
omni_create_dragon_hatchling = {
    create_character = {
        age = 1
        father = scope:omni_new_sovereign
        mother = scope:omni_dragon_bride
        culture = scope:omni_new_sovereign.culture
        religion = dragkhan_pantheon
        set_variable = { name = omni_dragon_descendant }
        save_scope_as = omni_dragon_hatchling
    }
    scope:omni_dragon_hatchling = {
        omni_name_dragon_hatchling = yes
        set_racial_trait_effect = yes
    }
}

# Country scope, with scope:omni_dragon_new_realm set. Queue omni_dragon.12.
omni_dragon_send_herald = {
    set_variable = { name = omni_dragon_herald_realm value = scope:omni_dragon_new_realm days = 30 }
    trigger_event = { id = omni_dragon.12 days = 3 }
}

# Country scope. Recheck after siege resolution before retiring the rebel unit.
omni_try_dragon_settlement = {
    if = {
        limit = { omni_dragon_controls_realm = yes }
        every_owned_province = {
            remove_variable = omni_dragon_occupier
            remove_variable = omni_dragon_scorched
            set_controller = owner
        }
        if = {
            limit = { global_var:omni_dragon_horde = { has_variable = omni_dragon_name_index } }
            set_variable = { name = omni_dragon_name_index value = global_var:omni_dragon_horde.var:omni_dragon_name_index }
        }
        global_var:omni_dragon_horde = {
            remove_variable = omni_wild_morrigon
            destroy_unit = yes
        }
        remove_global_variable = omni_dragon_horde
        omni_retire_wild_dragon_self = yes
        remove_global_variable = omni_dragon_in_field
        remove_global_variable = omni_dragon_hunt
        set_global_variable = { name = omni_dragon_roll_lock days = 365 }
        omni_release_dragon_scorched_land = yes
        change_government = omni_dragon_dominion
        omni_set_dragon_levies = yes
        if = { limit = { is_ai = yes } set_antagonist = yes }
        change_country_adjective = omni_morrigon_realm_adj
        capital_scope = { save_scope_as = omni_dragon_capital }
        omni_choose_dragon_name = { ACTION = omni_crown_named_dragon }
        remove_variable = omni_dragon_name_index
        trigger_event = { id = omni_dragon.11 }
        # Word of the coronation reaches every neighbour, then every other realm
        # holding land on the dragon's continent. The variable doubles as the
        # "already told" mark, so no realm hears it twice.
        save_scope_as = omni_dragon_new_realm
        every_neighbour_country = {
            omni_dragon_send_herald = yes
        }
        every_country = {
            limit = {
                NOT = { this = scope:omni_dragon_new_realm }
                NOT = { has_variable = omni_dragon_herald_realm }
                any_owned_province = { omni_dragon_same_continent_as = { PROV = scope:omni_dragon_capital } }
            }
            omni_dragon_send_herald = yes
        }
    }
}
