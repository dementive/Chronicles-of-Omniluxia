# Any scope. Prints VALUE as sign + four decimal digits, labelled NAME.
omni_census_print = {
    set_local_variable = {
        name = omni_census_v
        value = $VALUE$
    }
    if = {
        limit = { local_var:omni_census_v < 0 }
        debug_log = "OMNI_CENSUS $NAME$ sign -"
        set_local_variable = {
            name = omni_census_v
            value = {
                value = local_var:omni_census_v
                multiply = -1
            }
        }
    }
    set_local_variable = {
        name = omni_census_v
        value = {
            value = local_var:omni_census_v
            max = 9999
            round = yes
        }
    }
    set_local_variable = {
        name = omni_census_d
        value = {
            value = local_var:omni_census_v
            divide = 1000

        debug_log = "OMNI_CENSUS $NAME$ 1000 0"

# Character scope (the caster). Called from omni_spell_mana_cost_effect while
# omni_spell_base is still set. "battle" = caster's unit is fighting; "field" =
# caster commands a unit but is not in combat; the rest are court/civic casts.
omni_census_count_cast = {
    set_local_variable = {
        name = omni_census_cast_cost
        value = omni_scaled_spell_cost_value
    }
    if = {
        limit = { any_character_unit = { in_combat = yes } }
        set_local_variable = { name = omni_census_cast_kind value = 2 }
    }
    else_if = {
        limit = { any_character_unit = { always = yes } }
        set_local_variable = { name = omni_census_cast_kind value = 1 }
    }
    else = {
        set_local_variable = { name = omni_census_cast_kind value = 0 }
    }
    employer = {
        if = {
            limit = { NOT = { has_variable = omni_census_casts } }
            set_variable = { name = omni_census_casts value = 0 }
            set_variable = { name = omni_census_casts_battle value = 0 }
            set_variable = { name = omni_census_casts_field value = 0 }
            set_variable = { name = omni_census_mana_spent value = 0 }
        }
        change_variable = { name = omni_census_casts add = 1 }
        change_variable = { name = omni_census_mana_spent add = local_var:omni_census_cast_cost }
        if = {
            limit = { local_var:omni_census_cast_kind = 2 }
            change_variable = { name = omni_census_casts_battle add = 1 }
        }
        else_if = {
            limit = { local_var:omni_census_cast_kind = 1 }
            change_variable = { name = omni_census_casts_field add = 1 }
        }
    }
    # Per-cast trace, battle casts only (all casts produced ~700 MB in 70 years).
    if = {
        limit = { local_var:omni_census_cast_kind = 2 }
        debug_log = "OMNI_CAST begin"
        debug_log_scopes = yes
        omni_census_print = { NAME = cast_base VALUE = var:omni_spell_base }
        omni_census_print = { NAME = cast_cost VALUE = local_var:omni_census_cast_cost }
        omni_census_print = { NAME = cast_kind VALUE = local_var:omni_census_cast_kind }
        omni_census_print = { NAME = cast_mana_left VALUE = var:magic }
        debug_log = "OMNI_CAST end"
    }
    remove_local_variable = omni_census_cast_cost
    remove_local_variable = omni_census_cast_kind
}

# Country scope. Brackets one immediate block of the yearly AI chain and adds the
# gold it spent (same tick, so no income mixes in) to a per-country counter.
omni_census_spend_begin = {
    set_local_variable = {
        name = omni_census_t0
        value = treasury
    }
}
omni_census_spend_end = {
    if = {
        limit = {
            omni_ai_census_enabled > 0
            has_local_variable = omni_census_t0
        }
        if = {
            limit = { NOT = { has_variable = omni_census_spend_$KEY$ } }
            set_variable = { name = omni_census_spend_$KEY$ value = 0 }
        }
        change_variable = {
            name = omni_census_spend_$KEY$
            add = {
                value = local_var:omni_census_t0
                subtract = treasury
            }
        }
    }
    remove_local_variable = omni_census_t0
}
