# ----------------------------------------------------------------------------------------------------------------------
# Omniluxia - AI battle-magic auto-cast
#
# Lets AI mage-generals cast the same friendly army buffs a player can (Gale Step
# tiers, Battle Hymn, Blessing of War) when their army is in danger or in a fight.
# Driven monthly from me_zorgo_magic_focus.35. All effects run in the CASTER
# (general / character) scope and pay mana through omni_spell_mana_cost_effect, so
# the AI obeys the exact same mana rules and tier gating as the player.
# ----------------------------------------------------------------------------------------------------------------------

# Apply (or refresh) a single unit modifier on every army this general leads.
#   $MOD$ = the unit modifier name to apply.
omni_apply_unit_buff_effect = {
	every_character_unit = {
		limit = { is_army = yes }
		if = {
			limit = { has_unit_modifier = $MOD$ }
			remove_unit_modifier = $MOD$
		}
		add_unit_modifier = {
			name = $MOD$
			duration = 1825
		}
	}
}

# MOBILITY cast: best affordable Gale Step (archmages prefer Blessing of War, which
# also grants speed). Used to flee a stack-wipe or reposition. Skips entirely if the
# army already carries any speed buff, so the AI never wastes mana refreshing it.
omni_ai_cast_mobility_buff_effect = {
	if = {
		limit = {
			NOT = {
				any_character_unit = {
					is_army = yes
					OR = {
						has_unit_modifier = omni_battle_haste_spell_mod
						has_unit_modifier = omni_battle_haste_2_spell_mod
						has_unit_modifier = omni_battle_haste_3_spell_mod
						has_unit_modifier = omni_battle_haste_4_spell_mod
						has_unit_modifier = omni_battle_blessing_spell_mod
					}
				}
			}
		}
		if = {
			limit = {
				has_trait = archmage_trait
				omni_can_afford_spell = { BASE = 100 }
			}
			omni_spell_mana_cost_effect = { BASE = 100 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_haste_4_spell_mod }
		}
		else_if = {
			limit = { omni_can_afford_spell = { BASE = 75 } }
			omni_spell_mana_cost_effect = { BASE = 75 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_haste_3_spell_mod }
		}
		else_if = {
			limit = { omni_can_afford_spell = { BASE = 50 } }
			omni_spell_mana_cost_effect = { BASE = 50 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_haste_2_spell_mod }
		}
		else_if = {
			limit = { omni_can_afford_spell = { BASE = 25 } }
			omni_spell_mana_cost_effect = { BASE = 25 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_haste_spell_mod }
		}
	}
}

# COMBAT cast: best affordable survival buff for an army already locked in battle.
# Archmages cast Blessing of War (morale + discipline + speed); everyone else casts
# Battle Hymn (morale) to keep the stack from routing and being wiped. Skips if the
# army already has a combat buff.
omni_ai_cast_combat_buff_effect = {
	if = {
		limit = {
			NOT = {
				any_character_unit = {
					is_army = yes
					OR = {
						has_unit_modifier = omni_battle_blessing_spell_mod
						has_unit_modifier = omni_battle_rally_spell_mod
					}
				}
			}
		}
		if = {
			limit = {
				has_trait = archmage_trait
				omni_can_afford_spell = { BASE = 75 }
			}
			omni_spell_mana_cost_effect = { BASE = 75 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_blessing_spell_mod }
		}
		else_if = {
			limit = { omni_can_afford_spell = { BASE = 25 } }
			omni_spell_mana_cost_effect = { BASE = 25 }
			omni_apply_unit_buff_effect = { MOD = omni_battle_rally_spell_mod }
		}
	}
}

# Top-level decision for one eligible mage-general. Expects scope:caster_country to
# be the general's country (used to identify hostile armies).
#   1) Already fighting        -> combat buff (can't out-run a joined battle).
#   2) Hostile stack co-located-> mobility buff (the escape window).
#   3) Hostile stack adjacent  -> mobility buff (evade / concentrate).
omni_ai_battlecast_effect = {
	if = {
		limit = {
			any_character_unit = {
				is_army = yes
				in_combat = yes
			}
		}
		omni_ai_cast_combat_buff_effect = yes
	}
	else_if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = {
					any_unit_in_province = {
						is_army = yes
						unit_owner = { war_with = scope:caster_country }
					}
				}
			}
		}
		omni_ai_cast_mobility_buff_effect = yes
	}
	else_if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = {
					any_neighbor_province = {
						any_unit_in_province = {
							is_army = yes
							unit_owner = { war_with = scope:caster_country }
						}
					}
				}
			}
		}
		omni_ai_cast_mobility_buff_effect = yes
	}
}
