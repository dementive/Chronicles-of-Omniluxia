# Character scope. Resolve one accessible teaching site; never mix its access with another location.
omni_folk_resolve_teacher = {
 save_scope_as = omni_folk_student
 if = {
  limit = { location = { omni_folk_accessible_site = { COUNTRY = scope:omni_folk_student.employer } } }
  location = { save_scope_as = omni_folk_teacher }
 }
 else = {
  random_character_unit = {
   limit = { OR = { is_army = yes is_navy = yes } unit_location = { omni_folk_accessible_site = { COUNTRY = scope:omni_folk_student.employer } } }
   unit_location = { save_scope_as = omni_folk_teacher }
  }
 }
}
