# Historical holding bootstrap for major tags.
# This gives the big empires and key regional powers a small estate base at game start,
# so holdings are not completely empty in the opening setup.

omni_seed_family_holding_once = {
	every_family = {
		head_of_family = {
			save_scope_as = holding_seed_hof
		}

		ROOT = {
			random_owned_province = {
				limit = {
					is_capital = no
					OR = {
						NOT = {
							exists = holding_owner
						}
						AND = {
							exists = holding_owner
							holding_owner = {
								is_alive = no
							}
						}
					}
				}
				weight = {
					modifier = {
						factor = 10000
						any_neighbor_province = {
							trigger_if = {
								limit = {
									exists = holding_owner
									holding_owner = scope:holding_seed_hof
								}
							}
							trigger_else = {
								always = no
							}
						}
					}
				}
				save_scope_as = holding_seed_target
			}
		}

		head_of_family = {
			if = {
				limit = {
					exists = scope:holding_seed_target
				}
				add_holding = scope:holding_seed_target
			}
		}

		clear_saved_scope = holding_seed_target
		clear_saved_scope = holding_seed_hof
	}
}
