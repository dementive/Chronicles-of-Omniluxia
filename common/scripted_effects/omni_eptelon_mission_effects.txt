# Country scope. Province here means an area (the game's claim unit).
omni_eptelon_opening_claims_effect = {
    custom_tooltip = omni_eptelon_opening_claims_tt
    hidden_effect = {
        capital_scope.area = {
            save_scope_as = omni_eptelon_home_area
            every_area_province = { add_claim = root }
        }
        # Pick a province bordering the home province, within the home region.
        capital_scope.region = {
            random_region_province = {
                limit = {
                    has_owner = yes
                    NOT = { owner = root }
                    NOT = { area = scope:omni_eptelon_home_area }
                    any_neighbor_province = { area = scope:omni_eptelon_home_area }
                }
                area = {
                    every_area_province = { add_claim = root }
                }
            }
        }
    }
}

# Never empty a territory. If fewer than five can be spared, send what is available.

    # Guard the effect itself against duplicate prompts and direct repeated calls.
    if = {
        limit = {
            has_variable = omni_eptelon_slave_oath

            exists = scope:omni_eptelon_oath_destination
        }
        set_variable = { name = omni_eptelon_oath_payment_lock days = 1825 }
        set_variable = { name = omni_eptelon_oath_paid value = 0 }

            limit = { total_population > 1 }
