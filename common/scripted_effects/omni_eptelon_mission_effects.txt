# Country scope. Province here means an area (the game's claim unit).
omni_eptelon_opening_claims_effect = {
    custom_tooltip = omni_eptelon_opening_claims_tt
    hidden_effect = {
        capital_scope.area = {
            save_scope_as = omni_eptelon_home_area
            every_area_province = { add_claim = root }
        }
        # Pick a province bordering the home province, within the home region.
        capital_scope.region = {
            random_region_province = {
                limit = {
                    has_owner = yes
                    NOT = { owner = root }
                    NOT = { area = scope:omni_eptelon_home_area }
                    any_neighbor_province = { area = scope:omni_eptelon_home_area }
                }
                area = {
                    every_area_province = { add_claim = root }
                }
            }
        }
    }
}

# Country scope; destination is resolved from the persistent creditor at each due date.
# Invictus me_subject_independence.166 uses move_pop followed by set_pop_type.
# Never empty a territory. If fewer than five can be spared, send what is available.
omni_eptelon_slave_oath_payment_effect = {
    # Guard the effect itself against duplicate prompts and direct repeated calls.
    if = {
        limit = {
            has_variable = omni_eptelon_slave_oath
            NOT = { has_variable = omni_eptelon_oath_payment_lock }
            exists = scope:omni_eptelon_oath_destination
        }
        set_variable = { name = omni_eptelon_oath_payment_lock days = 1825 }
        set_variable = { name = omni_eptelon_oath_paid value = 0 }
        # Exactly five single-pop selections; recheck the source before each.
        random_owned_province = {
            limit = { total_population > 1 }
            random_pops_in_province = {
                set_pop_type = slaves
                move_pop = scope:omni_eptelon_oath_destination
                root = { change_variable = { name = omni_eptelon_oath_paid add = 1 } }
            }
        }
        random_owned_province = {
            limit = { total_population > 1 }
            random_pops_in_province = {
                set_pop_type = slaves
                move_pop = scope:omni_eptelon_oath_destination
                root = { change_variable = { name = omni_eptelon_oath_paid add = 1 } }
            }
        }
        random_owned_province = {
            limit = { total_population > 1 }
            random_pops_in_province = {
                set_pop_type = slaves
                move_pop = scope:omni_eptelon_oath_destination
                root = { change_variable = { name = omni_eptelon_oath_paid add = 1 } }
            }
        }
        random_owned_province = {
            limit = { total_population > 1 }
            random_pops_in_province = {
                set_pop_type = slaves
                move_pop = scope:omni_eptelon_oath_destination
                root = { change_variable = { name = omni_eptelon_oath_paid add = 1 } }
            }
        }
        random_owned_province = {
            limit = { total_population > 1 }
            random_pops_in_province = {
                set_pop_type = slaves
                move_pop = scope:omni_eptelon_oath_destination
                root = { change_variable = { name = omni_eptelon_oath_paid add = 1 } }
            }
        }
        trigger_event = { id = me_eptelon_gen.65 days = 1825 }
    }
}

# Masters of the Silphium: a grain, vegetable, game or timber territory on the northern slopes is
# terraced for silphium. Nothing happens if no such territory is owned.
omni_eptelon_new_silphium_terrace_effect = {
    if = {
        limit = {
            any_owned_province = {
                is_in_region = eptelon_north_region
                OR = {
                    trade_goods = grain
                    trade_goods = vegetables
                    trade_goods = wild_game
                    trade_goods = wood
                }
            }
        }
        random_owned_province = {
            limit = {
                is_in_region = eptelon_north_region
                OR = {
                    trade_goods = grain
                    trade_goods = vegetables
                    trade_goods = wild_game
                    trade_goods = wood
                }
            }
            custom_tooltip = omni_eptelon_new_silphium_terrace_tt
            hidden_effect = {
                set_trade_goods = silphium
            }
        }
    }
}

# Recorded once in on_start. Aeptelonians sit in the arteonian group, so they are checked by culture.
omni_eptelon_set_heritage_effect = {
    hidden_effect = {
        if = {
            limit = { primary_culture = aeptelonians }
            set_variable = eptelon_heritage_aeptelonian
        }
        else_if = {
            limit = { country_culture_group = eptelon }
            set_variable = eptelon_heritage_native
        }
        else_if = {
            limit = { country_culture_group = horteonian }
            set_variable = eptelon_heritage_heptelonian
        }
        else_if = {
            limit = {
                OR = {
                    country_culture_group = high_elves
                    country_culture_group = high_half_elves
                }
            }
            set_variable = eptelon_heritage_colonial
        }
    }
}

omni_eptelon_clear_heritage_effect = {
    hidden_effect = {
        remove_variable = eptelon_heritage_aeptelonian
        remove_variable = eptelon_heritage_native
        remove_variable = eptelon_heritage_heptelonian
        remove_variable = eptelon_heritage_colonial
    }
}
