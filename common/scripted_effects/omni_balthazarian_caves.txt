    if = { limit = { NOT = { has_variable = ob_cave_threat_applied } }
        set_variable = { name = ob_cave_threat_applied }
        if = { limit = { NOT = { has_variable = ob_early_launched } NOT = { has_variable = ob_early_resolved } }
            if = { limit = { NOT = { has_variable = ob_threat_adjustment } } set_variable = { name = ob_threat_adjustment value = 0 } }
            if = { limit = { has_variable = ob_cave_stores_burned } change_variable = { name = ob_threat_adjustment subtract = 20 } }

        }
    }
