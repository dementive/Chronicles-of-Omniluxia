# COUNTRY scope. The magic menu's student is saved as scope:target.
omni_pay_magic_study_cost = {
	if = {
		limit = { scope:target = { is_ruler = yes } }
		add_treasury = -$PRICE$
	}
	else = {
		scope:target = { add_gold = { value = $PRICE$ multiply = -1 } }
	}
}

# CHARACTER scope. During a commander city visit, the visible event is hosted
# by the ruler, while scope:omni_city_visitor remains the actual payer/payee.
omni_city_actor_change_gold = {
	if = {
		limit = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } }
		scope:omni_city_visitor = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
	}
	else = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
}
