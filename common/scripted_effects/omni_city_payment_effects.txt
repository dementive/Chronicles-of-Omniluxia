# COUNTRY scope. The magic menu's student is saved as scope:target.
omni_pay_magic_study_cost = {
	if = {
		limit = { scope:target = { is_ruler = yes } }
		add_treasury = -$PRICE$
	}
	else = {
		scope:target = { add_gold = { value = $PRICE$ multiply = -1 } }
	}
}

# Event-host scope. During a commander city visit, scope:omni_city_visitor is
# always the actual payer/payee, including country-hosted circus events.
omni_city_actor_change_gold = {
	if = {
		limit = { exists = scope:omni_city_visitor }
		scope:omni_city_visitor = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
	}
	else = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
}

# Uses treasury when a ruler has no separately saved visitor; otherwise charges
# the visiting character's personal wealth.
omni_city_actor_pay_magic_store_cost = {
	if = {
		limit = { exists = scope:omni_city_visitor }
		scope:omni_city_visitor = { add_gold = { value = $PRICE$ multiply = -1 } }
	}
	else_if = {
		limit = { is_ruler = yes }
		country = { add_treasury = -$PRICE$ }
	}
	else = {
		add_gold = { value = $PRICE$ multiply = -1 }
	}
}
