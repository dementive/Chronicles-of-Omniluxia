# COUNTRY scope. The magic menu's student is saved as scope:target.
omni_pay_magic_study_cost = {
	if = {
		limit = { scope:target = { is_ruler = yes } }
		custom_tooltip = omni_academy_ruler_payment_tt
		add_treasury = -$PRICE$
	}
	else = {
		custom_tooltip = omni_academy_personal_payment_tt
		scope:target = { add_gold = { value = $PRICE$ multiply = -1 } }
	}
}

# Event-host scope. During a commander city visit, scope:omni_city_visitor is
# always the actual payer/payee, including country-hosted circus events.
omni_city_actor_change_gold = {
	if = {
		limit = { exists = scope:omni_city_visitor }
		scope:omni_city_visitor = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
	}
	else = { add_gold = { value = $VALUE$ multiply = $MULTIPLY$ } }
}

# PRICE is evaluated on the visitor, even when the treasury pays.
omni_pay_mage_tower_price = {
	scope:omni_city_visitor = {
		if = {
			limit = { is_ruler = yes }
			custom_tooltip = omni_academy_ruler_payment_tt
			employer = { add_treasury = { value = scope:omni_city_visitor.$PRICE$ multiply = -1 } }
		}
		else = {
			custom_tooltip = omni_academy_personal_payment_tt
			add_gold = { value = $PRICE$ multiply = -1 }
		}
	}
}

# COUNTRY scope (the magic menu's employer). Pays for and starts one academy
# course for scope:target. Shared by every course option in events/magic_menu.txt.
#   PRICE  - gold, paid from the treasury by a ruler, otherwise personally
#   SCHOOL - suffix of the omni_studying_<SCHOOL>_var flag
# The first session goes through the generic study events only. The Diviner
# Accident (me_magic_studying.5) is chosen later by magic_studying_t1_effect,
# which limits it to Diviner students.
omni_start_academy_course = {
	omni_pay_magic_study_cost = { PRICE = $PRICE$ }
	scope:target = { set_variable = omni_studying_$SCHOOL$_var }
	hidden_effect = {
		if = {
			limit = { scope:target = { NOT = { has_variable = first_magic_event_blocker_var } } }
			trigger_event = { id = me_magic_studying.1 days = 15 }
		}
		else = {
			random_list = {
				1 = { trigger_event = { id = me_magic_studying.2 days = { 30 180 } } }
				1 = { trigger_event = { id = me_magic_studying.3 days = { 30 180 } } }
				1 = { trigger_event = { id = me_magic_studying.4 days = { 30 180 } } }
			}
		}
	}
}
