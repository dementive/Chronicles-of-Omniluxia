# Unit scope. End piracy and start the return/cooldown phase.
omni_finish_piracy_effect = {
	# Native completion and the backup event may both fire; clean up once.
	if = {
		limit = { has_variable = engage_in_piracy_active }
		remove_variable = engage_in_piracy_active
		# Reuse the cooldown window for the return trip; no recurring script scan.
		set_variable = { name = omni_piracy_returning_to_port value = yes days = 120 }
		if = {
			limit = {
				has_global_variable = omni_active_piracy_count
				global_var:omni_active_piracy_count >= 1
			}
			change_global_variable = { name = omni_active_piracy_count subtract = 1 }
		}
		if = {
			limit = {
				has_unit_modifier = engage_in_piracy_unit_modifier
			}
			remove_unit_modifier = engage_in_piracy_unit_modifier
		}
		set_variable = {
			name = engage_in_piracy_cooldown
			value = 1
			days = 120
		}
	}
}
