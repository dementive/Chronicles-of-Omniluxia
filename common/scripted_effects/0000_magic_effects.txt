magic_monthly_effects = {
	change_country_scope_magic_vars = yes
}

recalculate_character_magic_vars = {
	set_variable = {
		name = magic_monthly
		value = 0
	}
	change_variable = {
		name = magic_monthly
		add = modifier:great_work_tribals_workrate_character_modifier
	}
	set_variable = {
		name = magic_max
		value = 100
	}
	change_variable = {
		name = magic_max
		add = modifier:great_work_freemen_workrate_character_modifier
	}
}

magic_on_birth_effects = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = {
				integer_range = { min = 1 max = 10 }
			}
		}
	}
	recalculate_character_magic_vars = yes
}

magic_game_start_effects = {
	if = {
		limit = {
			NOT = { has_global_variable = magic_setup_done }
		}
		set_magic_vars = yes
	}
	set_global_variable = magic_setup_done
	trigger_event = {
		id = magic.2
		days = 1 # Remove vars next day
	}
}

set_magic_vars = {
	every_country = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_living_character = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		recalculate_character_magic_vars = yes
	}
}

change_country_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
}

change_province_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
}

clamp_character_magic_to_max = {
	recalculate_character_magic_vars = yes
	set_variable = {
		name = magic_clamp_check
		value = var:magic
	}
	change_variable = {
		name = magic_clamp_check
		subtract = 100
	}
	change_variable = {
		name = magic_clamp_check
		subtract = modifier:great_work_freemen_workrate_character_modifier
	}
	if = {
		limit = { var:magic_clamp_check > 0 }
		set_variable = {
			name = magic
			value = 100
		}
		change_variable = {
			name = magic
			add = modifier:great_work_freemen_workrate_character_modifier
		}
	}
	remove_variable = magic_clamp_check
}

change_character_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
	}
	clamp_character_magic_to_max = yes
}

change_character_magic_monthly = {
	change_variable = {
		name = magic
		add = modifier:great_work_tribals_workrate_character_modifier
	}
	clamp_character_magic_to_max = yes
}

change_country_scope_magic_vars = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = 0
		}
	}
	change_variable = {
		name = magic
		add = modifier:global_cohort_recruit_speed
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
	every_owned_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		change_variable = {
			name = magic
			add = modifier:local_cohort_recruit_speed
		}
		if = {
			limit = { var:magic > 100 }
			set_variable = {
				name = magic
				value = 100
			}
		}
	}
	every_character = {
		limit = {
			is_alive = yes
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		change_character_magic_monthly = yes
	}
}
