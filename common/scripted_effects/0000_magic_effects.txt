magic_monthly_effects = {
	change_country_scope_magic_vars = yes
}

recalculate_character_magic_vars = {
	if = {
		limit = { NOT = { has_variable = magic } }
		set_variable = {
			name = magic
			value = 0
		}
	}
	set_variable = {
		name = magic_monthly
		value = 0
	}
	change_variable = {
		name = magic_monthly
		add = modifier:great_work_tribals_workrate_character_modifier
	}
	set_variable = {
		name = magic_max
		value = 100
	}
	change_variable = {
		name = magic_max
		add = modifier:great_work_freemen_workrate_character_modifier
	}
}

refill_character_magic_to_max = {
	recalculate_character_magic_vars = yes
	set_variable = {
		name = magic
		value = var:magic_max
	}
	clamp_character_magic_to_max = yes
}

magic_on_birth_effects = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = {
				integer_range = { min = 1 max = 10 }
			}
		}
	}
	recalculate_character_magic_vars = yes
}

magic_game_start_effects = {
	if = {
		limit = {
			NOT = { has_global_variable = magic_setup_done }
		}
		set_magic_vars = yes
	}
	set_global_variable = magic_setup_done
}

set_magic_vars = {
	every_country = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_living_character = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		recalculate_character_magic_vars = yes
	}
}

clamp_character_magic_to_max = {
	recalculate_character_magic_vars = yes
	if = {
		limit = { var:magic > omni_character_magic_max_value }
		set_variable = {
			name = magic
			value = 100
		}
		change_variable = {
			name = magic
			add = modifier:great_work_freemen_workrate_character_modifier
		}
	}
}

change_country_scope_magic_vars = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = 0
		}
	}
	change_variable = {
		name = magic
		add = modifier:global_cohort_recruit_speed
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
	every_owned_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
			# Perf: skip provinces where this month's tick is provably a no-op:
			# var already initialized, zero regen to add, and no clamping needed.
			# Behavior identical to running the body.
			NAND = {
				has_variable = magic
				modifier:local_cohort_recruit_speed = 0
				var:magic <= 100
			}
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		change_variable = {
			name = magic
			add = modifier:local_cohort_recruit_speed
		}
		if = {
			limit = { var:magic > 100 }
			set_variable = {
				name = magic
				value = 100
			}
		}
	}
	every_character = {
		limit = {
			is_alive = yes
			# Perf: skip characters where the monthly tick is provably a no-op:
			# var initialized, zero regen, and magic <= 100 with a non-negative max
			# bonus means the clamp cannot trigger. magic_max/magic_monthly are only
			# ever read right after being recalculated, so skipping the recalc here
			# is unobservable. Behavior identical.
			NAND = {
				has_variable = magic
				modifier:great_work_tribals_workrate_character_modifier = 0
				modifier:great_work_freemen_workrate_character_modifier >= 0
				var:magic <= 100
			}
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		change_character_magic_monthly = yes
	}
}

		}
		change_variable = {
			name = magic

		}
		if = {

			set_variable = {
				name = magic
				value = 100
			}
		}
	}
	every_living_character = {
		change_character_magic_monthly = yes
	}
}
