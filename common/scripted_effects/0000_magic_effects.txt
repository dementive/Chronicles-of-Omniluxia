magic_monthly_effects = {
	if = {
		limit = {
			NOT = { has_global_variable = magic_monthly_done }
		}
		change_magic_vars = yes
	}
	set_global_variable = magic_monthly_done
	trigger_event = {
		id = magic.1
		days = 2 # Remove vars next day
	}
}

magic_on_birth_effects = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = {
				integer_range = { min = 1 max = 10 }
			}
		}
	}
}

magic_game_start_effects = {
	if = {
		limit = {
			NOT = { has_global_variable = magic_setup_done }
		}
		set_magic_vars = yes
	}
	set_global_variable = magic_setup_done
	trigger_event = {
		id = magic.2
		days = 1 # Remove vars next day
	}
}

set_magic_vars = {
	every_country = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_living_character = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
}

change_country_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
		max = {
			value = 100
		}
	}
}

change_province_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
		max = {
			value = 100
		}
	}
}

change_character_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
		max = {
			value = 100
			add = modifier:great_work_freemen_workrate_character_modifier
		}
	}
}

change_magic_vars = {
	set_magic_vars = yes
	every_country = {
		change_variable = {
			name = magic
			add = modifier:global_cohort_recruit_speed
			max = {
				value = 100
			}
		}
	}
	every_province = {
		limit = {
			is_sea = no
			is_inhabitable = yes
		}
		change_variable = {
			name = magic
			add = modifier:local_cohort_recruit_speed
			max = {
				value = 100
			}
		}
	}
	every_living_character = {
		change_variable = {
			name = magic
			add = modifier:great_work_tribals_workrate_character_modifier
			max = {
				value = 100
				add = modifier:great_work_freemen_workrate_character_modifier
			}
		}
	}
}
