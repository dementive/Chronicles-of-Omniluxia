magic_monthly_effects = {
	change_country_scope_magic_vars = yes
}

# NOTE (July 2026): magic_monthly and magic_max are display-only copies of the
# modifier-derived script values (omni_character_monthly_mana_value /
# omni_character_magic_max_value). The tick regenerates them from modifiers, so
# any hand-written change_variable bonus to them is wiped at the next refresh
# and never has a mechanical effect (see PERFORMANCE_AUDIT.md). They must stay
# STORED because this Imperator version's data system has no ScriptValue loc
# function - the mana tooltips can only read Var('magic_monthly')/Var('magic_max').
# They refresh monthly with the mana tick.
recalculate_character_magic_vars = {
	if = {
		limit = {
			has_trait = cramming_fatigue_trait
			NOT = { has_character_modifier = cramming_fatigue_timer_modifier }
		}
		remove_trait = cramming_fatigue_trait
	}
	if = {
		limit = { NOT = { has_variable = magic } }
		set_variable = {
			name = magic
			value = 0
		}
	}
	set_variable = {
		name = magic_monthly
		value = omni_character_monthly_mana_value
	}
	set_variable = {
		name = magic_max
		value = omni_character_magic_max_value
	}
	clamp_character_magic_to_max = yes
}

refill_character_magic_to_max = {
	recalculate_character_magic_vars = yes
	set_variable = {
		name = magic
		value = omni_character_magic_max_value
	}
}

magic_on_birth_effects = {
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = 0
		}
	}
	recalculate_character_magic_vars = yes
}

magic_game_start_effects = {
	if = {
		limit = {
			NOT = { has_global_variable = magic_setup_done }
		}
		set_magic_vars = yes
	}
	set_global_variable = magic_setup_done
}

set_magic_vars = {
	every_country = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
	}
	every_living_character = {
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		recalculate_character_magic_vars = yes
	}
}

change_country_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
}

change_province_magic = {
	change_variable = {
		name = magic
		add = $VALUE$
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
}

clamp_character_magic_to_max = {
	if = {
		limit = { has_variable = magic var:magic < 0 }
		set_variable = { name = magic value = 0 }
	}
	if = {
		limit = {
			has_variable = magic
			var:magic > omni_character_magic_max_value
		}
		set_variable = {
			name = magic
			value = omni_character_magic_max_value
		}
	}
}

change_character_magic = {
	if = {
		limit = { NOT = { has_variable = magic } }
		set_variable = {
			name = magic
			value = 0
		}
	}
	change_variable = {
		name = magic
		add = $VALUE$
	}
	clamp_character_magic_to_max = yes
}

change_country_scope_magic_vars = {
	# --- Country mana ---
	if = {
		limit = {
			NOT = { has_variable = magic }
		}
		set_variable = {
			name = magic
			value = 0
		}
	}
	change_variable = {
		name = magic
		add = modifier:global_cohort_recruit_speed
	}
	if = {
		limit = { var:magic > 100 }
		set_variable = {
			name = magic
			value = 100
		}
	}
	# ------------------------------------------------------------------
	# This effect runs from each country's monthly pulse. every_character is scoped
	# to that country, so its mana tick must not be behind a global cooldown: the
	# first country to claim one otherwise blocks every other country's casters.
	# ------------------------------------------------------------------
	every_character = {
		# PERF: skip non-casters. For a character with no mana income, no mana cap, no stored
		# mana/display values and no cramming fatigue, the full tick below only rewrites zeros.
		# Births and game start already initialise these vars, so the result is identical,
		# but most of the world's characters now cost two modifier reads instead of
		# ~6 script-value evaluations and 4 variable writes every month.
		limit = {
			is_alive = yes
			OR = {
				NOT = { has_variable = magic }
				has_trait = cramming_fatigue_trait
				omni_character_monthly_mana_value > 0
				omni_character_magic_max_value > 0
				var:magic > 0
				AND = { has_variable = magic_max var:magic_max > 0 }
				AND = { has_variable = magic_monthly var:magic_monthly > 0 }
			}
		}
		if = {
			limit = {
				NOT = { has_variable = magic }
			}
			set_variable = {
				name = magic
				value = 0
			}
		}
		# Refresh display values and expire fatigue before applying regeneration.
		recalculate_character_magic_vars = yes
		change_variable = {
			name = magic
			add = omni_character_monthly_mana_value
		}
		clamp_character_magic_to_max = yes
	}
}
