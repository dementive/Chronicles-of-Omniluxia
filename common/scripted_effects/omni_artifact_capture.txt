# Capture-on-death. When an artifact bearer dies, the dead bearer's country always
# loses the army modifier. If that country is at war, a war enemy seizes the
# artifact 25% of the time and the other 75% it is lost in the fighting. A bearer
# who dies in peacetime leaves the artifact to the realm's treasury.
# (Equipping destroys the treasure, so the artifact is rebuilt from
# omni_artifact_recreate_<key> rather than looked up in a treasury.)

# scope:omni_art_dead_country set. $TREASURE$ = treasure key, $ARMY$ = army modifier.
omni_artifact_lose_treasure = {
	scope:omni_art_dead_country = { remove_country_modifier = $ARMY$ }
	if = {
		limit = { scope:omni_art_dead_country = { war = yes } }
		random_list = {
			25 = {
				scope:omni_art_dead_country = { random_countries_at_war_with = { save_scope_as = omni_art_captor } }
				if = { limit = { exists = scope:omni_art_captor } scope:omni_art_captor = { omni_artifact_recreate_$TREASURE$ = yes } }
			}
			75 = { }
		}
	}
	else = { scope:omni_art_dead_country = { omni_artifact_recreate_$TREASURE$ = yes } }
}
