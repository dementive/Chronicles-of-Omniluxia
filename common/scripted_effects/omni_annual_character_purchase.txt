# Annual character equipment purchasing system
# Characters in their own country will check available equipment across all markets
# and purchase better gear if they can afford it with their personal wealth
# NPCs now spend actual gold on items based on realistic tier prices

# PRICE TIERS (base costs before market quality modifiers)
# Common items: 50-70 gold
# Uncommon items: 100-150 gold
# Rare items: 180-250 gold
# Unique items: 300+ gold

# Main effect - called once per character per year
omni_annual_character_equipment_check = {
	# Only run for characters who have wealth to spend and are in a country
	if = {
		limit = {
			wealth >= 40
			exists = employer
			NOT = { has_variable = omni_equipment_checked_recently }
		}
		set_variable = { name = omni_equipment_checked_recently days = 30 }
		# Initialize the country's average market quality variable if not already set
		employer = {
			if = {
				limit = { NOT = { has_variable = omni_country_avg_market_quality } }
				set_variable = {
					name = omni_country_avg_market_quality
					value = 0
				}
			}
		}

		# Build the available market items across all provinces of the character's country
		omni_build_country_market_average = yes

		# Now check each item category and see if we should buy an upgrade.
		# There are ten equip categories; this list used to cover six, so the
		# ring and emblem slots were structurally unreachable by AI purchase no
		# matter how rich the character got. curse and special are deliberately
		# still absent - those are narrative items, not merchandise.
		omni_check_and_buy_weapon = yes
		omni_check_and_buy_ranged = yes
		omni_check_and_buy_armour = yes
		omni_check_and_buy_helmet = yes
		omni_check_and_buy_shield = yes
		omni_check_and_buy_trinket = yes
		omni_check_and_buy_ring = yes
		omni_check_and_buy_emblem = yes
		# Last: one shot at the generated catalogue. See the effect for why this
		# is the branch that actually fills out a loadout.
		omni_check_and_buy_generated = yes
	}
}

omni_disloyal_character_equipment_check = {
	if = {
		limit = {
			loyalty <= 33
			NOT = { has_variable = omni_disloyal_equipment_checked }
			wealth >= 40
			exists = employer
			omni_character_has_local_market = yes
		}
		set_variable = omni_disloyal_equipment_checked
		omni_annual_character_equipment_check = yes
		trigger_event = { id = omni_item_autoequip.1 }
	}
	else_if = {
		limit = {
			loyalty > 33
			has_variable = omni_disloyal_equipment_checked
		}
		remove_variable = omni_disloyal_equipment_checked
	}
}

omni_build_country_market_average = {
	employer = {
		if = {
			limit = { has_variable = omni_market_avg_fresh }
			# Cached value from another character's check this year - skip rebuild.
		}
		else = {
			set_variable = { name = omni_market_avg_fresh days = 300 }
			omni_build_country_market_average_impl = yes
		}
	}
}

omni_build_country_market_average_impl = {
	# Runs in COUNTRY scope (called from the employer block above).
	set_variable = { name = omni_country_avg_market_quality value = 0 }

	# Evaluate buildings in the capital province (smooth O(1) check!)
	# The capital represents the nation's best market, preventing large empires from dragging the average down.
	capital_scope = {
			if = { limit = { num_of_commerce_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			if = { limit = { num_of_commerce_building >= 2 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			if = { limit = { num_of_commerce_building >= 3 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			if = { limit = { num_of_port_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
			if = { limit = { num_of_port_building >= 3 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			if = { limit = { num_of_workshop_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 8 } } }
			if = { limit = { num_of_foundry_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			if = { limit = { num_of_library_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
			if = { limit = { num_of_academy_building >= 1 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
			
			# Used Slots (Non-cumulative else_if)
			if = { limit = { num_of_used_building_slots >= 30 } prev = { change_variable = { name = omni_country_avg_market_quality add = 15 } } }
			else_if = { limit = { num_of_used_building_slots >= 16 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			else_if = { limit = { num_of_used_building_slots >= 8 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
			
			# Population (Non-cumulative else_if)
			if = { limit = { total_population >= 100 } prev = { change_variable = { name = omni_country_avg_market_quality add = 20 } } }
			else_if = { limit = { total_population >= 50 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			else_if = { limit = { total_population >= 20 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
			
			# Civilization Value (Non-cumulative else_if)
			if = { limit = { civilization_value >= 95 } prev = { change_variable = { name = omni_country_avg_market_quality add = 15 } } }
			else_if = { limit = { civilization_value >= 75 } prev = { change_variable = { name = omni_country_avg_market_quality add = 10 } } }
			else_if = { limit = { civilization_value >= 50 } prev = { change_variable = { name = omni_country_avg_market_quality add = 5 } } }
		}
}

# Check if should buy better weapon
omni_check_and_buy_weapon = {
	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 60 }
			# Only buy weapons that are meaningfully better
			# If you don't have a weapon, consider common ones
		}
		# High-quality markets: look for rare/unique weapons (320+ gold)
		# NOTE: only weapon-category items belong here. item_revenant_element
		# (special) and item_robe_of_wielkopan (armour) were removed - they ate
		# the weapon budget and blocked real weapon upgrades.
		if = {
			limit = {
				wealth >= 320
				NOT = { has_item = { item_ID = item_axe_of_thunder } }
			}
			omni_buy_character_item = { item_ID = item_axe_of_thunder price = 320 }
		}
	}

	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 45 }
		}
		# Mid-tier weapons for decent markets (130-160 gold)
		if = {
			limit = {
				wealth >= 140
				NOT = { has_item = { item_ID = item_masterwork_sword } }
				NOT = { has_item = { item_ID = item_masterwork_great_sword } }
			}
			random_list = {
				1 = { omni_buy_character_item = { item_ID = item_masterwork_great_sword price = 160 } }
				1 = { omni_buy_character_item = { item_ID = item_masterwork_sword price = 140 } }
			}
		}
	}

	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 30 }
		}
		# Common weapons for any market (50-70 gold)
		if = {
			limit = {
				wealth >= 60
				NOT = { has_item = { item_ID = item_sword } }
				NOT = { has_item = { item_ID = item_axe } }
				NOT = { has_item = { item_ID = item_mace } }
			}
			random_list = {
				1 = { omni_buy_character_item = { item_ID = item_sword price = 60 } }
				1 = { omni_buy_character_item = { item_ID = item_axe price = 65 } }
				1 = { omni_buy_character_item = { item_ID = item_long_axe price = 70 } }
				1 = { omni_buy_character_item = { item_ID = item_mace price = 60 } }
			}
		}
	}
}

# Check if should buy better trinket.
# NOTE: item_precious_gemstone and item_precious_jewelry used to live here, but
# both are RING category, not trinket - so the high tier of this check spent the
# budget filling a slot the trinket check could not use, and the trinket slot
# itself only ever had one reachable item. They now live in the ring check below.
omni_check_and_buy_trinket = {
	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 40 }
		}
		# Mid-tier trinkets (100-130 gold)
		if = {
			limit = {
				wealth >= 120
				NOT = { has_item = { item_ID = item_ring_of_confidence } }
			}
			omni_buy_character_item = { item_ID = item_ring_of_confidence price = 120 }
		}
	}

	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 25 }
		}
		# Common trinkets (70-85 gold). The books of skills are trinket category
		# and were previously unbuyable at any market quality, which is a large
		# part of why the trinket slot sat empty on most AI characters. The gate
		# is 25 rather than 40 because a scholarly keepsake should be reachable
		# in a modest town, unlike jewelry.
		if = {
			limit = {
				wealth >= 85
				NOT = { has_item = { item_ID = item_book_of_skills_military } }
				NOT = { has_item = { item_ID = item_book_of_skills_oratory } }
				NOT = { has_item = { item_ID = item_book_of_skills_fineese } }
				NOT = { has_item = { item_ID = item_book_of_skills_zeal } }
			}
			# Weighted toward the book that matches what the character is for.
			random_list = {
				3 = {
					trigger = { martial >= finesse charisma <= martial }
					omni_buy_character_item = { item_ID = item_book_of_skills_military price = 85 }
				}
				3 = {
					trigger = { charisma >= martial charisma >= finesse }
					omni_buy_character_item = { item_ID = item_book_of_skills_oratory price = 80 }
				}
				3 = {
					trigger = { finesse >= martial finesse >= charisma }
					omni_buy_character_item = { item_ID = item_book_of_skills_fineese price = 80 }
				}
				2 = { omni_buy_character_item = { item_ID = item_book_of_skills_zeal price = 75 } }
			}
		}
	}
}

# Check if should buy a ring. New - the ring slot had no purchase path at all.
omni_check_and_buy_ring = {
	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 70 }
		}
		# High-quality markets: fine jewelry (225-250 gold)
		if = {
			limit = {
				wealth >= 225
				NOT = { has_item = { item_ID = item_precious_gemstone } }
				NOT = { has_item = { item_ID = item_precious_jewelry } }
			}
			random_list = {
				1 = { omni_buy_character_item = { item_ID = item_precious_gemstone price = 250 } }
				1 = { omni_buy_character_item = { item_ID = item_precious_jewelry price = 225 } }
			}
		}
	}

	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 20 }
		}
		# Plain rings (45-50 gold). Cheap on purpose: a ring is the one thing
		# almost any employed character should be able to afford, and it is the
		# slot most visibly empty on AI characters today.
		if = {
			limit = {
				wealth >= 50
				NOT = { has_item = { item_ID = item_ring_1 } }
				NOT = { has_item = { item_ID = item_ring_2 } }
			}
			random_list = {
				1 = { omni_buy_character_item = { item_ID = item_ring_1 price = 50 } }
				1 = { omni_buy_character_item = { item_ID = item_ring_2 price = 45 } }
			}
		}
	}
}

# Check if should buy an emblem. New - the emblem slot had no purchase path.
# Gated on rank as well as gold: a crown or a royal emblem on a random courtier
# would read as nonsense, so only rulers and officeholders shop here.
omni_check_and_buy_emblem = {
	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 45 }
			OR = {
				is_ruler = yes
				is_governor = yes
				is_head_of_family = yes
			}
		}
		if = {
			limit = {
				wealth >= 130
				NOT = { has_item = { item_ID = item_crown } }
				NOT = { has_item = { item_ID = item_royal_emblem } }
			}
			random_list = {
				1 = {
					trigger = { is_ruler = yes }
					omni_buy_character_item = { item_ID = item_crown price = 150 }
				}
				2 = { omni_buy_character_item = { item_ID = item_royal_emblem price = 130 } }
			}
		}
	}
}

# One shot per year at the generated catalogue (item_gen_*).
#
# This is the branch that actually fills out a loadout. The hand-authored lists
# above total roughly a dozen purchasable items across ten slots; the generated
# catalogue is ~1,600 items spread across weapon, weapon_ranged, armour, shield,
# emblem and trinket, and until now had no way to reach a character except as
# adventure loot or a starting-kit roll. Giving it an economic path is what turns
# "the AI owns one sword" into a filled-out character.
#
# Cost: one random_in_global_list, then the shared equip path (a ten-way category
# chain plus the two-stage modifier dispatch), once per character per year, and
# only when the wealth gate passes. No iteration over the catalogue, no pulse.
omni_check_and_buy_generated = {
	if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 65 }
			wealth >= 200
		}
		add_gold = -200
		random_in_global_list = {
			variable = item_rare_rarity
			save_scope_as = omni_gen_item_pick
		}
		omni_give_and_equip_scoped_gen_item = yes
	}
	else_if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 40 }
			wealth >= 120
		}
		add_gold = -120
		random_in_global_list = {
			variable = item_uncommon_rarity
			save_scope_as = omni_gen_item_pick
		}
		omni_give_and_equip_scoped_gen_item = yes
	}
	else_if = {
		limit = {
			employer = { var:omni_country_avg_market_quality >= 15 }
			wealth >= 60
		}
		add_gold = -60
		random_in_global_list = {
			variable = item_common_rarity
			save_scope_as = omni_gen_item_pick
		}
		omni_give_and_equip_scoped_gen_item = yes
	}
}
