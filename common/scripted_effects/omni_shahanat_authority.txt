# Country scope. Only the incumbent office receives these modifiers.
omni_hc_clear_authority_modifiers_effect = {
    remove_country_modifier = omni_hc_authority_weak
    remove_country_modifier = omni_hc_authority_fragile
    remove_country_modifier = omni_hc_authority_established
    remove_country_modifier = omni_hc_authority_respected
    remove_country_modifier = omni_hc_authority_supreme
}

omni_hc_sync_authority_effect = {
    if = {
        limit = { omni_hc_shannah_trigger = yes }
        if = {
            limit = { omni_hc_authority >= 90 }
            if = {
                limit = { NOT = { has_country_modifier = omni_hc_authority_supreme } }
                omni_hc_clear_authority_modifiers_effect = yes
                add_country_modifier = { name = omni_hc_authority_supreme duration = -1 }
            }
        }
        else_if = {
            limit = { omni_hc_authority >= 75 }
            if = {
                limit = { NOT = { has_country_modifier = omni_hc_authority_respected } }
                omni_hc_clear_authority_modifiers_effect = yes
                add_country_modifier = { name = omni_hc_authority_respected duration = -1 }
            }
        }
        else_if = {
            limit = { omni_hc_authority >= 50 }
            if = {
                limit = { NOT = { has_country_modifier = omni_hc_authority_established } }
                omni_hc_clear_authority_modifiers_effect = yes
                add_country_modifier = { name = omni_hc_authority_established duration = -1 }
            }
        }
        else_if = {
            limit = { omni_hc_authority >= 25 }
            if = {
                limit = { NOT = { has_country_modifier = omni_hc_authority_fragile } }
                omni_hc_clear_authority_modifiers_effect = yes
                add_country_modifier = { name = omni_hc_authority_fragile duration = -1 }
            }
        }
        else_if = {
            limit = { omni_hc_authority >= 0 }
            if = {
                limit = { NOT = { has_country_modifier = omni_hc_authority_weak } }
                omni_hc_clear_authority_modifiers_effect = yes
                add_country_modifier = { name = omni_hc_authority_weak duration = -1 }
            }
        }
    }
    else = { omni_hc_clear_authority_modifiers_effect = yes }
}
