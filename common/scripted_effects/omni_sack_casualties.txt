# Territory scope. Resolve one casualty, favoring mages when present.
omni_sack_first_casualty = {
	random_list = {
		$MAGE_WEIGHT$ = {
			trigger = { num_of_mages >= 1 }
			random_pops_in_province = {
				limit = { pop_type = mages }
				kill_pop = yes
			}
		}
		25 = {
			random_pops_in_province = {
				kill_pop = yes
			}
		}
	}
}
