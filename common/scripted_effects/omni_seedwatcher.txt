omni_seedwatcher_install = {
	if = {
		limit = { exists = current_ruler }
		current_ruler = {
			if = {
				limit = { NOT = { has_trait = seedwatcher_trait } }
				force_add_trait = seedwatcher_trait
				recalculate_character_magic_vars = yes
			}
			if = {
				limit = {
					NOR = {
						has_variable = druid_t2_unlocked_var
						has_variable = druid_t3_unlocked_var
						has_variable = druid_t4_unlocked_var
						has_variable = druid_mastered_var
						has_trait = druid_trait
					}
				}
				# The academy tier-one milestone unlocks training for tier two.
				if = {
					limit = { NOT = { omni_character_is_eligible_caster = yes } }
					add_trait = mage_trait
				}
				set_variable = druid_t2_unlocked_var
				omni_sb_grant_spell_points = { FAMILY = amten AMOUNT = 1 }
				omni_gain_amten_affinity = yes
			}
		}
	}
}
