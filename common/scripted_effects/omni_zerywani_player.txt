z2_player_claim_effect = {
    save_scope_as = z2_player_host
    set_global_variable = { name = z2_player_claim value = scope:z2_player_host }
    add_treasury = z2_project_gold_payment
    add_political_influence = z2_project_influence_payment
    remove_variable = z2_player_offer_pending
    remove_variable = z2_player_generation_offered
    remove_variable = z2_player_claim_deadline
    remove_variable = z2_player_remounts
    remove_variable = z2_chief_menu_open
    remove_variable = z2_city_menu_open
    remove_variable = z2_negotiation_pending
    set_variable = { name = z2_player_project }
    set_variable = { name = z2_player_support value = 0 }
    if = { limit = { has_variable = zerywani_first_veterans } change_variable = { name = z2_player_support add = 1 } }
    trigger_event = { id = zerywani_second.101 days = 1 }
}
z2_player_release_claim_effect = {
    save_scope_as = z2_player_host
    every_country = {
        limit = { exists = var:z2_pledged_to var:z2_pledged_to = scope:z2_player_host }
        remove_variable = z2_pledged_to
        remove_variable = z2_privileges_promised
    }
    if = {
        limit = { exists = global_var:z2_player_claim global_var:z2_player_claim = this }
        remove_global_variable = z2_player_claim
    }
    remove_variable = z2_player_project
    remove_variable = z2_player_offer_pending
    set_variable = { name = z2_player_declined }
    set_variable = { name = z2_player_claim_cooldown days = 1825 }
    if = { limit = { has_variable = z2_preparing } z2_cancel_preparation_effect = yes }
}

# An optional contribution gives human tribes an actual route to veteran status,
# without declaring a war or forcing them into the first expedition's diplomacy.
z2_player_support_first_effect = {
    save_scope_as = z2_volunteer_country
    ordered_country = {
        limit = { has_variable = is_zerywani_horde_leader war = yes has_land = yes }
        order_by = num_of_cities
        position = 0
        add_treasury = 100
        add_manpower = 10
        scope:z2_volunteer_country = {
            add_treasury = -100
            add_manpower = -10
            set_variable = { name = zerywani_first_veterans }
            set_variable = { name = is_zerywani_horde_participant }
            current_ruler = { set_variable = { name = zerywani_first_generation } }
            every_owned_province = { set_variable = { name = zerywani_first_home_ground } }
            add_country_modifier = { name = zerywani_horde_participant_modifier duration = 7300 }
        }
    }
}
