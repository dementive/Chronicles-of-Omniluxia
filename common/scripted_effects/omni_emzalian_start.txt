# Global, run once from on_game_initialized. Swamp (borderlander) realms open the game
# with their ruler already practising the racial alligator summoning, which also makes
# the Spellbook seal appear for them (omni_sb_listable_mage reads folk_unlocked_var).
omni_emzalian_start_alligator_casters = {
    every_country = {
        limit = {
            country_culture_group = borderlander
            exists = current_ruler
        }
        current_ruler = {
            omni_folk_initiate_practitioner = yes
            omni_learn_spell = { SPELL = summon_emzalian_alligators }
        }
    }
}
