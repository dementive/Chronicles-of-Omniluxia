omni_schedule_bastard_adulthood_effect = {
	set_variable = { name = omni_royal_bastard_father value = scope:omni_bastard_father }
	if = {
		limit = { NOT = { has_variable = omni_bastard_adulthood_scheduled } }
		set_variable = { name = omni_bastard_adulthood_scheduled days = 7000 }
		trigger_event = { id = omni_seduction.30 days = { 5840 6570 } }
	}
}

omni_legitimize_bastard_effect = {
	remove_character_modifier = omni_disputed_parentage
	remove_character_modifier = omni_recognized_royal_bastard
	set_family = scope:omni_bastard_father.family
	add_character_modifier = { name = omni_legitimized_bastard duration = -1 }
	omni_schedule_bastard_adulthood_effect = yes
}

omni_bastard_legitimization_reaction_effect = {
	every_successor = {
		limit = { NOT = { this = scope:omni_bastard_child } }
		add_loyalty = loyalty_omni_bastard_legitimized
	}
	if = {
		limit = {
			exists = current_ruler.spouse
			NOT = { current_ruler.spouse = scope:omni_bastard_mother }
		}
		current_ruler.spouse = { add_loyalty = loyalty_omni_betrayed_spouse }
	}
}

omni_betrayed_spouse_effect = {
	if = {
		limit = { has_trait = forgiving }
		add_loyalty = loyalty_omni_betrayed_spouse_forgiving
	}
	else = {
		add_loyalty = loyalty_omni_betrayed_spouse
		add_character_modifier = { name = omni_humiliated_by_affair duration = 1825 }
		if = {
			limit = { is_friend = scope:omni_betrayal_culprit }
			remove_friend = scope:omni_betrayal_culprit
		}
		if = {
			limit = { can_get_rivals = yes scope:omni_betrayal_culprit = { can_get_rivals = yes } }
			add_rival = scope:omni_betrayal_culprit
		}
	}
}

omni_ai_seduce_character_effect = {
	# This scans the whole character list (every_character is a global iterator),
	# so it only needs to run ONCE per year, not once per AI country. It is called
	# from omni_perf.3 (per-AI-country); the global-variable lock below lets the
	# first AI country to reach it each year run the scan and suppresses the rest
	# for 300 days, removing the old O(AI-countries x characters) redundancy.
	if = {
	limit = { NOT = { has_global_variable = omni_ai_seduction_scan_done } }
	set_global_variable = { name = omni_ai_seduction_scan_done days = 300 }
	every_character = {
		limit = {
			# AI-controlled seducers only - never auto-initiate seductions from a
			# human player's court.
			employer = { is_ai = yes }
			is_alive = yes
			is_adult = yes
			prisoner = no
			NOT = { has_variable = omni_seduction_active }
			NOT = { has_variable = omni_seduction_cooldown }
			NOT = { has_character_modifier = indisposed_charmod }
			exists = employer
			OR = {
				has_trait = lustful
				AND = {
					is_married = yes
					spouse = {
						NOT = { has_variable = omni_recently_rejected_seducer }
					}
				}
			}
		}

		# Basic roll to prevent it happening constantly. E.g. 5% chance a year.
		# (This roll used to sit in the limit above as 'random = { percent = 5 }',
		# but random is an EFFECT, not a trigger - the engine logged
		# "Unknown trigger type: random" and the whole scan misparsed. Moved into
		# the body as the vanilla-style random = { chance = 5 } effect.)
		random = {
			chance = 5

		# Now try to find a target.
		# (random_character is a country-scope list; running it directly in
		# character scope logged "Inconsistent effect scopes (character vs.
		# country)". Searching from the seducer's employer matches the comment
		# below: own court, or their own spouse.)
		save_scope_as = omni_ai_seducer_candidate
		employer = {
			random_character = {
				limit = {
					is_alive = yes
					is_adult = yes
					prisoner = no
					is_mercenary = no
					NOT = { employer = { tag = REB } }
					NOT = { employer = { tag = BAR } }
					NOT = { employer = { tag = PIR } }
					NOT = { this = scope:omni_ai_seducer_candidate }
					NOT = { has_variable = omni_recently_rejected_seducer }

					# The interaction allowed_trigger allows own court or spouse of current ruler or allied/subject ruler's spouse.
					# To keep it performant and simple for AI: just allow in own court, or their own spouse.
					OR = {
						is_spouse_of = scope:omni_ai_seducer_candidate
						AND = {
							scope:omni_ai_seducer_candidate = { has_trait = lustful }
							NOT = { is_close_relative = scope:omni_ai_seducer_candidate }
							employer = scope:omni_ai_seducer_candidate.employer
						}
					}
				}
				# Found a target!
				save_scope_as = omni_ai_seduction_target
			}
		}
		
		if = {
			limit = {
				exists = scope:omni_ai_seduction_target
			}
			# Execute the effect as if they did it.
			save_scope_as = omni_seducer
			set_variable = { name = omni_seduction_active days = 400 }
			if = {
				limit = { is_married = yes }
				spouse = { save_scope_as = omni_seducer_spouse }
			}
			
			scope:omni_ai_seduction_target = {
				save_scope_as = omni_seduction_target
				if = {
					limit = { is_married = yes }
					spouse = { save_scope_as = omni_target_spouse }
				}
			}
			
			# omni_seduction.1 is a country_event: fire it on the seducer's
			# employer, not on the character scope we are standing in.
			employer = {
				trigger_event = { id = omni_seduction.1 days = { 1 3 } }
			}
		}
		} # closes random = { chance = 5
	} # closes every_character
	} # closes once-per-year global gate
}
