purge_court_open_hearing_effect = {
	employer = {
		save_scope_as = purge_country
		random_character = {
			limit = { purge_court_candidate_trigger = { RULER = root } }
			weight = {
				modifier = { factor = 3 loyalty < 33 }
				modifier = { factor = 2 power_base >= 5 }
			}
			save_scope_as = purge_target
		}
	}
	if = {
		limit = { exists = scope:purge_target }
		trigger_event = scheme_purge_court.1
	}
}

purge_court_cleanup_effect = {
	hidden_effect = {
		if = {
			limit = { has_variable = ongoing_scheme_target }
			remove_variable = ongoing_scheme_target
		}
		if = {
			limit = { has_character_modifier = ongoing_scheme }
			remove_character_modifier = ongoing_scheme
		}
		if = {
			limit = { has_variable = purge_court_pending_target }
			remove_variable = purge_court_pending_target
		}
		if = {
			limit = { has_variable = purge_court_opened_hearing }
			remove_variable = purge_court_opened_hearing
		}
		if = {
			limit = { has_variable = purge_court_resolved_hearing }
			remove_variable = purge_court_resolved_hearing
		}
		if = {
			limit = { has_variable = purge_court_removed_courtier }
			remove_variable = purge_court_removed_courtier
		}
		if = {
			limit = { has_variable_list = purge_court_heard }
			clear_variable_list = purge_court_heard
		}
	}
}

purge_court_record_hearing_effect = {
	hidden_effect = {
		add_to_variable_list = { name = purge_court_heard target = scope:purge_target }
		set_variable = purge_court_resolved_hearing
	}
}

purge_court_record_removal_effect = {
	hidden_effect = {
		if = {
			limit = {
				trigger_if = {
					limit = { exists = scope:purge_target }
					scope:purge_target = {
						OR = {
							is_alive = no
							prisoner = yes
							NOT = { employer = scope:purge_country }
						}
					}
				}
				trigger_else = { always = yes }
			}
			set_variable = purge_court_removed_courtier
			scope:purge_country = {
				set_variable = { name = purge_court_recent_removal days = 1095 }
			}
		}
	}
}

purge_court_abort_effect = {
	if = {
		limit = { has_variable = purge_court_resolved_hearing }
		set_variable = { name = purge_court_cooldown days = 1095 }
	}
	purge_court_cleanup_effect = yes
}
