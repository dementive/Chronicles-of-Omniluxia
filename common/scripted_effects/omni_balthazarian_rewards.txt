# Once-only small awards for milestones without a direct personal/economic reward.
ob_milestone_reward_5 = {
    if = { limit = { NOT = { has_variable = ob_milestone_rewarded_5 } }
        set_variable = { name = ob_milestone_rewarded_5 }
        add_political_influence = 5
    }
}
