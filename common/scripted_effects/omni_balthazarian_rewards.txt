# Country scope. The same treasure returns to the vault whenever its bearer relinquishes it.
ob_store_bastion_shield = {
    create_country_treasure = {
        key = ob_bastion_shield
        icon = "cr_treasure_shield_of_diomedes_01"
        modifier = { local_defensive = 0.10 }
    }
}

# Country scope. Once per campaign, including campaigns that finished before the artifact was added.
ob_award_bastion_shield = {
    if = {
        limit = { NOT = { has_global_variable = ob_bastion_shield_awarded } }
        hidden_effect = {
            set_global_variable = { name = ob_bastion_shield_awarded value = 1 }
            if = {
                limit = { NOT = { is_target_in_global_variable_list = { name = item_shield target = flag:item_artifact_ob_bastion_shield } } }
                assing_item_to_category = { item_category = shield item_ID = item_artifact_ob_bastion_shield }
            }
        }
        ob_store_bastion_shield = yes
    }
}

# Character scope.
ob_return_bastion_shield = {
    remove_character_item = { item_ID = item_artifact_ob_bastion_shield }
    country = {
        remove_country_modifier = item_artifact_ob_bastion_shield_army
        ob_store_bastion_shield = yes
    }
}

# Country scope. The covenant awards a single sword per campaign.
ob_award_lord_protector_sword = {
    if = {
        limit = { NOT = { has_global_variable = ob_lord_protector_sword_awarded } }
        hidden_effect = {
            set_global_variable = { name = ob_lord_protector_sword_awarded value = 1 }
            # Register the weapon when awarded in an existing campaign.
            if = {
                limit = {
                    NOT = { is_target_in_global_variable_list = { name = item_weapon target = flag:item_artifact_ob_lord_protector_sword } }
                }
                assing_item_to_category = { item_category = weapon item_ID = item_artifact_ob_lord_protector_sword }
            }
        }
        ob_store_lord_protector_sword = yes
    }
}

# Character scope.
ob_return_lord_protector_sword = {
    remove_character_item = { item_ID = item_artifact_ob_lord_protector_sword }
    country = {
        remove_country_modifier = item_artifact_ob_lord_protector_sword_army
        ob_store_lord_protector_sword = yes
    }
}

ob_finish_order_finale = {
    if = {
        limit = { NOT = { has_variable = ob_scene_24 } }
        hidden_effect = {
            set_variable = { name = ob_scene_24 }
            set_variable = { name = ob_tree_finished }
        }
        add_country_modifier = { name = ob_final duration = -1 }
        add_stability = 10
        show_as_tooltip = {
            create_country_treasure = {
                key = "ob_bastion_shield"
                icon = "cr_treasure_shield_of_diomedes_01"
                modifier = { local_defensive = 0.10 }
            }
        }
    }
}

        set_variable = { name = ob_milestone_rewarded_13 }
        current_ruler = { add_popularity = 5 }
    }
}
