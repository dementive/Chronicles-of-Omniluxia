# SUBJECT scope. Mirrors the country-level spouse variables onto the two
# characters, so the union can react to their DEATHS wherever they live
# (see omni_dynastic_union_on_character_death). Whether the engine moves a
# married ruler to the other court or not, the union now follows the spouse,
# not whoever happens to sit on the junior throne.
omni_dynastic_union_bind_spouses_effect = {
	save_scope_as = omni_du_bind_subject
	if = {
		limit = {
			has_variable = dynastic_union_junior_spouse
			var:dynastic_union_junior_spouse = { is_alive = yes }
		}
		var:dynastic_union_junior_spouse = {
			set_variable = {
				name = omni_du_junior_crown
				value = scope:omni_du_bind_subject
			}
		}
	}
	if = {
		limit = {
			exists = overlord
			has_variable = dynastic_union_senior_spouse
			var:dynastic_union_senior_spouse = { is_alive = yes }
		}
		overlord = { save_scope_as = omni_du_bind_overlord }
		var:dynastic_union_senior_spouse = {
			set_variable = {
				name = omni_du_senior_crown
				value = scope:omni_du_bind_overlord
			}
		}
	}
	set_variable = omni_du_spouses_bound
}

# SUBJECT scope, with scope:actor = the annexing senior crown.
# on_diplo_annexation_event_effect keeps only the junior's four strongest
# families and kills everyone else "of obscurity". That used to include the
# senior ruler's own spouse and the children of the union - the shared heir
# included. Move them to the senior court first.
omni_dynastic_union_protect_kin_effect = {
	if = {
		limit = { has_variable = dynastic_union_senior_spouse }
		var:dynastic_union_senior_spouse = { save_scope_as = omni_du_kin_senior_spouse }
	}
	if = {
		limit = { has_variable = dynastic_union_junior_spouse }
		var:dynastic_union_junior_spouse = { save_scope_as = omni_du_kin_junior_spouse }
	}
	scope:actor = {
		if = {
			limit = { exists = current_ruler }
			current_ruler = { save_scope_as = omni_du_kin_senior_ruler }
		}
	}
	every_character = {
		limit = {
			is_alive = yes
			OR = {
				AND = {
					exists = scope:omni_du_kin_senior_ruler
					OR = {
						is_child_of = scope:omni_du_kin_senior_ruler
						AND = {
							scope:omni_du_kin_senior_ruler = { exists = spouse }
							this = scope:omni_du_kin_senior_ruler.spouse
						}
					}
				}
				AND = {
					exists = scope:omni_du_kin_senior_spouse
					OR = {
						this = scope:omni_du_kin_senior_spouse
						is_child_of = scope:omni_du_kin_senior_spouse
					}
				}
				AND = {
					exists = scope:omni_du_kin_junior_spouse
					this = scope:omni_du_kin_junior_spouse
				}
			}
		}
		remove_all_positions = yes
		move_country = scope:actor
		set_home_country = scope:actor
	}
}

# SENIOR scope, with scope:dynastic_union_subject set.
omni_dynastic_union_integrate_subject_effect = {
	save_scope_as = actor
	scope:dynastic_union_subject = {
		omni_dynastic_union_protect_kin_effect = yes
		every_country_treasure = {
			add_to_list = annexed_treasures
		}
		# The diplomatic-annexation effect below only handles characters,
		# families, and treasures. Transfer the land explicitly so the junior
		# crown is actually absorbed and ceases to exist as a landed country.
		on_diplo_annexation_event_effect = yes
		every_owned_province = {
			set_owned_by = scope:actor
		}
	}
}

# SENIOR scope, with scope:dynastic_union_subject set.
# A player's junior crown is never annexed by someone else's decision: it is
# asked (omni_dynastic_union.13) and may refuse. AI junior crowns are integrated
# directly, as before.
omni_dynastic_union_request_or_integrate_effect = {
	save_scope_as = dynastic_senior_country
	if = {
		limit = { scope:dynastic_union_subject = { is_ai = yes } }
		omni_dynastic_union_integrate_subject_effect = yes
	}
	else = {
		scope:dynastic_union_subject = {
			trigger_event = { id = omni_dynastic_union.13 }
		}
	}
}

# SUBJECT scope. Re-points the union at whoever now holds each throne, so the
# compact always belongs to the two sitting rulers. Every way out of a
# succession event (.4 / .5 / .6 / .7 and the wars they lead to) must end here,
# or the union stays tied to a retired or deposed spouse, and that spouse's
# eventual death would replay a succession that was already settled.
omni_dynastic_union_rebind_effect = {
	if = {
		limit = { exists = current_ruler }
		set_variable = {
			name = dynastic_union_junior_spouse
			value = current_ruler
		}
	}
	if = {
		limit = {
			exists = overlord
			overlord = { exists = current_ruler }
		}
		set_variable = {
			name = dynastic_union_senior_country
			value = overlord
		}
		set_variable = {
			name = dynastic_union_senior_spouse
			value = overlord.current_ruler
		}
	}
	omni_dynastic_union_bind_spouses_effect = yes
}

# Any scope with the proposal scopes saved. Clears the "proposal in flight" lock
# on both courts. It used to be left to expire after 120 days on every outcome.
omni_dynastic_union_clear_proposal_effect = {
	if = {
		limit = { exists = scope:dynastic_senior_country }
		scope:dynastic_senior_country = {
			if = { limit = { has_variable = dynastic_marriage_proposal_ongoing } remove_variable = dynastic_marriage_proposal_ongoing }
		}
	}
	if = {
		limit = { exists = scope:dynastic_junior_country }
		scope:dynastic_junior_country = {
			if = { limit = { has_variable = dynastic_marriage_proposal_ongoing } remove_variable = dynastic_marriage_proposal_ongoing }
			if = { limit = { has_variable = omni_du_proposal_delivered } remove_variable = omni_du_proposal_delivered }
		}
	}
}

# SUBJECT scope. The compact lapses without blame - the senior crown stopped

# SUBJECT scope. The junior crown fights for its independence.
#
# FIX: this used to release the subject FIRST and then declare the war. But
# independence_wargoal has allow = { is_subject = yes }, so once released the
# war could not be declared - the junior simply walked away free. The war is now
# declared while the junior is still a subject (the pattern used by
# me_subject_independence.63 and me_eptelon_gen.61). Winning the war ends the
# subject relationship, which runs on_disable (colour + variable cleanup); if the
# senior wins, the union - and its variables - carry on.
omni_dynastic_union_independence_war_effect = {
	save_scope_as = dynastic_union_subject
	overlord = {
		save_scope_as = dynastic_senior_country
		add_opinion = {
			modifier = opinion_dynastic_union_disputed
			target = scope:dynastic_union_subject
		}
	}
	add_opinion = {
		modifier = opinion_dynastic_union_disputed
		target = scope:dynastic_senior_country
	}
	# If the senior crown wins, the union carries on under the sitting rulers.
	omni_dynastic_union_rebind_effect = yes
	declare_war_with_wargoal = {
		war_goal = independence_wargoal
		province = capital_scope
		target = scope:dynastic_senior_country
	}
}

# SUBJECT scope. A junior crown stronger than its new senior presses its own
# dynastic claim: an independence war fought with a claim on the senior capital
# (for any later conquest) and a modifier that makes the peace cheaper.
#
# FIX: this used to be the senior court's choice, released the subject before
# declaring (so the war never started, as above), and used the independence war
# goal against the SENIOR's capital. The junior now chooses (.12) and fights
# while it is still a subject.
omni_dynastic_union_reverse_claim_war_effect = {
	save_scope_as = dynastic_union_subject
	overlord = {
		save_scope_as = dynastic_senior_country
		add_opinion = {
			modifier = opinion_dynastic_union_disputed
			target = scope:dynastic_union_subject
		}
	}
	add_opinion = {
		modifier = opinion_dynastic_union_disputed
		target = scope:dynastic_senior_country
	}
	scope:dynastic_senior_country.capital_scope = {
		add_claim = scope:dynastic_union_subject
	}
	add_country_modifier = {
		name = dynastic_union_reverse_claim
		duration = 1825
	}
	# If the senior crown wins, the union carries on under the sitting rulers.
	omni_dynastic_union_rebind_effect = yes
	declare_war_with_wargoal = {
		war_goal = independence_wargoal
		province = capital_scope
		target = scope:dynastic_senior_country
	}
}
