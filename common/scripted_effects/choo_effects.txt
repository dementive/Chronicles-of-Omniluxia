disable_racial_assimilation_macro = {
	if = {
		limit = {
			NOR = {
				$COUNTRY_LIMIT$
			}
		}
		every_country_culture = {
			limit = {
				NOT = {
					has_country_culture_modifier = disable_cultural_assimilation
				}
				OR = {
					$CULTURE_LIMIT$
				}
			}
			add_country_culture_modifier = {
				name = disable_cultural_assimilation
				duration = -1
			}
		}
	}
}
