# Country scope; the battle caster has saved the land battle province.
omni_raise_emzalian_alligators = {
    set_variable = { name = omni_cd_summon_emzalian_alligators days = 365 }
    create_unit = {
        navy = no
        name = "Emzalian Alligators"
        location = scope:emzalian_summon_site
        hidden_effect = {
            while = { count = 3 add_subunit = omni_emzalian_alligators }
            add_food = 800
            add_morale = 5
        }
    }
}
