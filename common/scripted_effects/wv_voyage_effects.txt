# Stellarix Voyage to World's End: scripted effects. All run in the voyage country's scope (ROOT).

# ---------------------------------------------------------------- meters
wv_spend_stores = {
	change_variable = { name = wv_stores subtract = $AMOUNT$ }
	while = {
		limit = { var:wv_stores < 0 }
		change_variable = { name = wv_stores add = 1 }
		if = {
			limit = { var:wv_shortages < 1 }
			change_variable = { name = wv_company subtract = 8 }
		}
		else = {
			change_variable = { name = wv_company subtract = 10 }
		}
		change_variable = { name = wv_shortages add = 1 }
	}
	if = { limit = { var:wv_company < 0 } set_variable = { name = wv_company value = 0 } }
}

wv_passage = {
	wv_spend_stores = { AMOUNT = $AMOUNT$ }
	if = {
		limit = { var:wv_company < 50 }
		wv_spend_stores = { AMOUNT = 1 }
	}
}

wv_add_stores = {
	change_variable = { name = wv_stores add = $AMOUNT$ }
	if = { limit = { var:wv_stores > 18 } set_variable = { name = wv_stores value = 18 } }
}

wv_buy_stores = {
	change_variable = { name = wv_purse subtract = $COST$ }
	wv_add_stores = { AMOUNT = $AMOUNT$ }
}

wv_add_company = {
	change_variable = { name = wv_company add = $AMOUNT$ }
	if = { limit = { var:wv_company > 100 } set_variable = { name = wv_company value = 100 } }
}

wv_lose_company = {
	change_variable = { name = wv_company subtract = $AMOUNT$ }
	if = { limit = { var:wv_company < 0 } set_variable = { name = wv_company value = 0 } }
}

wv_add_hull = {
	change_variable = { name = wv_hull add = $AMOUNT$ }
	if = { limit = { var:wv_hull > 100 } set_variable = { name = wv_hull value = 100 } }
}

# Laden ships (full chest) take half again as much. The pilot's soundings take 15 off the first big blow.
wv_hull_damage = {
	set_variable = { name = wv_tmp value = $AMOUNT$ }
	if = {
		limit = { var:wv_chest = 2 }
		change_variable = { name = wv_tmp multiply = 1.5 }
	}
	if = {
		limit = {
			has_variable = wv_envoy_pilot
			NOT = { has_variable = wv_soundings_used }
			var:wv_tmp >= 12
		}
		change_variable = { name = wv_tmp subtract = 15 }
		set_variable = wv_soundings_used
		if = { limit = { var:wv_tmp < 0 } set_variable = { name = wv_tmp value = 0 } }
	}
	change_variable = { name = wv_hull subtract = var:wv_tmp }
	round_variable = { name = wv_hull nearest = 1 }
	if = { limit = { var:wv_hull < 0 } set_variable = { name = wv_hull value = 0 } }
}

wv_pay = {
	change_variable = { name = wv_purse subtract = $AMOUNT$ }
	if = { limit = { var:wv_purse < 0 } set_variable = { name = wv_purse value = 0 } }
}

wv_earn = {
	change_variable = { name = wv_purse add = $AMOUNT$ }
}

# Paid district and audience options. Charias halves them; a Gellazani charm (one use) pays for one of 40 or more.
wv_pay_district = {
	if = {
		limit = { has_variable = wv_charm }
		remove_variable = wv_charm
	}
	else_if = {
		limit = { has_variable = wv_c_appr }
		wv_pay = { AMOUNT = $HALF$ }
	}
	else = {
		wv_pay = { AMOUNT = $AMOUNT$ }
	}
}
wv_pay_district_small = {
	if = {
		limit = { has_variable = wv_c_appr }
		wv_pay = { AMOUNT = $HALF$ }
	}
	else = {
		wv_pay = { AMOUNT = $AMOUNT$ }
	}
}

# ---------------------------------------------------------------- ports

	set_variable = { name = wv_districts_here value = 0 }
	if = { limit = { has_variable = wv_c_vet } change_variable = { name = wv_ports_vet add = 1 } }
	if = { limit = { has_variable = wv_c_diver } change_variable = { name = wv_ports_diver add = 1 } }
	if = { limit = { has_variable = wv_c_pupil } change_variable = { name = wv_ports_pupil add = 1 } }
	if = { limit = { has_variable = wv_c_appr } change_variable = { name = wv_ports_appr add = 1 } }
	if = { limit = { has_variable = wv_c_gamb } change_variable = { name = wv_ports_gamb add = 1 } }

	if = { limit = { has_variable = wv_c_hunter } change_variable = { name = wv_ports_hunter add = 1 } }
	if = { limit = { has_variable = wv_c_prize } change_variable = { name = wv_ports_prize add = 1 } }

wv_district_visit = {
	if = {
		limit = { var:wv_districts_here >= 2 }
		wv_spend_stores = { AMOUNT = 1 }
	}
	change_variable = { name = wv_districts_here add = 1 }
	change_variable = { name = wv_wonders add = 1 }
}

# Called before every leg. Sends the fleet to $EVENT$, or to the wreck and the abandonment endings.
wv_next = {
	if = {
		limit = { has_variable = wv_c_diver var:wv_hull < 40 NOT = { has_variable = wv_diver_paid } }
		if = {
			limit = { var:wv_purse >= 20 }
			wv_pay = { AMOUNT = 20 }
			set_variable = wv_diver_paid
		}
		else = {
			wv_lose_diver = yes
		}
	}
	if = {
		limit = { var:wv_company <= 0 has_variable = wv_c_vet NOT = { has_variable = wv_vet_save_used } }
		set_variable = { name = wv_company value = 10 }
		set_variable = wv_vet_save_used
	}
	if = {
		limit = { var:wv_hull <= 0 }
		trigger_event = { id = wv.840 days = 3 }
	}
	else_if = {
		limit = { var:wv_company <= 0 }
		trigger_event = { id = wv.841 days = 3 }
	}
	else = {
		trigger_event = { id = $EVENT$ days = { $MIN$ $MAX$ } }
	}
}

# Sets wv_roll_hit if wv_roll is at or under the variable $TARGET$ (the engine cannot compare two variables directly).
wv_roll_against = {
	set_variable = { name = wv_tmp value = var:wv_roll }
	change_variable = { name = wv_tmp subtract = var:$TARGET$ }
	if = {
		limit = { var:wv_tmp <= 0 }
		set_variable = wv_roll_hit
	}
	else = {
		remove_variable = wv_roll_hit
	}
}

# Chance, in percent, that the ship fails at the last passage.
wv_compute_risk = {
	set_variable = { name = wv_risk value = 100 }
	change_variable = { name = wv_risk subtract = var:wv_hull }
	if = { limit = { var:wv_risk < 10 } set_variable = { name = wv_risk value = 10 } }
	if = { limit = { var:wv_risk > 65 } set_variable = { name = wv_risk value = 65 } }
	set_variable = { name = wv_tmp value = var:wv_bulky }
	change_variable = { name = wv_tmp multiply = 5 }
	change_variable = { name = wv_risk add = var:wv_tmp }
	if = { limit = { var:wv_risk > 80 } set_variable = { name = wv_risk value = 80 } }
	if = {
		limit = { has_variable = wv_towline }
		change_variable = { name = wv_risk multiply = 0.6 }
	}
	if = {
		limit = { NOT = { var:wv_chest = 2 } }
		change_variable = { name = wv_risk multiply = 0.25 }
	}
	round_variable = { name = wv_risk nearest = 1 }
}

# Chance, in percent, that the boats save the reduced chest.
wv_compute_rescue = {
	set_variable = { name = wv_rescue value = var:wv_company }
	change_variable = { name = wv_rescue multiply = 0.55 }
	if = { limit = { has_variable = wv_c_vet } change_variable = { name = wv_rescue add = 20 } }
	if = { limit = { var:wv_rescue > 95 } set_variable = { name = wv_rescue value = 95 } }
	round_variable = { name = wv_rescue nearest = 1 }
}

}

		limit = { has_variable = wv_c_vet }
		remove_variable = wv_c_vet

		}
	}
