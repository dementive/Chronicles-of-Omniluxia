engage_in_piracy_apply_privateering_effect = {
	save_scope_as = engage_in_piracy_fleet
	scope:engage_in_piracy_fleet.unit_owner = {
		save_scope_as = engage_in_piracy_raider_country
	}
	if = {
		limit = {
			unit_location = {
				any_neighbor_province = {
					engage_in_piracy_target_province_trigger = yes
					exists = scope:engage_in_piracy_fleet.unit_owner
					NOT = { owner = scope:engage_in_piracy_fleet.unit_owner }
					owner = {
						NOT = { alliance_with = scope:engage_in_piracy_fleet.unit_owner }
						NOT = { is_subject_of = scope:engage_in_piracy_fleet.unit_owner }
					}
				}
			}
		}
		custom_tooltip = engage_in_piracy_privateers_nearby_ports_tt
		unit_location = {
			ordered_neighbor_province = {
				limit = {
					engage_in_piracy_target_province_trigger = yes
					exists = scope:engage_in_piracy_fleet.unit_owner
					NOT = { owner = scope:engage_in_piracy_fleet.unit_owner }
					owner = {
						NOT = { alliance_with = scope:engage_in_piracy_fleet.unit_owner }
						NOT = { is_subject_of = scope:engage_in_piracy_fleet.unit_owner }
					}
				}
				order_by = total_population
				max = 2
				hidden_effect = {
					add_province_modifier = {
						name = engage_in_piracy_raided_port_modifier
						duration = 35
					}
					save_scope_as = engage_in_piracy_raided_port
					owner = {
						add_opinion = {
							modifier = opinion_engage_in_piracy_raided_us
							target = scope:engage_in_piracy_raider_country
						}
						if = {
							limit = {
								NOT = { has_variable = engage_in_piracy_raid_event_cooldown }
								NOT = { war_with = scope:engage_in_piracy_raider_country }
							}
							set_variable = {
								name = engage_in_piracy_raid_event_cooldown
								days = 180
							}
							trigger_event = {
								id = omni_piracy.1
							}
						}
					}
				}
			}
		}
	}
	else = {
		custom_tooltip = engage_in_piracy_no_valid_ports_tt
	}
}

engage_in_piracy_refresh_country_income_effect = {
	engage_in_piracy_clear_country_income_effect = yes
	set_variable = {
		name = engage_in_piracy_income_power
		value = 0
	}
	every_unit = {
		limit = {
			has_unit_modifier = engage_in_piracy_unit_modifier
			NOT = { has_variable = engage_in_piracy_active }
		}
		remove_unit_modifier = engage_in_piracy_unit_modifier
	}
	every_unit = {
		limit = {
			has_variable = engage_in_piracy_active
			is_navy = yes
			in_combat = no
			is_exiled = no
		}
		if = {
			limit = {
				NOT = { has_unit_modifier = engage_in_piracy_unit_modifier }
			}
			add_unit_modifier = {
				name = engage_in_piracy_unit_modifier
				duration = -1
			}
		}
		engage_in_piracy_add_fleet_raid_score_effect = yes
	}
	engage_in_piracy_apply_country_income_tier_effect = yes
}
