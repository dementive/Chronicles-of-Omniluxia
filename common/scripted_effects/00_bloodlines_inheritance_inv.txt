# Resolve all generations, even if children are iterated before their parents.
# A new pass is requested only when an eligible character actually gains a trait.
bloodline_setup_effect = {
    set_global_variable = { name = omni_bloodline_setup_running value = 1 }
    set_global_variable = { name = omni_bloodline_setup_pending value = 1 }
    while = {
        limit = { has_global_variable = omni_bloodline_setup_pending }
        remove_global_variable = omni_bloodline_setup_pending
        every_living_character = {
            bloodline_on_born_effect = yes
        }
    }
    remove_global_variable = omni_bloodline_setup_running
}

add_bloodline_setup_effect = {
    if = {
        limit = {
            exists = $CHAR$
        }
        every_living_character = {
            limit = {
                has_same_family = $CHAR$
            }
            omni_add_racial_bloodline_effect = { BLOODLINE = $BLOODLINE$ }
        }
    }
}

# Legitimacy awards propagate through all living generations, not just children
# and grandchildren. Keep this definition here rather than duplicating it.
add_bloodline_effect = {
    omni_add_racial_bloodline_effect = { BLOODLINE = $BLOODLINE$ }
    bloodline_setup_effect = yes
}

pass_down_bloodline_effect = {
    every_living_character = {
        pass_down_bloodline_character_effect = { BLOODLINE = $BLOODLINE$ }
    }
}

# Same as above but for an individual character
pass_down_bloodline_character_effect = {
    if = {
        limit = {
            exists = father
            father = {
                has_trait = $BLOODLINE$
            }                
            NOT = {
                has_trait = $BLOODLINE$
            }
        }
        omni_add_racial_bloodline_effect = { BLOODLINE = $BLOODLINE$ }
        if = {
            limit = {
                has_global_variable = omni_bloodline_setup_running
                has_trait = $BLOODLINE$
            }
            set_global_variable = { name = omni_bloodline_setup_pending value = 1 }
        }
    }
    if = {
        limit = {
            exists = mother
            mother = {
                has_trait = $BLOODLINE$
            }
            NOT = {
                has_trait = $BLOODLINE$
            }
        }
        omni_add_racial_bloodline_effect = { BLOODLINE = $BLOODLINE$ }
        if = {
            limit = {
                has_global_variable = omni_bloodline_setup_running
                has_trait = $BLOODLINE$
            }
            set_global_variable = { name = omni_bloodline_setup_pending value = 1 }
        }
    }
}
