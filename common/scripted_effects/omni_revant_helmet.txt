# Country scope. The discovery is unique even if several defeat hooks fire.
omni_offer_revant_helmet = {
    if = {
        limit = { NOT = { has_global_variable = omni_revant_helmet_offered } }
        set_global_variable = omni_revant_helmet_offered
        trigger_event = revant.13
    }
}

# Character scope. Register the slot here too for existing campaigns.
omni_equip_revant_helmet = {
    assing_item_to_category = { item_category = helmet item_ID = item_artifact_revant_helmet }
    give_character_an_artifact = { item_ID = item_artifact_revant_helmet }
}

# Country scope.
omni_artifact_recreate_revant_helmet = {
    create_country_treasure = { key = revant_helmet icon = "treasure_it_helmet_of_zalmoxis" modifier = { local_defensive = 0 } }
}

# Unit scope. Apply the multiplier once to the original damage amount.
omni_apply_spell_unit_damage = {
    damage_unit_percent = {
        value = $DAMAGE$
        if = {
            limit = { $CASTER$ = { omni_wears_revant_helmet = yes } }
            multiply = 2
        }
    }
}
