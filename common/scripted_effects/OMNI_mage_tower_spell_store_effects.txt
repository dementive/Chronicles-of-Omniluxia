# Province scope. Each tower shares one shelf among all visitors.
omni_refresh_mage_tower_stock = {
	if = {
		limit = { has_building = mage_tower_building NOT = { has_variable = omni_tower_restock_timer } }
		clear_variable_list = omni_tower_books
		set_variable = { name = omni_tower_restock_timer days = 90 }
		set_variable = { name = omni_mage_tower_quality value = 0 }
		if = { limit = { has_building = mage_tower_building } change_variable = { name = omni_mage_tower_quality add = 15 } }
		if = { limit = { num_of_magic_academy_building >= 1 } change_variable = { name = omni_mage_tower_quality add = 20 } }
		if = { limit = { num_of_magic_academy_building >= 2 } change_variable = { name = omni_mage_tower_quality add = 10 } }
		if = { limit = { num_of_academy_building >= 1 } change_variable = { name = omni_mage_tower_quality add = 8 } }
		if = { limit = { num_of_library_building >= 1 } change_variable = { name = omni_mage_tower_quality add = 8 } }
		if = { limit = { num_of_forum_building >= 1 } change_variable = { name = omni_mage_tower_quality add = 6 } }
		if = { limit = { num_of_court_building >= 1 } change_variable = { name = omni_mage_tower_quality add = 6 } }
		if = { limit = { num_of_used_building_slots >= 30 } change_variable = { name = omni_mage_tower_quality add = 15 } }
		else_if = { limit = { num_of_used_building_slots >= 16 } change_variable = { name = omni_mage_tower_quality add = 10 } }
		else_if = { limit = { num_of_used_building_slots >= 8 } change_variable = { name = omni_mage_tower_quality add = 5 } }
		if = { limit = { total_population >= 100 } change_variable = { name = omni_mage_tower_quality add = 20 } }
		else_if = { limit = { total_population >= 50 } change_variable = { name = omni_mage_tower_quality add = 10 } }
		else_if = { limit = { total_population >= 20 } change_variable = { name = omni_mage_tower_quality add = 5 } }
		if = { limit = { civilization_value >= 95 } change_variable = { name = omni_mage_tower_quality add = 15 } }
		else_if = { limit = { civilization_value >= 75 } change_variable = { name = omni_mage_tower_quality add = 10 } }
		else_if = { limit = { civilization_value >= 50 } change_variable = { name = omni_mage_tower_quality add = 5 } }
		if = { limit = { is_capital = yes } change_variable = { name = omni_mage_tower_quality add = 5 } }
		if = { limit = { is_holy_site = yes } change_variable = { name = omni_mage_tower_quality add = 10 } }
		if = { limit = { trade_goods = soulgems } change_variable = { name = omni_mage_tower_quality add = 20 } }
		if = { limit = { OR = { trade_goods = precious_metals trade_goods = gems trade_goods = papyrus } } change_variable = { name = omni_mage_tower_quality add = 10 } }
		omni_draw_mage_tower_t3_book = yes
		omni_draw_mage_tower_t3_book = yes
		if = {
			limit = { var:omni_mage_tower_quality >= 65 }
			omni_draw_mage_tower_t3_book = yes
		}
		if = {
			limit = { var:omni_mage_tower_quality >= 125 }
			omni_draw_mage_tower_t4_book = yes
		}
		else_if = {
			limit = { var:omni_mage_tower_quality >= 95 }
			random_list = { 35 = { omni_draw_mage_tower_t4_book = yes } 65 = {} }
		}
		else_if = {
			limit = { var:omni_mage_tower_quality >= 65 }
			random_list = { 10 = { omni_draw_mage_tower_t4_book = yes } 90 = {} }
		}
		remove_variable = omni_mage_tower_quality
	}
}

# Character event host. Select the visitor's tower, including a commander at a unit location.
omni_find_visited_mage_tower = {
	clear_saved_scope = omni_mage_tower_site
	scope:omni_city_visitor = {
		if = {
			limit = { location = { omni_mage_tower_territory_access = yes } }
			location = { save_scope_as = omni_mage_tower_site }
		}
		else = {
			random_character_unit = {
				limit = {
					OR = { is_army = yes is_navy = yes }
					unit_location = { omni_mage_tower_territory_access = yes }
				}
				unit_location = { save_scope_as = omni_mage_tower_site }
			}
		}
	}
}
