omni_mage_tower_add_spell_stock = {
	if = {
		limit = {
			NOT = { is_target_in_variable_list = { name = omni_mage_tower_spell_stock target = flag:$TIER$ } }
		}
		add_to_variable_list = { name = omni_mage_tower_spell_stock target = flag:$TIER$ }
	}
}

# CHARACTER scope. Builds a small random spellbook stock from the visitor's
# current city; better magical and civic centers roll rare master-tier books
# more often. Mage Towers only stock T3/T4 spellbooks.
omni_build_mage_tower_spell_stock = {
	clear_variable_list = omni_mage_tower_spell_stock
	set_variable = { name = omni_mage_tower_stock_timer days = 30 }
	set_variable = { name = omni_mage_tower_quality value = 0 }

	location = {
		if = { limit = { has_building = mage_tower_building } root = { change_variable = { name = omni_mage_tower_quality add = 15 } } }
		if = { limit = { num_of_magic_academy_building >= 1 } root = { change_variable = { name = omni_mage_tower_quality add = 20 } } }
		if = { limit = { num_of_magic_academy_building >= 2 } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
		if = { limit = { num_of_academy_building >= 1 } root = { change_variable = { name = omni_mage_tower_quality add = 8 } } }
		if = { limit = { num_of_library_building >= 1 } root = { change_variable = { name = omni_mage_tower_quality add = 8 } } }
		if = { limit = { num_of_forum_building >= 1 } root = { change_variable = { name = omni_mage_tower_quality add = 6 } } }
		if = { limit = { num_of_court_building >= 1 } root = { change_variable = { name = omni_mage_tower_quality add = 6 } } }
		if = { limit = { num_of_used_building_slots >= 30 } root = { change_variable = { name = omni_mage_tower_quality add = 15 } } }
		else_if = { limit = { num_of_used_building_slots >= 16 } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
		else_if = { limit = { num_of_used_building_slots >= 8 } root = { change_variable = { name = omni_mage_tower_quality add = 5 } } }
		if = { limit = { total_population >= 100 } root = { change_variable = { name = omni_mage_tower_quality add = 20 } } }
		else_if = { limit = { total_population >= 50 } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
		else_if = { limit = { total_population >= 20 } root = { change_variable = { name = omni_mage_tower_quality add = 5 } } }
		if = { limit = { civilization_value >= 95 } root = { change_variable = { name = omni_mage_tower_quality add = 15 } } }
		else_if = { limit = { civilization_value >= 75 } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
		else_if = { limit = { civilization_value >= 50 } root = { change_variable = { name = omni_mage_tower_quality add = 5 } } }
		if = { limit = { is_capital = yes } root = { change_variable = { name = omni_mage_tower_quality add = 5 } } }
		if = { limit = { is_holy_site = yes } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
		if = { limit = { trade_goods = soulgems } root = { change_variable = { name = omni_mage_tower_quality add = 20 } } }
		if = { limit = { OR = { trade_goods = precious_metals trade_goods = gems trade_goods = papyrus } } root = { change_variable = { name = omni_mage_tower_quality add = 10 } } }
	}

	if = {
		limit = { var:omni_mage_tower_quality < 35 }
		random_list = {
			80 = { }
			20 = { omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t3 } }
		}
	}
	else_if = {
		limit = { var:omni_mage_tower_quality < 65 }
		random_list = {
			45 = { }
			55 = { omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t3 } }
		}
	}
	else_if = {
		limit = { var:omni_mage_tower_quality < 95 }
		omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t3 }
		random_list = {
			90 = { }
			10 = { omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t4 } }
		}
	}
	else_if = {
		limit = { var:omni_mage_tower_quality < 125 }
		omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t3 }
		random_list = {
			65 = { }
			35 = { omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t4 } }
		}
	}
	else = {
		omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t3 }
		omni_mage_tower_add_spell_stock = { TIER = omni_mage_tower_t4 }
	}
}
