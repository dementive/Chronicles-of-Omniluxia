# Magical mercenary captains.
# PERF: nothing here runs on a pulse. The roll runs once a year for MER only
# (omni_mercenary_magic.1), and the arcane upkeep runs monthly ONLY for a
# country that currently employs a magical captain (omni_mercenary_magic.2).

# ---------------------------------------------------------------------------
# Per-tier spell unlocks. The spell-menu tier mapping starts at the *_t2
# variable, so tier N sets the *_t(N+1) variables. "_only" blocks grant just
# that tier so promotions do not re-roll lower tiers' spells.
# ---------------------------------------------------------------------------
omni_unlock_mercenary_magic_tier_one_only = {
	set_variable = diviner_t2_unlocked_var
	set_variable = cleric_t2_unlocked_var
	set_variable = elementalist_t2_unlocked_var
	set_variable = warlock_t2_unlocked_var
	omni_grant_random_omnic_t1 = yes
	omni_grant_random_aldic_t1 = yes
	omni_grant_random_amten_t1 = yes
	omni_grant_random_melodian_t1 = yes
}

omni_unlock_mercenary_magic_tier_two_only = {
	set_variable = diviner_t3_unlocked_var
	set_variable = cleric_t3_unlocked_var
	set_variable = elementalist_t3_unlocked_var
	set_variable = warlock_t3_unlocked_var
	omni_grant_random_omnic_t2 = yes
	omni_grant_random_aldic_t2 = yes
	omni_grant_random_amten_t2 = yes
	omni_grant_random_melodian_t2 = yes
}

omni_unlock_mercenary_magic_tier_three_only = {
	set_variable = diviner_t4_unlocked_var
	set_variable = cleric_t4_unlocked_var
	set_variable = elementalist_t4_unlocked_var
	set_variable = warlock_t4_unlocked_var
	omni_grant_random_omnic_t3 = yes
	omni_grant_random_aldic_t3 = yes
	omni_grant_random_amten_t3 = yes
	omni_grant_random_melodian_t3 = yes
}

omni_unlock_mercenary_magic_tier_four_only = {
	force_add_trait = archmage_trait
	omni_grant_random_omnic_t4 = yes
	omni_grant_random_aldic_t4 = yes
	omni_grant_random_amten_t4 = yes
	omni_grant_random_melodian_t4 = yes
	omni_check_magic_prestige = yes
}

# Cumulative versions (kept for any other callers).
omni_unlock_mercenary_magic_tier_one = {
	omni_unlock_mercenary_magic_tier_one_only = yes
}
omni_unlock_mercenary_magic_tier_two = {
	omni_unlock_mercenary_magic_tier_one_only = yes
	omni_unlock_mercenary_magic_tier_two_only = yes
}
omni_unlock_mercenary_magic_tier_three = {
	omni_unlock_mercenary_magic_tier_two = yes
	omni_unlock_mercenary_magic_tier_three_only = yes
}
omni_unlock_mercenary_magic_tier_four = {
	omni_unlock_mercenary_magic_tier_three = yes
	omni_unlock_mercenary_magic_tier_four_only = yes
}

# ---------------------------------------------------------------------------
# Tier bookkeeping. Prices/upkeep are cached as variables on the captain so
# the mercenary window can display them without any script-value evaluation.
# Tune the numbers in omni_magic_merc_hire_mult_svalue and
# omni_magic_merc_upkeep_svalue (common/script_values/omni_magical_mercenary_values.txt).
# ---------------------------------------------------------------------------
omni_set_magical_mercenary_tier = {
	set_variable = { name = omni_merc_magic_tier value = $TIER$ }
	set_variable = { name = omni_merc_magic_hire_mult value = omni_magic_merc_hire_mult_svalue }
	set_variable = { name = omni_merc_magic_upkeep value = omni_magic_merc_upkeep_svalue }
}

omni_register_magical_mercenary = {
	if = {
		limit = {
			NOT = {
				is_target_in_global_variable_list = {
					name = omni_magical_mercenaries
					target = this
				}
			}
		}
		add_to_global_variable_list = {
			name = omni_magical_mercenaries
			target = this
		}
	}
}

omni_make_magical_mercenary_base = {
	force_add_trait = mage_trait
	if = {
		limit = { NOT = { has_trait = omni_magical_mercenary_trait } }
		add_trait = omni_magical_mercenary_trait
	}
	omni_register_magical_mercenary = yes
}

omni_make_starting_magical_mercenary = {
	omni_make_magical_mercenary_base = yes
	omni_unlock_mercenary_magic_tier_one = yes
	omni_set_magical_mercenary_tier = { TIER = 1 }
	refill_character_magic_to_max = yes
}

omni_make_tier_two_magical_mercenary = {
	omni_make_magical_mercenary_base = yes
	omni_unlock_mercenary_magic_tier_two = yes
	omni_set_magical_mercenary_tier = { TIER = 2 }
	refill_character_magic_to_max = yes
}

omni_make_tier_three_magical_mercenary = {
	omni_make_magical_mercenary_base = yes
	omni_unlock_mercenary_magic_tier_three = yes
	omni_set_magical_mercenary_tier = { TIER = 3 }
	refill_character_magic_to_max = yes
}

omni_make_tier_four_magical_mercenary = {
	omni_make_magical_mercenary_base = yes
	omni_unlock_mercenary_magic_tier_four = yes
	omni_set_magical_mercenary_tier = { TIER = 4 }
	refill_character_magic_to_max = yes
}

# One step up, granting only the new tier's spells.
omni_promote_magical_mercenary = {
	if = {
		limit = { var:omni_merc_magic_tier = 1 }
		omni_unlock_mercenary_magic_tier_two_only = yes
		omni_set_magical_mercenary_tier = { TIER = 2 }
	}
	else_if = {
		limit = { var:omni_merc_magic_tier = 2 }
		omni_unlock_mercenary_magic_tier_three_only = yes
		omni_set_magical_mercenary_tier = { TIER = 3 }
	}
	else_if = {
		limit = { var:omni_merc_magic_tier = 3 }
		omni_unlock_mercenary_magic_tier_four_only = yes
		omni_set_magical_mercenary_tier = { TIER = 4 }
	}
	refill_character_magic_to_max = yes
}

# Era-weighted tier for a newly chosen captain. global_var:omni_merc_magic_year
# counts the yearly MER rolls since game start (same 50/100/150-year waves as
# the old design).
omni_roll_magical_mercenary_tier = {
	random_list = {
		70 = { omni_make_starting_magical_mercenary = yes }
		20 = {
			trigger = { global_var:omni_merc_magic_year >= 50 }
			omni_make_tier_two_magical_mercenary = yes
		}
		8 = {
			trigger = { global_var:omni_merc_magic_year >= 100 }
			omni_make_tier_three_magical_mercenary = yes
		}
		3 = {
			trigger = { global_var:omni_merc_magic_year >= 150 }
			omni_make_tier_four_magical_mercenary = yes
		}
	}
}

# ---------------------------------------------------------------------------
# Yearly MER-only maintenance of the roster. Scope: c:MER.
# ---------------------------------------------------------------------------
omni_mercenary_magic_yearly_effect = {
	if = {
		limit = { has_global_variable = omni_merc_magic_year }
		change_global_variable = { name = omni_merc_magic_year add = 1 }
	}
	else = {
		set_global_variable = { name = omni_merc_magic_year value = 1 }
	}

	# 1. Drop dead / no-longer-mercenary captains from the roster (rebuild the
	#    list from its own valid entries; at most a dozen entries).
	if = {
		limit = { has_global_variable_list = omni_magical_mercenaries }
		every_in_global_list = {
			variable = omni_magical_mercenaries
			limit = {
				is_alive = yes
				is_mercenary = yes
			}
			add_to_list = omni_valid_magic_mercs
		}
		clear_global_variable_list = omni_magical_mercenaries
		every_in_list = {
			list = omni_valid_magic_mercs
			add_to_global_variable_list = {
				name = omni_magical_mercenaries
				target = this
			}
			# Safety net: make sure whoever employs this captain pays upkeep
			# (covers bribes and any other employer change).
			if = {
				limit = {
					exists = employer
					NOT = { employer = c:MER }
				}
				employer = { omni_start_magic_merc_upkeep = yes }
			}
		}
	}

	# 2. Era waves: existing magical captains step up one tier.
	if = {
		limit = {
			has_global_variable_list = omni_magical_mercenaries
			OR = {
				global_var:omni_merc_magic_year = 50
				global_var:omni_merc_magic_year = 100
				global_var:omni_merc_magic_year = 150
			}
		}
		every_in_global_list = {
			variable = omni_magical_mercenaries
			limit = {
				has_variable = omni_merc_magic_tier
				var:omni_merc_magic_tier < 4
			}
			omni_promote_magical_mercenary = yes
		}
	}

	# 3. Each new captain is rolled exactly once in their life.
	every_character = {
		limit = {
			is_mercenary = yes
			is_general = yes
			NOT = { has_variable = omni_merc_magic_rolled }
		}
		set_variable = omni_merc_magic_rolled
		if = {
			limit = {
				OR = {
					NOT = { has_global_variable_list = omni_magical_mercenaries }
					global_variable_list_size = {
						name = omni_magical_mercenaries
						value < 12
					}
				}
			}
			random = {
				chance = 7
				omni_roll_magical_mercenary_tier = yes
			}
		}
	}
}

# ---------------------------------------------------------------------------
# Arcane upkeep. Scope: a country that just gained a magical captain.
# Starts a monthly self-rescheduling event that stops on its own as soon as
# the country no longer employs one.
# ---------------------------------------------------------------------------
omni_start_magic_merc_upkeep = {
	if = {
		limit = {
			NOT = { tag = MER }
			NOT = { has_variable = omni_magic_merc_upkeep_active }
		}
		set_variable = omni_magic_merc_upkeep_active
		trigger_event = { id = omni_mercenary_magic.2 days = 30 }
	}
}
