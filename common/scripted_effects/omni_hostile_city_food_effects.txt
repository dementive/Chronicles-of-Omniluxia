# Effects backing the "Black Market" and "Requisition Food" options available
# to a ruler or general (root) standing in a hostile/occupied city via
# omni_character_can_visit_hostile_city. Both options track a food reserve on
# the character's commanded unit (root.unit) rather than the state, since that
# reserve belongs to the visiting army, not the city itself.
#
# omni_unit_food_reserve is a simple 0-100 abstraction of "how well supplied
# is this army right now" - it does not hook into vanilla attrition directly,
# but is available for other scripted content (events, on_actions) to read.

omni_hostile_city_buy_food_effect = {
	if = {
		limit = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } }
		scope:omni_city_visitor = { add_gold = -150 }
	}
	else = { root.employer = { add_treasury = -150 } }
	custom_tooltip = omni_hostile_city_buy_food_effect_tt
}

omni_hostile_city_steal_food_effect = {
	if = {
		limit = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } }
		scope:omni_city_visitor = {
			location = {
				state = { add_state_food = -50 }
				add_province_modifier = { name = omni_food_requisitioned duration = 365 }
			}
		}
	}
	else = {
		root.location = {
			state = { add_state_food = -50 }
			add_province_modifier = { name = omni_food_requisitioned duration = 365 }
		}
	}
	custom_tooltip = omni_hostile_city_steal_food_effect_tt
}
