# National army loadout: 4 country slots (weapon / ranged / armour / shield).
# Buying an item outfits the realm: personal layer -> ruler, army layer -> country.
# One item per slot; a new purchase replaces the old one. Costs are paid from the
# NATIONAL TREASURY. Generated items are priced per-item by power (8 tiers):
# omni_price_tier_2..7 = 130 / 220 / 360 / 560 / 850 / 1250.
# Emblem/trinket generated items are personal-only (no slot, no army layer).

# Called from the shop (ROOT = buying ruler). Pays the item's tier price from the
# treasury, then routes the item into the country and (re)equips the matching slot.
omni_army_buy_slot_item = {
	if = {
		limit = { NOT = { is_ruler = yes } }
		if = {
			limit = {
				OR = {
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_2 target = var:omni_shop_slot_$SLOT$ } wealth >= 130 }
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_3 target = var:omni_shop_slot_$SLOT$ } wealth >= 220 }
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_4 target = var:omni_shop_slot_$SLOT$ } wealth >= 360 }
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_5 target = var:omni_shop_slot_$SLOT$ } wealth >= 560 }
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_6 target = var:omni_shop_slot_$SLOT$ } wealth >= 850 }
					AND = { is_target_in_global_variable_list = { name = omni_price_tier_7 target = var:omni_shop_slot_$SLOT$ } wealth >= 1250 }
					wealth >= 1750
				}
			}
			if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_2 target = var:omni_shop_slot_$SLOT$ } } add_gold = -130 }
			else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_3 target = var:omni_shop_slot_$SLOT$ } } add_gold = -220 }
			else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_4 target = var:omni_shop_slot_$SLOT$ } } add_gold = -360 }
			else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_5 target = var:omni_shop_slot_$SLOT$ } } add_gold = -560 }
			else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_6 target = var:omni_shop_slot_$SLOT$ } } add_gold = -850 }
			else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_7 target = var:omni_shop_slot_$SLOT$ } } add_gold = -1250 }
			else = { add_gold = -1750 }
			give_character_an_item_from_variable = { ITEM_VAR = omni_shop_slot_$SLOT$ }
			remove_list_variable = { name = omni_item_shop_stock target = var:omni_shop_slot_$SLOT$ }
			remove_list_variable = { name = omni_item_shop_stock_gen target = var:omni_shop_slot_$SLOT$ }
			remove_variable = omni_shop_slot_$SLOT$
		}
	}
	else = {
		omni_army_buy_slot_item_national = { SLOT = $SLOT$ }
	}
}

omni_army_buy_slot_item_national = {
	remove_variable = omni_army_purchase_succeeded
	if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_2 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 130 } } } set_variable = { name = omni_last_purchase_price value = 130 } country = { add_treasury = -130 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_3 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 220 } } } set_variable = { name = omni_last_purchase_price value = 220 } country = { add_treasury = -220 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_4 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 360 } } } set_variable = { name = omni_last_purchase_price value = 360 } country = { add_treasury = -360 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_5 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 560 } } } set_variable = { name = omni_last_purchase_price value = 560 } country = { add_treasury = -560 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_6 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 850 } } } set_variable = { name = omni_last_purchase_price value = 850 } country = { add_treasury = -850 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_7 target = var:omni_shop_slot_$SLOT$ } country = { omni_country_can_pay_for_luxury = { PRICE = 1250 } } } set_variable = { name = omni_last_purchase_price value = 1250 } country = { add_treasury = -1250 } set_variable = omni_army_purchase_succeeded }
	else_if = { limit = { country = { omni_country_can_pay_for_luxury = { PRICE = 1750 } } } set_variable = { name = omni_last_purchase_price value = 1750 } country = { add_treasury = -1750 } set_variable = omni_army_purchase_succeeded }

	if = {
		limit = { has_variable = omni_army_purchase_succeeded }
		save_scope_as = omni_army_buyer
		if = { limit = { is_target_in_global_variable_list = { name = item_weapon target = var:omni_shop_slot_$SLOT$ } } country = { set_variable = { name = omni_carried_slot_weapon value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_weapon value = scope:omni_army_buyer.var:omni_last_purchase_price } } }
		else_if = { limit = { is_target_in_global_variable_list = { name = item_weapon_ranged target = var:omni_shop_slot_$SLOT$ } } country = { set_variable = { name = omni_carried_slot_ranged value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_ranged value = scope:omni_army_buyer.var:omni_last_purchase_price } } }
		else_if = { limit = { is_target_in_global_variable_list = { name = item_armour target = var:omni_shop_slot_$SLOT$ } } country = { set_variable = { name = omni_carried_slot_armour value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_armour value = scope:omni_army_buyer.var:omni_last_purchase_price } } }
		else_if = { limit = { is_target_in_global_variable_list = { name = item_shield target = var:omni_shop_slot_$SLOT$ } } country = { set_variable = { name = omni_carried_slot_shield value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_shield value = scope:omni_army_buyer.var:omni_last_purchase_price } } }
		else_if = { limit = { is_target_in_global_variable_list = { name = item_emblem target = var:omni_shop_slot_$SLOT$ } } country = { set_variable = { name = omni_carried_slot_emblem value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_emblem value = scope:omni_army_buyer.var:omni_last_purchase_price } } }
		else = { country = { set_variable = { name = omni_carried_slot_trinket value = scope:omni_army_buyer.var:omni_shop_slot_$SLOT$ } set_variable = { name = omni_carried_price_trinket value = scope:omni_army_buyer.var:omni_last_purchase_price } } }

		remove_list_variable = { name = omni_item_shop_stock target = var:omni_shop_slot_$SLOT$ }
		remove_list_variable = { name = omni_item_shop_stock_gen target = var:omni_shop_slot_$SLOT$ }
		remove_variable = omni_shop_slot_$SLOT$
	}
	remove_variable = omni_army_purchase_succeeded
	remove_variable = omni_last_purchase_price
}

# Fixed / hand-made GEAR purchase (literal item_ID, explicit price).
# Rulers and AI use the national loadout; player-controlled non-rulers buy
# personally and receive only their own character item.
omni_army_buy_fixed = {
	save_scope_as = omni_fixed_purchase_host
	remove_variable = omni_fixed_purchase_succeeded
	if = {
		limit = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } }
		scope:omni_city_visitor = {
			if = {
				limit = { wealth >= $PRICE$ }
				add_gold = { value = $PRICE$ multiply = -1 }
				give_character_an_item = { item_ID = $item_ID$ }
				scope:omni_fixed_purchase_host = { set_variable = omni_fixed_purchase_succeeded }
			}
		}
	}
	else_if = {
		limit = { is_ruler = yes OR = { employer = { is_ai = yes } country = { is_ai = yes } } }
		if = {
			limit = { country = { omni_country_can_pay_for_luxury = { PRICE = $PRICE$ } } }
			country = { add_treasury = -$PRICE$ }
			# National gear must not call give_character_an_item here: that effect
			# wakes AI personal auto-equip, which would add the combat item's country
			# layer once before the national layer dispatcher adds it again.
			add_to_variable_list = { name = item_list target = flag:$item_ID$ }
			country = {
				set_variable = { name = omni_gen_fx_item value = flag:$item_ID$ }
				omni_army_replace_slot = { SLOTVAR = $SLOTVAR$ }
				remove_variable = omni_gen_fx_item
			}
			set_variable = omni_fixed_purchase_succeeded
		}
	}
	else = {
		if = {
			limit = { wealth >= $PRICE$ }
			add_gold = { value = $PRICE$ multiply = -1 }
			give_character_an_item = { item_ID = $item_ID$ }
			set_variable = omni_fixed_purchase_succeeded
		}
	}
	if = {
		limit = { has_variable = omni_fixed_purchase_succeeded }
		remove_list_variable = { name = omni_item_shop_stock target = flag:$item_ID$ }
	}
	remove_variable = omni_fixed_purchase_succeeded
}

# COUNTRY scope. var:omni_gen_fx_item = the item being equipped. Picks the slot.
omni_army_equip_current = {
	if = {
		limit = { is_target_in_global_variable_list = { name = item_weapon target = var:omni_gen_fx_item } }
		omni_army_replace_slot = { SLOTVAR = omni_army_slot_weapon }
	}
	else_if = {
		limit = { is_target_in_global_variable_list = { name = item_weapon_ranged target = var:omni_gen_fx_item } }
		omni_army_replace_slot = { SLOTVAR = omni_army_slot_ranged }
	}
	else_if = {
		limit = { is_target_in_global_variable_list = { name = item_armour target = var:omni_gen_fx_item } }
		omni_army_replace_slot = { SLOTVAR = omni_army_slot_armour }
	}
	else_if = {
		limit = { is_target_in_global_variable_list = { name = item_shield target = var:omni_gen_fx_item } }
		omni_army_replace_slot = { SLOTVAR = omni_army_slot_shield }
	}
	else = {
		# emblem / trinket: personal-only, no national slot
		omni_apply_army_item_layers = yes
	}
}

# COUNTRY scope. Replaces $SLOTVAR$: strips the old item's layers (if any),
# applies the new item's layers, and records the new item in the slot.
# On entry var:omni_gen_fx_item = NEW item.
omni_army_replace_slot = {
	if = {
		limit = { has_variable = $SLOTVAR$ }
		set_variable = { name = omni_gen_fx_new value = var:omni_gen_fx_item }
		set_variable = { name = omni_gen_fx_item value = var:$SLOTVAR$ }
		omni_remove_army_item_layers = yes
		set_variable = { name = omni_gen_fx_item value = var:omni_gen_fx_new }
		remove_variable = omni_gen_fx_new
	}
	omni_apply_army_item_layers = yes
	set_variable = { name = $SLOTVAR$ value = var:omni_gen_fx_item }
}

# COUNTRY scope. Hands the national loadout from the outgoing ruler to the
# incoming one. Called from on_ruler_change, where scope:former_ruler is the
# outgoing ruler and current_ruler is already the successor.
#
# This replaces a call to omni_army_clear_all, which was wrong twice over. It
# wiped the realm's whole armoury on every succession - gear the country paid for
# out of the treasury, described in its own tooltips as an "Army Standard" - and
# it stripped the personal layers via current_ruler, which at on_ruler_change is
# the NEW ruler. So the outgoing ruler kept every item's stat bonus for the rest
# of their life while the incoming one got nothing.
#
# The country _army layers are left alone: they belong to the realm, not to a
# person, and re-adding them here would stack a second copy.
omni_army_transfer_loadout = {
	if = {
		limit = { exists = scope:former_ruler }
		omni_army_queue_slots_on = { WHO = scope:former_ruler QUEUE = omni_gen_layer_strip_queue EVT = 2 }
	}
	if = {
		limit = { exists = current_ruler }
		omni_army_queue_slots_on = { WHO = current_ruler QUEUE = omni_gen_layer_sync_queue EVT = 3 }
	}
}

# COUNTRY scope. Re-asserts the current ruler's personal layers against whatever
# the realm actually has in its slots. Cheap to call and idempotent - the drain
# event removes before it applies - so it doubles as a repair for any save whose
# ruler and loadout have drifted apart.
omni_army_sync_ruler_layers = {
	if = {
		limit = { exists = current_ruler }
		omni_army_queue_slots_on = { WHO = current_ruler QUEUE = omni_gen_layer_sync_queue EVT = 3 }
	}
}

# COUNTRY scope. Copies every filled slot onto $WHO$'s $QUEUE$ and wakes the
# drain event. Queueing rather than dispatching inline matters here: the
# generated dispatch is a ~1,600-branch table, and calling it per slot per caller
# would inline it a dozen times over.
omni_army_queue_slots_on = {
	save_scope_as = omni_layer_realm
	$WHO$ = {
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_army_slot_weapon } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_army_slot_weapon } }
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_army_slot_ranged } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_army_slot_ranged } }
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_army_slot_armour } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_army_slot_armour } }
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_army_slot_shield } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_army_slot_shield } }
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_personal_slot_emblem } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_personal_slot_emblem } }
		if = { limit = { scope:omni_layer_realm = { has_variable = omni_personal_slot_trinket } } add_to_variable_list = { name = $QUEUE$ target = scope:omni_layer_realm.var:omni_personal_slot_trinket } }
		if = { limit = { has_variable_list = $QUEUE$ } trigger_event = { id = omni_item_dispatch.$EVT$ } }
	}
}

# COUNTRY scope. Clears the whole loadout and strips the army modifiers.
# No longer used on succession (see omni_army_transfer_loadout above); kept for
# callers that really do want the realm disarmed outright.
omni_army_clear_all = {
	if = { limit = { has_variable = omni_army_slot_weapon } set_variable = { name = omni_gen_fx_item value = var:omni_army_slot_weapon } omni_remove_army_item_layers = yes remove_variable = omni_army_slot_weapon }
	if = { limit = { has_variable = omni_army_slot_ranged } set_variable = { name = omni_gen_fx_item value = var:omni_army_slot_ranged } omni_remove_army_item_layers = yes remove_variable = omni_army_slot_ranged }
	if = { limit = { has_variable = omni_army_slot_armour } set_variable = { name = omni_gen_fx_item value = var:omni_army_slot_armour } omni_remove_army_item_layers = yes remove_variable = omni_army_slot_armour }
	if = { limit = { has_variable = omni_army_slot_shield } set_variable = { name = omni_gen_fx_item value = var:omni_army_slot_shield } omni_remove_army_item_layers = yes remove_variable = omni_army_slot_shield }
	if = { limit = { has_variable = omni_personal_slot_emblem } set_variable = { name = omni_gen_fx_item value = var:omni_personal_slot_emblem } omni_remove_army_item_layers = yes remove_variable = omni_personal_slot_emblem }
	if = { limit = { has_variable = omni_personal_slot_trinket } set_variable = { name = omni_gen_fx_item value = var:omni_personal_slot_trinket } omni_remove_army_item_layers = yes remove_variable = omni_personal_slot_trinket }
	remove_variable = omni_army_price_weapon
	remove_variable = omni_army_price_ranged
	remove_variable = omni_army_price_armour
	remove_variable = omni_army_price_shield
	remove_variable = omni_personal_price_emblem
	remove_variable = omni_personal_price_trinket
	if = { limit = { has_variable = omni_gen_fx_item } remove_variable = omni_gen_fx_item }
}

# COUNTRY scope. AI army-supply purchasing, called from the yearly pulse. Picks
# an item, then buys it at its tier price only if the treasury allows.
omni_army_ai_purchase = {
	if = {
		limit = { is_ai = yes exists = current_ruler omni_ai_luxury_spendable_treasury >= 130 }
		if = {
			limit = { omni_ai_luxury_spendable_treasury >= 500 }
			random_in_global_list = { variable = item_uncommon_rarity save_scope_as = omni_ai_item_pick }
		}
		else = {
			random_in_global_list = { variable = item_common_rarity save_scope_as = omni_ai_item_pick }
		}
		if = { limit = { exists = scope:omni_ai_item_pick } omni_army_ai_try_buy = yes }
	}
}

# Character-facing generated gear controls. These use four fixed category
# variables, keeping the menus compact regardless of the generated item count.
omni_character_unequip_army_slot = {
	country = {
		set_variable = { name = omni_gen_fx_item value = var:$SLOTVAR$ }
		omni_remove_army_item_layers = yes
		set_variable = { name = $CARRYVAR$ value = var:$SLOTVAR$ }

		remove_variable = $SLOTVAR$
		remove_variable = omni_gen_fx_item
	}
}

		remove_variable = $CARRYVAR$
	}
}

# COUNTRY scope. Sells var:omni_gen_fx_item for half its shop tier.

	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_3 target = var:omni_gen_fx_item } } add_treasury = 110 }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_4 target = var:omni_gen_fx_item } } add_treasury = 180 }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_5 target = var:omni_gen_fx_item } } add_treasury = 280 }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_6 target = var:omni_gen_fx_item } } add_treasury = 425 }
	else_if = { limit = { is_target_in_global_variable_list = { name = omni_price_tier_7 target = var:omni_gen_fx_item } } add_treasury = 625 }
	else = { add_treasury = 30 }
}

omni_character_sell_carried_slot = {
	country = {
		set_variable = { name = omni_gen_fx_item value = var:$CARRYVAR$ }
		if = { limit = { has_variable = $CARRYPRICE$ } add_treasury = { value = var:$CARRYPRICE$ multiply = 0.5 } remove_variable = $CARRYPRICE$ }
		else = { omni_sell_current_generated_gear = yes }
		remove_variable = omni_gen_fx_item
		remove_variable = $CARRYVAR$
	}
}

# One-time compatibility for saves made while carried slots were character-scoped.
omni_migrate_character_carried_gear = {
	save_scope_as = omni_gear_character
	if = { limit = { has_variable = omni_carried_slot_weapon } country = { set_variable = { name = omni_carried_slot_weapon value = scope:omni_gear_character.var:omni_carried_slot_weapon } } remove_variable = omni_carried_slot_weapon }
	if = { limit = { has_variable = omni_carried_slot_ranged } country = { set_variable = { name = omni_carried_slot_ranged value = scope:omni_gear_character.var:omni_carried_slot_ranged } } remove_variable = omni_carried_slot_ranged }
	if = { limit = { has_variable = omni_carried_slot_armour } country = { set_variable = { name = omni_carried_slot_armour value = scope:omni_gear_character.var:omni_carried_slot_armour } } remove_variable = omni_carried_slot_armour }
	if = { limit = { has_variable = omni_carried_slot_shield } country = { set_variable = { name = omni_carried_slot_shield value = scope:omni_gear_character.var:omni_carried_slot_shield } } remove_variable = omni_carried_slot_shield }
	if = { limit = { has_variable = omni_carried_slot_emblem } country = { set_variable = { name = omni_carried_slot_emblem value = scope:omni_gear_character.var:omni_carried_slot_emblem } } remove_variable = omni_carried_slot_emblem }
	if = { limit = { has_variable = omni_carried_slot_trinket } country = { set_variable = { name = omni_carried_slot_trinket value = scope:omni_gear_character.var:omni_carried_slot_trinket } } remove_variable = omni_carried_slot_trinket }
}
