# Dynamic availability gives the CB to present and future Helluvian countries.
# Superiority does not require fabricated territorial claims.
omni_shannah_penance_wargoal = {
    type = superiority
    allow = {
        is_subject = no
        religion = helluvian
        NOT = { has_country_modifier = omni_shannah_excommunicated }
        scope:defender = {
            religion = helluvian
            has_country_modifier = omni_shannah_excommunicated
        }
    }
    attacker = { conquer_cost = -0.1 }
    defender = { }
    ticking_war_score = 1
}

omni_shannah_contested_accession_wargoal = {
    type = superiority
    allow = {
        is_subject = no
        OR = {
            AND = {
                scope:defender = { omni_hc_shannah_trigger = yes }
                omni_hc_contests_shannah_trigger = { SHAHANAT = scope:defender }
            }
            AND = {
                omni_hc_shannah_trigger = yes
                scope:defender = {
                    omni_hc_contests_shannah_trigger = { SHAHANAT = prev }
                }
            }
        }
    }
    attacker = { conquer_cost = 0 }
    defender = { }
    ticking_war_score = 1
}
