generic_trade_expedition = {
	header = "lotr_generic_mission_ships_headers"
	icon = lotr_generic_ship_mission_icon

	repeatable = yes

	chance = {}

	ai_chance = {
		factor = 0
	}

	potential = {
		choo_generic_mission_blocker = yes
		has_civil_war = no
		civic_tech >= 2
		num_of_ports >= 1
		NOT = { has_variable = generic_trade_expedition_cooldown }
		any_owned_province = {
			generic_trade_expedition_valid_home_port_trigger = yes
		}
		any_country = {
			generic_trade_expedition_valid_partner_trigger = yes
		}
	}

	abort = {
		OR = {
			NOT = { exists = scope:generic_trade_expedition_partner }
			NOT = { exists = scope:generic_trade_expedition_home_port }
			NOT = { exists = scope:generic_trade_expedition_partner_port }
			scope:generic_trade_expedition_partner = {
				OR = {
					has_land = no
					war_with = ROOT
				}
			}
			scope:generic_trade_expedition_home_port = {
				NOT = { owner = ROOT }
			}
			scope:generic_trade_expedition_partner_port = {
				OR = {
					NOT = { owner = scope:generic_trade_expedition_partner }
					NOT = { is_port = yes }
					NOT = { has_building = commerce_building }
				}
			}
			AND = {
				has_variable = generic_trade_expedition_envoy_selected
				NOT = {
					any_character = {
						has_character_modifier = generic_trade_expedition_envoy_charmod
					}
				}
			}
			ai_mission_back_out_trigger = yes
		}
	}

	on_abort = {
		generic_trade_expedition_cleanup_effect = yes
		set_variable = {
			name = generic_trade_expedition_cooldown
			days = 730
		}
	}

	on_potential = {
		random_owned_province = {
			limit = {
				generic_trade_expedition_valid_home_port_trigger = yes
			}
			save_scope_as = generic_trade_expedition_home_port
		}
		random_country = {
			limit = {
				generic_trade_expedition_valid_partner_trigger = yes
			}
			save_scope_as = generic_trade_expedition_partner
			random_owned_province = {
				limit = {
					generic_trade_expedition_valid_partner_port_trigger = yes
				}
				save_scope_as = generic_trade_expedition_partner_port
			}
		}
	}

	on_start = {
		start_mission_ai_effect = yes
	}

	on_completion = {
		complete_mission_effect = yes
		generic_trade_expedition_cleanup_effect = yes
		set_variable = {
			name = generic_trade_expedition_cooldown
			days = 1825
		}
	}

		requires = { generic_trade_expedition_return_home }

		final = yes

		allow = {
			has_country_modifier = generic_trade_expedition_returned_home_cmod
		}

		highlight = {
			scope:province = {
				THIS = scope:generic_trade_expedition_home_port
			}
		}

		on_completion = {
			save_scope_as = generic_trade_expedition_sender
			if = {
				limit = { has_country_modifier = generic_trade_expedition_returned_home_cmod }
				remove_country_modifier = generic_trade_expedition_returned_home_cmod
			}
			if = {
				limit = { has_variable = generic_trade_expedition_cargo_bulk }
				add_treasury = generic_trade_expedition_event_profit_svalue
			}
			if = {
				limit = { has_variable = generic_trade_expedition_cargo_luxury }
				add_treasury = generic_trade_expedition_large_event_profit_svalue
			}
			if = {
				limit = { has_variable = generic_trade_expedition_cargo_knowledge }
				add_research = {
					technology = civic_tech
					value = 10
				}
			}
			if = {
				limit = { has_variable = generic_trade_expedition_bonded_warehouses }
				add_treasury = generic_trade_expedition_event_profit_svalue
			}
			if = {
				limit = { has_variable = generic_trade_expedition_guarded_warehouses }
				add_political_influence = 5
			}
			if = {
				limit = { has_variable = generic_trade_expedition_cramped_warehouses }
				add_research = {
					technology = civic_tech
					value = 5
				}
			}
			if = {
				limit = { has_variable = generic_trade_expedition_customs_recorded }
				add_political_influence = 5
			}
			if = {
				limit = { has_variable = generic_trade_expedition_factor_house_planted }
				scope:generic_trade_expedition_partner = {
					add_opinion = {
						modifier = opinion_trade_expedition_factor_house
						target = ROOT
					}
				}
			}
			add_country_modifier = {
				name = generic_trade_expedition_successful
				duration = 3650
			}
			scope:generic_trade_expedition_home_port = {
				add_province_modifier = {
					name = generic_trade_expedition_returned_merchants
					duration = 3650
				}
			}
			scope:generic_trade_expedition_partner = {
				add_opinion = {
					modifier = opinion_trade_expedition_success
					target = ROOT
				}
				hidden_effect = {
					trigger_event = { id = generic_trade_expedition.214 days = 1 }
				}
			}
			trigger_event = { id = generic_trade_expedition.3 }
		}
	}
}
