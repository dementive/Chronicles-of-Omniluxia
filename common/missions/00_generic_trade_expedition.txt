# THE TRADE EXPEDITION
#
# SHAPE: a short trunk, then three parallel tracks (The Ship, The Purse, The
# Court) that run at the same time and converge on the convoy, then the market,
# then home, then the Charter.
#
# FLEET GATES: only three tasks care where the fleet is - assemble it, arrive
# with it, come home with it. Those are the three moments the player actually
# moves something. Everything else is court and market work, and does not
# pretend the fleet's location matters.
#
# EVENTS: storms, pirates and market incidents are NOT tasks. They fire from
# monthly_on_action pulses attached to the sailing and trading tasks, so they
# happen while the player is genuinely at sea or genuinely in the market.

generic_trade_expedition = {
	header = "lotr_generic_mission_ships_headers"
	icon = lotr_generic_ship_mission_icon

	repeatable = yes

	chance = {}

	ai_chance = {
		factor = 0
	}

	potential = {
		# PERF: ai_chance is 0, so the AI never starts this mission; skip the AI before the world scan.
		is_ai = no
		choo_generic_mission_blocker = yes
		has_civil_war = no
		civic_tech >= 2
		num_of_ports >= 1
		NOT = { has_variable = generic_trade_expedition_cooldown }
		any_owned_province = {
			generic_trade_expedition_valid_home_port_trigger = yes
		}
		any_country = {
			generic_trade_expedition_valid_partner_trigger = yes
		}
	}

	abort = {
		OR = {
			NOT = { exists = scope:generic_trade_expedition_partner }
			NOT = { exists = scope:generic_trade_expedition_home_port }
			NOT = { exists = scope:generic_trade_expedition_partner_port }
			scope:generic_trade_expedition_partner = {
				OR = {
					has_land = no
					war_with = ROOT
				}
			}
			scope:generic_trade_expedition_home_port = {
				NOT = { owner = ROOT }
			}
			scope:generic_trade_expedition_partner_port = {
				OR = {
					NOT = { owner = scope:generic_trade_expedition_partner }
					NOT = { is_port = yes }
					NOT = { has_building = commerce_building }
				}
			}
			AND = {
				has_variable = generic_trade_expedition_envoy_selected
				NOT = {
					any_character = {
						has_character_modifier = generic_trade_expedition_envoy_charmod
					}
				}
			}
			ai_mission_back_out_trigger = yes
		}
	}

	on_abort = {
		generic_trade_expedition_cleanup_effect = yes
		set_variable = {
			name = generic_trade_expedition_cooldown
			days = 730
		}
	}

	on_potential = {
		random_owned_province = {
			limit = {
				generic_trade_expedition_valid_home_port_trigger = yes
			}
			save_scope_as = generic_trade_expedition_home_port
		}
		random_country = {
			limit = {
				generic_trade_expedition_valid_partner_trigger = yes
			}
			save_scope_as = generic_trade_expedition_partner
			random_owned_province = {
				limit = {
					generic_trade_expedition_valid_partner_port_trigger = yes
				}
				save_scope_as = generic_trade_expedition_partner_port
			}
		}
		# A competitor working the same marketplace. Used by market events.
		random_country = {
			limit = {
				generic_trade_expedition_valid_rival_trigger = yes
			}
			save_scope_as = generic_trade_expedition_rival
		}
	}

	on_start = {
		start_mission_ai_effect = yes
	}

	on_completion = {
		complete_mission_effect = yes
		generic_trade_expedition_count_run_effect = yes
		generic_trade_expedition_cleanup_effect = yes
		set_variable = {
			name = generic_trade_expedition_cooldown
			days = 1825
		}
	}

	# TRUNK

	generic_trade_expedition_departure_feast = {
		icon = "task_political"

			generic_trade_expedition_appoint_escort
			generic_trade_expedition_commission_merchants
			generic_trade_expedition_harbor_rights

		}

		highlight = {
			scope:province = {
				THIS = scope:generic_trade_expedition_home_port
			}
		}

		on_completion = {
			add_treasury = generic_trade_expedition_feast_pay_svalue
			current_ruler = {
				add_popularity = popularity_small
			}
			if = {
				limit = { exists = scope:generic_trade_expedition_envoy }
				scope:generic_trade_expedition_envoy = {
					add_popularity = popularity_small
				}
			}
			capital_scope = {
				add_province_modifier = {
					name = generic_trade_expedition_departure_feast_cmod
					duration = 365
				}
			}
			scope:generic_trade_expedition_partner = {
				add_opinion = {
					modifier = opinion_trade_expedition_departure_feast
					target = ROOT
				}
			}
			trigger_event = { id = generic_trade_expedition.219 }
		}
	}
