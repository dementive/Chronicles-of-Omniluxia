generic_vampire_dominion = {
	header = "generic_vampire_dominion_title"
	icon = lotr_generic_army_mission_icon

	repeatable = yes

	chance = {}

	ai_chance = {
		factor = 0
	}

	potential = {
		choo_generic_mission_blocker = yes
		has_civil_war = no
		current_ruler = { has_trait = vampire_trait }
		NOT = { has_variable = generic_vampire_dominion_cooldown }
	}

	abort = {
		OR = {
			current_ruler = { NOT = { has_trait = vampire_trait } }
			has_civil_war = yes
			ai_mission_back_out_trigger = yes
		}
	}

	on_abort = {
		set_variable = {
			name = generic_vampire_dominion_cooldown
			days = 730
		}
		if = {
			limit = { has_variable = generic_vampire_dominion_first_feeding }
			remove_variable = generic_vampire_dominion_first_feeding
		}
		if = {
			limit = { has_variable = generic_vampire_dominion_active }
			remove_variable = generic_vampire_dominion_active
		}
	}

	on_start = {
		start_mission_ai_effect = yes
		if = {
			limit = { has_variable = generic_vampire_dominion_first_feeding }
			remove_variable = generic_vampire_dominion_first_feeding
		}
		if = {
			limit = { has_variable = generic_vampire_dominion_active }
			remove_variable = generic_vampire_dominion_active
		}
	}

	on_completion = {
		complete_mission_effect = yes
		set_variable = {
			name = generic_vampire_dominion_cooldown
			days = 3650
		}
		if = {
			limit = { has_variable = generic_vampire_dominion_first_feeding }
			remove_variable = generic_vampire_dominion_first_feeding
		}
		if = {
			limit = { has_variable = generic_vampire_dominion_active }
			remove_variable = generic_vampire_dominion_active
		}
	}

	generic_vampire_dominion_first_thirst = {
		icon = "task_political"
		duration = 30

		bypass = {
			is_ai = yes
		}

		on_completion = {
			set_variable = generic_vampire_dominion_active
		}
	}
