			is_capital = no
			total_population >= 10

				any_state_province = {
					is_state_capital = yes

					NOT = { has_province_modifier = generic_vampire_dominion_guest_who_never_left_pmod }
					NOT = { has_province_modifier = generic_vampire_dominion_pale_retinue_pmod }
					NOT = { has_province_modifier = generic_vampire_dominion_court_of_whispers_pmod }
					NOT = { has_province_modifier = generic_vampire_dominion_unnatural_mist_pmod }
					NOT = { has_province_modifier = generic_vampire_dominion_wolves_grown_bold_pmod }
					NOT = { has_province_modifier = generic_vampire_dominion_where_crows_gather_pmod }

				}
			}
		}
	}

	abort = {
		OR = {
			NOT = { exists = scope:generic_vampire_dominion_target }
			scope:generic_vampire_dominion_target = {
				NOT = { owner = ROOT }
			}
			current_ruler = { NOT = { has_trait = vampire_trait } }
			has_civil_war = yes
			ai_mission_back_out_trigger = yes
		}
	}

	on_potential = {
		random_owned_province = {
			limit = {
				is_capital = no
				total_population >= 10
				state = {
					any_state_province = {
						is_state_capital = yes
						NOT = { has_province_modifier = generic_vampire_dominion_bloodless_nights_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_culled_doubters_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_night_tithe_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_thrall_levies_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_sealed_crypt_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_guest_who_never_left_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_pale_retinue_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_court_of_whispers_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_unnatural_mist_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_wolves_grown_bold_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_where_crows_gather_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_sleepless_village_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_quiet_hours_pmod }
						NOT = { has_province_modifier = generic_vampire_dominion_seat_of_pale_court_pmod }
					}
					NOT = {
						any_state_province = {
							has_province_modifier = generic_vampire_dominion_drained_herds_pmod
						}
					}
				}
			}
			save_scope_as = generic_vampire_dominion_target
			state = {
				random_state_province = {
					limit = {
						is_state_capital = yes
					}
					save_scope_as = generic_vampire_dominion_capital
				}
			}
		}
	}

	on_abort = {
		ROOT = {
			set_variable = {
				name = generic_vampire_dominion_cooldown
				days = 365
			}
			if = {
				limit = { has_variable = generic_vampire_dominion_row_picks }
				remove_variable = generic_vampire_dominion_row_picks
			}
			if = {
				limit = { has_variable = generic_vampire_dominion_active }
				remove_variable = generic_vampire_dominion_active
			}
		}
	}

	on_start = {
		start_mission_ai_effect = yes
		ROOT = {
			set_variable = {
				name = generic_vampire_dominion_row_picks
				value = 0
			}
			if = {
				limit = { has_variable = generic_vampire_dominion_active }
				remove_variable = generic_vampire_dominion_active
			}
		}
	}

	on_completion = {
		complete_mission_effect = yes
		ROOT = {
			set_variable = {
				name = generic_vampire_dominion_cooldown
				days = 365
			}
			if = {
				limit = { has_variable = generic_vampire_dominion_row_picks }

			}
			if = {
				limit = { has_variable = generic_vampire_dominion_active }
				remove_variable = generic_vampire_dominion_active
			}
		}
	}

				}
			}
			scope:generic_vampire_dominion_target = {
				state = {
					ordered_state_province = {
						limit = {
							has_owner = yes
							total_population > 0
						}
						order_by = total_population
						max = 1
						check_range_bounds = no
						save_scope_as = generic_vampire_dominion_pop_target
					}
				}
			}
			scope:generic_vampire_dominion_pop_target = {
				while = {
					count = 5
					random_pops_in_province = {
						kill_pop = yes
					}
				}
			}
			hidden_effect = {
				create_character = {
				female = no
				age = 35
				culture = undead
				religion = scope:generic_vampire_dominion_target.religion
				add_trait = wraith_trait
				save_scope_as = generic_vampire_dominion_undead_wraith_commander
			}
			}
			create_unit = {
				navy = no
				name = "Undead Wraiths"
				save_scope_as = generic_vampire_dominion_undead_wraiths_army_scope
				location = scope:generic_vampire_dominion_capital
				hidden_effect = {
					set_as_commander = scope:generic_vampire_dominion_undead_wraith_commander
				}
				hidden_effect = {
					while = {
						count = 5
						add_subunit = light_infantry
					}
					add_food = 100
					add_morale = 2
				}
			}
			scope:generic_vampire_dominion_capital = {
				add_province_modifier = {
					name = generic_vampire_dominion_thrall_levies_pmod
					duration = 1825
				}
			}
			current_ruler = {
				add_character_modifier = {
					name = generic_vampire_dominion_silken_thralls_charmod
					duration = 1825
				}
			}
			trigger_event = { id = me_vampire_dominion.2 }
			random_character = {
				limit = {
					NOT = { is_ruler = yes }
					employer = ROOT
				}
				add_loyalty = loyalty_granted_stipends
				add_popularity = 5
			}
		}
	}
