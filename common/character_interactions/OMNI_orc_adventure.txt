omniluxia_orc_adventure = {
	on_other_nation = no
	on_own_nation = yes

	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_scholar"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			employer = scope:actor
			is_ruler = yes
			is_alive = yes
			is_adult = yes
			NOT = { employer = { tag = REB } }
			NOT = { employer = { tag = BAR } }
		}
		scope:actor = {
			is_ai = no
			OR = {
				religion = orcish_ancestral_faith
				religion = orcish_new_faith
				religion = orcish_colony_faith
			}
			war = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			trigger_if = {
				limit = { is_adult = no }
				is_adult = yes
			}
			is_ruler = yes
			custom_tooltip = {
				text = is_on_omni_orc_adventure_tt
				NOT = { has_variable = omni_orc_adventure_cooldown }
			}
		}
	}

	effect = {
		scope:actor = {
			trigger_event = { id =  orc_flavor.5 }
		}
		scope:target = {
			set_variable = {
				name = omni_orc_adventure_cooldown
				days = 1825
			}
		}
	}
}
