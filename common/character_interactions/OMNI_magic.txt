omniluxia_magic = {
	on_other_nation = no
	on_own_nation = yes

	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		hidden:scope:target = {
			employer = scope:actor
			OR = {
				is_ruler = yes
				is_general = yes
				has_office = office_physician
				has_office = office_republic_physician
				has_office = office_apothecary
			}
			is_alive = yes
			is_adult = yes
			NOT = { employer = { tag = REB } }
			NOT = { employer = { tag = BAR } }
		}
		scope:actor = {
			is_ai = no
			#loyalty >= 50
			scope:target = {
				OR = {
					has_trait = mage_trait
					has_trait = archmage_trait
					has_trait = warlock_trait
					has_trait = cleric_trait
					has_trait = necromancer_trait
					has_trait = lich_trait
					has_trait = druid_trait
					has_trait = elementalist_trait
					has_trait = diviner_trait
					has_trait = hierophant_trait
					has_trait = illusionist_trait
					has_trait = vampire_trait
					has_trait = blood_angel_curse_trait
					has_variable = folk_unlocked_var
				}
				NOR = {
					has_trait = brain_damaged
					has_trait = goblin_trait
					has_variable = on_orc_adventure
				}
			}
		}
	}

	allowed_trigger = {
		scope:target = {
			trigger_if = {
				limit = { is_adult = no }
				is_adult = yes
			}
			OR = {
				is_ruler = yes
				is_general = yes
				has_office = office_physician
				has_office = office_republic_physician
				has_office = office_apothecary
			}
		}
	}

	effect = {
		scope:actor = {
			trigger_event = { id = me_zorgo_magic_focus.1 }
		}
	}
}

omniluxia_magic_plant_growth_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		hidden:scope:target = {
			employer = scope:actor
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_plant_growth_spell
			has_variable = druid_t2_unlocked_var
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		scope:target = {
			has_variable = omni_pending_plant_growth_spell
			has_variable = druid_t2_unlocked_var
		}
	}

	province_target_trigger = {
		has_owner = yes
		owner = scope:actor
		NOT = { is_sea = yes }
		is_uninhabitable = no
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_plant_growth_spell
			remove_25_mana_effect = yes
		}
		scope:recipient = {
			add_province_modifier = {
				name = amten_druid_spell_t1_mod
				duration = 180
			}
		}
	}
}

omniluxia_magic_summon_dryads_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		hidden:scope:target = {
			employer = scope:actor
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_summon_dryads_spell
			has_variable = druid_t2_unlocked_var
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		scope:target = {
			has_variable = omni_pending_summon_dryads_spell
			has_variable = druid_t2_unlocked_var
		}
	}

	province_target_trigger = {
		has_owner = yes
		OR = {
			owner = scope:actor
			controller = scope:actor
		}
		NOT = { is_sea = yes }
		is_uninhabitable = no
		OR = {
			terrain = forest
			terrain = jungle
			terrain = eldritch_forest
		}
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_summon_dryads_spell
			remove_50_mana_effect = yes
		}
		create_unit = {
			navy = no
			name = "Dryad Auxiliaries"
			location = scope:recipient
			hidden_effect = {
				while = {
					count = 4
					add_subunit = heavy_infantry
				}
				while = {
					count = 4
					add_subunit = light_infantry
				}
				add_food = 800
				add_morale = 5
			}
		}
	}
}
