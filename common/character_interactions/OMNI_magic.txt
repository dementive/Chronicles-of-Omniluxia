# The old "Cast A Spell" character interaction has been removed.
# Character magic now lives on the header magic icon in the character window.

omniluxia_magic_plant_growth_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			employer = scope:actor
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_plant_growth_spell
			has_variable = druid_t2_unlocked_var
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			has_variable = omni_pending_plant_growth_spell
			has_variable = druid_t2_unlocked_var
		}
	}

	province_target_trigger = {
		has_owner = yes
		owner = scope:actor
		NOT = { is_sea = yes }
		is_uninhabitable = no
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_plant_growth_spell
			change_variable = {
				name = magic
				subtract = var:omni_plant_growth_cost
			}
			clamp_character_magic_to_max = yes
		}
		scope:recipient = {
			add_province_modifier = {
				name = amten_druid_spell_t1_mod
				duration = 180
			}
		}
	}
}

omniluxia_magic_summon_dryads_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			employer = scope:actor
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_summon_dryads_spell
			has_variable = druid_t2_unlocked_var
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			has_variable = omni_pending_summon_dryads_spell
			has_variable = druid_t2_unlocked_var
		}
	}

	province_target_trigger = {
		has_owner = yes
		OR = {
			owner = scope:actor
			controller = scope:actor
		}
		NOT = { is_sea = yes }
		is_uninhabitable = no
		OR = {
			terrain = forest
			terrain = jungle
			terrain = eldritch_forest
		}
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_summon_dryads_spell
			change_variable = {
				name = magic
				subtract = var:omni_summon_dryads_cost
			}
			clamp_character_magic_to_max = yes
		}
		create_unit = {
			navy = no
			name = "Dryad Auxiliaries"
			location = scope:recipient
			hidden_effect = {
				while = {
					count = 4
					add_subunit = heavy_infantry
				}
				while = {
					count = 4
					add_subunit = light_infantry
				}
				add_food = 800
				add_morale = 5
			}
		}
		scope:recipient = {
			if = {
				limit = { has_province_modifier = summon_dryads_mod }
				remove_province_modifier = summon_dryads_mod
			}
			add_province_modifier = {
				name = summon_dryads_mod
				duration = 365
			}
		}
	}
}
