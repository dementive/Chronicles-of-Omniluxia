# The old "Cast A Spell" character interaction has been removed.
# Character magic now lives on the header magic icon in the character window.

omniluxia_magic_plant_growth_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			OR = { country = scope:actor employer = scope:actor }
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_plant_growth_spell
			omni_knows_spell = { SPELL = plant_growth }
			omni_can_afford_spell = { BASE = 25 }
			has_variable = druid_t2_unlocked_var
			omni_character_is_eligible_caster = yes
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			has_variable = omni_pending_plant_growth_spell
			omni_knows_spell = { SPELL = plant_growth }
			omni_can_afford_spell = { BASE = 25 }
			has_variable = druid_t2_unlocked_var
			omni_character_is_eligible_caster = yes
		}
	}

	province_target_trigger = {
		has_owner = yes
		owner = scope:actor
		NOT = { is_sea = yes }
		is_uninhabitable = no
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_plant_growth_spell
			omni_spell_mana_cost_effect = { BASE = 25 }
		}
		scope:recipient = {
			if = { limit = { has_province_modifier = amten_druid_spell_t1_mod } remove_province_modifier = amten_druid_spell_t1_mod }
			add_province_modifier = {
				name = amten_druid_spell_t1_mod
				duration = 180
			}
		}
	}
}

omniluxia_magic_summon_dryads_target = {
	on_other_nation = no
	on_own_nation = yes
	close_ui = yes
	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_corruption"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			OR = { country = scope:actor employer = scope:actor }
			is_alive = yes
			is_adult = yes
			has_variable = omni_pending_summon_dryads_spell
			omni_knows_spell = { SPELL = summon_dryads }
			omni_can_afford_spell = { BASE = 50 }
			has_variable = druid_t2_unlocked_var
			omni_character_is_eligible_caster = yes
		}
		scope:actor = {
			is_ai = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			has_variable = omni_pending_summon_dryads_spell
			omni_knows_spell = { SPELL = summon_dryads }
			omni_can_afford_spell = { BASE = 50 }
			has_variable = druid_t2_unlocked_var
			omni_character_is_eligible_caster = yes
		}
	}

	province_target_trigger = {
		has_owner = yes
		OR = {
			owner = scope:actor
			controller = scope:actor
		}
		NOT = { is_sea = yes }
		is_uninhabitable = no
		OR = {
			terrain = forest
			terrain = jungle
			terrain = eldritch_forest
		}
	}

	effect = {
		scope:target = {
			remove_variable = omni_pending_summon_dryads_spell
			omni_spell_mana_cost_effect = { BASE = 50 }
		}
		# create_unit is a COUNTRY effect; the interaction root is a character.
		scope:actor = {
			create_unit = {
				navy = no
				name = "Dryad Auxiliaries"
				location = scope:recipient
				hidden_effect = {
					while = {
						count = omni_dryad_cohort_count
						add_subunit = heavy_infantry
					}
					add_food = 800
					add_morale = 5
				}
			}
		}
		scope:recipient = {
			if = {
				limit = { has_province_modifier = summon_dryads_mod }
				remove_province_modifier = summon_dryads_mod
			}
			add_province_modifier = {
				name = summon_dryads_mod
				duration = 365
			}
		}
	}
}
