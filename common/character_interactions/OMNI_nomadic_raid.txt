	on_other_nation = no
	on_own_nation = yes

	message = no

	sound = "event:/SFX/UI/Character/Generic/sfx_ui_character_scholar"

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		hidden:scope:target = {
			employer = scope:actor
			is_ruler = yes
			is_alive = yes
			is_adult = yes
			NOT = { employer = { tag = REB } }
			NOT = { employer = { tag = BAR } }
		}
		scope:actor = {
			is_ai = no
			OR = {
				government = nomadic_tribe
				government = tribal_chiefdom
				country_culture_group = orcish
				country_culture_group = eptelon
			}
			war = no
		}
	}

	allowed_trigger = {
		exists = scope:target
		scope:target = {
			trigger_if = {
				limit = { is_adult = no }
				is_adult = yes
			}
			is_ruler = yes
			wealth >= 0
			custom_tooltip = {
				text = is_on_omni_raid_tt
				NOT = { has_variable = omni_nomadic_raid_cooldown }
			}
		}
	}

	country_target_trigger = {
		has_land = yes
		is_subject = no
		in_diplomatic_range = scope:actor
		NOT = { this = scope:actor }
		NOT = { alliance_with = scope:actor }
		NOT = { has_variable = stop_raiding_me_every_day_please_var }
	}

	effect = {
		scope:actor = {
			trigger_event = { id =  me_zorgoomni.31 }
		}
		scope:target = {
			set_variable = {
				name = omni_nomadic_raid_cooldown
				days = 1825
			}
		}
	}
}
