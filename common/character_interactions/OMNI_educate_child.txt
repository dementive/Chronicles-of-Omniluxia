omni_educate_child = {
	on_own_nation = yes
	on_other_nation = no
	message = no

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		scope:target = {
			is_alive = yes
			is_adult = no
			employer = scope:actor
			NOT = { has_variable = has_been_educated }
		}
	}

	allowed_trigger = {
		scope:target = { loyalty >= 35 }
	}

	effect = {
		scope:target = {
			set_variable = has_been_educated
			# The four general focuses bank a modest mana reserve. The magical
			# focus is capacity-only, so it never grants free spells.
			change_variable = { name = magic add = 15 }
			change_variable = { name = magic_max add = 25 }
			change_variable = { name = magic_monthly add = 1 }
			clamp_character_magic_to_max = yes
		}
	}
}
