omni_educate_child = {
	on_own_nation = yes
	on_other_nation = no
	message = no

	potential_trigger = {
		exists = scope:actor
		exists = scope:target
		scope:target = {
			is_alive = yes
			is_adult = no
			employer = scope:actor
			NOR = { 
				has_variable = ec_land_var
				has_variable = ec_sea_var
				has_variable = ec_sophistry_var
				has_variable = ec_law_var
				has_variable = ec_admin_var
				has_variable = ec_trade_var
				has_variable = ec_secular_var
				has_variable = ec_religious_var
				has_variable = ec_omnic_var
				has_variable = ec_aldic_var
				has_variable = ec_amten_var
				has_variable = ec_melodian_var
			}
		}
	}

	allowed_trigger = {
		scope:target = { loyalty >= 35 }
	}

	effect = {
		scope:actor = {
			trigger_event = {
				id = omni_educate_child.1
			}
		}
	}
}
