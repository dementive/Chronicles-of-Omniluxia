# Generic Trade Expedition - script values
#
# Design notes:
#   Costs scale with OUR income (a large empire should feel the outlay).
#   Profits scale with the PARTNER's wealth and the DISTANCE sailed, so the
#   interesting decision is "where do we sail", not "how rich are we".
#   Every payout is multiplied by generic_trade_expedition_repeat_mult_svalue,
#   which decays with each completed expedition. The mission is repeatable, so
#   the cash must not be farmable; the Charter is the real reward.

########################################
# Scaling helpers
########################################

# 0.75 near / 1.0 same region / 1.5 far. Distance is the main profit lever.
generic_trade_expedition_distance_mult_svalue = {
	value = 1.0
	if = {
		limit = {
			exists = scope:generic_trade_expedition_home_port
			exists = scope:generic_trade_expedition_partner_port
			scope:generic_trade_expedition_home_port = {
				exists = area
				area = scope:generic_trade_expedition_partner_port.area
			}
		}
		multiply = 0.75
	}
	else_if = {
		limit = {
			exists = scope:generic_trade_expedition_home_port
			exists = scope:generic_trade_expedition_partner_port
			scope:generic_trade_expedition_home_port = {
				exists = region
				region = scope:generic_trade_expedition_partner_port.region
			}
		}
		multiply = 1.0
	}
	else = {
		multiply = 1.5
	}
}

# Diminishing returns across repeat runs. -15% per completed expedition,
# floored at 40%. This is the primary anti-farm guard on cash rewards.
generic_trade_expedition_repeat_mult_svalue = {
	value = 1
	if = {
		limit = { has_variable = generic_trade_expedition_runs }
		subtract = {
			value = var:generic_trade_expedition_runs
			multiply = 0.15
		}
	}
	min = 0.4
	max = 1
}

# How rich the destination is. Bounded so a superpower partner cannot be milked.
generic_trade_expedition_partner_wealth_svalue = {
	value = 0
	if = {
		limit = { exists = scope:generic_trade_expedition_partner }
		add = scope:generic_trade_expedition_partner.has_monthly_income
	}
	min = 10
	max = 400
}

########################################
# Risk
########################################

# Rolled against 100 at each voyage risk check. Mitigation tasks buy this down;
# distance and a hostile partner push it up. Floored at 3 so nothing is ever
# perfectly safe, capped at 60 so a long voyage is never hopeless.
generic_trade_expedition_risk_svalue = {
	value = 20
	multiply = generic_trade_expedition_distance_mult_svalue
	if = {
		limit = { has_country_modifier = generic_trade_expedition_provisioned_convoy_cmod }
		subtract = 5
	}
	if = {
		limit = { has_variable = generic_trade_expedition_charts_prepared }
		subtract = 4
	}
	if = {
		limit = { has_variable = generic_trade_expedition_escort_assigned }
		subtract = 8
	}
	if = {
		limit = { has_variable = generic_trade_expedition_stopover_secured }
		subtract = 4
	}
	if = {
		limit = { has_variable = generic_trade_expedition_partner_tariffs }
		add = 6
	}
	if = {
		limit = { has_variable = generic_trade_expedition_rival_present }
		add = 5
	}
	if = {
		limit = {
			exists = var:generic_trade_expedition_envoy_character
			var:generic_trade_expedition_envoy_character = { martial >= 8 }
		}
		subtract = 3
	}
	min = 3
	max = 60
}
