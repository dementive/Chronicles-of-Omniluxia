# Character scope. Best martial first, but each artifact already held this war
# counts as -10 martial so artifacts spread across commanders instead of piling up.
omni_ai_artifact_holder_score = {
	value = martial
	if = {
		limit = { has_variable = omni_ai_artifacts_held }
		subtract = {
			value = var:omni_ai_artifacts_held
			multiply = 10
		}
	}
}
