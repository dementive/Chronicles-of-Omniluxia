state_food_percent = {
    value = 0
    # Empty states have no capacity: do not evaluate their food ratio.
    if = {
        limit = { has_state_food_capacity > 0 }
        value = has_state_food
        divide = has_state_food_capacity
    }
    min = 0
    max = 1
}

unused_trade_routes_percent = {
    if = {
        limit = {
            trade_routes <= 0
        }
        value = 0
    }
    else = {
        value = unused_trade_routes
        divide = trade_routes
    }
}

state_population_svalue = {
    value = 0
    every_state_province = {
        add = total_population
    }
}
