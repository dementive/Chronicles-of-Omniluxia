    value = 0

    if = { limit = { omni_sf_relationship_trigger = yes has_variable = omni_sf_favor }

        value = var:omni_sf_favor

    }

    min = 0 max = 100

}

omni_sf_favor_fraction = { value = omni_sf_favor_value divide = 100 }
