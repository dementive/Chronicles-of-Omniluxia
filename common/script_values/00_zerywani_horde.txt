# Power/wealth score used by the zerywani horde to pick its invasion target:
# "the most powerful and wealthy and strongest nation" on their continent.
zerywani_horde_power_score_svalue = {
	value = 0
	add = {
		value = num_of_cities
		multiply = 20
	}
	add = {
		value = treasury
		multiply = 0.1
	}
	add = {
		value = max_manpower
		multiply = 0.05
	}
}

# How many eligible zerywani tribes exist in total, used to scale the size of the horde.
zerywani_horde_tribe_count_svalue = {
	value = 0
	every_country = {
		limit = { has_variable = is_zerywani_horde_participant }
		add = 1
	}
}
