omni_lich_ascension_required_months = { value = 300 }

# Country scope, used by the decision report.
omni_lich_ascension_progress = {
	value = 0
	if = {
		limit = { current_ruler = { has_variable = omni_lich_ascension_months } }
		add = current_ruler.var:omni_lich_ascension_months
	}
}

omni_lich_ascension_percent = {
	value = omni_lich_ascension_progress
	multiply = 100
	divide = omni_lich_ascension_required_months
}
