# Dynamic mana costs for magic spells.
# Evaluated in the CASTER (character) scope.
# Cost scales with the terrain the caster's army occupies and the caster's rank:
#   harsh terrain  x1.5 (+50%)   fertile terrain x0.85 (-15%)   normal x1.0
#   archmage rank  x0.6 (-40%)

# --- Mana modifier readers -------------------------------------------------
# great_work_*_workrate_character_modifier are PERCENT-type modifiers in the
# engine, so their values are STORED divided by 100 (see
# common/modifiers/0000_magic_modifiers.txt). Always read them through these
# two script values so the x100 is applied in exactly one place.
#
# REGEN RATE (rebalanced July 2026): the multipliers here carry a x5 rate factor
# on top of the x100 percent-decode. Before the mana-tick fix in
# change_country_scope_magic_vars, the mana sweep ran once PER COUNTRY, so every
# caster sat pinned at magic_max and the rate was irrelevant. With the sweep
# correctly running once per month, the raw trait values would have given a
# mage_trait caster (0.01) just 1 mana per month - more than two years to afford
# a single 25-mana tier-1 spell.
#
# The x5 lives here rather than on the traits because
# great_work_*_workrate_character_modifier are real engine modifiers used
# elsewhere; scaling at the reader keeps trait balance untouched and keeps the
# displayed and granted rate identical, so the tooltip stays truthful.
# Resulting rates (max mana is 100 + freemen bonus):
#   mage_trait      0.01 -> 5/month  (60/yr)
#   diviner/etc.    0.02 -> 10/month (120/yr)
#   archmage_trait  0.03 -> 15/month (180/yr)
# Battle spells cost 15-75 and have a 1-day cooldown, so mana - not the
# cooldown - is now the real limiter on AI and player casting. That is intended.
# To retune the whole system's pace, change the x5 in both values together.
omni_character_monthly_mana_value = {
	value = modifier:great_work_tribals_workrate_character_modifier
	multiply = 500
}

# Retained for scripts that may use a quarterly grant. The normal regeneration
# path uses omni_character_monthly_mana_value above.
omni_character_quarterly_mana_value = {
	value = modifier:great_work_tribals_workrate_character_modifier
	multiply = 1500
}

omni_character_max_mana_bonus_value = {
	value = modifier:great_work_freemen_workrate_character_modifier
	multiply = 100
}

omni_character_magic_max_value = {
	value = 100
	add = omni_character_max_mana_bonus_value
}

# Generic scaled cost used by every spell. Reads the flat base cost from
# var:omni_spell_base (set immediately before by omni_spell_mana_cost_effect)
# and applies the same terrain + rank scaling as the bespoke spell costs.
omni_scaled_spell_cost_value = {
	value = var:omni_spell_base
	if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = { omni_harsh_nature_terrain_trigger = yes }
			}
		}
		multiply = 1.5
	}
	else_if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = { omni_fertile_nature_terrain_trigger = yes }
			}
		}
		multiply = 0.85
	}
	if = {
		limit = { has_trait = archmage_trait }
		multiply = 0.6
	}
}

# Size of the dryad army raised by Summon Dryads (cohorts; ~500 dryads each).
# Evaluated in COUNTRY scope - reads the caster via scope:target.
#   plain mage          3 cohorts  (~1500 dryads)
#   druid OR archmage   6 cohorts  (~3000 dryads)
#   druid AND archmage  10 cohorts (~5000 dryads)
omni_dryad_cohort_count = {
	value = 3
	if = {
		limit = {
			scope:target = {
				OR = {
					has_trait = druid_trait
					has_trait = archmage_trait
				}
			}
		}
		add = 3
	}
	if = {
		limit = {
			scope:target = {
				has_trait = druid_trait
				has_trait = archmage_trait
			}
		}
		add = 4
	}
}


# How much BASE spell cost the caster can currently afford. This is var:magic
# with the SAME terrain + rank scaling as omni_scaled_spell_cost_value applied
# in reverse (divide instead of multiply), so that:
#   omni_spell_affordable_budget >= BASE   <=>   magic >= scaled cost of BASE
# Used by the omni_can_afford_spell trigger to gate spell options. Evaluated in
# the CASTER (character) scope.
omni_spell_affordable_budget = {
	value = 0
	if = {
		limit = { has_variable = magic }
		add = var:magic
	}
	if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = { omni_harsh_nature_terrain_trigger = yes }
			}
		}
		divide = 1.5
	}
	else_if = {
		limit = {
			any_character_unit = {
				is_army = yes
				unit_location = { omni_fertile_nature_terrain_trigger = yes }
			}
		}
		divide = 0.85
	}
	if = {
		limit = { has_trait = archmage_trait }
		divide = 0.6
	}
}
