omni_hc_authority = {
        value = 0
        # Incumbents read their own institution, independent of a cached see link.
        if = { limit = { omni_hc_shannah_trigger = yes }
            value = 50
            if = { limit = { has_variable = omni_hc_authority_base } value = var:omni_hc_authority_base }
            if = { limit = { NOT = { owns = 2706 } } subtract = 10 }
        }
        else_if = { limit = { exists = var:omni_hc_see var:omni_hc_see = { omni_hc_shannah_trigger = yes } }
            value = 50
            if = { limit = { var:omni_hc_see = { has_variable = omni_hc_authority_base } }
                value = var:omni_hc_see.var:omni_hc_authority_base
            }
            if = { limit = { var:omni_hc_see = { NOT = { owns = 2706 } } } subtract = 10 }
        }
        min = 0 max = 100
    }
    omni_hc_authority_fraction = { value = omni_hc_authority divide = 100 }
