# Direct country metrics, following 00_senate_objective and 00_zerywani_horde.
# Territories provide a peacetime floor when levies are unraised. No unit scan.
omni_crusade_strength_svalue = {
    value = num_of_cohorts
    multiply = 2
    add = num_of_cities
    min = 1
}

# Council reach for non-border targets, in map-distance units (base diplo range: 800).
omni_crusade_target_distance_svalue = { value = 250 }

# Offerings reflect the host's territorial size, snapshotted at confirmation.
omni_crusade_offering_svalue = {
    # Floor keeps the offering a real choice against pledging troops or declining.
    value = 25
    if = { limit = { num_of_cities >= 200 } value = 150 }
    else_if = { limit = { num_of_cities >= 100 } value = 100 }
    else_if = { limit = { num_of_cities >= 50 } value = 80 }
    else_if = { limit = { num_of_cities >= 25 } value = 60 }
    else_if = { limit = { num_of_cities >= 10 } value = 40 }
}
# Old expeditions retain their original 100-gold price and refund amount.
omni_crusade_current_offering_svalue = {
    value = 100
    if = {
        limit = { scope:omni_crusade_actor = { has_variable = omni_crusade_offering } }
        value = scope:omni_crusade_actor.var:omni_crusade_offering
    }
}
omni_crusade_paid_offering_svalue = {
    value = 100
    if = { limit = { has_variable = omni_crusade_paid_offering } value = var:omni_crusade_paid_offering }
}
omni_crusade_negative_paid_offering_svalue = { value = omni_crusade_paid_offering_svalue multiply = -1 }
