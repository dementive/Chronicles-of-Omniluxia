# Character scope. A living AI mage who knows the Rite of Lichdom, is not yet
# a lich, can pay its 350 mana, and is old enough to fear death.
omni_ai_lichdom_candidate = {
	# Cheapest and rarest test first: almost no one knows the rite.
	has_variable = rite_of_lichdom_known_var
	is_alive = yes
	NOT = { has_trait = lich_trait }
	omni_knows_spell = { SPELL = rite_of_lichdom }
	omni_character_is_eligible_caster = yes
	omni_can_afford_spell = { BASE = 350 }
	OR = {
		age >= 50
		health < 30
	}
}

# Character scope. A living AI mage who knows the Rite of the Oracle, is not
# yet an Oracle and can pay its 350 mana. A realm needs only one Oracle.
omni_ai_oracle_candidate = {
	has_variable = rite_of_the_oracle_known_var
	is_alive = yes
	NOT = { has_trait = oracle_trait }
	omni_character_is_eligible_caster = yes
	omni_can_afford_spell = { BASE = 350 }
	employer = { NOT = { has_variable = omni_has_oracle } }
}
