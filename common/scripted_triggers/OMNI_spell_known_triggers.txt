# Per-spell knowledge gates for the Magic Learning Rework.
# SPELL is substituted by the caller (for example fireball_known_var).
omni_knows_spell = {
	has_variable = $SPELL$_known_var
}

# Jomini's variable audit does not connect the parameter-expanded setter in
# omni_learn_spell with parameter-expanded consumers. These literal consumers
# keep the audit accurate and are ready for the eventual spell options.
omni_knows_bloodlust = { has_variable = bloodlust_known_var }
omni_knows_berserker_rage = { has_variable = berserker_rage_known_var }
omni_knows_water_walker = { has_variable = water_walker_known_var }
omni_knows_dreaming = { has_variable = dreaming_known_var }
omni_knows_summon_eagle_of_zanis = { has_variable = summon_eagle_of_zanis_known_var }
omni_knows_helluvius_peace = { has_variable = helluvius_peace_known_var }
omni_knows_hellas_providence = { has_variable = hellas_providence_known_var }
omni_knows_palpan = { has_variable = palpan_known_var }

# NOTE: palpan_known_var is intentionally not gated by any omni_knows_spell = { SPELL = palpan }
# check. "palpan" is granted (omni_learn_spell) in folk_travel_events.txt, magic_studying_events.txt
# and zorgoomni.txt, but it has no dedicated spell-effect option in the Melodian School event chain
# (unlike mended_anima_soul/astral_search_soul/silver_tongue_charm, which now correctly gate their
# _school options via the bare id). Learning palpan currently only flips palpan_unlocked_var (a
# separate, already-used school-access flag) and this lore flag; treat palpan_known_var as a
# lore-only placeholder until a real Melodian spell effect is designed for it. Do not delete.

omni_discipline_mastered = {
	has_variable = $DISCIPLINE$_mastered_var
}

omni_any_discipline_mastered = {
	OR = {
		has_variable = diviner_mastered_var
		has_variable = hierophant_mastered_var
		has_variable = cleric_mastered_var
		has_variable = illusionist_mastered_var
		has_variable = elementalist_mastered_var
		has_variable = druid_mastered_var
		has_variable = necromancer_mastered_var
		has_variable = warlock_mastered_var
	}
}

omni_school_entered = {
	var:$SCHOOL$_affinity_var > 0
}

# FAMILY_CODE: 1 = omnic, 2 = aldic, 3 = amten, 4 = melodian.
# Reads omni_best_affinity_var, resolved in effect scope by
# omni_update_best_affinity. Comparing one affinity var against another directly
# is illegal in Jomini triggers ("Illegal use of operator >=").
omni_primary_school = {
	has_variable = omni_best_affinity_var
	var:omni_best_affinity_var = $FAMILY_CODE$
}
