# The Goblin Invasion of the World's Lungs (namespace goblin_invasion).
# The Root of the First Spring (great work + root_of_first_spring_mod) stands at p:2356, Venerurich.

goblin_invasion_root_stands_trigger = {
	NOT = { has_global_variable = root_of_first_spring_felled }
	p:2356 = {
		has_owner = yes
		has_province_modifier = root_of_first_spring_mod
	}
}

# Country scope: a goblin realm that can lead the march on the Root.
goblin_invasion_eligible_trigger = {
	country_culture_group = goblin
	is_subject = no
	num_of_cities >= 1
	NOT = { has_variable = is_goblin_invasion_leader }
	NOT = { owns = 2356 }
}

# Country scope. The same realm must directly own 10 of 15 and 6 of 10 territories.
goblin_invasion_controls_skyreach_trigger = {
	save_temporary_scope_as = skyreach_claimant
	p:3180 = { # passages_of_skyreach_1
		area = {
			any_area_province = {
				count >= 10
				owner = scope:skyreach_claimant
			}
		}
	}
	p:3340 = { # passages_of_skyreach_3
		area = {
			any_area_province = {
				count >= 6
				owner = scope:skyreach_claimant
			}
		}
	}
}
