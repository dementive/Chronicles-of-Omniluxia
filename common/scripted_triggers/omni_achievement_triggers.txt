### OMNILUXIA CUSTOM ACHIEVEMENTS - SHARED CONDITION TRIGGERS ###
#
# Single source of truth for each achievement's unlock condition. Referenced by:
#   - common/scripted_effects/00_achievement_effects.txt  (the auto-unlock check)
#   - common/scripted_guis/sguis_omni_achievements.txt     (window "still possible")
# Keeping the condition in one place means the window and the unlock logic can
# never drift apart. All triggers evaluate in COUNTRY scope.

# --- Race / culture-group helpers ---
omni_is_elven_realm = {
    OR = {
        country_culture_group = high_elves
        country_culture_group = high_half_elves
        country_culture_group = western_half_elves
        country_culture_group = forest_elves
        country_culture_group = southern_forest_elves
        country_culture_group = morrigon_forest_elves
        country_culture_group = sea_elves
        country_culture_group = sunset_elves
        country_culture_group = drow
    }
}

omni_is_dwarven_realm = {
    OR = {
        country_culture_group = common_dwarves
        country_culture_group = gold_dwarves
        country_culture_group = silver_dwarves
        country_culture_group = castaway_dwarves
    }
}

omni_is_orcish_realm = {
    OR = {
        country_culture_group = orcish
        country_culture_group = orcish_humano
        country_culture_group = hobgoblin
        country_culture_group = goblin
    }
}

omni_has_court_mage = {
    any_character = {
        OR = {
            has_trait = mage_trait
            has_trait = archmage_trait
        }
    }
}

# --- Per-achievement unlock conditions (omni_ach_<key>_trigger) ---
omni_ach_royal_progress_trigger = {
    has_variable = omni_tut_done
}

omni_ach_court_mage_trigger = {
    omni_has_court_mage = yes
}

omni_ach_awakening_trigger = {
    has_variable = magic
    var:magic >= 50
}

omni_ach_elven_realm_trigger = {
    omni_is_elven_realm = yes
    num_of_cities >= 25
}

omni_ach_dwarven_hold_trigger = {
    omni_is_dwarven_realm = yes
    num_of_cities >= 20
}

        has_trait = archmage_trait
    }
}

    has_variable = magic

}

}

}

    omni_is_orcish_realm = yes

}

        has_trait = archmage_trait
    }
}

}

    has_variable = magic

    }
}
