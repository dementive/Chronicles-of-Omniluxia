# Province scope. Visiting armies and navies follow the same access rules as city visits.
omni_mage_tower_territory_access = {
	has_building = mage_tower_building
	has_owner = yes
	OR = {
		owner = scope:omni_city_visitor.employer
		controller = scope:omni_city_visitor.employer
		scope:omni_city_visitor.employer = { has_military_access = prev.owner }
	}
}

# Character event host. Recheck access against the specific tower on the open shelf.
omni_mage_tower_visit_is_current = {
	exists = scope:omni_mage_tower_site
	scope:omni_city_visitor = {
		is_alive = yes
		exists = employer
		employer = root.employer
		OR = {
			location = { this = scope:omni_mage_tower_site }
			any_character_unit = {
				OR = { is_army = yes is_navy = yes }
				unit_location = { this = scope:omni_mage_tower_site }
			}
		}
	}
	scope:omni_mage_tower_site = { omni_mage_tower_territory_access = yes }
}

# PRICE names a script value evaluated on the buyer's training.
omni_mage_tower_can_pay = {
	scope:omni_city_visitor = {
		trigger_if = {
			limit = { is_ruler = yes }
			employer = { treasury >= scope:omni_city_visitor.$PRICE$ }
		}
		trigger_else = { wealth >= $PRICE$ }
	}
}
