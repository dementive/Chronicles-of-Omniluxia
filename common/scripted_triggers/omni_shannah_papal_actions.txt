# Shared interaction eligibility. TARGET is a character scope expression.
omni_shannah_excommunicate_potential_trigger = {

        exists = scope:actor
        exists = $TARGET$
        scope:actor = {
            has_land = yes
            religion = helluvian
            exists = current_ruler
            current_ruler = { has_trait = shannah_trait }
        }
        $TARGET$ = {
            is_alive = yes
            is_ruler = yes
            exists = employer
            NOT = { employer = scope:actor }
            employer = {
                has_land = yes
                religion = helluvian
                NOT = { tag = REB }
                NOT = { tag = BAR }
                NOT = { tag = MER }
                NOT = { tag = PIR }
            }
        }
        $TARGET$.employer = { NOT = { has_country_modifier = omni_shannah_excommunicated } }
    
}

omni_shannah_excommunicate_allowed_trigger = {
        scope:actor = { omni_ruler_available_for_court_action_trigger = yes }
        scope:actor = {
            political_influence >= 75 # BALANCE: was 25
            NOT = { has_variable = omni_shannah_excommunication_cooldown }
        }
    
}

omni_shannah_crown_offer_current_trigger = {
    exists = scope:omni_crown_offer_see
    exists = scope:omni_crown_offer_serial
    scope:omni_crown_offer_see = {
        has_variable = omni_shannah_crown_offer_pending
        has_variable = omni_shannah_crown_offer_serial
        var:omni_shannah_crown_offer_serial = scope:omni_crown_offer_serial
    }
}

omni_shannah_crown_offer_accept_trigger = {
    omni_shannah_crown_offer_current_trigger = yes
    exists = scope:omni_crown_offer_shannah
    exists = scope:omni_crown_offer_ruler
    current_ruler = scope:omni_crown_offer_ruler
    scope:omni_crown_offer_see = {
        current_ruler = scope:omni_crown_offer_shannah
        save_temporary_scope_as = actor
        # An AI council may need to honor its reserve days after making the offer.
        OR = { is_ai = no political_influence >= 60 }
    }
    scope:omni_crown_offer_ruler = { save_temporary_scope_as = target }
    omni_shannah_recognize_crown_potential_trigger = { TARGET = scope:target }
    omni_shannah_recognize_crown_allowed_trigger = yes
}
