# School + tier gating for the battle, siege, and friendly spell menus.
# Evaluated in the CASTER (character) scope.
#
# Tier mapping: a "tier 1 spell" needs the school's T2 study unlock (the first
# spell-granting rank), tier 2 needs T3, tier 3 needs T4. Tier 4 spells
# additionally require archmage_trait (checked at the option).
# School specialties (Magonis, Charm, etc.) count as tier 1 of their school.

# ----- Omnic (Diviner / Hierophant) -----
omni_caster_omnic_t1 = {
	OR = {
		has_variable = diviner_t2_unlocked_var
		has_variable = hierophant_t2_unlocked_var
		has_variable = magonis_unlocked_var
		has_variable = taanka_unlocked_var
		has_variable = dragonscale_unlocked_var
		has_variable = helluvian_holy_unlocked_var
	}
}
omni_caster_omnic_t2 = {
	OR = {
		has_variable = diviner_t3_unlocked_var
		has_variable = hierophant_t3_unlocked_var
	}
}
omni_caster_omnic_t3 = {
	OR = {
		has_variable = diviner_t4_unlocked_var
		has_variable = hierophant_t4_unlocked_var
	}
}

# Tier 1 including folk practitioners - folk magic counts as basic training,
# so hedge-mages keep access to tier 1 buffs and siege workings.

		omni_caster_any_school_t1 = yes
