# Character scope. Academy access follows the character's current employer.
omni_can_learn_academy_magic = {
	is_alive = yes
	exists = employer
	employer = { NOT = { has_law = magic_banned } }
	trigger_if = {
		limit = { employer = { has_law = magic_mages_guilds } }
		OR = { is_ruler = yes has_doctor_office_trigger = yes }
	}
}

# Country scope. The AI enroller saves its student before the first event.
omni_academy_course_student_allowed = {
	trigger_if = {
		limit = { exists = scope:omni_study_target }
		scope:omni_study_target = { employer = root omni_can_learn_academy_magic = yes }
	}
	trigger_else = {
		exists = scope:target
		scope:target = { employer = root omni_can_learn_academy_magic = yes }
	}
}

# True while a character has an active magic-study course. Course effects use
# this after awarding progress: when a tier-up clears the study flags, no follow-up
# event is queued and the course ends immediately.
omni_has_active_magic_study_course = {
	OR = {
		has_variable = omni_studying_diviner_var
		has_variable = omni_studying_hierophant_var
		has_variable = omni_studying_cleric_var
		has_variable = omni_studying_illusionist_var
		has_variable = omni_studying_elementalist_var
		has_variable = omni_studying_druid_var
		has_variable = omni_studying_necromancer_var
		has_variable = omni_studying_warlock_var
		has_variable = omni_studying_magonis_var
		has_variable = omni_studying_taanka_var
		has_variable = omni_studying_dragonscale_var
		has_variable = omni_studying_helluvian_holy_var
		has_variable = omni_studying_charm_magic_var
		has_variable = omni_studying_silver_dwarf_magic_var
		has_variable = omni_studying_palpan_curse_var
		has_variable = omni_studying_soul_magic_var
	}
}

# Character scope. Recurring Spellbook point study uses its own timer.
omni_academy_can_start_course = {
	omni_can_learn_academy_magic = yes
	NOT = { has_variable = omni_studying_magic_cooldown }
	NOT = { omni_has_active_magic_study_course = yes }
}

# Country scope. Ignore any queued course event after its student finishes.
omni_academy_course_is_active = {
	omni_academy_course_student_allowed = yes
	trigger_if = {
		limit = { exists = scope:omni_study_target }
		scope:omni_study_target = { omni_has_active_magic_study_course = yes }
	}
	trigger_else = {
		scope:target = { omni_has_active_magic_study_course = yes }
	}
}

# Character scope. Recurring Study Spells pays out only for a trained mage
# (Tier I access in any family) whose employer owns a Magic Academy. Shared by
# the player menu option, the yearly omni_items.90 tick and the AI auto-enrol.
omni_can_continue_spell_study = {
	omni_can_learn_academy_magic = yes
	omni_sb_any_t1_access = yes
	employer = { any_owned_province = { has_building = magic_academy_building } }
}
