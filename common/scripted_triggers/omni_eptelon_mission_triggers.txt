# Country scope. Triggers for 03_eptelon_generic (The Beleaguered Isle).

# Heritage for 03_eptelon_generic, recorded in on_start. The fallback covers trees started earlier.
omni_eptelon_heritage_recorded_trigger = {
    OR = {
        has_variable = eptelon_heritage_aeptelonian
        has_variable = eptelon_heritage_native
        has_variable = eptelon_heritage_heptelonian
        has_variable = eptelon_heritage_colonial
    }
}

omni_eptelon_heritage_aeptelonian_trigger = {
    OR = {
        has_variable = eptelon_heritage_aeptelonian
        AND = {
            omni_eptelon_heritage_recorded_trigger = no
            primary_culture = aeptelonians
        }
    }
}

omni_eptelon_heritage_native_trigger = {
    OR = {
        has_variable = eptelon_heritage_native
        AND = {
            omni_eptelon_heritage_recorded_trigger = no
            country_culture_group = eptelon
            NOT = { primary_culture = aeptelonians }
        }
    }
}

omni_eptelon_heritage_heptelonian_trigger = {
    OR = {
        has_variable = eptelon_heritage_heptelonian
        AND = {
            omni_eptelon_heritage_recorded_trigger = no
            country_culture_group = horteonian
            NOT = { primary_culture = aeptelonians }
        }
    }
}

omni_eptelon_heritage_colonial_trigger = {
    OR = {
        has_variable = eptelon_heritage_colonial
        AND = {
            omni_eptelon_heritage_recorded_trigger = no
            OR = {
                country_culture_group = high_elves
                country_culture_group = high_half_elves
            }
            NOT = { primary_culture = aeptelonians }
        }
    }
}

# Country-culture scope. The seven native Eptelonian cultures.
omni_eptelon_native_country_culture_trigger = {
    OR = {
        is_culture = ghedemi
        is_culture = kuazien
        is_culture = suru
        is_culture = ghergian
        is_culture = edertanian
        is_culture = dagaison
        is_culture = tolesucani
    }
}
