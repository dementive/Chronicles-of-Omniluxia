# Country scope. The sovereign flag is set when a wild dragon claims a realm.
omni_has_casting_dragon_sovereign = {
    government = omni_dragon_dominion
    exists = current_ruler
    current_ruler = {
        is_alive = yes
        has_variable = omni_dragon_sovereign
    }
}

omni_can_cast_dragon_magic = {
    omni_has_casting_dragon_sovereign = yes
    NOT = { has_variable = omni_dragon_magic_recovery }
    NOT = { has_variable = omni_dragon_cd_$SPELL$ }
}

omni_can_cast_dragon_cinderstorm = {
    omni_can_cast_dragon_magic = { SPELL = cinderstorm }
    OR = { war = yes has_civil_war = yes }
}

omni_can_cast_dragon_aegis_of_scales = {
    omni_can_cast_dragon_magic = { SPELL = aegis_of_scales }
    OR = { war = yes has_civil_war = yes }
}

omni_can_cast_dragon_hearthfire_blessing = {
    omni_can_cast_dragon_magic = { SPELL = hearthfire_blessing }
    war = no
    has_civil_war = no
}
