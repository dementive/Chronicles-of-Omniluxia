# Province scope. Shared by the menu, destination resolver, and final effect.
omni_kinweaving_valid_province = {
    has_owner = yes
    owner = $COUNTRY$
    controller = $COUNTRY$
    NOT = { is_sea = yes }
    is_uninhabitable = no
    NOT = { has_province_modifier = omni_kinweaving_cooldown }
    any_pops_in_province = {
        NOT = { pop_culture = $COUNTRY$.culture }
    }
}

# Character scope inside the country-rooted self-casting menu.
omni_can_cast_kinweaving = {
    is_alive = yes
    is_adult = yes
    employer = root
    omni_character_is_eligible_caster = yes
    omni_knows_spell = { SPELL = kinweaving }
    omni_can_afford_spell = { BASE = 100 }
    OR = {
        any_character_unit = {
            is_army = yes
            unit_location = { omni_kinweaving_valid_province = { COUNTRY = root } }
        }
        location = { omni_kinweaving_valid_province = { COUNTRY = root } }
    }
}
