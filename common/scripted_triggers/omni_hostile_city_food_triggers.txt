	OR = {
		AND = { exists = scope:omni_city_visitor NOT = { this = scope:omni_city_visitor } scope:omni_city_visitor = { wealth >= 150 } }
		AND = { OR = { NOT = { exists = scope:omni_city_visitor } this = scope:omni_city_visitor } exists = root.employer root.employer = { treasury >= 150 } }
	}
}
