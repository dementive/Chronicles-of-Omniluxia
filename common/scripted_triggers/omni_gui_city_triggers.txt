# Country-rooted GUI checks retain the visiting character for access rules.
omni_gui_character_can_visit_city = {
	is_alive = yes
	exists = employer
	OR = {
		location = {
			has_owner = yes
			OR = {
				owner = scope:omni_gui_character.employer
				controller = scope:omni_gui_character.employer
				scope:omni_gui_character.employer = { has_military_access = prev.owner }
			}
		}
		any_character_unit = {
			OR = {
				is_army = yes
				is_navy = yes
			}
			unit_location = {
				has_owner = yes
				OR = {
					owner = scope:omni_gui_character.employer
					controller = scope:omni_gui_character.employer
					scope:omni_gui_character.employer = { has_military_access = prev.owner }
				}
			}
		}
	}
}

omni_gui_character_local_city_has_building = {
	omni_gui_character_can_visit_city = yes
	OR = {
		location = {
			has_building = $BUILDING$
		}
		any_character_unit = {
			OR = {
				is_army = yes
				is_navy = yes
			}
			unit_location = {
				has_building = $BUILDING$
			}
		}
	}
}
