omni_portrait_scene_eligible = {
    is_alive = yes
    prisoner = no
    is_general = no
    exists = location
}

# Province-rooted. New rebuilds persist across later conquests; old saves use the owner flag.
omni_tiuban_ziggurat_is_rebuilt = {
    OR = {
        has_variable = omni_tiuban_ziggurat_rebuilt
        AND = { has_owner = yes owner = { has_variable = rebuilt_azar_ziggurat_var } }
    }
}

# Character-rooted: read the decision flag directly, without a fixed province re-scope.
omni_portrait_tiuban_rebuilt = {
    OR = {
        AND = { exists = employer employer = { has_variable = rebuilt_azar_ziggurat_var } }
        location = { omni_tiuban_ziggurat_is_rebuilt = yes }
    }
}
