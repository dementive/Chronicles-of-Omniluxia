# Country scope; actor/target scopes are reconstructed from country variables.
# Invictus flavor_cap.3 supplies the war membership/leadership pattern.
omni_shannah_office_trigger = {
    has_land = yes
    religion = helluvian
    government = papacy
    NOT = { has_variable = omni_shannah_deposed }
}

omni_crusade_target_trigger = {
    has_land = yes
    is_subject = no
    has_civil_war = no
    NOT = { religion = helluvian }
    NOT = { this = scope:omni_crusade_actor }
    in_diplomatic_range = scope:omni_crusade_actor
    # Neighbors remain eligible even when a sprawling realm's capital is far.
    # Other targets must have a nearby capital; this also allows coastal reach.
    OR = {
        any_neighbour_country = { this = scope:omni_crusade_actor }
        capital_scope = {
            distance_from = {
                province = scope:omni_crusade_actor.capital_scope
                value <= omni_crusade_target_distance_svalue
            }
        }
    }
    NOT = { alliance_with = scope:omni_crusade_actor }
    NOT = { has_truce_with = scope:omni_crusade_actor }
    NOT = { war_with = scope:omni_crusade_actor }
}

omni_crusade_invitee_trigger = {
    NOR = { has_variable = omni_crusade_defense_invited has_variable = omni_crusade_defense_pledged has_variable = omni_crusade_defense_refused }
    has_land = yes
    religion = helluvian
    is_subject = no
    has_civil_war = no
    NOT = { this = scope:omni_crusade_actor }
    in_diplomatic_range = scope:omni_crusade_actor
    in_diplomatic_range = scope:omni_crusade_target
    NOT = { war_with = scope:omni_crusade_actor }
    NOT = { war_with = scope:omni_crusade_target }
    NOT = { alliance_with = scope:omni_crusade_target }
    NOT = { has_truce_with = scope:omni_crusade_target }
}

omni_crusade_host_ready_trigger = {
    has_land = yes
    religion = helluvian
    is_subject = no
    has_civil_war = no
    war = no
    exists = current_ruler
    current_ruler = { has_trait = shannah_trait }
}

# Preparation survives temporary launch blockers. Peace, diplomatic range,
# civil wars and treaties are rechecked when actually declaring, not used as
# reasons to discard months of preparation on every country pulse.
omni_crusade_preparation_valid_trigger = {
    has_land = yes
    religion = helluvian
    is_subject = no
    exists = current_ruler
    current_ruler = { has_trait = shannah_trait }
    exists = var:omni_crusade_target
    var:omni_crusade_target = {
        has_land = yes
        is_subject = no
        NOT = { religion = helluvian }
    }
}

# Used before declaration, both at confirmation and after preparation.
omni_crusade_target_still_valid_trigger = {
    exists = var:omni_crusade_target
    var:omni_crusade_target = { omni_crusade_target_trigger = yes }
}

# No generic random war fallback. The original host must still lead the war
# against its recorded target. Concurrent crusades are globally locked out.
omni_crusade_matching_war_trigger = {
    is_war_leader = scope:omni_crusade_actor
    any_war_attacker = { this = scope:omni_crusade_actor }
    any_war_defender = { this = scope:omni_crusade_target }
}

omni_crusade_invitation_live_trigger = {
    has_variable = omni_crusade_invited
    exists = var:omni_crusade_host
    var:omni_crusade_host = {
        this = scope:omni_crusade_actor
        has_variable = omni_crusade_preparing
        NOT = { has_variable = omni_crusade_declaring }
        exists = var:omni_crusade_target
        var:omni_crusade_target = { this = scope:omni_crusade_target }
    }
}

# Defensive eligibility is independent of attacker faith; commitments are exclusive.
omni_crusade_defender_trigger = {
    has_land = yes
    is_subject = no
    has_civil_war = no
    NOT = { this = scope:omni_crusade_actor }
    NOT = { this = scope:omni_crusade_target }
    in_diplomatic_range = scope:omni_crusade_target
    in_diplomatic_range = scope:omni_crusade_actor
    NOT = { war_with = scope:omni_crusade_actor }
    NOT = { war_with = scope:omni_crusade_target }
    NOT = { alliance_with = scope:omni_crusade_actor }
    NOT = { has_truce_with = scope:omni_crusade_actor }
    NOR = { has_variable = omni_crusade_pledged has_variable = omni_crusade_donated }
}

        this = scope:omni_crusade_actor
        has_variable = omni_crusade_preparing
        NOT = { has_variable = omni_crusade_declaring }
        exists = var:omni_crusade_target
        var:omni_crusade_target = { this = scope:omni_crusade_target }
    }
}
