# Direct parent checks also identify children born before lineage was recorded.
omni_has_dragon_identity = {
    OR = {
        has_variable = omni_dragon_sovereign
        has_variable = omni_dragon_descendant
        has_variable = omni_wild_dragon_self
    }
}

omni_is_dragon_character = {
    OR = {
        omni_has_dragon_identity = yes
        AND = { exists = father father = { omni_has_dragon_identity = yes } }
        AND = { exists = mother mother = { omni_has_dragon_identity = yes } }
    }
}
