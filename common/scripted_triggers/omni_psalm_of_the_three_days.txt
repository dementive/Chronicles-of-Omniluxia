# Country scope. A realm whose masters sing the Psalm of the Three Days.
omni_psalm_order_realm = {
	OR = {
		tag = BLZ
		ob_country = yes
	}
}

# Country scope. A master of the Order has not yet learned the Psalm.
omni_psalm_master_needs_teaching = {
	omni_psalm_order_realm = yes
	OR = {
		AND = {
			exists = current_ruler
			current_ruler = { NOT = { omni_knows_spell = { SPELL = psalm_of_the_three_days } } }
		}
		AND = {
			exists = current_co_ruler
			current_co_ruler = { NOT = { omni_knows_spell = { SPELL = psalm_of_the_three_days } } }
		}
	}
}
