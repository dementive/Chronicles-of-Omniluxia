# Country scope. The personal faith AND state faith match existing power recipients.

omni_sf_petitioner_trigger = {

    has_land = yes

    religion = helluvian

    exists = current_ruler

    current_ruler = { is_alive = yes has_religion = helluvian NOT = { has_trait = shannah_trait } }

    NOT = { tag = REB } NOT = { tag = BAR } NOT = { tag = MER } NOT = { tag = PIR }

}

omni_sf_shannah_trigger = {

    is_alive = yes is_adult = yes is_ruler = yes prisoner = no

    has_trait = shannah_trait

    exists = employer

    employer = { has_land = yes religion = helluvian }

}

# Saved personal balance is valid only for these two living officeholders.

omni_sf_relationship_trigger = {
    trigger_if = {
        limit = {
            exists = current_ruler
            exists = var:omni_sf_patron
            exists = var:omni_sf_shannah
            exists = var:omni_sf_see
            has_variable = omni_sf_bound_epoch
        }

        # Character interactions do not guarantee a root country.
        save_temporary_scope_as = omni_sf_relationship_country

        omni_sf_petitioner_trigger = yes

        has_variable = omni_sf_patron

        exists = var:omni_sf_patron

        current_ruler = var:omni_sf_patron

        has_variable = omni_sf_shannah

        exists = var:omni_sf_shannah

        exists = var:omni_sf_see

        trigger_if = {
            limit = {
                exists = var:omni_sf_shannah
                exists = var:omni_sf_see
                has_variable = omni_sf_bound_epoch
            }
            var:omni_sf_shannah = { employer = scope:omni_sf_relationship_country.var:omni_sf_see }

            var:omni_sf_shannah = {

                omni_sf_shannah_trigger = yes

                NOT = { employer = scope:omni_sf_relationship_country }

                employer = { trigger_if = { limit = { has_variable = omni_sf_reign_epoch } var:omni_sf_reign_epoch = scope:omni_sf_relationship_country.var:omni_sf_bound_epoch } trigger_else = { always = no } }

            }

        }
        trigger_else = { always = no }

    }
    trigger_else = { always = no }
}
