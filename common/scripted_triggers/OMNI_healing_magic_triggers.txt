# Character-scope eligibility for the shared Healing Light cure effects.

omni_has_minor_curable_ailment = {
	OR = {
		has_trait = pneumonia
		has_trait = pox
		has_trait = dysentery
		has_trait = infection
		has_trait = inflammation
		has_trait = arthritis
		has_trait = gout
		has_trait = drow_mushroom_spots_trait
	}
}

omni_has_moderate_curable_ailment = {
	OR = {
		has_trait = vampiric_fever
		has_trait = schizophrenic
		has_trait = epileptic
		has_trait = invalid
		has_trait = blind
		has_trait = mentagra
	}
}

omni_has_major_curable_ailment = {
	OR = {
		has_trait = plague
		has_trait = hemorrhagic_fever
		has_trait = cancer
		has_trait = gangrene
		has_trait = diphtheria
		has_trait = leper
		has_trait = brain_damaged
		has_trait = dementia
		has_trait = paralysis
	}
}

# Worth spending Vitality (add_health) on: hurt or sick rather than maimed.
# health is a 0-100 scale in this engine version - compare retire.txt's
# "health < 15" and 00_generic.txt's "health <= 25". Vitality grants add_health = 3,
# so 75 is the point where a 3-point top-up is still meaningful.
omni_needs_vitality_healing = {
	OR = {
		has_trait = wounded
		has_trait = infection
		health < 75
	}
}

# The characters an AI ruler should care about keeping alive and functional, in the
# order the healing event prioritises them.
omni_is_healing_priority_character = {
	OR = {
		is_ruler = yes
		is_primary_heir = yes
		is_general = yes
		is_admiral = yes
		prominence >= 60
	}
}

# Caster checks for the selected AI patient.
omni_ai_can_cast_minor_cure = {
	omni_can_afford_spell = { BASE = 50 }
	has_variable = cleric_t2_unlocked_var
	omni_knows_spell = { SPELL = healing_light }
	scope:omni_heal_target = { omni_has_minor_curable_ailment = yes }
}
