omni_adventure_companion_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	prisoner = no
	NOT = { is_mercenary = yes }
	NOT = { has_variable = on_omni_adventure }
	NOR = {
		has_trait = brain_damaged
		has_trait = lunatic
	}
}

omni_adventure_can_start_trigger = {
	has_land = yes
	war = no
	exists = current_ruler
	current_ruler = {
		is_alive = yes
		is_adult = yes
		prisoner = no
		NOR = {
			has_variable = omni_adventure_cooldown
			has_variable = on_omni_adventure
		}
	}
	any_owned_province = {
		num_of_adventurer_guild_building >= 1
	}
	any_character = {
		count >= 3
		omni_adventure_companion_candidate_trigger = yes
		NOT = { is_ruler = yes }
	}
}
