omni_adventure_companion_candidate_trigger = {
	is_alive = yes
	is_adult = yes
	prisoner = no
	NOT = { is_mercenary = yes }
	is_general = no
	is_admiral = no
	is_governor = no
	char_is_unoccupied_trigger = yes
	loyalty >= 40
	NOT = { has_variable = on_omni_adventure }
	NOR = {
		has_trait = brain_damaged
		has_trait = lunatic
	}
}

omni_adventure_has_valid_companion_trigger = {
	any_character = {
		count >= 3
		omni_adventure_companion_candidate_trigger = yes
		NOT = { is_ruler = yes }
	}
}

omni_adventure_can_start_trigger = {
	has_land = yes
	war = no
	exists = current_ruler
	current_ruler = {
		is_alive = yes
		is_adult = yes
		prisoner = no
		custom_tooltip = { text = adventure_at_guild_tt location = { has_building = adventurer_guild_building } }
		custom_tooltip = { text = adventure_cooldown_ready_tt NOR = { has_variable = on_omni_adventure has_variable = omni_adventure_cooldown } }
	}
	custom_tooltip = { text = adventure_requires_party_tt omni_adventure_has_valid_companion_trigger = yes }
}

# True when this country's ruler is currently away on an adventure.
# Evaluated in COUNTRY scope.
omni_ruler_on_adventure_trigger = {
	exists = current_ruler
	current_ruler = { has_variable = on_omni_adventure }
}

# Court-action availability gate. While the acting country's ruler is away on an
# adventure this blocks the interaction and shows a red "ruler is away" line in
# the tooltip; when the ruler is home it adds nothing to the tooltip.
# Put inside an allowed_trigger, in COUNTRY (scope:actor) scope.
omni_ruler_available_for_court_action_trigger = {
	trigger_if = {
		limit = { omni_ruler_on_adventure_trigger = yes }
		custom_tooltip = {
			text = omni_ruler_on_adventure_block_tt
			always = no
		}
	}
}

# Character-scope variant: blocks (with the same tooltip) when THIS character is
# the one away on an adventure. Use on the acting/target character as appropriate.
omni_char_available_for_court_action_trigger = {
	trigger_if = {
		limit = { has_variable = on_omni_adventure }
		custom_tooltip = {
			text = omni_ruler_on_adventure_block_tt
			always = no
		}
	}
}
