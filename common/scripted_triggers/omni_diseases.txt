# Supports events/omni_disease_outbreaks.txt. Per-disease target triggers gate on terrain (and season where
# noted) only - no hardcoded regions, same philosophy as omni_disasters.txt. See agent.md "Disease Outbreak
# System" for the full pattern.

# Shared helper: is there a living, currently-uninfected governor in the scoped province to catch this?
# Keeps character infection scoped to one already-selected province, never a realm-wide character scan.
omni_province_has_infectable_governor_trigger = {
	governor = {
		is_alive = yes
		NOT = { has_character_modifier = omni_disease_infected }
	}
}
