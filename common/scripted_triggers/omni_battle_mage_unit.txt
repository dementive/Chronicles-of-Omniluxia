omni_battle_mage_unit = {
	in_combat = yes
	unit_owner = scope:player
	OR = {
		AND = { is_army = yes root = { is_army = yes } }
		AND = { is_navy = yes root = { is_navy = yes } }
	}
	exists = commander
	commander = {
		is_alive = yes
		OR = {
			is_general = yes
			is_admiral = yes
		}
		NOT = { has_variable = is_the_revant }
		omni_character_is_eligible_caster = yes
	}
}
