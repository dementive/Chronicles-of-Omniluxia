# Affordability gate for magic spells. Evaluated in the CASTER (character) scope.
#   $BASE$ = the spell's base mana cost (same number passed to omni_spell_mana_cost_effect).
# True only when the caster has enough banked MP to pay the fully scaled cost, so
# players and AI can never deficit-spend mana on a spell.
omni_can_afford_spell = {
	omni_spell_affordable_budget >= $BASE$
}

# True when the country has at least one ELIGIBLE caster (ruler/general/admiral/physician/
# apothecary with a magic trait, not brain-damaged/goblin/on an orc adventure).
# Evaluated in COUNTRY scope.
omni_player_has_eligible_mage = {
	any_character = {
		OR = {
			is_ruler = yes
			is_general = yes
			is_admiral = yes
			has_office = office_physician
			has_office = office_republic_physician
			has_office = office_apothecary
		}
		OR = {
			has_trait = mage_trait
			has_trait = archmage_trait
			has_trait = warlock_trait
			has_trait = cleric_trait
			has_trait = necromancer_trait
			has_trait = lich_trait
			has_trait = druid_trait
			has_trait = elementalist_trait
			has_trait = diviner_trait
			has_trait = hierophant_trait
			has_trait = illusionist_trait
			has_trait = vampire_trait
			has_trait = blood_angel_curse_trait
			has_variable = folk_unlocked_var
		}
		NOR = {
			has_trait = brain_damaged
			has_trait = goblin_trait
			has_variable = on_orc_adventure
			has_variable = on_omni_adventure
		}
	}
}

# True when the country has at least one eligible caster who can afford a spell
# of base cost $BASE$. Used to enable magic buttons only when casting is possible.
# Evaluated in COUNTRY scope.
omni_player_has_affordable_mage = {
	any_character = {
		OR = {
			is_ruler = yes
			is_general = yes
			is_admiral = yes
			has_office = office_physician
			has_office = office_republic_physician
			has_office = office_apothecary
		}
		OR = {
			has_trait = mage_trait
			has_trait = archmage_trait
			has_trait = warlock_trait
			has_trait = cleric_trait
			has_trait = necromancer_trait
			has_trait = lich_trait
			has_trait = druid_trait
			has_trait = elementalist_trait
			has_trait = diviner_trait
			has_trait = hierophant_trait
			has_trait = illusionist_trait
			has_trait = vampire_trait
			has_trait = blood_angel_curse_trait
			has_variable = folk_unlocked_var
		}
		NOR = {
			has_trait = brain_damaged
			has_trait = goblin_trait
			has_variable = on_orc_adventure
			has_variable = on_omni_adventure
		}
		omni_can_afford_spell = { BASE = $BASE$ }
	}
}

omni_player_has_character_magic_ready = {
	any_character = {
		OR = {
			is_ruler = yes
			is_general = yes
			is_admiral = yes
			has_office = office_physician
			has_office = office_republic_physician
			has_office = office_apothecary
		}
		omni_character_is_eligible_caster = yes
		OR = {
			omni_can_afford_spell = { BASE = 15 }
			AND = {
				has_trait = vampire_trait
				NOT = { has_variable = omni_vampiric_feeding_cooldown }
			}
		}
	}
}

# True when a UNIT's commander can cast a spell of the given BASE cost right now:
# unit has a commander who is an eligible caster, is not the Revant, and can
# afford the mana. UNIT scope. Used by the AI auto-cast events (me_zorgo_magic_
# focus.31/.35) so they scan a country's few armies instead of its whole court.
omni_unit_commander_can_cast_trigger = {
	exists = commander
	commander = {
		NOT = { has_variable = is_the_revant }
		omni_character_is_eligible_caster = yes
		omni_can_afford_spell = { BASE = $BASE$ }
	}
}
