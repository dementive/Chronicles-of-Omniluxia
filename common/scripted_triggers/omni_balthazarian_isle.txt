# Balthazarian Order task 32 (the night landing). See common/scripted_effects/omni_balthazarian_isle.txt.

# Province scope. The landing takes an isle that is unowned or held by an AI country other than the Order.
ob_isle_takeable = {
	OR = {
		NOT = { exists = owner }
		AND = {
			NOT = { owner = root }
			owner = { is_ai = yes }
		}
	}
}

# Character scope, inside the Order. Who can lead the landing when there is no Sword Master.
ob_isle_sword_candidate = {
	is_alive = yes
	is_adult = yes
	is_ruler = no
	prisoner = no
	trigger_if = {
		limit = { exists = scope:ob_isle_lord }
		NOT = { this = scope:ob_isle_lord }
	}
}
