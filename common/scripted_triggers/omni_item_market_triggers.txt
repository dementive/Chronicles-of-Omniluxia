omni_character_has_local_market = {
	is_alive = yes
	root.location = {
		has_building = commerce_building
		OR = {
			owner = root.employer
			controller = root.employer
		}
	}
}

omni_ai_character_can_use_country_market = {
	is_alive = yes
	is_adult = yes
	employer = { is_ai = yes }
	NOT = { has_variable = omni_ai_shopped_market_this_year }
	OR = {
		is_ruler = yes
		is_general = yes
		is_governor = yes
		prominence >= 60
		has_office = office_steward
		has_office = office_marshal
		has_office = office_master_of_the_guard
		has_office = office_foreign_minister
		has_office = office_censor
		has_office = office_praetor
		has_office = office_high_priest_monarchy
		has_office = office_augur
		has_office = office_pontifex
		has_office = office_high_priest
	}
}
