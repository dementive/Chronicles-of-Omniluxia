omni_character_can_visit_city = {
	is_alive = yes
	exists = employer
	OR = {
		location = {
			has_owner = yes
			OR = {
				owner = root.employer
				controller = root.employer
				root.employer = { has_military_access = prev.owner }
			}
		}
		any_character_unit = {
			OR = {
				is_army = yes
				is_navy = yes
			}
			unit_location = {
				has_owner = yes
				OR = {
					owner = root.employer
					controller = root.employer
					root.employer = { has_military_access = prev.owner }
				}
			}
		}
	}
}

omni_character_local_city_has_building = {
	omni_character_can_visit_city = yes
	OR = {
		location = {
			has_building = $BUILDING$
		}
		any_character_unit = {
			OR = {
				is_army = yes
				is_navy = yes
			}
			unit_location = {
				has_building = $BUILDING$
			}
		}
	}
}

omni_character_has_local_market = {
	omni_character_local_city_has_building = { BUILDING = commerce_building }

# A character may enter a city they do NOT own, control, or hold military
# access to - provided it is a valid, owned settlement and the character (or
# a leading commander/ruler) is physically standing there. This covers both a
# city occupied by the character's own side after a siege, and a city merely
# being passed through/besieged by a hostile army that has not taken it.
# Unlike omni_character_can_visit_city, no traditional buildings are required
# - a hostile army only ever interacts with what lies outside the walls.
omni_character_can_visit_hostile_city = {
	is_alive = yes
	exists = employer
	OR = {
		is_ruler = yes
		is_general = yes
		is_governor = yes
	}
	OR = {
		location = {
			has_owner = yes
		}
		any_character_unit = {
			OR = {
				is_army = yes
				is_navy = yes
			}
			unit_location = {
				has_owner = yes
			}
		}
	}
	NOT = { omni_character_can_visit_city = yes }
}

omni_ai_character_can_use_country_market = {
	is_alive = yes
	is_adult = yes
	employer = { is_ai = yes }
	NOT = { has_variable = omni_ai_shopped_market_this_year }
	OR = {
		is_ruler = yes
		is_general = yes
		is_governor = yes
		prominence >= 60
		has_office = office_steward
		has_office = office_marshal
		has_office = office_master_of_the_guard
		has_office = office_foreign_minister
		has_office = office_censor
		has_office = office_praetor
		has_office = office_high_priest_monarchy
		has_office = office_augur
		has_office = office_pontifex
		has_office = office_high_priest
	}
}
