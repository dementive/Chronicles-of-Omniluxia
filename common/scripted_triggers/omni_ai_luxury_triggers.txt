# COUNTRY scope.  Humans only need the listed price; AI countries must still have
# their protected war/construction reserve after paying it.
omni_country_can_pay_for_luxury = {
	trigger_if = {
		limit = { is_ai = yes }
		omni_ai_luxury_spendable_treasury >= $PRICE$
	}
	trigger_else = {
		treasury >= $PRICE$
	}
}
