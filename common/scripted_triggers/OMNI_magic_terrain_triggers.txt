# Terrain difficulty for nature (Amten) magic - evaluated in PROVINCE scope.
# Harsh terrain is hostile to plant/forest magic and raises the mana cost.
omni_harsh_nature_terrain_trigger = {
	OR = {
		terrain = mountain
		terrain = mountain
		terrain = desert
		terrain = cavern
		terrain = impassable_terrain
	}
}

# Fertile terrain is friendly to nature magic and lowers the mana cost.
omni_fertile_nature_terrain_trigger = {
	OR = {
		terrain = farmland
		terrain = plains
		terrain = flood_plain
		terrain = marsh
		terrain = riverine_terrain
		terrain = forest
		terrain = eldritch_forest
		terrain = jungle
	}
}
