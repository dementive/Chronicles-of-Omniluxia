# Country scope unless noted. Triggers for 03_arteon_generic (The Battle For Arteon).

# The three faiths with a dedicated branch. Every other faith takes the old-faith branch.
omni_arteon_named_faith_trigger = {
    OR = {
        religion = artheous_pantheon
        religion = yedidyhan
        religion = transcendence_enlighten
        religion = victory_pantheon
    }
}

# Routes are recorded by each branch capstone. Saves that completed a branch before the
# variables existed fall back to the faith that branch serves.
omni_arteon_route_recorded_trigger = {
    OR = {
        has_variable = arteon_route_artheousan
        has_variable = arteon_route_ascendic
        has_variable = arteon_route_viktreic
        has_variable = arteon_route_old_faith
    }
}

omni_arteon_route_ascendic_trigger = {
    OR = {
        has_variable = arteon_route_ascendic
        AND = {
            omni_arteon_route_recorded_trigger = no
            religion = transcendence_enlighten
        }
    }
}

omni_arteon_route_viktreic_trigger = {
    OR = {
        has_variable = arteon_route_viktreic
        AND = {
            omni_arteon_route_recorded_trigger = no
            religion = victory_pantheon
        }
    }
}

omni_arteon_route_old_faith_trigger = {
    OR = {
        has_variable = arteon_route_old_faith
        AND = {
            omni_arteon_route_recorded_trigger = no
            omni_arteon_named_faith_trigger = no
        }
    }
}

# Province scope, root = the mission country. A territory across the capital area's boundary.
omni_arteon_home_ring_border_trigger = {
    NOT = { area = root.capital_scope.area }
    any_neighbor_province = { area = root.capital_scope.area }
}

# Province scope. Root is the country receiving the claim.
omni_arteon_can_gain_claim_trigger = {
    NOR = {
        owned_or_subject_owned = root
        root = { has_claim = prev }
    }
}

# Province scope, inside the home region.
omni_arteon_boundary_claim_target_trigger = {
    NOT = { area = root.capital_scope.area }
    omni_arteon_can_gain_claim_trigger = yes
}

omni_arteon_two_altars_trigger = {
    OR = {
        has_variable = arteon_kept_both_altars
        # Saves in which the choice predates the variable.
        has_country_modifier = arteon_two_altars_mod
        has_country_modifier = arteon_both_altars_compact_mod
    }
}

omni_arteon_four_shores_held_trigger = {
    owns_or_subject_owns_region = arteon_shout_region
    owns_or_subject_owns_region = arteon_eastern_region
    owns_or_subject_owns_region = arteon_western_region
    owns_or_subject_owns_region = arteon_north_region
}

# Province scope, root = the mission country.
omni_arteon_iron_seam_candidate_trigger = {
    OR = {
        trade_goods = stone
        trade_goods = base_metals
    }
    OR = {
        is_in_region = arteon_shout_region
        is_in_region = arteon_eastern_region
        is_in_region = arteon_north_region
        is_in_region = arteon_western_region
    }
}

# Province scope, root = the mission country. A neighbor held by an independent country we are
# at peace with: the target for Kin on Both Sides of the Boundary.
omni_arteon_kin_province_trigger = {
    has_owner = yes
    is_capital = no
    NOT = { owner = root }
    owner = {
        is_subject = no
        NOT = { war_with = root }
    }
}
