omni_city_actor_has_wealth = {
	OR = {
		AND = {
			exists = scope:omni_city_visitor
			NOT = { this = scope:omni_city_visitor }
			scope:omni_city_visitor = { wealth >= $PRICE$ }
		}
		AND = {
			OR = { NOT = { exists = scope:omni_city_visitor } this = scope:omni_city_visitor }
			is_ruler = yes
			country = { treasury >= $PRICE$ }
		}
	}
}
