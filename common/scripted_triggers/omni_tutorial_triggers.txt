# Omniluxia tutorial "A Royal Progress" (tutorial/00_tutorial.txt).

# CHARACTER scope. Bad-Darinfur reserves one ram for the tutorial ruler.
omni_tutorial_ram_allowance_available = {
	is_ruler = yes
	country = {
		tag = ROM
		is_ai = no
		is_tutorial_active = yes
		has_variable = omni_tut_on
		NOR = {
			has_variable = omni_tut_s10
			has_variable = omni_tut_used_ram
			has_variable = omni_tut_ram_allowance_used
		}
	}
	NOT = { omni_kit_has = { KIT = ram } }
	OR = {
		location = { province_id = 1409 }
		any_character_unit = { is_army = yes unit_location = { province_id = 1409 } }
	}
}

# PROVINCE scope. Carries any spell ward.
omni_tutorial_province_warded = {
	OR = {
		has_province_modifier = omni_folk_hearth_ward_mod
		has_province_modifier = omnic_diviner_spell_t2_mod
		has_province_modifier = dragonscale_dome_mod
	}
}

# COUNTRY scope (the player). The tutorial raid has happened and its war is over
# (raiders destroyed, or peace made). Lets the battle/siege lessons bypass
# instead of softlocking.
omni_tutorial_raid_war_over = {
	has_variable = omni_tut_war_phase
	trigger_if = {
		limit = { exists = var:omni_tut_raiders }
		var:omni_tut_raiders = { NOT = { war_with = root } }
	}
	trigger_else = {
		always = yes
	}
}
