# Balthazarian Order endgame triggers.
ob_can_seek_see_charter = {
    omni_sf_petitioner_trigger = yes
    any_country = {
        NOT = { this = root }
        exists = current_ruler
        current_ruler = { omni_sf_shannah_trigger = yes }
    }
}

ob_has_three_sworn_brothers = {
    any_character = {
        count >= 3
        employer = root
        is_alive = yes
        has_character_modifier = ob_seal_sworn
    }
}

# State scope: frontier states whose captains could follow the Sword seal.
ob_rebel_state_candidate = {
    is_capital_state = no
    NOT = { any_state_province = { province_id = 870 } }
    any_state_province = { is_in_region = norrfalian_borderland has_building = fortress_building }
}

ob_deep_candidate = {
    country_culture_group = drow
    is_ai = yes
    is_subject = no
    has_land = yes
    has_civil_war = no
    NOT = { this = root }
    NOT = { alliance_with = root }
}

ob_lanuthua_chained = {
    exists = var:ob_returned
    var:ob_returned = { is_alive = yes prisoner = yes employer = root }
}
