# A combat-category item (weapon/weapon_ranged/armour/shield/helmet) also grants
# the wearer's country a matching item_$item_ID$_army modifier while worn - see
# choo_items_modifiers.txt. Non-combat items (trinkets/rings/emblems) simply have
# no such modifier defined, so this check quietly does nothing for them.
is_combat_category_item = {
    OR = {
        is_target_in_global_variable_list = { name = item_weapon target = flag:$item_ID$ }
        is_target_in_global_variable_list = { name = item_weapon_ranged target = flag:$item_ID$ }
        is_target_in_global_variable_list = { name = item_armour target = flag:$item_ID$ }
        is_target_in_global_variable_list = { name = item_shield target = flag:$item_ID$ }
        is_target_in_global_variable_list = { name = item_helmet target = flag:$item_ID$ }
    }
}

has_item = {

    is_target_in_variable_list = {  
        name = item_list   
        target = flag:$item_ID$
                                }
}
has_any_item = {

    has_variable_list = item_list

}

has_equiped_item = {

    is_target_in_variable_list = {  
        name = item_list   
        target = flag:$item_ID$
                                }
    OR = {
    has_variable = $item_ID$_equiped
    has_character_modifier = $item_ID$_modifier
    }
    

}

is_item_in_item_category = {

    AND = {
        
        is_target_in_global_variable_list = {
            name = item_$item_category$
            target = flag:$item_ID$
    
        }
        is_target_in_variable_list = {
            name = item_list
            target = flag:$item_ID$
    
        }

    }


}

has_any_other_same_category_item_equipped = {

        # True when the character has NOTHING else occupying category
        # $item_category$, i.e. the slot is free for $item_ID$. Also true (with a
        # different tooltip) when $item_ID$ is already worn, because in that case
        # the caller is offering an unequip and there is nothing to block.

        trigger_if = {

            limit = {

                NOT = {
                    is_target_in_variable_list = {
                        name = equiped_item_list
                        target = flag:$item_ID$
                    }
                }

            }
            custom_tooltip = {
                text = no_other_same_category_item_equipped
            }

            # FIXED: this was
            #     root = { NOT = { is_target_in_variable_list = ... } }
            # in the limit above, and
            #     any_in_list = { ... count < $amount_per_category$ }
            # here. Both were wrong.
            #
            # 1) ROOT is the caller's root, not this character. From the GUI that
            #    happens to be the same thing, but from a mission reward it is
            #    the country, so the check silently evaluated against a scope
            #    that has no equiped_item_list at all.
            # 2) any_in_list with "count < 1" evaluates false when the list does
            #    not exist yet. Handle that case explicitly, then use the caller's
            #    amount_per_category argument for existing lists. Keeping that
            #    argument in the scripted trigger's body is also required by the
            #    Jomini argument compiler; all item-menu and auto-equip callers
            #    pass it.
            #
            # Inside any_in_list the scope is the list element (a bare flag),
            # which cannot host a trigger, so step back to the owning character
            # with prev = { ... } and refer to the element as PREV from there.
            OR = {
                NOT = { has_variable_list = equiped_item_list }
                any_in_list = {
                    variable = equiped_item_list
                    count < $amount_per_category$
                    prev = {
                        is_target_in_global_variable_list = {
                            name = item_$item_category$
                            target = PREV
                        }
                    }
                }
            }
        }
        trigger_else = {

            custom_tooltip = {
                text = item_always_can_be_unequiped
                always = yes
            }

        }

}
