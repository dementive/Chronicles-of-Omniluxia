# COUNTRY scope. True when the realm army/personal loadout slot holds $ITEM$.
omni_country_slots_hold_item = {
	OR = {
		AND = { has_variable = omni_army_slot_weapon var:omni_army_slot_weapon = flag:$ITEM$ }
		AND = { has_variable = omni_army_slot_ranged var:omni_army_slot_ranged = flag:$ITEM$ }
		AND = { has_variable = omni_army_slot_armour var:omni_army_slot_armour = flag:$ITEM$ }
		AND = { has_variable = omni_army_slot_shield var:omni_army_slot_shield = flag:$ITEM$ }
		AND = { has_variable = omni_personal_slot_emblem var:omni_personal_slot_emblem = flag:$ITEM$ }
		AND = { has_variable = omni_personal_slot_trinket var:omni_personal_slot_trinket = flag:$ITEM$ }
	}
}
