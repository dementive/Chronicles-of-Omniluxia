construct_pirate_haven = {

	duration = 0
	enable = yes
	toggle = no
	soundeffect = "event:/SFX/UI/Unit/sfx_ui_unit_remove_pirate"
  
	army_only = yes
	
	hidden = {
		unit_owner = {
			NOT = { construct_pirate_haven_country_trigger = yes }
		}
	}

	allow = {
		NOR = {
			has_variable = construct_pirate_haven_cooldown
		}
		unit_owner = {
			construct_pirate_haven_country_trigger = yes
		}
		is_army = yes
		in_combat = no
		is_moving = no
		in_siege = no
		is_exiled = no
		unit_location = {
			has_owner = yes
			is_port = yes
			OR = {
				controller = root.unit_owner
				controller = {
					is_subject_of = root.unit_owner
				}
			}
			NOT = {
				has_province_modifier = pirate_haven_modifier
			}
		}
		trigger_if = {
			limit = {
				exists = commander
			}
			commander = {
				loyalty > 33
			}
		}
	}
	
	
	start_effect = {
		unit_location = {
			add_province_modifier = {
				name = pirate_haven_modifier
				duration = -1
			}
		}
		unit_owner = {
			add_aggressive_expansion = 5
		}
		root = {
			set_variable = {
  			name = construct_pirate_haven_cooldown
    		days = 3650
   		}	
		}
	}
	
	ai_will_do = {
		modifier = {
			trigger = {
				unit_location = {
					NOT = {
						has_province_modifier = pirate_haven_modifier
					}
					has_owner = yes
					controller = root.unit_owner
				}
				unit_owner = {
					construct_pirate_haven_country_trigger = yes
				}
			}
			add = {
				value = 100
			}
		}
	}
}
