dynastic_union_subject = {
	subject_pays = subject_pays_tributary
	
	joins_overlord_in_war = yes
	protected_when_attacked = yes
	has_overlords_ruler = no
	can_be_integrated = yes
	costs_diplomatic_slot = no
	subject_can_cancel = no
	has_limited_diplomacy = yes
	only_trade_with_overlord = yes
	
	overlord_modifier = {
		monthly_legitimacy = 0.01
		subject_loyalty = 5
	}
	subject_modifier = {
		loyalty_to_overlord = 25
		ruler_popularity_gain = 0.02
	}
	
	allow = {
		is_monarchy = yes
		is_subject = no
		exists = scope:future_overlord
		scope:future_overlord = {
			is_monarchy = yes
			is_subject = no
		}
	}
	
	on_enable = {
		set_variable = {
			name = dynastic_union_subject_started
			days = 3650
		}
		set_variable = {
			name = dynastic_union_integration_cooldown
			days = 3650
		}
		add_opinion = {
			modifier = opinion_dynastic_union_subject
			target = scope:future_overlord
		}
		scope:future_overlord = {
			add_opinion = {
				modifier = opinion_dynastic_union_overlord
				target = root
			}
		}
	}
	
	on_disable = {
		hidden_effect = {
			omni_dynastic_union_former_colour_effect = yes
			omni_dynastic_union_clear_variables_effect = yes
		}
	}
	
	diplo_chance = {
		base = {
			offset = -150
		}
		in_debt = {
			scale = -1
		}
		negative_stability = {
			scale = 20
		}
		current_strength = {
			scale = 0.25
		}
		potential_strength = {
			scale = 0.25
		}
		border_distance = {
			scale = 3
		}
		rank = {
			scale = -10
		}
		junior_to = {
			scale = -40
		}
	}
}
