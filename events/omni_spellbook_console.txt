namespace = omni_spellbook_console

# With a mage selected in the Spellbook, run: event omni_spellbook_console.1
omni_spellbook_console.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_ai = no
				has_variable = omni_sb_selected_mage
				var:omni_sb_selected_mage = { is_alive = yes }
			}
			var:omni_sb_selected_mage = {
				if = {
					limit = { NOT = { has_variable = omni_spell_points } }
					set_variable = { name = omni_spell_points value = 0 }
				}
				change_variable = { name = omni_spell_points add = 10 }
			}
		}
	}
}
