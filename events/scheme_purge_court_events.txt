namespace = scheme_purge_court

scheme_purge_court.1 = {
	type = character_event
	title = scheme_purge_court.1_title
	desc = {
		random_valid = {
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.3 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.1 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.2 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = deceitful } } desc = scheme_purge_court.desc.deceitful.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = deceitful } } desc = scheme_purge_court.desc.deceitful.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = deceitful } } desc = scheme_purge_court.desc.deceitful.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = cruel } } desc = scheme_purge_court.desc.cruel.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = cruel } } desc = scheme_purge_court.desc.cruel.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = cruel } } desc = scheme_purge_court.desc.cruel.3 }
			triggered_desc = { trigger = { scope:purge_target = { NOT = { purge_court_charge_trigger = yes } } } desc = scheme_purge_court.desc.generic.1 }
			triggered_desc = { trigger = { scope:purge_target = { NOT = { purge_court_charge_trigger = yes } } } desc = scheme_purge_court.desc.generic.2 }
			triggered_desc = { trigger = { scope:purge_target = { NOT = { purge_court_charge_trigger = yes } } } desc = scheme_purge_court.desc.generic.3 }
		}
	}
	picture = assassination
	left_portrait = root
	right_portrait = scope:purge_target

	trigger = {
		purge_court_active_trigger = yes
		NOT = { has_variable = purge_court_pending_target }
		exists = scope:purge_country
		employer = scope:purge_country
		exists = scope:purge_target
		scope:purge_target = {
			purge_court_candidate_trigger = { RULER = root }
		}
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = purge_court_hearing_serial } }
			set_variable = { name = purge_court_hearing_serial value = 0 }
		}
		change_variable = { name = purge_court_hearing_serial add = 1 }
		var:purge_court_hearing_serial = { save_scope_as = purge_hearing_serial }
		set_variable = { name = purge_court_pending_target value = scope:purge_target }
		set_variable = purge_court_opened_hearing

		random_country = {
			limit = {
				purge_court_exile_country_trigger = yes
				in_diplomatic_range = scope:purge_country
			}
			save_scope_as = purge_destination
		}
		if = {
			limit = { NOT = { exists = scope:purge_destination } }
			random_country = {
				limit = { purge_court_exile_country_trigger = yes }
				save_scope_as = purge_destination
			}
		}
	}

	option = {
		name = scheme_purge_court.dismiss
		ai_chance = { factor = 1 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
			}
			purge_court_record_hearing_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.restitution
		trigger = {
			scope:purge_target = { has_trait = corrupt wealth >= 10 }
		}
		custom_tooltip = scheme_purge_court.restitution_tt
		ai_chance = { factor = 2 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				scope:purge_target = { has_trait = corrupt wealth >= 10 }
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = { add_gold = -10 }
			scope:purge_country = { add_treasury = 10 }
		}
	}

	option = {
		name = scheme_purge_court.banish
		trigger = {
			purge_court_destination_trigger = yes
		}
		ai_chance = { factor = 2 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				purge_court_destination_trigger = yes
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = {
				banish = scope:purge_destination
				set_home_country = scope:purge_destination
			}
			scope:purge_country = { add_tyranny = 2 }
			add_corruption = 1
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.imprison
		ai_chance = { factor = 3 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
			}
			purge_court_record_hearing_effect = yes
			scope:purge_country = { imprison = { target = scope:purge_target } }
			scope:purge_country = { add_tyranny = 2 }
			add_corruption = 1
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.execute
		ai_chance = { factor = 1 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = {
				death = { death_reason = death_execution killer = root }
			}
			scope:purge_country = { add_tyranny = 5 }
			add_corruption = 2
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.torture
		ai_chance = { factor = 0 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
			}
			purge_court_record_hearing_effect = yes
			scope:purge_country = { imprison = { target = scope:purge_target } }
			scope:purge_target = { add_health = -10 }
			scope:purge_country = { add_tyranny = 4 }
			add_corruption = 2
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.martial
		trigger = {
			martial >= 10
		}
		highlight = yes
		custom_tooltip = scheme_purge_court.martial_tt
		ai_chance = { factor = 1 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				martial >= 10
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = {
				if = {
					limit = { is_minor_character = no exists = family }
					family = { add_prestige = -25 }
				}
				death = { death_reason = death_execution killer = root }
			}
			scope:purge_country = { add_tyranny = 4 }
			add_corruption = 2
			add_popularity = 5
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.finesse
		trigger = {
			finesse >= 10
			purge_court_destination_trigger = yes
		}
		highlight = yes
		custom_tooltip = scheme_purge_court.finesse_tt
		ai_chance = { factor = 2 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				finesse >= 10
				purge_court_destination_trigger = yes
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = {
				banish = scope:purge_destination
				set_home_country = scope:purge_destination
			}
			scope:purge_country = { add_tyranny = 1 }
			add_corruption = 2
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.charisma
		trigger = {
			charisma >= 10
		}
		highlight = yes
		custom_tooltip = scheme_purge_court.charisma_tt
		ai_chance = { factor = 2 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				charisma >= 10
			}
			purge_court_record_hearing_effect = yes
			scope:purge_country = { imprison = { target = scope:purge_target } }
			scope:purge_country = { add_tyranny = 2 }
			add_corruption = 2
			add_popularity = 5
			purge_court_record_removal_effect = yes
		}
	}

	option = {
		name = scheme_purge_court.zeal
		trigger = {
			zeal >= 10
			scope:purge_country = { political_influence >= 5 }
			scope:purge_target = {
				OR = {
					has_trait = vampire_trait
					has_trait = corrupt
					has_trait = cruel
					has_trait = deceitful
				}
			}
		}
		highlight = yes
		custom_tooltip = scheme_purge_court.zeal_tt
		ai_chance = { factor = 1 }
		if = {
			limit = {
				purge_court_live_hearing_trigger = yes
				zeal >= 10
				scope:purge_country = { political_influence >= 5 }
				scope:purge_target = {
					OR = {
						has_trait = vampire_trait
						has_trait = corrupt
						has_trait = cruel
						has_trait = deceitful
					}
				}
			}
			purge_court_record_hearing_effect = yes
			scope:purge_target = {
				death = { death_reason = death_execution killer = root }
			}
			scope:purge_country = { add_political_influence = -5 }
			scope:purge_country = { add_tyranny = 3 }
			add_popularity = 5
			purge_court_record_removal_effect = yes
		}
	}

	after = {
		if = {
			limit = { purge_court_pending_hearing_trigger = yes }
			remove_variable = purge_court_pending_target
		}
	}
}
