	type = character_event
	title = scheme_purge_court.1_title
	desc = {
		random_valid = {
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = vampire_trait } } desc = scheme_purge_court.desc.vampire.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = corrupt } } desc = scheme_purge_court.desc.corrupt.3 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.1 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.2 }
			triggered_desc = { trigger = { scope:purge_target = { loyalty < 33 } } desc = scheme_purge_court.desc.disloyal.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = lunatic } } desc = scheme_purge_court.desc.lunatic.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = reckless } } desc = scheme_purge_court.desc.reckless.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = arrogant } } desc = scheme_purge_court.desc.arrogant.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.2 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = gluttonous } } desc = scheme_purge_court.desc.gluttonous.3 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = deceitful } } desc = scheme_purge_court.desc.deceitful.1 }

			triggered_desc = { trigger = { scope:purge_target = { has_trait = cruel } } desc = scheme_purge_court.desc.cruel.1 }
			triggered_desc = { trigger = { scope:purge_target = { has_trait = cruel } } desc = scheme_purge_court.desc.cruel.2 }

			triggered_desc = { desc = scheme_purge_court.desc.generic.2 }
			triggered_desc = { desc = scheme_purge_court.desc.generic.3 }
		}
	}
	picture = assassination
	left_portrait = root
	right_portrait = scope:purge_target
	trigger = {
		is_ruler = yes
		has_ambition = scheme_purge_court
		employer = {
			any_character = {
				employer = root.employer
				is_alive = yes
				is_ruler = no
				is_co_ruler = no
				prisoner = no
				OR = {
					has_trait = vampire_trait
					has_trait = corrupt
					loyalty < 33
					has_trait = lunatic
					has_trait = reckless
					has_trait = arrogant
					has_trait = gluttonous
					has_trait = deceitful
					has_trait = cruel
				}
			}
		}
	}
	immediate = {
		root = {
			save_scope_as = purge_actor
		}
		root.employer = {
			random_character = {
				limit = {
					employer = root.employer
					is_alive = yes
					is_ruler = no
					is_co_ruler = no
					prisoner = no
					OR = {
						has_trait = vampire_trait
						has_trait = corrupt
						loyalty < 33
						has_trait = lunatic
						has_trait = reckless
						has_trait = arrogant
						has_trait = gluttonous
						has_trait = deceitful
						has_trait = cruel
					}
				}
				save_scope_as = purge_target
			}
			random_country = {
				limit = {
					in_diplomatic_range = root.employer
					NOT = { this = root.employer }
				}
				save_scope_as = purge_destination
			}
		}
		# The pick above finds nothing for an isolated court with no one in
		# diplomatic range, and the banish/finesse options then used an unset
		# scope:purge_destination - "Undefined event target 'purge_destination'"
		# followed by "set_home_country (target) was null". Fall back to any
		# other country so every option stays valid rather than gating the
		# options away and risking an event with nothing to click.
		if = {
			limit = { NOT = { exists = scope:purge_destination } }
			random_country = {
				limit = {
					NOT = { this = root.employer }
				}
				save_scope_as = purge_destination
			}
		}
	}
	option = {
		name = scheme_purge_court.banish
		scope:purge_target = {
			banish = scope:purge_destination
			set_home_country = scope:purge_destination
		}
		root = {
			employer = { add_tyranny = 2 }
			add_corruption = 1
		}
	}
	option = {
		name = scheme_purge_court.martial
		trigger = {
			martial >= 10
		}
		highlight = yes
		custom_tooltip = high_martial_character_tt
		custom_tooltip = scheme_purge_court.martial_tt
		scope:purge_target = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		root = {
			add_popularity = 5
			employer = { add_tyranny = 4 }
			add_corruption = 1
		}
	}
	option = {
		name = scheme_purge_court.finesse
		trigger = {

		}

		custom_tooltip = high_finesse_character_tt
		custom_tooltip = scheme_purge_court.finesse_tt
		scope:purge_target = {

			set_home_country = scope:purge_destination
		}
		root = {
			employer = { add_tyranny = 1 }
			add_corruption = 2
		}
	}
	option = {
		name = scheme_purge_court.charisma
		trigger = {
			charisma >= 10
		}
		highlight = yes
		custom_tooltip = high_charisma_character_tt
		custom_tooltip = scheme_purge_court.charisma_tt
		root.employer = {
			imprison = { target = scope:purge_target }
		}
		root = {
			add_popularity = 5
			employer = { add_tyranny = 1 }
			add_corruption = 1
		}
	}
	option = {
		name = scheme_purge_court.zeal
		trigger = {
			zeal >= 10
			scope:purge_target = {
				OR = {
					has_trait = vampire_trait
					has_trait = corrupt
					has_trait = cruel
					has_trait = deceitful
				}
			}
		}
		highlight = yes
		custom_tooltip = high_zeal_character_tt
		custom_tooltip = scheme_purge_court.zeal_tt
		scope:purge_target = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		root = {
			add_popularity = 5
			employer = { add_tyranny = 3 }
		}
	}
	option = {
		name = scheme_purge_court.imprison
		root.employer = {
			imprison = { target = scope:purge_target }
		}
		root = {
			employer = { add_tyranny = 2 }
			add_corruption = 1
		}
	}
	option = {
		name = scheme_purge_court.execute
		scope:purge_target = {
			death = {
				death_reason = death_execution
				killer = root
			}
		}
		root = {

			add_corruption = 2
		}

		scope:purge_target = {
			add_health = -10

			employer = { add_tyranny = 1 }
			add_corruption = 1
		}
	}
