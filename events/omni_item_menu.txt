		trigger_event = { id = omni_items.12 }
	}

		name = omni_items.7.magic_academy

		trigger_event = { id = omni_items.13 }
	}
	option = {
		name = omni_items.7.mage_tower
		trigger = { OR = { omni_character_has_local_mage_tower = yes scope:omni_city_visitor = { omni_character_has_local_mage_tower = yes } } }
		trigger_event = { id = omni_items.16 }
	}
	option = {
		name = omni_items.7.black_market
		trigger = {
			OR = { omni_character_can_visit_hostile_city = yes scope:omni_city_visitor = { omni_character_can_visit_hostile_city = yes } }
			omni_character_can_afford_black_market_food = yes
		}
		show_as_tooltip = {
			if = {
				limit = {
					OR = { omni_character_can_visit_hostile_city = yes scope:omni_city_visitor = { omni_character_can_visit_hostile_city = yes } }
					NOT = { omni_character_can_afford_black_market_food = yes }
				}
				custom_tooltip = omni_items.7.black_market_cannot_afford_tt
			}
		}
		omni_hostile_city_buy_food_effect = yes
		trigger_event = { id = omni_items.7 }
	}
	option = {
		name = omni_items.7.steal_food
		trigger = {
			OR = { omni_character_can_visit_hostile_city = yes scope:omni_city_visitor = { omni_character_can_visit_hostile_city = yes } }
			OR = { omni_character_can_steal_food = yes scope:omni_city_visitor = { omni_character_can_steal_food = yes } }
		}
		show_as_tooltip = {
			if = {
				limit = {
					OR = { omni_character_can_visit_hostile_city = yes scope:omni_city_visitor = { omni_character_can_visit_hostile_city = yes } }
					NOT = { OR = { omni_character_can_steal_food = yes scope:omni_city_visitor = { omni_character_can_steal_food = yes } } }
				}
				custom_tooltip = omni_items.7.steal_food_none_available_tt
			}
		}
		omni_hostile_city_steal_food_effect = yes
		trigger_event = { id = omni_items.7 }
	}
	option = {
		name = omni_items.back
		trigger_event = { id = omni_items.1 }
	}
	option = {
		name = omni_items.1.close
	}
}
