namespace = omni_items

# Event-based character inventory menu. Opened by the Items button on the
# character window (omni_character_items_button scripted GUI). Root scope is the
# character whose window is open. Each list event shows one option per item,
# gated so only relevant items appear.

### HUB
omni_items.1 = {
	type = character_event
	picture = war_chest
	title = omni_items.1.t
	desc = omni_items.1.desc
	left_portrait = root

	option = {
		name = omni_items.1.equip
		trigger = { has_any_item = yes }
		trigger_event = { id = omni_items.2 }
	}
	option = {
		name = omni_items.1.unequip
		trigger = {
			has_variable_list = equiped_item_list
			any_in_list = { variable = equiped_item_list count >= 1 }
		}
		trigger_event = { id = omni_items.3 }
	}
	option = {
		name = omni_items.1.buy
		trigger = {
			omni_character_has_local_market = yes
		}
		clear_variable_list = omni_item_shop_stock
		remove_variable = omni_item_shop_open
		remove_variable = omni_market_quality
		trigger_event = { id = omni_items.6 }
	}
	option = {
		name = omni_items.1.equip_artifact
		trigger = {
			is_ruler = yes
			employer = {
				any_country_treasure = { exists = yes }
			}
		}
		trigger_event = { id = omni_artifacts.1 }
	}

		name = omni_items.1.unequip_artifact
