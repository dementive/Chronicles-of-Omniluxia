namespace = omni_dynastic_union

omni_dynastic_union.1 = {
	type = country_event
	title = omni_dynastic_union.1.t
	desc = omni_dynastic_union.1.desc
	picture = throneroom
	
	left_portrait = scope:dynastic_junior_ruler
	right_portrait = scope:dynastic_senior_ruler
	goto_location = capital_scope
	
	trigger = {
		scope:dynastic_senior_country = {
			has_land = yes
			is_monarchy = yes
			is_subject = no
			current_ruler = {
				is_adult = yes
				is_married = no
				NOT = { is_same_gender = scope:dynastic_junior_ruler }
			}
		}
		scope:dynastic_junior_ruler = {
			is_ruler = yes
			is_adult = yes
			is_married = no
			employer = root
		}
		is_monarchy = yes
		is_subject = no
		war = no
		has_civil_war = no
	}
	
	option = {
		name = omni_dynastic_union.1.a
		ai_chance = {
			factor = 90
		}
		scope:dynastic_senior_ruler = {
			marry_character = scope:dynastic_junior_ruler
		}
		add_opinion = {
			target = scope:dynastic_senior_country
			modifier = opinion_dynastic_marriage
		}
		scope:dynastic_senior_country = {
			add_opinion = {
				target = root
				modifier = opinion_dynastic_marriage
			}
			set_variable = {
				name = dynastic_union_senior_spouse
				value = scope:dynastic_senior_ruler
			}
			make_subject = {
				target = root
				type = dynastic_union_subject
			}
		}
		set_variable = {
			name = dynastic_union_senior_country
			value = scope:dynastic_senior_country
		}
		set_variable = {
			name = dynastic_union_senior_spouse
			value = scope:dynastic_senior_ruler
		}
		set_variable = {
			name = dynastic_union_junior_spouse
			value = scope:dynastic_junior_ruler
		}
		omni_dynastic_union_new_colour_effect = yes
		add_stability_effect = yes
		current_ruler = {
			add_popularity = popularity_medium
		}
		scope:dynastic_senior_country = {
			current_ruler = {
				add_popularity = popularity_medium
			}
		}
		scope:dynastic_senior_country = {
			trigger_event = {
				id = omni_dynastic_union.2
				days = 3
			}
		}
	}
	
	option = {
		name = omni_dynastic_union.1.b
		ai_chance = {
			factor = 10
		}
		add_opinion = {
			target = scope:dynastic_senior_country
			modifier = opinion_dynastic_marriage_refused
		}
		scope:dynastic_senior_country = {
			add_opinion = {
				target = root
				modifier = opinion_dynastic_marriage_refused
			}
			trigger_event = { id = omni_dynastic_union.9 }
		}
	}
}

omni_dynastic_union.4 = {
	type = country_event
	title = omni_dynastic_union.4.t
	desc = omni_dynastic_union.4.desc
	picture = throneroom
	
	left_portrait = current_ruler
	right_portrait = scope:former_ruler
	goto_location = capital_scope
	
	trigger = {
		is_subject_type = dynastic_union_subject
		has_land = yes
		exists = scope:former_ruler
		has_variable = dynastic_union_junior_spouse
		var:dynastic_union_junior_spouse = scope:former_ruler
		overlord = {
			has_land = yes
			is_monarchy = yes
		}
	}
	
	immediate = {
		overlord = { save_scope_as = dynastic_senior_country }
		save_scope_as = dynastic_union_subject
	}
	
	option = {
		name = omni_dynastic_union.4.a
		trigger = {
			has_subject_loyalty >= 85
			overlord = {
				stability >= 50
				war = no
				has_civil_war = no
			}
			war = no
			has_civil_war = no
		}
		ai_chance = {
			factor = 20
		}
		scope:dynastic_senior_country = {
			omni_dynastic_union_integrate_subject_effect = yes
		}
	}
	
	option = {
		name = omni_dynastic_union.4.b
		ai_chance = {
			factor = 70
		}
		omni_dynastic_union_renew_subject_effect = yes
	}
	
	option = {
		name = omni_dynastic_union.4.c
		trigger = {
			OR = {
				has_subject_loyalty < 50
				overlord = { stability < 35 }
			}
			war = no
			overlord = { war = no }
		}
		ai_chance = {
			factor = 10
		}
		omni_dynastic_union_independence_war_effect = yes
	}
	
	option = {
		name = omni_dynastic_union.4.d
		trigger = {
			is_ai = no
		}
		ai_chance = {
			factor = 0
		}
		omni_dynastic_union_dissolve_subject_effect = yes
	}
}

omni_dynastic_union.5 = {
	type = country_event
	title = omni_dynastic_union.5.t
	desc = omni_dynastic_union.5.desc
	picture = throneroom
	
	left_portrait = current_ruler
	right_portrait = scope:dynastic_union_subject.current_ruler
	goto_location = scope:dynastic_union_subject.capital_scope
	
	trigger = {
		is_monarchy = yes
		has_land = yes
		exists = scope:former_ruler
		any_subject = {
			is_subject_type = dynastic_union_subject
			has_land = yes
			has_variable = dynastic_union_senior_spouse
			var:dynastic_union_senior_spouse = scope:former_ruler
		}
	}
	
	immediate = {
		random_subject = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				var:dynastic_union_senior_spouse = scope:former_ruler
			}
			save_scope_as = dynastic_union_subject
		}
		save_scope_as = dynastic_senior_country
		current_ruler = { save_scope_as = dynastic_new_senior_ruler }
		scope:former_ruler = { save_scope_as = dynastic_dead_senior_ruler }
	}
	
	option = {
		name = omni_dynastic_union.5.a
		ai_chance = {
			factor = 70
		}
		every_subject = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				var:dynastic_union_senior_spouse = scope:dynastic_dead_senior_ruler
			}
			omni_dynastic_union_renew_subject_effect = yes
			trigger_event = { id = omni_dynastic_union.10 }
		}
	}
	
	option = {
		name = omni_dynastic_union.5.b
		trigger = {
			war = no
			OR = {
				stability < 35
				any_subject = {
					is_subject_type = dynastic_union_subject
					has_land = yes
					has_variable = dynastic_union_senior_spouse
					var:dynastic_union_senior_spouse = scope:former_ruler
					has_subject_loyalty < 50
					war = no
				}
			}
		}
		ai_chance = {
			factor = 20
		}
		every_subject = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				var:dynastic_union_senior_spouse = scope:dynastic_dead_senior_ruler
			}
			trigger_event = { id = omni_dynastic_union.11 }
			omni_dynastic_union_dissolve_subject_effect = yes
		}
	}
	
	option = {
		name = omni_dynastic_union.5.c
		trigger = {
			war = no
			any_subject = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				var:dynastic_union_senior_spouse = scope:former_ruler
				war = no
				num_of_cohorts >= root.num_of_cohorts
			}
		}
		ai_chance = {
			factor = 10
		}
		every_subject = {
			limit = {
				is_subject_type = dynastic_union_subject
				has_land = yes
				has_variable = dynastic_union_senior_spouse
				var:dynastic_union_senior_spouse = scope:dynastic_dead_senior_ruler
			}
			if = {
				limit = {
					war = no
					num_of_cohorts >= scope:dynastic_senior_country.num_of_cohorts
				}
				trigger_event = { id = omni_dynastic_union.12 }
				omni_dynastic_union_reverse_claim_war_effect = yes
			}
			else = {
				omni_dynastic_union_renew_subject_effect = yes
				trigger_event = { id = omni_dynastic_union.10 }
			}
		}
	}
}

# The senior court chooses its succession policy in .5. These follow-up events
# make that outcome visible to every junior crown and give a player junior a
# measured opportunity to contest an otherwise routine renewal.
omni_dynastic_union.10 = {
	type = country_event
	title = omni_dynastic_union.10.t
	desc = omni_dynastic_union.10.desc
	picture = throneroom

	left_portrait = current_ruler
	right_portrait = scope:dynastic_new_senior_ruler
	goto_location = capital_scope

	trigger = {
		is_subject_type = dynastic_union_subject
		has_land = yes
		exists = scope:dynastic_senior_country
		scope:dynastic_senior_country = { has_land = yes }
	}

	option = {
		name = omni_dynastic_union.10.a
		ai_chance = { factor = 70 }
	}

	option = {
		name = omni_dynastic_union.10.b
		ai_chance = { factor = 25 }
		add_country_modifier = {
			name = dynastic_union_succession_concessions
			duration = 1825
		}
		add_opinion = {
			modifier = opinion_dynastic_union_concessions
			target = scope:dynastic_senior_country
		}
		scope:dynastic_senior_country = {
			add_legitimacy = -5
			add_opinion = {
				modifier = opinion_dynastic_union_concessions
				target = root
			}
		}
	}

	option = {
		name = omni_dynastic_union.10.c
		trigger = {
			war = no
			scope:dynastic_senior_country = { war = no }
			OR = {
				has_subject_loyalty < 50
				scope:dynastic_senior_country = { stability < 35 }
				num_of_cohorts >= scope:dynastic_senior_country.num_of_cohorts
			}
		}
		ai_chance = { factor = 5 }
		omni_dynastic_union_independence_war_effect = yes
	}
}
