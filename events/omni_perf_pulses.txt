# PERF: the yearly AI pass used to be ~20 heavy effects in a single immediate
# block. Spreading it across countries (days = { 0 300 }) fixed the Jan 1 spike,
# but each individual firing was still one big lump in a single tick.
#
# It is now split into three chained chunks 20 days apart. This is safe because:
#   - create_list_of_ownable_territories builds a GLOBAL variable list that
#     persists between events and only rebuilds when empty (omni_perf.4 already
#     relied on this), so its consumers can run in a later chunk;
#   - every effect below self-throttles on its own has_variable / set_variable
#     guard with a ~1 year duration, so exact ordering within the year is not
#     load-bearing.
# Chunk 1 keeps list-building and caches first, so later chunks read fresh data.
omni_perf.3 = {
	type = country_event
	hidden = yes

	immediate = {
		# --- chunk 1: caches, lists, economy ---
		omni_court_arcana_yearly = yes
		create_list_of_ownable_territories = yes
		cache_adjusted_research_efficiency = yes
		ai_determine_economic_policies = yes
		ai_import_food_into_states_in_need = yes  # Global, food is lowest during winter
		ai_recruit_mercenaries = yes

		trigger_event = { id = omni_perf.6 days = 20 }
	}

	option = {}
}

# Chunk 3: politics, characters, military and magic.
omni_perf.7 = {
	type = country_event
	hidden = yes

	immediate = {
		ai_revoke_free_hands = yes
		ai_fix_character_loyalty = yes
		ai_choose_and_enact_law = yes
		ai_perform_divine_sacrifice = yes
		ai_press_hold_games_button = yes
		ai_enshrine_social_equity = yes           # Only if Mixed Gender Rules are enabled
		ai_destroy_leftover_navies = yes
		omni_ai_study_magic_effect = yes          # Uses Magic Academies to train useful mages
		omni_ai_seduce_character_effect = yes     # Lustful AI characters and spouses seduce

		# Non-combat AI magic. Both are hidden, AI-only, variable-guarded to once
		# per year, and ordered cheapest-trigger-first - see the header comment in
		# events/omni_ai_magic_events.txt. Deliberately on this yearly chunk rather
		# than a monthly on_action: healing and civic buffing are not time-critical,
		# and the magic system does not get new monthly ticks.
		trigger_event = { id = omni_ai_magic.1 days = { 5 40 } }   # Court cleric heals ruler / heir / commanders
		trigger_event = { id = omni_ai_magic.2 days = { 45 80 } }  # Court mage casts a civic spell in peacetime
	}

	option = {}
}
