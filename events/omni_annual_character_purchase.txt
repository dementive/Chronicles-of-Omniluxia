namespace = omni_annual_purchase

# Annual character equipment purchase event
# This event triggers once per year for each character and allows them to
# evaluate and purchase better equipment based on their country's available markets
omni_annual_purchase.1 = {
	type = character_event
	hidden = yes

	trigger = {
		# Must be in a country to participate
		exists = employer
		# Must have some wealth to potentially spend
		wealth >= 30
		# Don't run for very young characters
		age >= 16
	}

	immediate = {
		# Run the annual equipment check
		omni_annual_character_equipment_check = yes
	}

	option = {
		name = omni_annual_purchase.1.opt
		ai_chance = { factor = 100 }
	}
}
