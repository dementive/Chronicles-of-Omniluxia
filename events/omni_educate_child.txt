# Graduation

omni_educate_child.7 = { # Martial Graduation
	type = country_event
	title = omni_educate_child.7.t
	desc = omni_educate_child.7.desc
	picture = ascetic
	
	left_portrait = current_ruler
	right_portrait = scope:educated_child
	goto_location = capital_scope

	trigger = {
		any_character = {
			is_adult = no
			OR = {
				has_variable = ec_land_var
				has_variable = ec_sea_var
			}
		}
	}

	immediate = {
		random_character = {
			limit = {
				is_adult = no
				OR = {
					has_variable = ec_land_var
					has_variable = ec_sea_var
				}
			}
			save_scope_as = educated_child
		}
	}
	
	option = { # Land
		trigger = {
			scope:educated_child = {
				has_variable = ec_land_var
			}
		}
		name = omni_educate_child.7.a
		random_list = {
			1 = {
				scope:educated_child = {
					add_martial = 1
				}
			}
			1 = {
				scope:educated_child = {
					add_martial = 2
					if = {
						limit = { NOT = { scope:educated_child = { has_trait = comradely } } }
						add_trait = comradely
					}
				}
			}
			1 = {
				scope:educated_child = {
					add_martial = 3
					if = {
						limit = { NOT = { scope:educated_child = { has_trait = tactician } } }
						add_trait = tactician
					}
				}
			}
		}
	}
	option = { # Sea
		trigger = {
			scope:educated_child = {
				has_variable = ec_sea_var
			}
		}
		name = omni_educate_child.7.b
		random_list = {
			1 = {
				scope:educated_child = {
					add_martial = 1
				}
			}
			1 = {
				scope:educated_child = {
					add_martial = 2
					if = {
						limit = { NOT = { scope:educated_child = { has_trait = comradely } } }
						add_trait = comradely
					}
				}
			}
			1 = {
				scope:educated_child = {
					add_martial = 3
					if = {
						limit = { NOT = { scope:educated_child = { has_trait = seadog } } }
						add_trait = seadog
					}
				}
			}
		}
	}
}
