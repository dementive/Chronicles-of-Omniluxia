# Learn Magic Main Menu
me_magic_menu.1 = {
    type = country_event

    trigger = { omni_academy_course_student_allowed = yes }

    title = me_magic_menu.1_title
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { scope:target = { omni_has_active_magic_study_course = yes } }
                desc = omni_academy_course_in_progress_desc
            }
            triggered_desc = {
                trigger = { scope:target = { has_variable = omni_studying_magic_cooldown } }
                desc = omni_academy_course_cooldown_desc
            }
            triggered_desc = { desc = me_magic_menu.1_visitor_desc }
        }
    }
    picture = ascetic
    left_portrait = scope:target

    option = { # Omnic
        name = "me_magic_menu.1.a"
        custom_tooltip = me_magic_menu.1a_tooltip
        trigger_event = { id = me_magic_menu.2 }
    }

    option = { # Aldic
        name = "me_magic_menu.1.b"
        custom_tooltip = me_magic_menu.1b_tooltip
        trigger_event = { id = me_magic_menu.3 }
    }

    option = { # Amten
        name = "me_magic_menu.1.c"
        custom_tooltip = me_magic_menu.1c_tooltip
        trigger_event = { id = me_magic_menu.4 }
    }

    option = { # Melodian
        trigger = {
            OR = { 
                has_law = magic_permitted
                country_culture_group = orcish
                country_culture_group = orcish_humano
                country_culture_group = goblin
                country_culture_group = hobgoblin
                country_culture_group = drow
                country_culture_group = deep_gnome
                country_culture_group = dissolved
                country_culture_group = dissolved_revant
            }
        }

        name = "me_magic_menu.1.d"
        custom_tooltip = me_magic_menu.1d_tooltip
        trigger_event = { id = me_magic_menu.5 }
    }

    option = { # Leave
        name = "me_magic_menu.1.e"
        custom_tooltip = me_magic_menu.1e_tooltip
    }
}
