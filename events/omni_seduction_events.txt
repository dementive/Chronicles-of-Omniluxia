	type = country_event
	title = omni_seduction.1.t
	desc = omni_seduction.1.desc
	picture = greek_court
	left_portrait = scope:omni_seducer
	right_portrait = scope:omni_seduction_target

	trigger = {
		exists = scope:omni_seducer
		exists = scope:omni_seduction_target
		scope:omni_seducer = { is_alive = yes is_ruler = yes }
		scope:omni_seduction_target = { is_alive = yes prisoner = no }
	}

	option = {
		name = omni_seduction.1.a
		trigger = { scope:omni_seducer = { wealth >= 25 } }
		scope:omni_seducer = { add_gold = -25 }
		scope:omni_seducer = { set_variable = { name = omni_seduction_gift days = 300 } }
		custom_tooltip = omni_seduction_gift_tt
	}
	option = {
		name = omni_seduction.1.b
		scope:omni_seducer = { add_popularity = -2 }
		scope:omni_seducer = { set_variable = { name = omni_seduction_flattery days = 300 } }
		custom_tooltip = omni_seduction_flattery_tt
	}
	option = {
		name = omni_seduction.1.c
		scope:omni_seducer = { add_corruption = 1 }
		scope:omni_seducer = { set_variable = { name = omni_seduction_bold days = 300 } }
		custom_tooltip = omni_seduction_bold_tt
	}
	option = {
		name = omni_seduction.1.d
		scope:omni_seducer = { remove_variable = omni_seduction_active }
		custom_tooltip = omni_seduction_abandoned_tt
	}
	after = {
		if = {
			limit = { scope:omni_seducer = { has_variable = omni_seduction_active } }
			trigger_event = { on_action = omni_seduction_stage_two days = { 30 60 } }
		}
	}
}

	type = country_event
	title = omni_seduction.7.t
	desc = omni_seduction.7.desc

	left_portrait = scope:omni_seducer
	right_portrait = scope:omni_seduction_target
	option = {
		name = omni_seduction.7.a
		custom_tooltip = omni_seduction_exposure_tt
		scope:omni_seducer = { add_popularity = -10 add_corruption = 5 }
		if = {
			limit = { scope:omni_seducer = { has_variable = omni_seduction_father } }
			scope:omni_seducer = { set_variable = { name = omni_seduction_discovered days = 400 } }
		}
		if = {
			limit = { scope:omni_seduction_target = { has_variable = omni_seduction_father } }
			scope:omni_seduction_target = { set_variable = { name = omni_seduction_discovered days = 400 } }
		}
		if = {
			limit = { exists = scope:omni_target_spouse scope:omni_target_spouse = { is_alive = yes } }
			scope:omni_seducer = { save_scope_as = omni_betrayal_culprit }
			scope:omni_target_spouse = {
				omni_betrayed_spouse_effect = yes
				if = {
					limit = { OR = { has_trait = vengeful has_trait = cruel has_trait = deceitful } }
					save_scope_as = omni_avenging_spouse
				}
			}
		}
		if = {
			limit = {
				exists = scope:omni_seducer_spouse
				scope:omni_seducer_spouse = { is_alive = yes }
				NOT = { scope:omni_seducer_spouse = scope:omni_target_spouse }
			}
			scope:omni_seduction_target = { save_scope_as = omni_betrayal_culprit }
			scope:omni_seducer_spouse = {
				omni_betrayed_spouse_effect = yes
				if = {
					limit = { OR = { has_trait = vengeful has_trait = cruel has_trait = deceitful } }
					save_scope_as = omni_avenging_spouse
				}
			}
		}
		if = {
			limit = { exists = scope:omni_avenging_spouse }
			random = { chance = 25 trigger_event = { id = omni_seduction.8 days = { 90 240 } } }
		}
		if = {
			limit = { NOT = { scope:omni_seducer.employer = scope:omni_seduction_target.employer } }
			scope:omni_seduction_target.employer = {
				add_opinion = { target = scope:omni_seducer.employer modifier = omni_foreign_royal_affair }
			}
		}
	}
}
