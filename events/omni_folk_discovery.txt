namespace = omni_folk_discovery
# Entry only: one new local teacher per visit, retained through lesson purchases.
omni_folk_discovery.1 = {
 type = character_event
 hidden = yes
 immediate = {
  if = {
   limit = { exists = scope:omni_city_visitor scope:omni_city_visitor = { omni_character_has_folk_teacher = yes } }
   scope:omni_city_visitor = { omni_folk_resolve_teacher = yes }
   if = {
    limit = { exists = scope:omni_folk_teacher }
    omni_folk_prepare_teacher_character = yes
    if = {
     limit = { exists = scope:omni_folk_tutor }
     trigger_event = { id = omni_folk_discovery.3 }
    }
   }
  }
 }
}

 option = {
  name = omni_folk_discovery.learn_dreaming
  trigger = { omni_folk_can_learn = { SPELL = dreaming } }
  custom_tooltip = omni_folk_discovery.effect_dreaming
  if = {
   limit = { scope:omni_city_visitor = { omni_folk_would_become_practitioner = yes } }
   custom_tooltip = omni_folk_discovery.gain_practitioner
  }
  # The price is on the button; keep gold and bookkeeping out of the tooltip.
  hidden_effect = {
   if = {
    limit = { omni_folk_can_learn = { SPELL = dreaming } }
    scope:omni_city_visitor = {
     add_gold = -50
     omni_folk_initiate_practitioner = yes
     omni_learn_spell = { SPELL = dreaming }
    }
   }
  }
  trigger_event = { id = omni_folk_discovery.3 }
 }
