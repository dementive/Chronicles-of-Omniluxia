namespace = omni_piracy

omni_piracy.1 = {
	type = country_event
	title = omni_piracy.1.t
	desc = omni_piracy.1.desc
	picture = trade_port

	left_portrait = current_ruler

	trigger = {
		exists = scope:engage_in_piracy_raider_country
		exists = scope:engage_in_piracy_raided_port
		NOT = { this = scope:engage_in_piracy_raider_country }
		NOT = { war_with = scope:engage_in_piracy_raider_country }
	}

	option = {
		name = omni_piracy.1.a
		trigger = {
			is_subject = no
			scope:engage_in_piracy_raider_country = {
				is_subject = no
			}
			current_ruler = {
				OR = {
					has_trait = brave
					has_trait = reckless
					has_trait = vengeful
					has_trait = assertive
					has_trait = cruel
					has_trait = ambitious
				}
			}
		}
		ai_chance = {
			factor = 60
			modifier = {
				factor = 2
				current_ruler = { has_trait = vengeful }
			}
			modifier = {
				factor = 2
				current_ruler = { has_trait = reckless }
			}
			modifier = {
				factor = 1.5
				current_ruler = { has_trait = brave }
			}
			modifier = {
				factor = 0.5
				current_ruler = { has_trait = cautious }
			}
		}
		declare_war_with_wargoal = {
			war_goal = naval_wargoal
			target = scope:engage_in_piracy_raider_country
		}
	}

	option = {
		name = omni_piracy.1.b
		trigger = {
			is_subject = no
			scope:engage_in_piracy_raider_country = {
				is_subject = no
			}
			current_ruler = {
				NOR = {
					has_trait = brave
					has_trait = reckless
					has_trait = vengeful
					has_trait = assertive
					has_trait = cruel
					has_trait = ambitious
				}
			}
		}
		ai_chance = {
			factor = 25
			modifier = {
				factor = 2
				current_ruler = { has_trait = brave }
			}
			modifier = {
				factor = 0.25
				current_ruler = { has_trait = cautious }
			}
			modifier = {
				factor = 0.25
				current_ruler = { has_trait = content }
			}
		}
		declare_war_with_wargoal = {
			war_goal = naval_wargoal
			target = scope:engage_in_piracy_raider_country
		}
	}

	option = {
		name = omni_piracy.1.c
		ai_chance = {
			factor = 40
			modifier = {
				factor = 2
				current_ruler = { has_trait = cautious }
			}
			modifier = {
				factor = 2
				current_ruler = { has_trait = content }
			}
			modifier = {
				factor = 0.5
				current_ruler = { has_trait = vengeful }
			}
			modifier = {
				factor = 0.5
				current_ruler = { has_trait = reckless }
			}
		}
		add_stability = -2
	}
}

omni_piracy.10 = {
	type = country_event
	title = omni_piracy.10.t
	desc = omni_piracy.10.desc
	picture = trade_port

	option = {
		name = omni_piracy.10.a
		if = {
			limit = { var:omni_piracy_num_raids >= 25 }
			custom_tooltip = omni_piracy.10.a.tt_25
			hidden_effect = {
				add_treasury = 15
				add_military_experience = 5
			}
		}
		else_if = {
			limit = { var:omni_piracy_num_raids >= 20 }
			custom_tooltip = omni_piracy.10.a.tt_20
			hidden_effect = {
				add_treasury = 13
				add_military_experience = 5
			}
		}
		else_if = {
			limit = { var:omni_piracy_num_raids >= 15 }
			custom_tooltip = omni_piracy.10.a.tt_15
			hidden_effect = {
				add_treasury = 11
				add_military_experience = 4
			}
		}
		else_if = {
			limit = { var:omni_piracy_num_raids >= 10 }
			custom_tooltip = omni_piracy.10.a.tt_10
			hidden_effect = {
				add_treasury = 9
				add_military_experience = 3
			}
		}
		else_if = {
			limit = { var:omni_piracy_num_raids >= 5 }
			custom_tooltip = omni_piracy.10.a.tt_5
			hidden_effect = {
				add_treasury = 7
				add_military_experience = 2
			}
		}
		else = {
			custom_tooltip = omni_piracy.10.a.tt_default
			hidden_effect = {
				add_treasury = 5
				add_military_experience = 1
			}
		}
		remove_variable = omni_piracy_num_raids
	}
}
