# Resolve the rite even if the envoy died or the village changed hands.
omni_wenelru.2 = {
	type = country_event
	hidden = yes

	immediate = {
		remove_variable = omni_wenelru_leaf_pending
		omni_wenelru_load_rite_scopes = yes
		if = {
			limit = {
				omni_wenelru_covenant = yes
				exists = scope:wenelru_hearth
				scope:wenelru_hearth = {
					owner = root
					omni_wenelru_first_leaf_site = yes
				}
			}
			scope:wenelru_hearth = {
				add_province_modifier = { name = omni_wenelru_covenant_hearth duration = -1 }
			}
			if = {
				limit = {
					OR = {
						NOT = { exists = scope:wenelru_delegate }
						NOT = { scope:wenelru_delegate = { omni_wenelru_delegate = yes } }
					}
					any_character = { omni_wenelru_delegate = yes }
				}
				random_character = {
					limit = { omni_wenelru_delegate = yes }
					root = { set_variable = { name = omni_wenelru_delegate_target value = prev } }
				}
			}
			if = {
				limit = { var:omni_wenelru_delegate_target = { omni_wenelru_delegate = yes } }
				trigger_event = { id = omni_wenelru.3 }
			}
			else = {
				remove_variable = omni_wenelru_hearth_target
				remove_variable = omni_wenelru_delegate_target
			}
		}
		else_if = {
			limit = { exists = scope:wenelru_hearth }
			trigger_event = { id = omni_wenelru.9 }
		}
	}
}
