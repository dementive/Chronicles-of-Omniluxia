namespace = omni_item_autoequip

# Event boundary for the large AI item dispatcher. Keeping this in one event
# compiles omni_ai_auto_equip_items once instead of once per give-item call.
omni_item_autoequip.1 = {
	type = character_event
	hidden = yes

	immediate = {
        if = {
            limit = { omni_item_registry_ready = yes }
            omni_ai_auto_equip_items = yes
        }
        else_if = {
            limit = { exists = employer }
            employer = { omni_ensure_item_registry = yes }
            trigger_event = { id = omni_item_autoequip.1 days = 2 }
        }
	}

	option = {}
}
