namespace = omni_ai_magic

# ----------------------------------------------------------------------------------------------------------------------
# AI non-combat magic drivers.
#
# The spellcasting events below are hidden, AI-only, and use the chunked AI pulse
# (omni_perf.7 in events/omni_perf_pulses.txt) rather than a monthly on_action -
# neither healing nor civic buffing is time-critical, and per the performance rules
# in MAGIC_AI_AUDIT.md we do not add new monthly ticks for the magic system.
#
# The spellcasting triggers are ordered cheapest-first:
#   is_ai -> need checks -> the bounded any_character court scan.
# The court scan is the only real cost and it is the last thing evaluated.
# ----------------------------------------------------------------------------------------------------------------------

omni_ai_magic.1 = { # AI: a court cleric heals the ruler / heir / commanders
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		any_character = {
			country = root
			is_alive = yes
			omni_is_healing_priority_character = yes
			OR = {
				omni_has_major_curable_ailment = yes
				omni_has_moderate_curable_ailment = yes
				omni_has_minor_curable_ailment = yes
				omni_needs_vitality_healing = yes
			}
		}
	}

	immediate = {
		save_scope_as = caster_country
		# Patient priority: the ruler first, then anyone else who matters. A dying
		# ruler is a succession crisis, so they outrank a sick general.
		if = {
			limit = {
				current_ruler = {
					OR = {
						omni_has_major_curable_ailment = yes
						omni_has_moderate_curable_ailment = yes
						omni_has_minor_curable_ailment = yes
						omni_needs_vitality_healing = yes
					}
				}
			}
			current_ruler = { save_scope_as = omni_heal_target }
		}
		else = {
			random_character = {
				limit = {
					country = root
					is_alive = yes
					omni_is_healing_priority_character = yes
					OR = {
						omni_has_major_curable_ailment = yes
						omni_has_moderate_curable_ailment = yes
						omni_has_minor_curable_ailment = yes
						omni_needs_vitality_healing = yes
					}
				}
				save_scope_as = omni_heal_target
			}
		}
		random_character = {
			limit = {
				country = root
				is_alive = yes
				omni_character_is_eligible_caster = yes
				OR = {
					omni_ai_can_cast_major_cure = yes
					omni_ai_can_cast_moderate_cure = yes
					omni_ai_can_cast_minor_cure = yes
					omni_ai_can_cast_vitality = yes
				}
			}
			weight = {
				modifier = {
					factor = 40
					omni_ai_can_cast_major_cure = yes
				}
				modifier = {
					factor = 20
					omni_ai_can_cast_moderate_cure = yes
				}
				modifier = {
					factor = 10
					omni_ai_can_cast_minor_cure = yes
				}
			}
			omni_ai_cast_healing_effect = yes
		}
	}

	option = {}
}

omni_ai_magic.2 = { # AI: a court mage casts a civic spell in peacetime
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		# Peacetime only - at war, mana belongs to the battle and siege casters,
		# which run on the monthly pulse and have first call on it.
		war = no
		any_character = {
			country = root
			is_alive = yes

			omni_character_is_eligible_caster = yes
			omni_can_afford_spell = { BASE = 50 }
			OR = {
				omni_knows_spell = { SPELL = unrest_cleansing }
				omni_knows_spell = { SPELL = weathercall_growth }
				omni_knows_spell = { SPELL = harbor_ward }
				omni_knows_spell = { SPELL = migration_beacon }
			}
		}
	}

	immediate = {
		save_scope_as = caster_country
		omni_ai_choose_civic_spell_effect = yes
	}

	option = {}
}

omni_ai_magic.6 = { # AI: raise one army through wartime summoning
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		war = yes
	}

	immediate = {
		save_scope_as = player
		clear_saved_scope = selected_mage
		clear_saved_scope = target

		# Tier V first: a master of Summon Dragon binds one while at war.
		# The cast event rechecks everything (mana, one dragon per realm, cooldown).
		if = {
			limit = {
				NOT = { has_variable = omni_cd_summon_dragon }
				omni_can_bind_dragon = yes
			}
			random_character = {
				limit = {
					# Cheapest and rarest test first: almost no one knows the spell.
					has_variable = summon_dragon_known_var
					OR = { employer = root country = root }
					is_alive = yes
					omni_character_is_eligible_caster = yes
					omni_can_afford_spell = { BASE = 350 }
				}
				save_scope_as = selected_mage
				root = { trigger_event = { id = omni_spellbook_cast.146 } } # summon_dragon
			}
		}

		if = {
			limit = {
				NOT = { exists = scope:selected_mage }
				NOT = { has_variable = omni_cd_summon_animal_auxiliaries }
				capital_scope = { owner = root omni_spell_target_land = yes }
			}
			random_character = {
				limit = {
					OR = { employer = root country = root }
					is_alive = yes
					omni_can_cast_summon_animal_auxiliaries_char = yes
					omni_can_afford_spell = { BASE = 75 }
				}
				save_scope_as = selected_mage
				save_scope_as = target
				root = { trigger_event = { id = omni_spellbook_cast.115 } }
			}
		}

		if = {
			limit = {
				NOT = { exists = scope:selected_mage }
				NOT = { has_variable = omni_cd_summon_warborn }
				capital_scope = { owner = root omni_spell_target_land = yes }
			}
			random_character = {
				limit = {
					OR = { employer = root country = root }
					omni_can_select_territory_summon_warborn = yes
					omni_can_afford_spell = { BASE = 75 }
				}
				save_scope_as = selected_mage
				root = {
					capital_scope = { save_scope_as = target }
					trigger_event = { id = omni_spellbook_cast.120 }
				}
			}
		}

		if = {
			limit = {
				NOT = { exists = scope:selected_mage }
				NOT = { has_variable = omni_summon_dryads_cooldown }
			}
			random_character = {
				limit = {
					OR = { employer = root country = root }
					omni_can_select_territory_summon_dryads = yes
					omni_can_afford_spell = { BASE = 100 }
				}
				save_scope_as = selected_mage
				root = {
					random_owned_province = {
						limit = { omni_spell_target_forest = yes }
						save_scope_as = target
					}
					if = {
						limit = { exists = scope:target }
						trigger_event = { id = omni_spellbook_cast.116 }
					}
				}
			}
		}
	}

	option = {}
}

omni_ai_magic.5 = { # AI: fund the country's first Magic Academy in its capital
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		invention = organized_mage_guilds_inv
		NOT = { has_country_modifier = omni_funding_magic_academy }
		exists = capital_scope
		OR = {
			# Powerful countries are guaranteed an academy (see immediate).
			omni_ai_academy_priority_country = yes
			AND = {
				NOT = { has_variable = omni_ai_war_threatened }
				capital_scope = {
					has_city_status = yes
					controller = owner
					NOT = { has_building = magic_academy_building }
					num_of_mages >= 6
					OR = {
						free_building_slots >= 1
						AND = {
							free_building_slots = 0
							has_construction = no
							omni_province_has_removable_low_value_building = yes
						}
					}
				}
				has_monthly_balance >= omni_ai_magic_academy_monthly_cost
			}
		}
		# Check other cities only after the cheaper gates pass.
		omni_country_has_no_magic_academy = yes
	}

	immediate = {
		if = {
			limit = { omni_ai_academy_priority_country = yes }
			# 1) Capital with a free slot.
			if = {
				limit = {
					capital_scope = {
						has_city_status = yes
						controller = owner
						free_building_slots >= 1
					}
				}
				capital_scope = { start_building_construction = magic_academy_building }
			}
			# 2) Largest other controlled city with a free slot.
			else_if = {
				limit = {
					any_owned_province = {
						has_city_status = yes
						controller = owner
						free_building_slots >= 1
					}
				}
				ordered_owned_province = {
					limit = {
						has_city_status = yes
						controller = owner
						free_building_slots >= 1
					}
					order_by = total_population
					start_building_construction = magic_academy_building
				}
			}
			# 3) Capital is full: clear a low-value building, else place it
			#    directly (overflowing the slot cap) so it can never stall.
			else_if = {
				limit = {
					capital_scope = {
						has_city_status = yes
						controller = owner
					}
				}
				capital_scope = {
					if = {
						limit = { omni_province_has_removable_low_value_building = yes }
						omni_remove_one_low_value_building = yes
					}
					if = {
						limit = { free_building_slots >= 1 }
						start_building_construction = magic_academy_building
					}
					else = {
						add_building_level = magic_academy_building
					}
				}
			}
			if = {
				limit = { omni_country_has_no_magic_academy = no }
				if = {
					limit = { has_law = magic_banned }
					change_law = magic_only_for_entrusted
				}
				add_country_modifier = { name = omni_funding_magic_academy duration = 1825 }
			}
		}
		else = {
		capital_scope = {
			if = {
				limit = { free_building_slots = 0 }
				omni_remove_one_low_value_building = yes
			}
			if = {
				limit = { free_building_slots >= 1 }
				# Scripted construction uses the funding modifier as its cost.
				start_building_construction = magic_academy_building
			}
		}
		if = {
			limit = { capital_scope = { has_specific_construction = magic_academy_building } }
			if = {
				limit = { has_law = magic_banned }
				change_law = magic_only_for_entrusted
			}
			add_country_modifier = { name = omni_funding_magic_academy duration = 1825 }
		}
		}
	}

	option = {}
}

omni_ai_magic.7 = { # AI: an aging mage who knows the Rite of Lichdom performs it
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		any_character = {
			country = root
			omni_ai_lichdom_candidate = yes
		}
	}

	immediate = {
		random_character = {
			limit = {
				country = root
				omni_ai_lichdom_candidate = yes
			}
			save_scope_as = omni_lich_aspirant
		}
		# Fear of death drives the rite: the older or sicker, the likelier.
		if = {
			limit = { scope:omni_lich_aspirant = { OR = { age >= 90 health < 30 } } }
			random = { chance = 75 set_local_variable = omni_lich_go }
		}
		else_if = {
			limit = { scope:omni_lich_aspirant = { age >= 70 } }
			random = { chance = 40 set_local_variable = omni_lich_go }
		}
		else = {
			random = { chance = 20 set_local_variable = omni_lich_go }
		}
		if = {
			limit = { has_local_variable = omni_lich_go }
			scope:omni_lich_aspirant = {
				omni_spell_mana_cost_effect = { BASE = 350 }
				force_add_trait = lich_trait
			}
			save_scope_as = omni_lich_country
			every_country = {
				limit = { is_ai = no }
				trigger_event = { id = omni_ai_magic.9 }
			}
		}
	}

	option = {}
}
