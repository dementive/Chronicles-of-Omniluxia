orc_flavor.1 = { # Downgrade to Independent Shogun
    type = country_event

    title = orc_flavor_1_title
    desc = orc_flavor_1_desc
    picture = event_mountain_village
    left_portrait = current_ruler

    trigger = {
        OR = {
            country_culture_group = orcish
            country_culture_group = orcish_humano
        }
        government = shogun
        any_subject = {
			count < 2
		}
		is_subject = no
    }

    option = {
        name = "orc_flavor_1.a"
        custom_tooltip = orc_flavor_1_tooltip.a
        change_government = independent_shogun
        add_stability = -12
        add_legitimacy = -50
    }
}

orc_flavor.9 = { # High Piety
    type = country_event

    title = orc_flavor_9_title
    desc = orc_flavor_9_desc
    picture = city_country
    left_portrait = current_ruler
    left_portrait = char:173
    right_portrait = char:137
    right_portrait = char:175

    fire_only_once = yes

    immediate = {
		hidden_effect = {
		    char:175 = {
	            save_scope_as = gevanni_rebel_scum
	        }
			p:2470 = {
				create_country = {
	                set_country_religion = the_first_emperor
	                if = {
						limit = {
							NOT = {
								primary_culture = eagelian
							}
						}
						set_primary_culture_cleanup_effect = {
							NEW_PRIMARY_CULTURE = eagelian
							MAKE_OLD_CULTURE_INTEGRATED = yes
						}
					}
	                change_country_name = "Republic of Gevanni"
	                change_country_color = "kanai_confed_purple"
	                change_government = democratic_republic
	                change_country_adjective = "Gevannic"
	                change_country_flag = EGA
	                save_scope_as = gevanni_revolt
				}
			}
			scope:gevanni_rebel_scum = {
				remove_all_positions = yes
				move_country = scope:gevanni_revolt
				set_home_country = scope:gevanni_revolt
				remove_character_modifier = foreign_citizen
			}
			p:2470.area = {
				every_area_province = {
					limit = {
						NOT = { owner = scope:gevanni_revolt }
						owner = c:EGA
					}
					set_owned_by = scope:gevanni_revolt
				}
			}
			scope:gevanni_revolt = {
				add_treasury = 1000
				add_manpower = 20
				set_as_ruler = scope:gevanni_rebel_scum
				if = {
					limit = {
						NOR = {
							has_law = republican_recruitment_2 # legions capital
							has_law = republican_recruitment_4 # legions enabled
						}
					}
					change_law = republican_recruitment_2 # capital
				}
			}
			p:2470 = { # capital fort
				state = { set_state_capital = p:2470 }
				if = {
					limit = { can_build_building = fortress_building }
					add_building_level = fortress_building
				}
			}
		}
		char:173 = {
        	death = {
        		death_reason = death_execution
        	}
        }
    }

    option = {
        name = "orc_flavor_9.a"
        add_stability = -3
        add_legitimacy = -5
		scope:gevanni_revolt = {
	        declare_war_with_wargoal = {
				war_goal = independence_wargoal
				province = 2470
				target = EGA
			}
		}
        c:EGA = {
			trigger_event = {
		        id = orc_flavor.10
		        days = { 90 180 }
		    }
	    }
    }
}

orc_flavor.13 = { # Embracing the Black Flame
    type = country_event

    title = orc_flavor_13_title
    desc = orc_flavor_13_desc
    picture = city_country
    left_portrait = scope:evil_orc

    trigger = {
        any_character = {
                AND = {
                    OR = {
            		  has_trait = vampire_trait
            		  has_trait = necromancer_trait
            		  has_trait = warlock_trait
            		  has_trait = lich_trait
            	   }
            	   has_trait = orc_trait
                    NOT = { has_trait = black_flame_trait }
                }
        }
    }

    immediate = {
        random_character = {
        	limit = {
	        	OR = {
	        		has_trait = vampire_trait
	        		has_trait = necromancer_trait
	        		has_trait = warlock_trait
	        		has_trait = lich_trait
	        	}
	        	has_trait = orc_trait
                NOT = { has_trait = black_flame_trait }
        	}
        	save_scope_as = evil_orc
        }

        hidden_effect = {
	        scope:evil_orc = {
	        	remove_trait = white_flame_trait
	        	remove_trait = yellow_flame_trait
	        	remove_trait = orange_flame_trait
	        	remove_trait = red_flame_trait
                remove_trait = crimson_flame_trait
	        }
        }
    }

    option = {
        name = "orc_flavor_13.a"
        scope:evil_orc = {
        	force_add_trait = black_flame_trait
        }
    }
}
