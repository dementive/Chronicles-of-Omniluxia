            has_variable = integrate_halfling_var
            has_variable = integrate_high_elf_var
            has_variable = integrate_hobgoblin_var
            has_variable = integrate_human_var
            has_variable = integrate_kinones_var
            has_variable = integrate_mycelar_var
            has_variable = integrate_orc_var
            has_variable = integrate_sea_elf_var
            has_variable = integrate_snakefolk_var
            has_variable = integrate_vampire_var
            has_variable = integrate_wood_elf_var
            has_variable = integrate_wraith_var
            has_variable = integrate_zorg_var
        }
    }

    immediate = {
        choose_race_integrate_effect = yes
        if = {
            limit = { exists = scope:chosen_culture }

        ordered_owned_province = {
            limit = {
                any_pops_in_province = {
                    pop_culture = scope:chosen_culture.culture
                }
                is_capital = no
            }
            save_scope_as = village_hometown
        }

        if = {
            limit = { exists = scope:village_hometown }
            scope:village_hometown = {
                ordered_pops_in_province = {
                    limit = {
                        pop_culture = scope:chosen_culture.culture
                    }
                    max = {
                        value = total_population
                        divide = 4
                    }
                    order_by = pop_hapiness
                    move_pop = root.capital_scope
                    check_range_bounds = no
                }
            }
        }

        add_stability = -2
        }
    }

    # Great them With Open Arms
    option = {
        name = "me_racial_events_21.a"
        add_treasury = {
            value = 0
            subtract = has_yearly_income_svalue
        }

        root.capital_scope = {
            add_province_modifier = {
                name = "cap_migrants_mod"
                duration = 365
            }
        }


        every_country_culture = {
            limit = {
                this.culture = root.culture
            }
            add_country_culture_modifier = {
                name = "cap_migration_angry_mod"
                duration = 1825
            }
        }

        race_integrate_positive_level_effect = yes
        race_integrate_events_effect = yes
    }
    # Restrict their Migration

        name = "me_racial_events_21.b"
        add_stability = 3

        every_country_culture = {
            limit = {
                this.culture = root.culture
            }
            add_country_culture_modifier = {
                name = "cap_migration_denied_mod"
                duration = 1825
            }
        }

        race_integrate_negative_level_effect = yes
        race_integrate_events_effect = yes
    }
    option = {
        name = "me_racial_events_21.c"
        custom_tooltip = end_focus_tooltip

    }
