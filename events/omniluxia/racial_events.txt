    type = country_event

    title = me_racial_events_24_title
    desc = me_racial_events_24_desc

    immediate = {
        add_stability = -2
        ordered_owned_province = {
            limit = {
                dominant_province_culture = scope:target_culture.culture
                any_neighbor_province = {
                    count > 1
                    dominant_province_culture = scope:target_culture.culture
                }
                state = {
                    NOT = {
                        any_state_province = {
                            is_capital = yes
                        }
                    }
                }
                AND = {
                    any_neighbor_province = {
                        owner = ROOT
                    }
                    any_neighbor_province = {
                        exists = owner
                        NOT = {
                            owner = ROOT
                        }
                    }
                }
            }
            save_scope_as = expel_hometown
            random_neighbor_province = {
                limit = {
                    exists = owner
                    NOT = {
                        owner = ROOT
                    }
                }
                save_scope_as = expel_target
            }
            order_by = {
                value = 0
                add = population_cap
                subtract = total_population
            }
            add_province_modifier = {
                name = "local_expulsion"
                duration = 732
            }
            add_state_loyalty = -2
            custom_tooltip = colonia_here_tt

        }
        culture_state_loyalty_effect = yes
        scope:target_culture = {
            add_country_culture_modifier = {
                name = "culture_expulsed" 
                duration = 365
            }
        }
        add_country_modifier = {
            name = "culture_expulsed_country"
            duration = 182
        }
    }

        name = "me_racial_events_2.a"

            ordered_pops_in_province = {

                    value = total_population

                }
                min = {
                    value = 0
                }
                move_pop = scope:expel_target
            }
            ordered_pops_in_province = {
                limit = {
                    culture = scope:target_culture.culture
                }
