    					}
    				}
                    kill_pop = yes
                }

            save_scope_as = settler_colony

                    pop_culture = root.culture
                }
            }
            save_scope_as = settler_home

                exists = scope:settler_home

                        pop_culture = root.culture

                        value = total_population
                        divide = 3

                }

                is_capital = no
                AND = {
                    any_neighbor_province = {
                        owner = ROOT

                        exists = owner
                        NOT = {
                            owner = ROOT
                        }
                    }

                    exists = owner

                        owner = ROOT
                    }

        }
        }
    }

        name = "me_racial_events_27.a"

                    }

                15 = {
                    trigger_event = {
                        id = me_racial_events.26
                        days = { 30 180 }
                    }
                }
                25 = {

                        id = me_racial_events.27
                        days = { 30 180 }

            }
        }

        add_legitimacy = -12

                exists = scope:settler_colony
