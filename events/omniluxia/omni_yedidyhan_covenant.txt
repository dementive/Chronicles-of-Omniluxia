namespace = omni_yedidyhan_covenant

omni_yedidyhan_covenant.1 = {
	type = country_event
	title = omni_yedidyhan_covenant.1.t
	desc = omni_yedidyhan_covenant.1.desc
	picture = event_barrow_white
	left_portrait = scope:selected_mage
	right_portrait = scope:target

	trigger = {
		omni_sb_can_cast_yedidyhan_covenant = yes
		omni_sb_can_target_yedidyhan_covenant = yes
	}

	option = {
		name = omni_yedidyhan_covenant.1.a
		trigger = {
			omni_sb_can_cast_yedidyhan_covenant = yes
			omni_sb_can_target_yedidyhan_covenant = yes
		}
		custom_tooltip = omni_yedidyhan_covenant.1.a_tt
		hidden_effect = {
			set_variable = { name = omni_yedidyhan_covenant_cooldown years = 5 }
			if = {
				limit = { scope:target = { loyalty >= 70 } }
				scope:target = {
					force_add_trait = vampiric_fever
					trigger_event = { id = omni_vampire_spread.1 days = 30 }
				}
				trigger_event = { id = omni_yedidyhan_covenant.2 days = 31 }
			}
			else = {
				scope:target = { add_loyalty = loyalty_omni_yedidyhan_covenant_refused }
				trigger_event = { id = omni_yedidyhan_covenant.3 }
			}
		}
	}

	option = {
		name = omni_yedidyhan_covenant.1.b
	}
}
