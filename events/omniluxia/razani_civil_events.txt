razani_civil_war.4 = { # WGA Intervene in Civil War
    type = country_event

    title = razani_civil_war_4_title
    desc = razani_civil_war_4_desc
    picture = city_country
    left_portrait = current_ruler
    left_portrait = scope:nazani_pretender
    right_portrait = char:137
    right_portrait = char:138

    fire_only_once = yes

    immediate = {
    	hidden_effect = {
	    	random_family = {
	            limit = {
	                this = root.char:18.family
	            }
	            save_scope_as = nazani_family
	        }
	        create_character = {
		        first_name = "O-Unse"
		       	age = 34
		        culture = weagelian
		        religion = the_first_emperor
		        female = no
		        good_popularity_trait_effect = yes
		        save_scope_as = nazani_rebel
		    }
		    scope:nazani_rebel = {
                save_scope_as = target
                set_family = scope:nazani_family
                add_character_modifier = {
                    name = adopted_into_great_family
                    duration = -1
                }
                if = { #Bastards are easier to make family members.
                    limit = { is_child_of = ROOT }
                    custom_tooltip = bastard_adoption_event_tooltip
                }
                else_if = { #Siblings are easier to make full family members.
                    limit = { is_sibling_of = ROOT }
                    custom_tooltip = sibling_adoption_event_tooltip
                }
                else_if = { #Extra Marital Relations are harder to make full family members.
                    limit = {
                        any_child = {
                            ROOT = { is_parent_of = PREV }
                        }
                    }
                    custom_tooltip = lover_adoption_event_tooltip
                }
                else_if = { #Friends are easier to make full family members.
                    limit = { is_friend = ROOT }
                    custom_tooltip = friend_adoption_event_tooltip
                }
                else = {
                    custom_tooltip = adoption_explanation_event_tooltip
                }

                scope:nazani_family = { add_prestige = adoption_cost_event_pay_svalue }
                save_scope_as = picked_adoption_candidate
            }
        }
    }

    option = { # Support Pretender
        name = "razani_civil_war_4.a"
        custom_tooltip = razani_civil_war_4_tooltip.a
        add_stability = -2
        add_treasury = -150
        add_manpower = -10
        c:EGA = {
			trigger_event = razani_civil_war.5
        }
    }
    option = { # Support Razani
        name = "razani_civil_war_4.b"
        custom_tooltip = razani_civil_war_4_tooltip.b
        add_stability = -2
        add_treasury = -50
        add_manpower = -5
        c:EGA = {
			trigger_event = razani_civil_war.6
        }
    }
    option = { # Support Neither
        name = "razani_civil_war_4.c"
        add_stability = 3
        c:YGA = {
			trigger_event = {
		        id = razani_civil_war.7
		        days = { 15 30 }
		    }
        }
    }
}

razani_civil_war.5 = { # EGA Nazani Back a Pretender
    type = country_event

    title = razani_civil_war_5_title
    desc = razani_civil_war_5_desc
    picture = city_country
    left_portrait = current_ruler
    right_portrait = scope:nazani_pretender
    right_portrait = c:WGA.current_ruler

    immediate = {
    	hidden_effect = {
			p:2348 = {
				create_country = {
	                set_country_religion = the_first_emperor

	                change_country_color = "nazani_client_color"
	                change_government = imperium
	                change_country_adjective = "Nazani"
	                change_country_flag = "WGA"
	                save_scope_as = nazani_revolter
				}
			}
			scope:nazani_rebel = {
				remove_all_positions = yes
				move_country = scope:nazani_revolter
				set_home_country = scope:nazani_revolter
				remove_character_modifier = foreign_citizen
			}
			scope:nazani_revolter = {
				add_treasury = 1000
				add_manpower = 20
				set_as_ruler = scope:nazani_rebel

					limit = {
						NOR = {
							has_law = republican_recruitment_2 # legions capital
							has_law = republican_recruitment_4 # legions enabled
						}
					}
					change_law = republican_recruitment_2 # capital
				}
			}
			scope:nazani_revolter = {
				set_primary_culture_cleanup_effect = {
					NEW_PRIMARY_CULTURE = weagelian
					MAKE_OLD_CULTURE_INTEGRATED = yes
				}
			}
			p:2348 = { # capital fort
				state = { set_state_capital = p:2348 }
				if = {
					limit = { can_build_building = fortress_building }
					add_building_level = fortress_building
				}
			}
			p:2339 = {
				limit = {
					NOT = { owner = scope:nazani_revolter }
					owner = c:EGA
				}
				set_owned_by = scope:nazani_revolter
			}
			p:2306 = {
				limit = {
					NOT = { owner = scope:nazani_revolter }
					owner = c:EGA
				}
				set_owned_by = scope:nazani_revolter
			}

			c:WGA = {
				make_subject = {
	                type = feudatory
	                target = scope:nazani_revolter
	            }
            }
		}
    }

    option = {
        name = "razani_civil_war_5.a"
        custom_tooltip = razani_civil_war_5_tooltip

        scope:nazani_revolter = {
			declare_war_with_wargoal = {

				target = EGA
			}
		}
        c:YGA = {
			trigger_event = {
		        id = razani_civil_war.7
		        days = { 15 30 }
		    }
        }
    }
}
