    type = country_event

    desc = {
        first_valid = {
            triggered_desc = { trigger = { has_variable = z2_test_setup_ok } desc = z2_player_test_desc }
            triggered_desc = { trigger = { is_ai = yes } desc = z2_test_not_player }
            triggered_desc = { trigger = { has_land = no } desc = z2_test_no_land }
            triggered_desc = { trigger = { is_tribal = no } desc = z2_test_not_tribal }
            triggered_desc = { trigger = { is_subject = yes } desc = z2_test_subject }
            triggered_desc = { trigger = { has_civil_war = yes } desc = z2_test_civil_war }
            triggered_desc = { trigger = { capital_scope = { z2_homeland_trigger = no } } desc = z2_test_wrong_region }
            triggered_desc = { trigger = { capital_scope = { z2_mainland_trigger = no } } desc = z2_test_wrong_landmass }
            triggered_desc = { trigger = { war = yes } desc = z2_test_at_war }
            triggered_desc = { trigger = { OR = { has_global_variable = z2_warned has_global_variable = z2_ended } } desc = z2_test_already_used }
            triggered_desc = { trigger = { has_global_variable = zerywani_horde_formed NOT = { has_global_variable = zerywani_first_ended } } desc = z2_test_first_active }
            triggered_desc = { trigger = { z2_tribe_count < 1 } desc = z2_test_too_few_tribes }
            triggered_desc = { trigger = { z2_tribal_land < 24 } desc = z2_test_too_little_land }
            triggered_desc = { trigger = { always = yes } desc = z2_test_other_failure }
        }
    }
    picture = war_council
    immediate = {
        remove_variable = z2_test_setup_ok
        if = {
            limit = {
                is_ai = no war = no z2_tribe_trigger = yes z2_base_survives_trigger = yes
                NOT = { has_global_variable = z2_warned }
                NOT = { has_global_variable = z2_ended }
                OR = { NOT = { has_global_variable = zerywani_horde_formed } has_global_variable = zerywani_first_ended }
            }
            if = {
                limit = { NOT = { has_global_variable = zerywani_first_ended } }
                set_global_variable = { name = zerywani_horde_formed }
                set_global_variable = { name = zerywani_first_ended }
                set_global_variable = { name = zerywani_first_capital_not_taken }
            }
            set_global_variable = { name = z2_debug_testing }
            add_treasury = 2000
            add_political_influence = 200
            add_manpower = 100
            set_variable = { name = z2_test_setup_ok }
        }
    }
    option = { name = z2_player_test.a trigger = { has_variable = z2_test_setup_ok } }
    option = { name = z2_player_test.blocked trigger = { NOT = { has_variable = z2_test_setup_ok } } }
}
