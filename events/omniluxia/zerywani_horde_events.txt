# DEBUG/CONSOLE ONLY: fire on the currently played tag with the `event zerywani_horde.999`
# console command to force-trigger the horde formation immediately, bypassing eligibility,
# the decade-pulse roll and the 50-100 year window. Useful for testing.
zerywani_horde.999 = {
	type = country_event
	hidden = yes

	immediate = {
		zerywani_horde_formation_effect = yes
	}
}

# Periodic monitor: checks whether the invasion war is still ongoing, reschedules
# itself while it is, and resolves the horde's fate once it ends.
zerywani_horde.4 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = is_zerywani_horde_leader
	}

	immediate = {
		if = {
			limit = {
				exists = scope:zerywani_horde_target
				war_with = scope:zerywani_horde_target
			}
			zerywani_horde_reinforcement_effect = yes
			trigger_event = {
				id = zerywani_horde.4
				days = { 300 400 }
			}
		}
		else = {
			trigger_event = {
				id = zerywani_horde.5
				days = 1
			}
		}
	}
}

# The invasion war has ended - resolve victory or defeat and clean everything up.
zerywani_horde.5 = {
	type = country_event
	title = zerywani_horde_5_title
	picture = city_siege
	hidden = no

	left_portrait = current_ruler

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:zerywani_horde_target
					scope:zerywani_horde_target = {
						capital_scope = { owner = root }
					}
				}
				desc = zerywani_horde_5_victory_desc
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = zerywani_horde_5_defeat_desc
			}
		}
	}

	immediate = {
		zerywani_horde_cleanup_effect = yes
		remove_variable = is_zerywani_horde_leader
		remove_variable = zerywani_horde_invasion_origin
	}

	option = {
		name = "zerywani_horde_5.a"
		trigger = {
			exists = scope:zerywani_horde_target
			scope:zerywani_horde_target = {
				capital_scope = { owner = root }
			}
		}
		add_stability = 15
	}
	option = {
		name = "zerywani_horde_5.b"
		trigger = {
			NOT = {
				AND = {
					exists = scope:zerywani_horde_target
					scope:zerywani_horde_target = {
						capital_scope = { owner = root }
					}
				}
			}
		}
		add_stability = -10
	}
}
