	trigger = {
		employer = {
			OR = {
				country_culture_group = dissolved 
				country_culture_group = dissolved_revant
			}
			is_ai = yes
		}
		NOT = { employer = { tag = BAR } }
		scope:target = {
			OR = {
				ROOT = {
					is_mercenary = yes
				}
				NOT = { owner = root.employer }
			}
		}
	}

		scope:target = {
			add_city_smoke_effect = yes

                }

                    value = 0
                }
                kill_pop = yes
            }
		}
	}
	option = {      
        name = revant.3.a
    }
