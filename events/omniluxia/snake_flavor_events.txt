namespace = snake_flavor

snake_flavor.1 = { # New Mage Offered
    type = country_event

    title = snake_flavor_1_title
    desc = snake_flavor_1_desc
    picture = city_country
    left_portrait = current_ruler
    right_portrait = scope:illu_snake_mage

    trigger = {
    	NOT = { has_variable = illuminati_angry }
        any_character = { # FIXXXXXXXXXXX
			is_ruler = no
			OR = {
				has_office = office_physician
				has_office = office_republic_physician
				has_office = office_apothecary
			}
			NOT = { has_variable = illuminati_mage }
		}
        is_ai = no
    }

    immediate = {
    	create_character = {
            female = no
            culture = ishtari
            religion = shamanist_beliefs
            set_variable = illuminati_mage
            save_scope_as = illu_snake_mage
        }
        hidden_effect = {
	        random_list = { # Give Magic
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = diviner_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = hierophant_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = illusionist_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = cleric_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = elementalist_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	            1 = {
	                scope:illu_snake_mage = {
	                	add_trait = mage_trait
	                    set_variable = druid_t2_unlocked_var
	                    set_variable = folk_unlocked_var
	                }
	            }
	        }
	        random_character = {
				limit = {
					is_ruler = no
					OR = {
						has_office = office_physician
						has_office = office_republic_physician
						has_office = office_apothecary
					}
				}
				save_scope_as = poor_sap
			}
		}
    }

    option = {
        name = "snake_flavor_1.a"
        custom_tooltip = snake_flavor_1_tooltip.a
        add_stability = 3
        current_ruler = {
        	add_legitimacy = 5
        }
        hidden_effect = {
	        scope:poor_sap = {
				if = {
					limit = {
						has_office = office_physician
					}
					remove_office = office_physician
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_physician
				}
			}
			scope:poor_sap = {
				if = {
					limit = {
						has_office = office_republic_physician
					}
					remove_office = office_republic_physician
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_republic_physician
				}
			}
			scope:poor_sap = {
				if = {
					limit = {
						has_office = office_apothecary
					}
					remove_office = office_apothecary
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_apothecary
				}
			}
		}
		scope:poor_sap = {
			if = {
				limit = {
					can_get_rivals = yes
				}
				add_rival = scope:illu_snake_mage
			}
		}
    }
    option = {
        name = "snake_flavor_2.a"
        custom_tooltip = snake_flavor_2_tooltip.a
        add_stability = -2
        current_ruler = {
        	add_legitimacy = -12
        }
        hidden_effect = {
            scope:illu_snake_mage = {
	            death = {
	                death_reason = death_illuminati
	                killer = root.current_ruler
	            }
	        }
        }
    }
}
