namespace = snake_flavor

snake_flavor.1 = { # New Mage Offered
    type = country_event

    title = snake_flavor_1_title
    desc = snake_flavor_1_desc
    picture = event_snake_cult
    left_portrait = current_ruler
    right_portrait = scope:illu_snake_mage
    right_portrait = scope:poor_sap

    trigger = {
        country_culture_group = snakefolk
    	has_variable = illuminati_happy
        NOT = { 
        	any_character = {
				has_culture = ishtari
				has_trait = mage_trait
			}
		}
        is_ai = no
    }

    immediate = {
    	create_ishtari_mage_effect = yes
        random_character = {
			limit = {
				is_ruler = no
				OR = {
					has_office = office_physician
					has_office = office_republic_physician
					has_office = office_apothecary
				}
			}
			save_scope_as = poor_sap
		}
    }

    option = {
        name = "snake_flavor_1.a"
        custom_tooltip = snake_flavor_1_tooltip.a
        add_stability = 3
        add_legitimacy = 5
        hidden_effect = {
	        scope:poor_sap = {
				if = {
					limit = {
						has_office = office_physician
					}
					remove_office = office_physician
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_physician
				}
			}
			scope:poor_sap = {
				if = {
					limit = {
						has_office = office_republic_physician
					}
					remove_office = office_republic_physician
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_republic_physician
				}
			}
			scope:poor_sap = {
				if = {
					limit = {
						has_office = office_apothecary
					}
					remove_office = office_apothecary
				}
			}
			scope:illu_snake_mage = {
				if = {
					limit = {
						has_any_office = no
					}
					give_office = office_apothecary
				}
			}
		}
		scope:poor_sap = {
			if = {
				limit = {
					can_get_rivals = yes
				}
				add_rival = scope:illu_snake_mage
			}
		}
    }
    option = {
        name = "snake_flavor_1.b"
        custom_tooltip = snake_flavor_1_tooltip.b
        add_stability = -2
        add_tyranny = -3
        random_country_culture = {
            limit = {
                this.culture = culture:ishtari
            }
            set_country_culture_right = freemen
        }
        hidden_effect = {
        	remove_variable = illuminati_happy
            scope:illu_snake_mage = {
	            death = {
	                death_reason = death_illuminati
	            }
	        }
        }
    }
}

snake_flavor.3 = { # Bad Events
	type = country_event
    hidden = yes

    trigger = {
        country_culture_group = snakefolk
    	NOT = { has_variable = illuminati_happy }
        any_owned_province = {
            has_province_modifier = snakefolk_lodge_mod
        }
        is_ai = no
    }

    immediate = {
        random_list = {
            65 = { # Nothing Happens

            }
            20 = { # Assassination Fails
                trigger_event = {
		            id = snake_flavor.4
		            days = { 3 5 }
		        }
            }
            5 = { # Assassination Successful
                trigger_event = {
		            id = snake_flavor.5
		            days = { 3 5 }
		        }
            }
            10 = { # Inspired Revolt
                trigger_event = {
		            id = snake_flavor.6
		            days = { 3 5 }
		        }
            }
        }
    }
}

snake_flavor.7 = { # Lodge Destruction
    type = country_event

    title = snake_flavor_7_title
    desc = snake_flavor_7_desc
    picture = looting
    left_portrait = current_ruler
    right_portrait = scope:illu_local_leader

    immediate = {
    	hidden_effect = {
	    	random_owned_province = {
                limit = {
                    has_province_modifier = snakefolk_lodge_mod
                }
                save_scope_as = lodge_revolt_cap
            }

            create_character = {
            	first_name = "Unnus"
		        culture = ishtari
		        religion = shamanist_beliefs
		        female = no
		        good_popularity_trait_effect = yes
		        save_scope_as = illu_local_leader
		    }
			scope:lodge_revolt_cap = {
				create_country = {
	                set_country_religion = shamanist_beliefs
	                if = {
						limit = {
							NOT = {
								primary_culture = ishtari
							}
						}
						set_primary_culture_cleanup_effect = {
							NEW_PRIMARY_CULTURE = ishtari
							MAKE_OLD_CULTURE_INTEGRATED = yes
						}
					}
	                change_country_name = "Ishtari Lodge"
	                change_country_color = "holy_bhaxian"
	                change_government = theocratic_monarchy
	                change_country_adjective = "Ishtari"
	                change_country_flag = IP3_FLAG
	                save_scope_as = lodge_revolt
				}
				remove_province_modifier = snakefolk_lodge_mod
			}
			scope:illu_local_leader = {
				remove_all_positions = yes
				move_country = scope:lodge_revolt
				set_home_country = scope:lodge_revolt
				remove_character_modifier = foreign_citizen
			}
			scope:lodge_revolt = {
				add_treasury = 1000
				add_manpower = 20
				set_as_ruler = scope:illu_local_leader
				set_country_heritage = lodge_heritage
				if = {
					limit = {
						NOR = {
							has_law = monarchy_recruitment_2 # legions capital
							has_law = monarchy_recruitment_4 # legions enabled
						}
					}
					change_law = monarchy_recruitment_2 # capital
				}
				create_unit = {
	                navy = no
	                name = "Lodge Members"
	                location = capital_scope
	                hidden_effect = {
	                    add_subunit = heavy_infantry
	                    add_subunit = heavy_infantry
	                    add_subunit = heavy_infantry
	                    add_subunit = heavy_infantry
	                    add_subunit = heavy_infantry
	                    add_subunit = heavy_infantry                                        
	                    add_food = 150
	                    add_morale = 8
	                }
	            }
			}
			scope:lodge_revolt_cap = { # capital fort
				state = { set_state_capital = scope:lodge_revolt_cap }
				if = {
					limit = { can_build_building = fortress_building }
					add_building_level = fortress_building
				}
			}
		}
    }

    option = {
        name = "snake_flavor_7.a"
        custom_tooltip = snake_flavor_7_tooltip.a
        add_stability = -2
        scope:lodge_revolt = {
	        declare_war_with_wargoal = {
				war_goal = independence_wargoal
				province = scope:lodge_revolt_cap
				target = ROOT
			}
		}
    }
}
