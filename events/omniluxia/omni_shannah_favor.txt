omni_shannah_favor.10 = {

    type = country_event hidden = yes

    immediate = {

        if = {

            limit = {

                omni_sf_relationship_trigger = yes has_variable = omni_sf_active

                current_ruler = scope:omni_sf_chain_patron

                trigger_if = { limit = { has_variable = omni_sf_shannah } var:omni_sf_shannah = scope:omni_sf_chain_shannah } trigger_else = { always = no }

                current_ruler = { is_adult = yes prisoner = no }

                var:omni_sf_shannah = { omni_char_available_for_court_action_trigger = yes }

            }

            # Rotating random start gives three distinct scenes per cultivation.

            if = { limit = { trigger_if = { limit = { has_variable = omni_sf_scene } var:omni_sf_scene = 0 } trigger_else = { always = no } }

                random_list = {

                    1 = { set_variable = { name = omni_sf_scene value = 1 } }

                    1 = { set_variable = { name = omni_sf_scene value = 2 } }

                    1 = { set_variable = { name = omni_sf_scene value = 3 } }

                }

            }

            else = {

                change_variable = { name = omni_sf_scene add = 1 }

                if = { limit = { trigger_if = { limit = { has_variable = omni_sf_scene } var:omni_sf_scene > 3 } trigger_else = { always = no } } set_variable = { name = omni_sf_scene value = 1 } }

            else = { trigger_event = { id = omni_shannah_favor.14 } }

        }

        else_if = {

            limit = { has_variable = omni_sf_chain_waiting }

            remove_variable = omni_sf_active

            remove_variable = omni_sf_chain_waiting

            trigger_event = { id = omni_shannah_favor.15 }

        }

    }

}
