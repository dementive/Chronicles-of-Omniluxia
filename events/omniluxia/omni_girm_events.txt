omni_girm.3 = {
    type = country_event
    title = omni_girm.3.t
    desc = {
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = omni_girm_captured } desc = omni_girm.3.desc_taken }
            triggered_desc = { trigger = { has_global_variable = omni_girm_frontier } desc = omni_girm.3.desc_frontier }
            triggered_desc = { trigger = { always = yes } desc = omni_girm.3.desc_held }
        }
    }
    picture = event_human_council
    left_portrait = current_ruler
    right_portrait = scope:girm_peace_envoy
    goto_location = p:4864

    immediate = {
        create_character = { save_scope_as = girm_peace_envoy }
    }

    option = {
        name = omni_girm.3.a
        trigger = { has_global_variable = omni_girm_captured treasury >= 80 p:4864 = { owner = root } }
        custom_tooltip = omni_girm.3.a_tt
        add_treasury = -80
        p:4864 = { add_province_modifier = { name = omni_girm_settlement duration = 1825 } }
    }

    option = {
        name = omni_girm.3.b
        trigger = { NOR = { has_global_variable = omni_girm_captured has_global_variable = omni_girm_frontier } treasury >= 50 }
        custom_tooltip = omni_girm.3.b_tt
        add_treasury = -50
        current_ruler = { add_popularity = 5 }
        add_military_experience = 10
    }

    option = {
        name = omni_girm.3.c
        custom_tooltip = omni_girm.3.c_tt
        add_stability = 3
    }
}
