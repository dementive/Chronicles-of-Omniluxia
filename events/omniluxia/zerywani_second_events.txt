# Notices are delivered to humans; the autonomous state machine lives in on_actions.

    left_portrait = current_ruler
    type = country_event
    title = z2_warning_title
    desc = {
        first_valid = {
            triggered_desc = { trigger = { has_global_variable = zerywani_first_capital_taken } desc = z2_warning_victory_desc }
            triggered_desc = { trigger = { has_global_variable = zerywani_first_capital_sacked } desc = z2_warning_withdrawal_desc }
            triggered_desc = { trigger = { always = yes } desc = z2_warning_defeat_desc }
        }
    }
    picture = war_council
    trigger = { exists = global_var:z2_host NOT = { has_global_variable = z2_ended } }
    immediate = { z2_restore_effect = yes }
    right_portrait = scope:z2_founder
    option = { name = z2_warning.a }
}
zerywani_second.2 = {
    type = country_event
    hidden = yes
    trigger = { has_variable = z2_preparing NOT = { has_global_variable = z2_ended } }
    immediate = {
        z2_restore_effect = yes
        if = {
            # A foreign war postpones the council; it does not erase the crisis.
            limit = { war = yes has_civil_war = no }
            trigger_event = { id = zerywani_second.2 days = 30 }
        }
        else_if = {
            limit = {
                z2_base_survives_trigger = yes
                z2_crown_host_trigger = yes
                exists = var:z2_founder
                var:z2_founder = { is_alive = yes }
                exists = var:z2_goal
                var:z2_goal = { has_owner = yes }
            }
            if = {
                limit = { is_ai = no }
                set_variable = { name = z2_player_coronation_unlocked }
                if = {
                    limit = { NOT = { has_variable = z2_coronation_pending } }
                    set_variable = { name = z2_coronation_pending }
                    trigger_event = { id = zerywani_second.140 days = 1 }
                }
            }
            else = { z2_start_effect = yes }
        }
        else = { z2_cancel_preparation_effect = yes }
    }
}

# Console helpers: .999 begins the warnings, .998 skips the four-year wait,
# .996 breaks an active confederacy. None reset the one-time global gates.
zerywani_second.999 = {
    type = country_event
    hidden = yes
    immediate = {
        if = {
            limit = {
                NOT = { has_global_variable = z2_warned }
                z2_base_survives_trigger = yes
            }
            set_global_variable = { name = z2_debug_testing }
            ordered_country = {
                limit = { z2_host_trigger = yes }
                order_by = z2_host_score
                position = 0
                z2_prepare_effect = yes
            }
        }
    }
}

# Resolve after character-death/succession processing has finished, not inside it.
zerywani_second.50 = {
    type = country_event
    hidden = yes
    immediate = {
        if = { limit = { has_variable = z2_active NOT = { z2_founder_rules_trigger = yes } } z2_collapse_effect = yes }
        if = {
            limit = {
                has_variable = z2_preparing
                OR = { NOT = { exists = var:z2_founder } var:z2_founder = { is_alive = no } }
            }
            z2_cancel_preparation_effect = yes
        }
    }
}
