# Candidate confirmation is free. Costs and the global lock begin on assent.
omni_crusade.1 = {
    type = country_event
    title = omni_crusade.1.t
    desc = omni_crusade.1.desc
    picture = omni_shannah_crusade
    left_portrait = current_ruler
    immediate = {
        save_scope_as = omni_crusade_actor
        if = {
            limit = {
                has_variable = omni_crusade_pending
                exists = var:omni_crusade_target
            }
            # The council has already selected a country. Do not reroll it.
            var:omni_crusade_target = { save_scope_as = omni_crusade_target }
        }
        else = {
            # Retain the direct event .1 testing entry point.
            random_country = {
                limit = { omni_crusade_target_trigger = yes }
                save_scope_as = omni_crusade_target
            }
        }
        if = {
            limit = { exists = scope:omni_crusade_target }
            set_variable = { name = omni_crusade_target value = scope:omni_crusade_target }
            set_variable = omni_crusade_pending
        }
    }
    option = {
        name = omni_crusade.1.a
        trigger = {
            has_variable = omni_crusade_pending
            NOT = { has_global_variable = omni_crusade_world_busy }
            NOT = { has_variable = called_helluvian_crusade_var }
            omni_crusade_host_ready_trigger = yes
            political_influence >= 50
            treasury >= 250
            # Compare with this event's candidate, not a subsequent proposal.
            exists = var:omni_crusade_target
            var:omni_crusade_target = { this = scope:omni_crusade_target }
            scope:omni_crusade_target = { omni_crusade_target_trigger = yes }
        }
        ai_chance = { factor = 4 }
        custom_tooltip = omni_crusade_preparation_tt
        hidden_effect = { omni_crusade_begin_preparation_effect = yes }
    }
    option = {
        name = omni_crusade.1.b
        ai_chance = {
            factor = 1
            modifier = {
                factor = 0
                is_ai = yes
                has_variable = omni_crusade_pending
                omni_crusade_host_ready_trigger = yes
                treasury >= 250
                political_influence >= 50
                NOT = { has_global_variable = omni_crusade_world_busy }
                NOT = { has_variable = called_helluvian_crusade_var }
                omni_crusade_target_still_valid_trigger = yes
            }
        }
        if = {
            limit = {
                NOT = { has_variable = omni_crusade_preparing }
                NOT = { has_variable = omni_crusade_active }
            }
            remove_variable = omni_crusade_pending
            remove_variable = omni_crusade_target
        }
    }
}

# Capture the war on the next day, after the declaration has registered.
# Membership and recruitment follow Invictus flavor_cap.3.
omni_crusade.9 = {
    type = country_event
    hidden = yes
    trigger = { has_variable = omni_crusade_preparing }
    immediate = {
        omni_crusade_restore_scopes_effect = yes
        if = {
            limit = { any_current_war = { omni_crusade_matching_war_trigger = yes } }
            random_current_war = {
                limit = { omni_crusade_matching_war_trigger = yes }
                save_scope_as = omni_crusade_war
            }
            remove_variable = omni_crusade_preparing
            set_variable = omni_crusade_active
            debug_log = "OMNI_CRUSADE: matching host-led war found; activating expedition and recruiting pledgers."
            # Preserve the original expedition's nine subunits, but only after
            # a real declaration. Cancelling cannot repeatedly create armies.
            create_unit = {
                navy = no
                name = "Warriors of the Omostthanat"
                location = capital_scope
                hidden_effect = {
                    add_subunit = heavy_infantry
                    add_subunit = heavy_infantry
                    add_subunit = heavy_infantry
                    add_subunit = light_infantry
                    add_subunit = light_infantry
                    add_subunit = heavy_cavalry
                    add_subunit = heavy_cavalry
                    add_subunit = supply_train
                    add_subunit = supply_train
                    add_food = 100
                    add_morale = 2
                }
            }
            omni_crusade_recruit_effect = yes
            omni_crusade_recruit_defenders_effect = yes
            omni_crusade_announce_launch_effect = yes
        }
        else = {
            if = {
                limit = { war = no }
                debug_log = "OMNI_CRUSADE: declaration check failed; host is still at peace."
            }
            else = {
                debug_log = "OMNI_CRUSADE: declaration check failed; host is at war but no matching host-led war was found."
            }
            omni_crusade_cancel_effect = yes
        }
    }
}

# Reconstruct from persistent country references after reload. No assumed war
# outcome or contribution ranking. Leadership loss also releases our state.
omni_crusade.10 = {
    type = country_event
    hidden = yes
    trigger = { has_variable = omni_crusade_active }
    immediate = {
        omni_crusade_restore_scopes_effect = yes
        if = {
            limit = {
                OR = {
                    NOT = { exists = var:omni_crusade_target }
                    NOT = { any_current_war = { omni_crusade_matching_war_trigger = yes } }
                }
            }
            # Losing leadership alone ends administration, but is not victory.
            if = {
                limit = { NOT = { exists = var:omni_crusade_target } }
                omni_crusade_convert_conquests_effect = yes
            }
            else_if = {
                limit = {
                    NOT = {
                        any_current_war = {
                            any_war_attacker = { this = scope:omni_crusade_actor }
                            any_war_defender = { this = scope:omni_crusade_target }
                        }
                    }
                }
                omni_crusade_convert_conquests_effect = yes
            }
            omni_crusade_finish_effect = yes
        }
    }
}
