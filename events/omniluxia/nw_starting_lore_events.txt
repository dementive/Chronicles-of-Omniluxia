lorestart.36 = { # Saurthi
	type = country_event
	title = lorestart.36.t
	desc = lorestart.36.desc
	picture = event_swamp
	left_portrait = c:Q01.current_ruler
	left_portrait = c:Q38.current_ruler
	right_portrait = c:X00.current_ruler
	right_portrait = c:X13.current_ruler

	mean_time_to_happen = {
		days = 1
	}

	fire_only_once = yes

	trigger = {
		is_ai = no
		country_culture_group = saurthi
	}

	option = {
		name = lorestart.36.a
		hidden_effect = {
			set_variable = nw_lore_received
			current_ruler = {
				force_add_trait = blood_of_saurthi
			}
			c:Q15 = {
				set_antagonist = yes
			}
			c:Q16 = {
				set_antagonist = yes
			}
			c:Q39 = {
				set_antagonist = yes
			}
		}
	}
}
