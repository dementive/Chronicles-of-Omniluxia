# Sixty days after the summons: the council opens and the host chooses its agenda.
omni_pastoral_council.5 = {
    type = country_event
    title = omni_pastoral_council.5.t
    desc = {
        desc = omni_pastoral_council.5.desc
        first_valid = {
            triggered_desc = { trigger = { NOT = { omni_pc_turnout_trigger = { N = 1 } } } desc = omni_pastoral_council.5.empty }
            triggered_desc = { trigger = { always = yes } desc = omni_pastoral_council.5.attendance }
        }
        first_valid = {
            triggered_desc = { trigger = { trigger_if = { limit = { has_variable = omni_pc_spies_caught } var:omni_pc_spies_caught >= 1 } trigger_else = { always = no } } desc = omni_pastoral_council.5.spies }
        }
        first_valid = {
            triggered_desc = { trigger = { exists = scope:omni_pc_dissenter } desc = omni_pastoral_council.5.dissenter }
        }
        desc = omni_pastoral_council.5.rules
    }
    picture = omni_shannah_petition
    left_portrait = current_ruler
    trigger = { has_variable = omni_pc_summons_pending NOT = { has_variable = omni_pc_ballot_open } }

    immediate = {
        omni_pc_count_effect = yes
        omni_pc_catch_spies_effect = yes
        if = {
            limit = { exists = scope:omni_pc_dissenter }
            scope:omni_pc_dissenter = { current_ruler = { save_scope_as = omni_pc_dissenter_ruler } }
        }
    }

    option = { # Canons of Unity
        name = omni_pastoral_council.5.a
        trigger = { omni_pc_agenda_available_trigger = yes }
        custom_tooltip = omni_pc_proposal_tt
        omni_pc_propose_effect = { AGENDA = 1 }
        ai_chance = { factor = 40 }
    }
    option = { # Discipline the Clergy
        name = omni_pastoral_council.5.b
        trigger = { omni_pc_agenda_available_trigger = yes }
        custom_tooltip = omni_pc_proposal_tt
        omni_pc_propose_effect = { AGENDA = 2 }
        ai_chance = { factor = 20 modifier = { factor = 2 omni_hc_authority < 50 } }
    }
    option = { # Embassy to the Dissenters
        name = omni_pastoral_council.5.c
        trigger = { omni_pc_agenda_available_trigger = yes exists = scope:omni_pc_dissenter scope:omni_pc_dissenter = { omni_pc_dissenter_trigger = yes } }
        custom_tooltip = omni_pc_proposal_tt
        omni_pc_propose_effect = { AGENDA = 3 }
        ai_chance = { factor = 20 }
    }
    option = { # Endow the Faith
        name = omni_pastoral_council.5.d
        trigger = { omni_pc_endow_available_trigger = yes }
        custom_tooltip = omni_pc_proposal_tt
        omni_pc_propose_effect = { AGENDA = 4 }
        ai_chance = { factor = 20 modifier = { factor = 0 treasury < 350 } }
    }
    option = { # Dismiss
        name = omni_pastoral_council.5.e
        omni_pc_dismiss_effect = yes
        ai_chance = { factor = 0.01 modifier = { factor = 0 omni_hc_shannah_trigger = yes } }
    }
    option = {
        name = omni_pastoral_council.5.f
        trigger = { omni_hc_shannah_trigger = yes NOT = { omni_pc_agenda_available_trigger = yes } }
        custom_tooltip = omni_pastoral_council.5.f_tt
        hidden_effect = { trigger_event = { id = omni_pastoral_council.9 days = 15 } }
        ai_chance = { factor = 100 }
    }
}

# Incident dispatcher, 30-60 days after the council. Trouble is likelier for a weak Shahanat.
omni_pastoral_council.10 = {
    type = country_event
    hidden = yes
    trigger = { omni_hc_shannah_trigger = yes }
    immediate = {
        omni_pc_count_effect = yes
        current_ruler = { save_scope_as = omni_pc_shannah }
        clear_saved_scope = omni_pc_rival_a
        clear_saved_scope = omni_pc_rival_b
        if = {
            limit = { omni_pc_turnout_trigger = { N = 2 } }
            random_country = { limit = { omni_pc_attending_trigger = yes } save_scope_as = omni_pc_rival_a }
            random_country = { limit = { omni_pc_attending_trigger = yes NOT = { this = scope:omni_pc_rival_a } } save_scope_as = omni_pc_rival_b }
        }
        random_list = {
            25 = {
                modifier = { factor = 2 omni_hc_authority < 50 }
                modifier = { factor = 0.5 omni_hc_authority >= 75 }
                trigger_event = { id = omni_pastoral_council.11 }
            }
            25 = {
                modifier = { factor = 0 NOT = { exists = scope:omni_pc_rival_b } }
                trigger_event = { id = omni_pastoral_council.12 }
            }
            20 = {
                modifier = { factor = 2 omni_hc_authority >= 75 }
                modifier = { factor = 0.5 omni_hc_authority < 25 }
                trigger_event = { id = omni_pastoral_council.13 }
            }
            20 = {
                modifier = { factor = 1.5 omni_hc_authority < 50 }
                trigger_event = { id = omni_pastoral_council.14 }
            }
            10 = { debug_log = "OMNI_COUNCIL: the council closed without incident." }
        }
    }
}
