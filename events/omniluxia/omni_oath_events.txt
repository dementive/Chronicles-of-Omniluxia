namespace = omni_oath

omni_oath.1 = {
    type = country_event
    title = omni_oath.1.t
    desc = omni_oath.1.desc
    picture = event_rohan_riders
    left_portrait = current_ruler
    left_portrait = scope:oath_chief_left
    right_portrait = scope:sunaksatre
    right_portrait = scope:oath_chief_right
    goto_location = capital_scope

    immediate = {
        hidden_effect = {
            random_character = {
                limit = {
                    is_alive = yes
                    is_adult = yes
                    is_ruler = no
                    prisoner = no
                    OR = { is_clan_chief = yes is_head_of_family = yes }
                }
                save_scope_as = oath_chief_left
            }
            if = {
                limit = { exists = scope:oath_chief_left }
                random_character = {
                    limit = {
                        is_alive = yes
                        is_adult = yes
                        is_ruler = no
                        prisoner = no
                        OR = { is_clan_chief = yes is_head_of_family = yes }
                        NOT = { this = scope:oath_chief_left }
                    }
                    save_scope_as = oath_chief_right
                }
            }
            create_character = {
                first_name = "Sunaksatre"
                culture = sind
                religion = dragkhan_pantheon
                female = no
                age = 28
                no_stats = yes
                add_martial = 10
                add_finesse = 6
                add_charisma = 10
                add_zeal = 11
                add_trait = founder
                add_trait = tactician
                save_scope_as = sunaksatre
            }
        }
    }

    option = {
        name = omni_oath.1.a
        trigger = { political_influence >= 30 }
        custom_tooltip = omni_oath.1.a_tt
        add_political_influence = -30
        set_as_ruler = scope:sunaksatre
        scope:sunaksatre = { add_popularity = 10 }
        hidden_effect = { omni_oath_found = yes }
    }

    option = {
        name = omni_oath.1.b
        custom_tooltip = omni_oath.1.b_tt
        add_manpower = 1
        hidden_effect = { omni_oath_found = yes }
    }
}

omni_oath.3 = {
    type = country_event
    title = omni_oath.3.t
    desc = omni_oath.3.desc
    picture = event_corsair_ships
    left_portrait = current_ruler
    right_portrait = scope:oath_luxterran_envoy
    goto_location = scope:oath_port

    trigger = {
        has_variable = omni_oath_morrigon_open
        NOT = { has_variable = omni_oath_luxterra_open }
        p:4864 = { owner = root }
        any_owned_province = { is_port = yes is_in_region = dragon_lands_west }
    }

    immediate = {
        random_owned_province = {
            limit = { is_port = yes is_in_region = dragon_lands_west }
            save_scope_as = oath_port
        }
        random_province = {
            limit = {
                is_port = yes
                omni_oath_luxterra_province = yes
                has_owner = yes
                is_capital = no
                owner = { is_subject = no NOR = { this = root war_with = root } }
            }
            save_scope_as = oath_landing_port
        }
        if = {
            limit = { exists = scope:oath_landing_port }
            scope:oath_landing_port.owner.current_ruler = { save_scope_as = oath_luxterran_envoy }
        }
        else = {
            create_character = { religion = beatepian_pantheon save_scope_as = oath_luxterran_envoy }
        }
        set_variable = omni_oath_luxterra_open
        if = {
            limit = { has_variable = omni_oath_guarantees }
            every_owned_province = {
                limit = { is_in_region = dragon_lands_west }
                add_province_modifier = { name = omni_oath_guaranteed_towns duration = 3650 }
            }
        }
    }

    option = {
        name = omni_oath.3.a
        trigger = {
            has_variable = omni_oath_guarantees
            exists = scope:oath_landing_port
            treasury >= 100
            political_influence >= 20
        }
        custom_tooltip = omni_oath.3.a_tt
        add_treasury = -100
        add_political_influence = -20
        add_manpower = 1
        hidden_effect = {
            omni_oath_luxterra_claims = yes
            root = { save_scope_as = oath_patron }
            scope:oath_landing_port.owner = {
                trigger_event = { id = omni_oath.6 days = 14 }
            }
        }
    }

    option = {
        name = omni_oath.3.b
        trigger = { treasury >= 150 }
        custom_tooltip = omni_oath.3.b_tt
        add_treasury = -150
        create_unit = {
            name = "Fleet of the Oath"
            navy = yes
            location = scope:oath_port
            add_subunit = liburnian
            add_subunit = liburnian
            add_subunit = liburnian
            add_subunit = liburnian
            add_subunit = liburnian
            add_subunit = liburnian
        }
        hidden_effect = { omni_oath_luxterra_claims = yes }
    }

    option = {
        name = omni_oath.3.c
        custom_tooltip = omni_oath.3.c_tt
        hidden_effect = { omni_oath_luxterra_claims = yes }
    }
}
