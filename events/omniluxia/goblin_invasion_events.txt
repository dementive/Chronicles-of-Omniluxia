namespace = goblin_invasion

# The Goblin Invasion of the World's Lungs. Script: common/scripted_effects/omni_goblin_invasion.txt
#   .1  keeper of the Root warned (60 days before the march)
#   .2  hidden: the march begins, or is called off
#   .3/.4/.5/.6  launch: attacker / defender (and overlord) / Aneic onlookers / other onlookers
#   .10 hidden monitor (30 days)  .11 hidden reinforcements  .12 hidden: the host goes home
#   .20/.21/.22/.23  the Root falls: burner / keeper / Aneic realms / other onlookers
#   .30/.31/.32  the Root stands: keeper / goblins / onlookers
# Portraits: scopes gi_leader, gi_keeper, gi_target, gi_burner, gi_mourner are saved by the
# effects and carried into the triggered events.

# DEBUG/CONSOLE ONLY: `event goblin_invasion.999` forces the strongest eligible goblin realm to
# march now (the warning still runs 60 days ahead of the war).
goblin_invasion.999 = {
	type = country_event
	hidden = yes
	immediate = {
		ordered_country = {
			order_by = goblin_invasion_leader_score_svalue
			position = 0
			limit = { goblin_invasion_eligible_trigger = yes }
			goblin_invasion_formation_effect = yes
		}
	}
}

		}
		ai_chance = { factor = 50 }
	}
	option = {
		name = goblin_invasion.1.c
		ai_chance = { factor = 10 }
	}
}

# Leader: 60 days on, march or stand down.
goblin_invasion.2 = {
	type = country_event
	hidden = yes
	trigger = { has_variable = is_goblin_invasion_leader }
	immediate = {
		save_scope_as = gi_self
		if = {
			limit = {
				goblin_invasion_root_stands_trigger = yes
				NOT = { owns = 2356 }
				p:2356 = {
					owner = {
						NOT = {
							AND = {
								is_subject = yes
								overlord = { this = scope:gi_self }
							}
						}
					}
				}
			}
			goblin_invasion_launch_effect = yes
		}
		else = {
			goblin_invasion_abort_effect = yes
		}
	}
}

# Monitor (leader, every 30 days).
goblin_invasion.10 = {
	type = country_event
	hidden = yes
	trigger = { has_variable = is_goblin_invasion_leader }
	immediate = {
		save_scope_as = gi_leader
		if = {
			limit = { exists = var:gi_keeper }
			var:gi_keeper = { save_scope_as = gi_keeper }
		}
		if = {
			limit = { has_global_variable = root_of_first_spring_felled }
			# Burned by us through the decision, or by somebody else. Either way the host goes home.
			trigger_event = { id = goblin_invasion.12 days = 90 }
		}
		else_if = {
			limit = { p:2356 = { controller = root } }
			if = {
				limit = { is_ai = yes }
				root_of_first_spring_fell_effect = yes
				trigger_event = { id = goblin_invasion.12 days = 90 }
			}
			else = {
				# A player leader decides with omni_fell_root_of_first_spring.
				trigger_event = { id = goblin_invasion.10 days = 30 }
			}
		}
		else_if = {
			limit = {
				exists = var:gi_target
				war_with = var:gi_target
			}
			trigger_event = { id = goblin_invasion.10 days = 30 }
		}
		else = {
			# The war is over and the Root still stands.
			trigger_event = { id = goblin_invasion.31 days = 1 }
			if = {
				limit = { exists = scope:gi_keeper }
				scope:gi_keeper = {
					trigger_event = { id = goblin_invasion.30 days = 1 }
				}
			}
			every_country = {
				limit = {
					is_ai = no
					NOT = { this = scope:gi_leader }
					NOT = { this = scope:gi_keeper }
				}
				trigger_event = { id = goblin_invasion.32 days = 2 }
			}
		}
	}
}
