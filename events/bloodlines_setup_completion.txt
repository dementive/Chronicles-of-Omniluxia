## If adding bloodlines no changes to this file are needed. Change bloodline_setup_effect in file common/scripted_effects/00_bloodlines_inheritance_inv.txt 

namespace = bloodlines_setup_completion

bloodlines_setup_completion.1 = {
	type = country_event
	hidden = yes

	# PERF: this event sits in main_event_pulse_country's events list, which
	# re-fires it for every country on every main pulse forever. The run_* variable
	# machine below only stops the SELF-retrigger chain, not the pulse re-fire, so
	# without this trigger every country repeated the full bloodline setup pass
	# (a whole-court character scan) once a year for the entire campaign.
	# The three setup passes still run for every country, including newly spawned
	# ones (no vars set -> trigger passes). Inheritance after setup is handled by
	# bloodline_on_born_effect on birth, so re-running the pass added nothing.
	trigger = {
		NOT = { has_variable = run_once }
	}

	immediate = {

		bloodline_setup_effect = yes

		if = {
			limit = {
				NOT = {
					has_variable = run_two_more_times
				} 
				NOT = {
					has_variable = run_one_more_time
				}
				NOT = {
					has_variable = run_once
				}
			}
			set_variable = run_two_more_times
		}

		else_if = {
			limit = {
				has_variable = run_two_more_times
			}
			set_variable = run_one_more_time
			remove_variable = run_two_more_times
		}

		else_if = {
			limit = {
				has_variable = run_one_more_time
			}
			set_variable = run_once
			remove_variable = run_one_more_time
		}

		hidden_effect = {
			if = {
				limit = {
					OR = {
						has_variable = run_two_more_times
						has_variable = run_one_more_time
					}
				}
				trigger_event = {
					id = bloodlines_setup_completion.1
				}
			}
		}
	}
}
