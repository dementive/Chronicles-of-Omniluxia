# Compatibility for events already queued in existing saves and new-country hooks.
# The old implementation ran three passes per country, which could still miss
# deeper ancestry and did not initialize missing racial traits.
namespace = bloodlines_setup_completion

bloodlines_setup_completion.1 = {
    type = country_event
    hidden = yes
    immediate = {
        if = {
            limit = { NOT = { has_global_variable = omni_bloodline_initialization_v2 } }
            every_living_character = { set_racial_trait_effect = yes }
            bloodline_setup_effect = yes
            set_global_variable = { name = omni_bloodline_initialization_v2 value = 1 }
        }
    }
}
