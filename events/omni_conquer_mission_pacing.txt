namespace = omni_conquer_pacing

# Lifts the start-of-game blocker on generic_mission_conquer_region for this country.
# See common/on_action/omni_conquer_mission_pacing.txt.
omni_conquer_pacing.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = omni_conquer_mission_delay }
			remove_variable = omni_conquer_mission_delay
		}
	}

	option = {}
}

# Ends this country's start-of-game early peace (common/ai_plan_goals/omni_early_peace.txt).
omni_conquer_pacing.2 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { has_variable = omni_ai_early_peace }
			remove_variable = omni_ai_early_peace
		}
	}

	option = {}
}
