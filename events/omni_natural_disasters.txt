namespace = omni_disaster

# Ported from Invictus (disasters_inv.txt), reskinned for Omniluxia:
# - Floods no longer require the Invictus-only "flood_plain" terrain (which Omniluxia's map doesn't have painted).
#   They now trigger on farmland/plains, which every country already has.
# - Snowstorm no longer hardcodes real-world Invictus regions (scandinavia_region, germania_region, etc).
#   It now fires on any owned province currently experiencing winter (has_winter = yes) - i.e. wherever it's
#   actually snowy season right now, wherever that happens to be on the Omniluxia map.
# - Locusts no longer hardcode Arabia/Nubia/Egypt. They fire on desert-terrain provinces, and only during the
#   in-game "summer" window (harvest_summer_var, set by the harvest season system in 000_harvest_seasons.txt).
# All buildings/modifiers/effects referenced are vanilla or already exist elsewhere in Omniluxia.

###Floods
omni_disaster.1 = { #flood on farmland
	type = country_event
	title = omni_disaster.1.t
	desc = omni_disaster.1.desc
	picture = farming

	left_portrait = current_ruler
	right_portrait = scope:the_day_after.governor
	goto_location = scope:the_day_after

	trigger = {
		any_owned_province = {
			OR = {
				terrain = farmland
				terrain = plains
			}
			total_population >= 4
		}
		NOT = {
			has_variable = floods_cooldown
		}
	}

	immediate = {
		random_owned_province = {
			limit = {
				OR = {
					terrain = farmland
					terrain = plains
				}
				total_population >= 4
			}
			save_scope_as = the_day_after
			add_province_modifier = {
				name = omni_flood_initial_damage
				duration = 730
			}
			ordered_pops_in_province = {
				order_by = pop_hapiness
				max = {
					value = total_population
					multiply = 0.15
				}
				kill_pop = yes
			}
			custom_tooltip = omni_disaster_destroyed_building
			hidden_effect = {
				random_list = {
					1 = {
						trigger = { has_building = commerce_building }
						remove_building_level = commerce_building
					}
					1 = {
						trigger = { has_building = library_building }
						remove_building_level = library_building
					}
					1 = {
						trigger = { has_building = theathre_building }
						remove_building_level = theathre_building
					}
					1 = {
						trigger = { has_building = academy_building }
						remove_building_level = academy_building
					}
					1 = {
						trigger = { has_building = population_building } # granary
						remove_building_level = population_building
					}
					1 = {
						trigger = { has_building = forum_building }
						remove_building_level = forum_building
					}
					1 = {
						trigger = { has_building = workshop_building }
						remove_building_level = workshop_building
					}
					1 = {
						trigger = { has_building = port_building }
						remove_building_level = port_building
					}
					8 = {}
				}
			}
		}
	}

	option = { #secure the area
		name = omni_disaster.1.a
		current_ruler = {
			add_popularity = 5
		}
		add_treasury = {
			subtract = has_monthly_income
			multiply = 6
			floor = yes
			max = -50
			min = -200
		}
		trigger_event = {
			id = omni_disaster.3
			days = { 10 20 }
		}
	}

	option = { #greedy bastard
		name = omni_disaster.1.b
		hidden_effect = {
			if = {
				limit = {
					scope:the_day_after = {
						any_neighbor_province = {
							exists = owner
							owner = root
							OR = {
								terrain = farmland
								terrain = plains
							}
						}
					}
				}
				random = {
					chance = 50
					trigger_event = {
						id = omni_disaster.2
						days = { 1 7 }
					}
				}
			}
		}
	}

	after = {
		set_variable = {
			name = floods_cooldown
			days = 9125 #25yrs
		}
	}
}

###Avalanche
omni_disaster.4 = { #Avalanche
	type = country_event
	title = omni_disaster.4.t
	desc = omni_disaster.4.desc
	picture = natural_disaster

	left_portrait = current_ruler
	right_portrait = scope:snowball.governor
	goto_location = scope:snowball

	trigger = {
		any_owned_province = {
			terrain = mountain
			total_population >= 4
		}
		NOT = {
			has_variable = avalanche_cooldown
		}
	}

	immediate = {
		random_owned_province = {
			limit = {
				terrain = mountain
				total_population >= 4
			}
			if = {
				limit = {
					NOT = {
						has_variable = starting_point
					}
				}
				set_variable = {
					name = starting_point
					days = 2
				}
			}
			switch = {
				trigger = has_variable
				starting_point = {
					save_scope_as = snowball
				}
				recurring_disaster = {}
			}
			add_province_modifier = {
				name = omni_hit_by_avalanche
				duration = 730
			}
			ordered_pops_in_province = {
				order_by = pop_hapiness
				max = {
					value = total_population
					multiply = 0.35
				}
				kill_pop = yes
			}
			if = {
				limit = {
					any_unit_in_province = {
						exists = yes
					}
				}
				every_unit_in_province = {
					damage_unit_percent = 0.06
					damage_unit_morale_percent = 0.12
				}
			}
			custom_tooltip = omni_disaster_destroyed_building
			hidden_effect = {
				random_list = {
					1 = {
						trigger = { has_building = commerce_building }
						remove_building_level = commerce_building
					}
					1 = {
						trigger = { has_building = library_building }
						remove_building_level = library_building
					}
					1 = {
						trigger = { has_building = theathre_building }
						remove_building_level = theathre_building
					}
					1 = {
						trigger = { has_building = academy_building }
						remove_building_level = academy_building
					}
					1 = {
						trigger = { has_building = population_building } # granary
						remove_building_level = population_building
					}
					1 = {
						trigger = { has_building = forum_building }
						remove_building_level = forum_building
					}
					1 = {
						trigger = { has_building = workshop_building }
						remove_building_level = workshop_building
					}
					1 = {
						trigger = { has_building = port_building }
						remove_building_level = port_building
					}
					8 = {}
				}
			}
		}
	}

	option = {
		name = omni_disaster.4.a
	}

	after = {
		if = {
			limit = {
				scope:snowball = {
					any_neighbor_province = {
						terrain = mountain
						NOT = {
							has_province_modifier = omni_hit_by_avalanche
						}
					}
				}
			}
			scope:snowball = {
				clear_saved_scope = snowball
				random_neighbor_province = {
					limit = {
						terrain = mountain
						NOT = {
							has_province_modifier = omni_hit_by_avalanche
						}
					}
					save_scope_as = snowball
					set_variable = {
						name = recurring_disaster
						days = 7
					}
					random = {
						chance = 50
						owner = {
							trigger_event = {
								id = omni_disaster.4
								days = { 3 5 }
							}
						}
					}
				}
			}
		}
		else = {
			set_variable = {
				name = avalanche_cooldown
				days = 9125 #25 yrs
			}
		}
	}
}

###Drought - May turn into a famine
omni_disaster.7 = { #drought
	type = country_event
	title = omni_disaster.7.t
	desc = omni_disaster.7.desc
	picture = farming

	left_portrait = current_ruler
	right_portrait = scope:i_am_thirsty.governor
	goto_location = scope:i_am_thirsty

	trigger = {
		any_country_state = {
			OR = {
				state_monthly_food_income < 0
				has_state_food <= 0
			}
		}
		any_owned_province = {
			OR = {
				terrain = desert
				terrain = desert_hills
			}
		}
		NOT = {
			has_variable = drought_cooldown
		}
	}

	immediate = {
		random_country_state = {
			limit = {
				OR = {
					state_monthly_food_income < 0
					has_state_food <= 0
				}
			}
			save_scope_as = i_am_thirsty
			add_state_modifier = {
				name = omni_grow_a_pear
				duration = 1825 #5 yrs
			}
		}
	}

	option = { #try to save and feed your people
		name = omni_disaster.7.a
		scope:i_am_thirsty = {
			area = {
				every_neighbor_area = {
					every_area_state = {
						add_state_food = -100
						add_state_modifier = {
							name = omni_i_bet_you_wish_you_had_my_water
							duration = 365
						}
					}
				}
			}
			add_state_food = {
				value = 100
				multiply = 2 #count neighboring states
			}
		}
		add_treasury = {
			subtract = has_monthly_income
			multiply = 16
			min = -300
			max = -50
		}
		current_ruler = {
			add_trait = benefactor
		}
		hidden_effect = {
			if = {
				limit = {
					scope:i_am_thirsty.area = {
						any_neighbor_area = {
							any_area_state = {
								has_state_food <= 0
							}
						}
					}
				}
				scope:i_am_thirsty.area = {
					every_neighbor_area = {
						every_area_state = {
							limit = {
								has_state_food <= 0
							}
							add_state_modifier = {
								name = omni_grow_a_pear
								duration = 1825 # 5yrs
							}
						}
					}
				}
			}
		}
	}

	option = { #ignore the area
		name = omni_disaster.7.b
		current_ruler = {
			add_trait = harsh
		}
		hidden_effect = {
			trigger_event = {
				id = omni_disaster.8 #drought spreads
				days = { 30 100 }
			}
			scope:i_am_thirsty.area = {
				every_neighbor_area = {
					every_area_state = {
						limit = {
							owner = root
						}
						add_state_food = -150
					}
				}
			}
		}
	}
}
