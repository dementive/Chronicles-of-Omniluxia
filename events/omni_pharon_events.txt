namespace = omni_pharon

# Country scope, after ruler-change processing has settled. This deliberately
# selects current_ruler at execution time instead of caching a possibly stale heir.
omni_pharon.1 = {
    type = country_event
    hidden = yes

    trigger = {
        government = egyptian_dynasty
        exists = current_ruler
        current_ruler = { NOT = { has_trait = Pharon } }
    }

    immediate = {
        omni_pharon_ensure_ruler_trait_effect = yes
    }
}
