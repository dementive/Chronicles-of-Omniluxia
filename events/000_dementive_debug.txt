namespace = dementive_debug

dementive_debug.1 = {
	type = country_event

	title = dementive_debug.1_title
	desc = dementive_debug.1_desc
	picture = ascetic
	left_portrait = current_ruler

	option = {
		name = dementive_debug.1.a
		custom_tooltip = dementive_debug.1.a.tt
		current_ruler = {
			if = {
				limit = {
					NOT = { has_variable = magic }
				}
				set_variable = {
					name = magic
					value = 0
				}
			}
			recalculate_character_magic_vars = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.b
		custom_tooltip = dementive_debug.1.b.tt
		current_ruler = {
			remove_character_modifier = chararacter_magic_4
			add_character_modifier = {
				name = chararacter_magic_4
				duration = -1
			}
			recalculate_character_magic_vars = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.c
		custom_tooltip = dementive_debug.1.c.tt
		current_ruler = {
			add_character_modifier = {
				name = magic_debug_character_max_25
				duration = -1
				mode = add
			}
			recalculate_character_magic_vars = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.d
		custom_tooltip = dementive_debug.1.d.tt
		current_ruler = {
			change_character_magic_monthly = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.e
		custom_tooltip = dementive_debug.1.e.tt
		current_ruler = {
			if = {
				limit = {
					NOT = { has_variable = magic }
				}
				set_variable = {
					name = magic
					value = 0
				}
			}
			change_variable = {
				name = magic
				add = 25
			}
			clamp_character_magic_to_max = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.f
		custom_tooltip = dementive_debug.1.f.tt
		current_ruler = {
			remove_character_modifier = chararacter_magic_4
			remove_character_modifier = magic_debug_character_max_25
			set_variable = {
				name = magic
				value = 0
			}
			recalculate_character_magic_vars = yes
		}
		trigger_event = { id = dementive_debug.1 }
	}

	option = {
		name = dementive_debug.1.g

	}
}
