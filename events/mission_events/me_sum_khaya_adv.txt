namespace = sum_khaya_adv

# ═══════════════════════════════════════════════════════════════════════
# SUM KHAYA ADVENTURE CHAIN
# A personal ruler excursion into the infamous slums of Province 653
# Three branching paths based on the ruler's attributes:
#   .1 → choice event
#   .2 → stealth path (finesse, discovers the Rogue's Ledger)
#   .3 → open path (martial, dockside test)
#   .4 → gold path (coin, audience with the King of Thieves)
# ═══════════════════════════════════════════════════════════════════════

# Event 1 — "The City of Thieves" (branching choice)
sum_khaya_adv.1 = {
	type = country_event

	title = sum_khaya_adv.1.t
	desc = sum_khaya_adv.1.desc
	picture = celtic_village
	left_portrait = current_ruler

	option = {
		name = sum_khaya_adv.1.a
		trigger = {
			current_ruler = { finesse >= 7 }
		}
		custom_tooltip = sum_khaya_adv.1.a.tt
		trigger_event = {
			id = sum_khaya_adv.2
			days = { 5 10 }
		}
	}

	option = {
		name = sum_khaya_adv.1.b
		trigger = {
			current_ruler = { martial >= 6 }
		}
		custom_tooltip = sum_khaya_adv.1.b.tt
		trigger_event = {
			id = sum_khaya_adv.3
			days = { 5 10 }
		}
	}

	option = {
		name = sum_khaya_adv.1.c
		trigger = {
			treasury >= 100
		}
		custom_tooltip = sum_khaya_adv.1.c.tt
		add_treasury = -100
		trigger_event = {
			id = sum_khaya_adv.4
			days = { 5 10 }
		}
	}

	option = {
		name = sum_khaya_adv.1.d
		add_political_influence = 5
	}
}

# ───────────────────────────────────────────────────────
# Event 2 — STEALTH PATH: "The Unrecognized Ruler"
# ───────────────────────────────────────────────────────
sum_khaya_adv.2 = {
	type = country_event

	title = sum_khaya_adv.2.t
	desc = sum_khaya_adv.2.desc
	picture = assassination
	left_portrait = current_ruler

	# Option A: Take the ledger and vanish
	option = {
		name = sum_khaya_adv.2.a
		current_ruler = {
			give_character_an_item = {
				item_ID = item_sum_khaya_rogue_ledger
			}
			add_finesse = 1
		}
		add_political_influence = 20
		add_country_modifier = {
			name = sum_khaya_shadows_compact_mod
			duration = 7300
		}
	}

	# Option B: Reveal yourself and turn the knowledge into leverage
	option = {
		name = sum_khaya_adv.2.b
		current_ruler = {
			add_finesse = 1
			add_popularity = 10
		}
		add_political_influence = 35
		add_stability = 5
	}
}

# ───────────────────────────────────────────────────────
# Event 3 — OPEN PATH: "The Dockside Test"
# ───────────────────────────────────────────────────────
sum_khaya_adv.3 = {
	type = country_event

	title = sum_khaya_adv.3.t
	desc = sum_khaya_adv.3.desc
	picture = great_work_riot
	left_portrait = current_ruler

	# Option A: Accept the challenge and win (reputation through victory)
	option = {
		name = sum_khaya_adv.3.a
		current_ruler = {
			add_popularity = 15
			add_martial = 1
		}
		p:653 = {
			add_permanent_province_modifier = {
				name = sum_khaya_street_respect_mod
				duration = -1
			}
		}
		add_military_experience = 10
	}

	# Option B: Decline the challenge but assert authority with force of command
	option = {
		name = sum_khaya_adv.3.b
		current_ruler = {
			force_add_trait = assertive
			add_popularity = -5
		}
		p:653 = {
			add_permanent_province_modifier = {
				name = sum_khaya_cold_authority_mod
				duration = -1
			}
		}
		add_political_influence = 20
	}
}

# ───────────────────────────────────────────────────────
# Event 4 — COIN PATH: "An Audience with the King of Thieves"
# ───────────────────────────────────────────────────────
sum_khaya_adv.4 = {
	type = country_event

	title = sum_khaya_adv.4.t
	desc = sum_khaya_adv.4.desc
	picture = war_chest
	left_portrait = current_ruler

	# Option A: Accept the compact (mutual protection arrangement)
	option = {
		name = sum_khaya_adv.4.a
		add_political_influence = 30
		p:653 = {
			add_permanent_province_modifier = {
				name = sum_khaya_kings_compact_mod
				duration = -1
			}
		}
		add_country_modifier = {
			name = sum_khaya_underworld_network_mod
			duration = 7300
		}
	}

	# Option B: Take the information, decline the arrangement
	option = {
		name = sum_khaya_adv.4.b
		add_political_influence = 50
		add_country_modifier = {
			name = sum_khaya_shadows_compact_mod
			duration = 7300
		}
		current_ruler = {
			add_charisma = 1
		}
	}
}
