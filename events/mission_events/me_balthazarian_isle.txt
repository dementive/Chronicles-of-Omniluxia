# Balthazarian Order, task 32: the night landing on the Qasuda Isles (Qivuk 6068, Qasuda 6069, Inqar 6070).
# Chain: 1 (task start, choose the plan) -> 2 (the fleet sails dark) -> 3 (the landing) -> 4 (the sentinels)
#        -> 5 (the fight for the Amber Quay) -> 6 (the isle lord) -> 7 (dawn, the isles pass to the Order).
# Choices are stored in variables on the Order: ob_isle_plan_harbor / ob_isle_plan_caves, ob_isle_silent / ob_isle_loud,
# ob_isle_mercy / ob_isle_hardline. All are cleared in ob_isle.7 (ob_isle_clear_vars).

# ---------------------------------------------------------------- 1. the chart table
ob_isle.1 = {
	type = country_event
	title = ob_isle.1.t
	desc = ob_isle.1.desc
	picture = omni_event_scene_northern_war_tent
	left_portrait = current_ruler
	right_portrait = scope:ob_isle_aux
	immediate = {
		ob_isle_save_scopes = yes
	}
	option = {
		name = ob_isle.1.a
		custom_tooltip = ob_isle.1.a.tt
		hidden_effect = {
			remove_variable = ob_isle_plan_caves
			set_variable = { name = ob_isle_plan_harbor }
		}
	}
	option = {
		name = ob_isle.1.b
		custom_tooltip = ob_isle.1.b.tt
		hidden_effect = {

# ---------------------------------------------------------------- 6. the isle lord
ob_isle.6 = {
	type = country_event
	title = ob_isle.6.t
	desc = ob_isle.6.desc
	picture = omni_city_night_moonlight
	left_portrait = scope:ob_isle_sword
	right_portrait = scope:ob_isle_lord
	immediate = {
		ob_isle_save_scopes = yes
	}
	option = {
		name = ob_isle.6.a
		custom_tooltip = ob_isle.6.a.tt
		hidden_effect = {
			# Spared: the lord joins the Order's court as a guest of the chapter.
			ob_isle_take_lord = yes
			remove_variable = ob_isle_hardline
			set_variable = { name = ob_isle_mercy }
			trigger_event = { id = ob_isle.7 days = 1 }
		}
	}
	option = {
		name = ob_isle.6.b
		custom_tooltip = ob_isle.6.b.tt
		hidden_effect = {
			# Sent north in chains: brought into the Order, imprisoned, and hurt on the way.
			ob_isle_take_lord = yes
			if = {
				limit = {
					exists = scope:ob_isle_lord
					scope:ob_isle_lord = { is_alive = yes employer = root }
				}
				imprison = { target = scope:ob_isle_lord }
				scope:ob_isle_lord = { add_health = -25 }
			}
			remove_variable = ob_isle_mercy
			set_variable = { name = ob_isle_hardline }
			trigger_event = { id = ob_isle.7 days = 1 }
		}
	}
}
