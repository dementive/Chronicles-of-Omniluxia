# Task 14 — Form Eptelon (final)
me_eptelon_gen.47 = {
	type = country_event
	title = me_eptelon_gen.47.t
	desc = me_eptelon_gen.47.desc
	picture = diadochi_coronation
	left_portrait = current_ruler

	option = {
		name = me_eptelon_gen.47.a
		add_country_modifier = {
            name = eptelon_island_at_peace_mod
            duration = -1
        }
        add_5_free_province_investments = yes
        if = {
            limit = {
                OR = {
                    has_completed_mission_task = 03_eptelon_generic_task_6a
                    tag = EPT
                    country_culture_group = eptelon
                }
            }
        current_ruler = {
            force_add_trait = blood_of_the_eptelonian_chiefs
        	}
        }
        else_if = {
            limit = {
                OR = {
                    has_completed_mission_task = 03_eptelon_generic_task_8b
                    tag = ASE
                    country_culture_group = high_elves
                    }
            }
        current_ruler = {
            force_add_trait = blood_of_atheusian_shores
            }
        }
        else = {
            current_ruler = {
            force_add_trait = blood_of_the_harbor_founders
            }
        }
        current_ruler = {
            add_popularity = 25
            family = {
                add_prestige = 200
            }
        }
	}
}
