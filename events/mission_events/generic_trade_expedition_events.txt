		exists = scope:generic_trade_expedition_home_port

			order_by = charisma

				generic_trade_expedition_eligible_envoy_trigger = yes

			order_by = charisma
			limit = {
				generic_trade_expedition_eligible_envoy_trigger = yes
				NOT = { this = scope:generic_trade_expedition_envoy_1 }
				NOT = { this = scope:generic_trade_expedition_envoy_2 }

		}

			order_by = charisma

				generic_trade_expedition_eligible_envoy_trigger = yes

				NOT = { this = scope:generic_trade_expedition_envoy_2 }

		name = "generic_trade_expedition.4.a"

				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_prominence = prominence_small
			add_popularity = popularity_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_1
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
	}

	option = {
		name = "generic_trade_expedition.4.b"
		trigger = { exists = scope:generic_trade_expedition_envoy_2 }
		scope:generic_trade_expedition_envoy_2 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {
				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_prominence = prominence_small
			add_popularity = popularity_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_2
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
	}

	option = {
		name = "generic_trade_expedition.4.c"
		trigger = { exists = scope:generic_trade_expedition_envoy_3 }
		scope:generic_trade_expedition_envoy_3 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {

			add_popularity = popularity_small
		}
