namespace = generic_trade_expedition

# Choosing the envoy. The four candidates are no longer "the four most
# charismatic courtiers" but four different political bets: a governor you
# would like out of the country, a relative of the ruler, a loyal client, and
# a merchant with no family behind him.
generic_trade_expedition.4 = {
	type = country_event
	title = "generic_trade_expedition.4.t"
	desc = "generic_trade_expedition.4.desc"
	picture = greek_court

	goto_location = scope:generic_trade_expedition_home_port
	left_portrait = scope:generic_trade_expedition_envoy_1
	left_portrait = scope:generic_trade_expedition_envoy_2
	right_portrait = scope:generic_trade_expedition_envoy_3
	right_portrait = scope:generic_trade_expedition_envoy_4

	trigger = {
		exists = scope:generic_trade_expedition_home_port
		NOT = { has_variable = generic_trade_expedition_envoy_selected }
		OR = {
			any_character = { generic_trade_expedition_envoy_rival_trigger = yes }
			any_character = { generic_trade_expedition_envoy_kin_trigger = yes }
			any_character = { generic_trade_expedition_eligible_envoy_trigger = yes }
		}
	}

	immediate = {
		ordered_character = {
			order_by = prominence
			limit = { generic_trade_expedition_envoy_rival_trigger = yes }
			save_scope_as = generic_trade_expedition_envoy_1
		}
		ordered_character = {
			order_by = charisma
			limit = {
				generic_trade_expedition_envoy_kin_trigger = yes
				NOT = { this = scope:generic_trade_expedition_envoy_1 }
			}
			save_scope_as = generic_trade_expedition_envoy_2
		}
		ordered_character = {
			order_by = charisma
			limit = {
				generic_trade_expedition_envoy_client_trigger = yes
				NOT = { this = scope:generic_trade_expedition_envoy_1 }
				NOT = { this = scope:generic_trade_expedition_envoy_2 }
			}
			save_scope_as = generic_trade_expedition_envoy_3
		}
		ordered_character = {
			order_by = finesse
			limit = {
				generic_trade_expedition_envoy_lowborn_trigger = yes
				NOT = { this = scope:generic_trade_expedition_envoy_1 }
				NOT = { this = scope:generic_trade_expedition_envoy_2 }
				NOT = { this = scope:generic_trade_expedition_envoy_3 }
			}
			save_scope_as = generic_trade_expedition_envoy_4
		}
		# Fallback so the event can never dead-end with no options.
		if = {
			limit = {
				NOT = { exists = scope:generic_trade_expedition_envoy_3 }
				NOT = { exists = scope:generic_trade_expedition_envoy_4 }
			}
			ordered_character = {
				order_by = charisma
				limit = {
					generic_trade_expedition_eligible_envoy_trigger = yes
					NOT = { this = scope:generic_trade_expedition_envoy_1 }
					NOT = { this = scope:generic_trade_expedition_envoy_2 }
				}
				save_scope_as = generic_trade_expedition_envoy_3
			}
		}
	}

	# A. The overmighty governor. Four years abroad is four years his
	# powerbase does not grow at home - but a man that far away, holding our
	# cargo and our charts, has options.
	option = {
		name = "generic_trade_expedition.4.a"
		trigger = { exists = scope:generic_trade_expedition_envoy_1 }
		scope:generic_trade_expedition_envoy_1 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {
				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_character_modifier = {
				name = generic_trade_expedition_envoy_rival_charmod
				duration = 1825
			}
			add_prominence = prominence_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_1
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
		set_variable = generic_trade_expedition_envoy_rival
		# The faction he was building goes quiet while he is at sea.
		add_political_influence = 10
	}

	# B. The ruler's kin. Prestige at court, and a dynasty with a merchant
	# fortune of its own when he gets back.
	option = {
		name = "generic_trade_expedition.4.b"
		trigger = { exists = scope:generic_trade_expedition_envoy_2 }
		scope:generic_trade_expedition_envoy_2 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {
				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_character_modifier = {
				name = generic_trade_expedition_envoy_kin_charmod
				duration = 1825
			}
			add_prominence = prominence_small
			add_popularity = popularity_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_2
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
		set_variable = generic_trade_expedition_envoy_kin
		current_ruler = {
			add_popularity = popularity_small
		}
	}

	# C. The loyal client. Nothing surprising will happen, in either direction.
	option = {
		name = "generic_trade_expedition.4.c"
		trigger = { exists = scope:generic_trade_expedition_envoy_3 }
		scope:generic_trade_expedition_envoy_3 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {
				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_character_modifier = {
				name = generic_trade_expedition_envoy_client_charmod
				duration = 1825
			}
			add_prominence = prominence_small
			add_popularity = popularity_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_3
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
		set_variable = generic_trade_expedition_envoy_client
	}

	# D. The merchant. No house behind him, so no house grows off this - but
	# he will not be received as an equal in a foreign court either.
	option = {
		name = "generic_trade_expedition.4.d"
		trigger = { exists = scope:generic_trade_expedition_envoy_4 }
		scope:generic_trade_expedition_envoy_4 = {
			save_scope_as = generic_trade_expedition_envoy
			add_character_modifier = {
				name = generic_trade_expedition_envoy_charmod
				duration = 1825
			}
			add_character_modifier = {
				name = generic_trade_expedition_envoy_lowborn_charmod
				duration = 1825
			}
			add_popularity = popularity_small
		}
		set_variable = {
			name = generic_trade_expedition_envoy_character
			value = scope:generic_trade_expedition_envoy_4
			days = -1
		}
		set_variable = generic_trade_expedition_envoy_selected
		set_variable = generic_trade_expedition_envoy_lowborn
	}
}

generic_trade_expedition.5 = {
	type = country_event
	title = "generic_trade_expedition.5.t"
	desc = "generic_trade_expedition.5.desc"
	picture = greek_court

	goto_location = scope:generic_trade_expedition_partner_port
	left_portrait = current_ruler
	right_portrait = scope:generic_trade_expedition_partner.current_ruler

	trigger = {
		exists = scope:generic_trade_expedition_partner
		exists = scope:generic_trade_expedition_partner_port
		has_variable = generic_trade_expedition_envoy_character
	}

	immediate = {
		var:generic_trade_expedition_envoy_character = {
			save_scope_as = generic_trade_expedition_envoy
		}
	}

	option = {
		name = "generic_trade_expedition.5.a"
		trigger = {
			treasury >= generic_trade_expedition_lavish_gift_cost_svalue
		}
		save_scope_as = generic_trade_expedition_sender
		add_treasury = generic_trade_expedition_lavish_gift_pay_svalue
		set_variable = generic_trade_expedition_gift_lavish
		scope:generic_trade_expedition_partner = {
			add_opinion = {
				modifier = opinion_trade_expedition_lavish_gifts
				target = ROOT
			}
			hidden_effect = {
				trigger_event = { id = generic_trade_expedition.202 days = 1 }
			}
		}
		current_ruler = {
			add_popularity = popularity_small
		}
	}

	option = {
		name = "generic_trade_expedition.5.b"

			treasury >= generic_trade_expedition_gift_cost_svalue
		}
		save_scope_as = generic_trade_expedition_sender

			add_province_modifier = {
				name = generic_trade_expedition_home_market_survey_pmod

			}
		}
	}

	option = {
		name = "generic_trade_expedition.5.c"
		trigger = {
			treasury >= generic_trade_expedition_small_cost_svalue
		}
		save_scope_as = generic_trade_expedition_sender
		add_treasury = generic_trade_expedition_small_pay_svalue

		scope:generic_trade_expedition_partner = {
			add_opinion = {
				modifier = opinion_trade_expedition_ceremonial_gifts
				target = ROOT
			}
			hidden_effect = {

################################################################
# THE CHARTER - the terminal choice
#
# Three permanent, mutually exclusive endings. Each converts the work of the
# expedition into a different permanent institution, and each is paid for in
# a different currency: the Company costs internal politics, the Monopoly
# costs legitimacy, the Colony costs diplomacy.
#
# A country holds exactly ONE Charter. Choosing a different one on a later
# expedition tears up the old one at real cost. That is what stops a
# repeatable mission from compounding into a broken economy.
################################################################

			limit = { NOT = { exists = scope:generic_trade_expedition_charter_seat } }

		if = {

				has_variable = generic_trade_expedition_charter

			}
			custom_tooltip = generic_trade_expedition_charter_replace_tt

		}
		generic_trade_expedition_grant_colony_charter_effect = yes
	}
