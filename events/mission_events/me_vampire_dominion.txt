###########################################################################
# THE SAPPING OF THE PROVINCE
#
# Flavor events fired on the completion of each task in the
# generic_vampire_dominion mission tree. All of these rely on the scopes
# saved in the mission's on_potential block:
#	scope:generic_vampire_dominion_target   - the chosen province
#	scope:generic_vampire_dominion_capital  - the state capital of that state
###########################################################################

# The First Thirst
me_vampire_dominion.1 = {
	type = country_event

	title = me_vampire_dominion_1_title
	desc = me_vampire_dominion_1_desc
	picture = night_sky

	goto_location = scope:generic_vampire_dominion_capital
	left_portrait = current_ruler

	option = {
		name = me_vampire_dominion_1.a
		show_as_tooltip = {
			scope:generic_vampire_dominion_capital = {
				add_province_modifier = {
					name = generic_vampire_dominion_bloodless_nights_pmod
					duration = 1825
				}
			}
			current_ruler = {
				add_character_modifier = {
					name = generic_vampire_dominion_vigor_of_the_drained_charmod
					duration = 1825
				}
			}
		}
	}
}

# The Undying Court
me_vampire_dominion.15 = {
	type = country_event

	title = me_vampire_dominion_15_title
	desc = me_vampire_dominion_15_desc
	picture = event_empty_thrones

	goto_location = scope:generic_vampire_dominion_capital
	left_portrait = current_ruler
	right_portrait = scope:generic_vampire_dominion_get_1
	right_portrait = scope:generic_vampire_dominion_get_2
	right_portrait = scope:generic_vampire_dominion_get_3

	immediate = {
		hidden_effect = {
			random_character = {
				limit = {
					employer = ROOT
					is_ruler = no
					is_adult = yes
					is_alive = yes
					NOT = { has_trait = vampire_trait }
					NOT = { has_trait = wraith_trait }
				}
				save_scope_as = generic_vampire_dominion_get_1
			}
			random_character = {
				limit = {
					employer = ROOT
					is_ruler = no
					is_adult = yes
					is_alive = yes
					NOT = { has_trait = vampire_trait }
					NOT = { has_trait = wraith_trait }
					NOT = { this = scope:generic_vampire_dominion_get_1 }
				}
				save_scope_as = generic_vampire_dominion_get_2
			}
			random_character = {
				limit = {
					employer = ROOT
					is_ruler = no
					is_adult = yes
					is_alive = yes
					NOT = { has_trait = vampire_trait }
					NOT = { has_trait = wraith_trait }
					NOT = { this = scope:generic_vampire_dominion_get_1 }
					NOT = { this = scope:generic_vampire_dominion_get_2 }
				}
				save_scope_as = generic_vampire_dominion_get_3
			}
		}
	}

	option = {
		name = me_vampire_dominion_15.a
		show_as_tooltip = {
			scope:generic_vampire_dominion_capital = {
				add_province_modifier = {
					name = generic_vampire_dominion_seat_of_pale_court_pmod
					duration = 1825
				}
			}
			current_ruler = {
				add_character_modifier = {
					name = generic_vampire_dominion_lord_of_the_sapped_domain_charmod
					duration = 1825
				}
			}
		}
		if = {
			limit = { exists = scope:generic_vampire_dominion_get_1 }
			scope:generic_vampire_dominion_get_1 = {
				force_add_trait = vampire_trait
				add_loyalty = loyalty_granted_stipends
			}
		}
		if = {
			limit = { exists = scope:generic_vampire_dominion_get_2 }
			scope:generic_vampire_dominion_get_2 = {
				force_add_trait = vampire_trait
				add_loyalty = loyalty_granted_stipends
			}
		}
		if = {
			limit = { exists = scope:generic_vampire_dominion_get_3 }
			scope:generic_vampire_dominion_get_3 = {
				force_add_trait = vampire_trait
				add_loyalty = loyalty_granted_stipends
			}
		}
	}
}
