    title = "me_zaraken.142.t"
    desc = "me_zaraken.142.desc"
    picture = omni_wheat_harvest
    left_portrait = current_ruler

    option = {
        name = "me_zaraken.142.a"
        if = {
            limit = {
                owns = 4666
                exists = scope:zaraken_grain_buyer

        }
    }
}

    type = country_event
    title = "me_zaraken.143.t"
    desc = "me_zaraken.143.desc"
    picture = event_bread_basket
    left_portrait = current_ruler

    immediate = {
        if = {
            limit = {
                owns = 4662
                p:4699 = { owner = root has_province_modifier = zaraken_militia_grain_contract_mod }
                p:4704 = { owner = root has_province_modifier = zaraken_militia_grain_contract_mod }
            }
            p:4662 = { state = { add_state_food = 50 } }
        }
    }

    option = {
        name = "me_zaraken.143.a"
        every_character = {
            limit = { is_head_of_family = yes from_ruler_family = no }
            add_loyalty = zaraken_houses_hosted_feast
        }
    }
    option = {
        name = "me_zaraken.143.b"
        trigger = { treasury >= omni_rank_gold_cost_60_svalue }
        add_treasury = { value = omni_rank_gold_cost_60_svalue multiply = -1 }
        current_ruler = { add_popularity = 5 }
        every_owned_province = {
            limit = {
                is_in_region = golden_foothills_eteradica
                is_state_capital = yes
            }
            state = { add_state_food = 50 }
        }
    }
}
