# ==============================================================================
# GROMMASH'S REWARD OR REVOLT
# Follow-up on Grommash's demand to be made Governor of the Borderlands.
# me_nazani.6 : hidden watcher - waits for the scripted war to end
# me_nazani.7 : hidden reckoning - fires ~6 months after the war, checks his status
# me_nazani.8 : Grommash is NOT governor -> he leads a revolt
# me_nazani.9 : Grommash IS governor -> he is content
# ==============================================================================

# Event 6: Hidden watcher. Re-checks until the Borderlands war is over.
me_nazani.6 = {
	type = country_event
	hidden = yes

	immediate = {
		# Count how long we have been waiting, so we never loop forever.
		if = {
			limit = { NOT = { has_variable = nazani_grommash_watch_count } }
			set_variable = {
				name = nazani_grommash_watch_count
				value = 0
			}
		}
		change_variable = {
			name = nazani_grommash_watch_count
			add = 1
		}

		if = {
			# Grommash is dead or gone - nothing to reward, nothing to fear.
			limit = {
				NOT = {
					any_character = {
						is_alive = yes
						has_variable = grommash_the_orc
					}
				}
			}
			remove_variable = nazani_grommash_governor_promise
			remove_variable = nazani_grommash_watch_count
		}
		else_if = {
			# Give up quietly after ~5 years of waiting on the war.
			limit = { var:nazani_grommash_watch_count >= 60 }
			remove_variable = nazani_grommash_governor_promise
			remove_variable = nazani_grommash_watch_count
		}
		else_if = {
			# The Borderlands war is still going - keep waiting.
			# The scripted war is fought over the stronghold at province 2187,
			# so "won and at peace" is the signal that the war has concluded.
			limit = {
				OR = {
					war = yes
					NOT = { owns = 2187 }
				}
			}
			trigger_event = {
				id = me_nazani.6
				days = 30
			}
		}
		else = {
			# The war is over and the Borderlands stronghold is held. Grommash
			# expects his governorship. Give the player ~six months to grant it.
			remove_variable = nazani_grommash_watch_count
			trigger_event = {
				id = me_nazani.7
				days = 180
			}
		}
	}
}

# Event 7: Hidden reckoning. Six months on, is Grommash the governor?
me_nazani.7 = {
	type = country_event
	hidden = yes

	immediate = {
		remove_variable = nazani_grommash_governor_promise

		if = {
			# Grommash died in the interim - the matter is closed.
			limit = {
				NOT = {
					any_character = {
						is_alive = yes
						has_variable = grommash_the_orc
					}
				}
			}
		}
		else = {
			random_character = {
				limit = {
					is_alive = yes
					has_variable = grommash_the_orc
				}
				save_scope_as = orcish_captain
			}
			if = {
				limit = {
					# Is Grommash the governor of any Borderlands state we own?
					region:border_lands_morrigon = {
						any_region_province = {
							owner = ROOT
							state = {
								governor = scope:orcish_captain
							}
						}
					}
				}
				trigger_event = { id = me_nazani.9 }
			}
			else = {
				trigger_event = { id = me_nazani.8 }
			}
		}
	}
}

me_nazani.20 = {
	type = country_event
	title = "me_nazani.20.t"
	desc = "me_nazani.20.desc"
	picture = event_octopus
	left_portrait = current_ruler

	immediate = {
		set_variable = porphyros_active
		hidden_effect = {
			random_owned_province = {
				limit = { num_of_port_building >= 1 }
				save_scope_as = porphyros_target
			}
			if = {
				limit = { NOT = { exists = scope:porphyros_target } }
				capital_scope = {
					save_scope_as = porphyros_target
				}
			}
			# Begin the cycle of attacks, and set the beast's doom ~30 years hence.
			trigger_event = {
				id = me_nazani.21
				days = { 1825 2555 }
			}
			trigger_event = {
				id = me_nazani.25
				days = 10950
			}
		}
	}

	option = { # Do nothing - the panic spreads and men are lost.
		name = "me_nazani.20.a"
		ai_chance = { factor = 40 }
		if = {
			limit = { exists = scope:porphyros_target }
			scope:porphyros_target = {
				# Smash the harbour to splinters - strip every level of the port.
				while = {
					limit = { has_building = port_building }
					remove_building_level = port_building
				}
			}
		}
		add_stability = -5
		if = {
			limit = { exists = scope:porphyros_target }
			scope:porphyros_target = {
				while = {
					count = 2
					random_pops_in_province = { kill_pop = yes }
				}
			}
		}
	}
	option = { # Pay to rally the harbor and salvage what we can.
		name = "me_nazani.20.b"
		ai_chance = { factor = 60 }
		add_treasury = -150
	}
}

# me_nazani.201: The Heir's Ultimatum - Main decision event
me_nazani.201 = {
	type = country_event
	title = "me_nazani.201.t"
	desc = "me_nazani.201.desc"
	picture = event_umbar_units

	left_portrait = root.current_ruler
	right_portrait = char:185

	option = {
		name = "me_nazani.201.a"
		ai_chance = { factor = 25 }
		custom_tooltip = "me_nazani.201.a.tt"

		hidden_effect = {
			# Renos ascends. set_as_ruler is a country-scoped effect that takes
			# the new ruler as its target, so it must be called on ROOT.
			root = {
				set_as_ruler = char:185
			}
		}

		trigger_event = {
			id = me_nazani.202
			days = 180
		}
	}

	option = {
		name = "me_nazani.201.b"
		ai_chance = { factor = 75 }

		custom_tooltip = "me_nazani.201.b.tt"

		hidden_effect = {
			# Renos refuses to accept the refusal and starts a civil war
			char:185 = {
			}

			# Find Martios (the current emperor) to start the war against
			if = {
				limit = { exists = root.current_ruler }
				root.current_ruler = {
					save_scope_as = emperor_martios
				}
				root = {
					start_civil_war = char:185
				}
			}
		}
	}
}

me_nazani.202 = {
	type = country_event
	title = "me_nazani.202.t"
	desc = "me_nazani.202.desc"
	picture = event_boat_storm

	left_portrait = root.current_ruler
	right_portrait = char:18

	immediate = {
		# Martios is a fixed historical character in this chain.
		char:18 = {
			save_scope_as = former_emperor_martios
		}
		# Remember the Nazani Empire itself so the exile country can later be
		# granted a claim on its capital (see me_nazani.203).
		save_scope_as = nazani_empire
	}

	option = {
		name = "me_nazani.202.a"

		custom_tooltip = "me_nazani.202.a.tt"

		hidden_effect = {
			if = {
				limit = {
					exists = scope:former_emperor_martios
					scope:former_emperor_martios = { is_alive = yes }
				}

				# Put out his eyes. The vanilla trait is "blind", not "blinded".
				scope:former_emperor_martios = {
					add_trait = blind
				}

				if = {
					limit = { exists = p:2478.owner }
					p:2478.owner = {
						save_scope_as = exile_destination_owner
					}
					scope:former_emperor_martios = {
						remove_all_positions = yes
						set_home_country = scope:exile_destination_owner
						set_as_minor_character = THIS
					}

					# A week later, the court that received him must decide his fate.
					scope:exile_destination_owner = {
						trigger_event = {
							id = me_nazani.203
							days = 7
						}
					}
				}
			}
		}
		current_ruler = {
			force_add_trait = cruel
			add_nickname = NICKNAME_THE_CRUEL
		}
	}
}


me_nazani.203 = {
	type = country_event
	title = "me_nazani.203.t"
	desc = "me_nazani.203.desc"
	picture = greek_court

	left_portrait = root.current_ruler
	right_portrait = char:18

	trigger = {
		char:18 = {
			is_alive = yes
		}
	}

	option = {
		name = "me_nazani.203.a"
		ai_chance = { factor = 40 }
		char:18 = {
			death = {
				death_reason = death_murder
			}
		}
		current_ruler = {
			add_popularity = 15
		}
	}

	option = {
		name = "me_nazani.203.b"
		ai_chance = { factor = 40 }

		add_treasury = -25
		current_ruler = {
			add_popularity = 10
		}
		add_legitimacy = 10
	}

	option = {
		name = "me_nazani.203.c"
		ai_chance = { factor = 20 }

		custom_tooltip = "me_nazani.203.c.tt"

		hidden_effect = {
			char:18 = {
				set_home_country = ROOT
				add_loyalty = loyalty_offered_refuge
			}
			if = {
				limit = { exists = scope:nazani_empire }
				scope:nazani_empire.capital_scope = {
					add_claim = ROOT
				}
			}
		}
	}
}
