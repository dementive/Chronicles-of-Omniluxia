namespace = omni_cmd_gear

# Commander's gear. Generals and admirals manage their own fixed combat items and
# siege supplies. Shown on the RULER (non-ruler character events auto-resolve);
# the commander is scope:omni_gear_holder. A commander's item gives its personal
# bonus to them and its army bonus (item_X_unit) to the army or fleet they lead.

omni_cmd_gear.1 = {
	type = character_event
	picture = war_chest
	title = omni_cmd_gear.1.t
	desc = omni_cmd_gear.1.desc
	left_portrait = scope:omni_gear_holder

	immediate = {
		country = { save_scope_as = omni_gear_country }
		scope:omni_gear_holder = { omni_kit_init = yes }
	}

	option = {
		name = omni_cmd_gear.1.equip
		trigger = { scope:omni_gear_holder = { has_any_item = yes } }
		trigger_event = { id = omni_cmd_gear.2 }
	}
	option = {
		name = omni_cmd_gear.1.unequip
		trigger = { scope:omni_gear_holder = { has_variable_list = equiped_item_list any_in_list = { variable = equiped_item_list count >= 1 } } }
		trigger_event = { id = omni_cmd_gear.3 }
	}
	option = {
		name = omni_cmd_gear.1.siege
		trigger = {
			scope:omni_gear_holder = {
				omni_kit_has_any = yes
				omni_character_is_besieging = { COUNTRY = scope:omni_gear_country }
			}
		}
		trigger_event = { id = omni_cmd_gear.4 }
	}
	option = {
		name = omni_items.1.close
	}
}

# Use carried siege supplies on the settlement this commander is besieging.
omni_cmd_gear.4 = {
	type = character_event
	picture = city_siege
	title = omni_cmd_gear.4.t
	desc = omni_cmd_gear.4.desc
	left_portrait = scope:omni_gear_holder

	immediate = {
		if = { limit = { NOT = { exists = scope:omni_gear_country } } country = { save_scope_as = omni_gear_country } }
		scope:omni_gear_holder = { omni_kit_init = yes }
	}

	option = {
		name = omni_cmd_gear.4.ram
		trigger = {
			scope:omni_gear_holder = {
				omni_kit_has = { KIT = ram }
				omni_character_is_besieging = { COUNTRY = scope:omni_gear_country }
			}
		}
		custom_tooltip = omni_kit_ram_use_tt
		hidden_effect = {
			scope:omni_gear_holder = {
				omni_kit_find_target = yes
				omni_kit_spend = { KIT = ram }
			}
			scope:omni_kit_target = {
				siege = { add_breach = 1 }
				if = {
					limit = { owner = { is_ai = no NOT = { this = scope:omni_gear_country } } }
					owner = { trigger_event = { id = omni_siege_supply.10 } }
				}
			}
		}
		trigger_event = { id = omni_cmd_gear.4 }
	}
	option = {
		name = omni_cmd_gear.4.sapper
		trigger = {
			scope:omni_gear_holder = {
				omni_kit_has = { KIT = sapper }
				omni_character_is_besieging = { COUNTRY = scope:omni_gear_country }

				omni_supply_refresh_province_mod = { MOD = omni_kit_sapper_mod DAYS = 30 }

					limit = { owner = { is_ai = no NOT = { this = scope:omni_gear_country } } }
					owner = { trigger_event = { id = omni_siege_supply.10 } }
				}
			}
		}
