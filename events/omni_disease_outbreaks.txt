namespace = omni_disease

# Disease Outbreak System. Structurally a sibling of omni_disaster (events/omni_natural_disasters.txt) - same
# shape (opening event -> contain/ignore choice -> optional spread -> resolution, terrain/season gated, no
# hardcoded regions, 25yr cooldown per disease). See agent.md "Disease Outbreak System" for the full pattern
# and why this shares omni_yearly_disaster_pulse's roll instead of adding a second on_action.
#
# Three chain lengths, matching the "mild to epidemic" ask:
#  - Short (2 events): onset -> resolution. Never escalates. omni_disease_resolve_mild_effect (~2% death).
#  - Medium (3 events): onset -> spread/worsen -> resolution. omni_disease_resolve_moderate_effect (~10% death).
#  - Long (4 events): onset -> worsens -> spreads/epidemic -> resolution. omni_disease_resolve_severe_effect
#    (~25% death), but only if left unchecked through every stage - acting against it early still short-circuits
#    straight to resolution, same as the locust chain (omni_disaster.11-14).
#
# Character infection: only the sick province's governor (if any, alive, uninfected) ever catches it - see
# omni_infect_province_governor_effect. No realm-wide character scans.

############################################
# GREY COUGH (short chain) - crowd respiratory illness, farmland
############################################
omni_disease.1 = { #Grey Cough - onset
	type = country_event
	title = omni_disease.1.t
	desc = omni_disease.1.desc
	picture = city_country

	left_portrait = current_ruler
	right_portrait = scope:grey_cough_province.governor
	goto_location = scope:grey_cough_province

	trigger = {
		any_owned_province = {
			omni_grey_cough_target_trigger = yes
			total_population >= 4
		}
		NOT = {
			has_variable = grey_cough_cooldown
		}
	}

	immediate = {
		random_owned_province = {
			limit = {
				omni_grey_cough_target_trigger = yes
				total_population >= 4
			}
			save_scope_as = grey_cough_province
			add_province_modifier = {
				name = omni_grey_cough_outbreak
				duration = 180
			}
			omni_infect_province_governor_effect = yes
		}
	}

	option = { #quarantine the sick
		name = omni_disease.1.a
		add_treasury = {
			subtract = has_monthly_income
			multiply = 4
			min = -150
			max = -30
		}
	}

	option = { #let it pass
		name = omni_disease.1.b
	}

	after = {
		set_variable = {
			name = grey_cough_cooldown
			days = 9125 #25yrs
		}
		hidden_effect = {
			trigger_event = {
				id = omni_disease.2
				days = { 150 210 }
			}
		}
	}
}
