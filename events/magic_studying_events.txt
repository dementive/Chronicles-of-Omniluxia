# Overworking
me_magic_studying.2 = {
    type = country_event

    title = me_magic_studying.2_title
    desc = me_magic_studying.2_desc
    picture = event_dwarven_hall
    left_portrait = scope:target

    # The studying character can die between this event being queued and it
    # resolving, which left add_health below firing on a corpse
    # ("add_health : scope was dead"). No target, no overwork event.
    trigger = {
        exists = scope:target
        scope:target = {
            is_alive = yes
        }
    }

    immediate = {
        check_mage_effect = yes
    }


    option = {
        name = "me_magic_studying.2.a"
        add_political_influence = -8
        scope:target = {
            if = {
                limit = { NOT = { has_trait = stressed } }
                add_trait = stressed
            }
            else_if = {
                limit = { NOT = { has_trait = infection } }
                add_trait = infection
            }
            else = {
                add_health = -0.025
            }
        }
        custom_tooltip = me_magic_studying_bad_tooltip

        hidden_effect = {
            magic_studying_t1_effect = yes
        }
    }
}
