namespace = tribal_flavour_events

tribal_flavour_events.1 = {
	type = country_event
	title = "tribal_flavour_event.1.t"
	desc =  "tribal_flavour_event.1.desc"
	picture = celtic_diplomacy
	
	left_portrait = root.current_ruler
	right_portrait = scope:duelist_1
	right_portrait = scope:duelist_2
	
	trigger = {
		is_tribal = yes
	}
	
	immediate = {
		random_character = {
			limit = {
				is_ruler = no
				is_adult = yes
			}
			save_scope_as = duelist_1
		}
		random_character = {
			limit = {
				is_ruler = no
				is_adult = yes
				NOT = { this = scope:duelist_1 }
			}
			save_scope_as = duelist_2
		}
	}
	
	option = {
		name = "tribal_flavour_event.1.a"
		scope:duelist_1 = {
			add_loyalty = loyalty_picked_side
			add_gold = add_gold_small
			
		}
		scope:duelist_2 = {
			add_loyalty = loyalty_picked_other_side
		}
	}
	
	option = {
		name = "tribal_flavour_event.1.b"
		scope:duelist_2 = {
			add_loyalty = loyalty_picked_side
			add_gold = add_gold_small
			
		}
		scope:duelist_1 = {
			add_loyalty = loyalty_picked_other_side
		}
	}
	
	option = {
		name = "tribal_flavour_event.1.c"
		add_tyranny = 5
		add_treasury = add_gold_small
		scope:duelist_1 = {
			add_loyalty = loyalty_chided_bickering
			
		}
		scope:duelist_2 = {
			add_loyalty = loyalty_chided_bickering
		}
	}
	
	option = {
		name = "tribal_flavour_event.1.d"
		add_stability_effect = yes
		
		scope:duelist_1 = {
			add_loyalty = loyalty_chided_bickering_2
		}
		scope:duelist_2 = {
			add_loyalty = loyalty_chided_bickering_2
		}	
	}
}

# Replaces the original immediate 80-territory breakup. Eligibility, rise,
# regional transfers and cleanup follow Invictus's tribal crisis chain.
tribal_flavour_events.2 = {
	type = country_event
	hidden = yes
	trigger = { omni_tribal_cycle_can_start_trigger = yes }
	weight_multiplier = {
		modifier = { factor = 1.35 num_of_cities >= 150 }
		modifier = { factor = 1.25 num_of_cities >= 200 }
		modifier = { factor = 1.2 num_of_cities >= 300 }
		modifier = { factor = 1.2 num_of_cities >= 400 }
		modifier = { factor = 1.2 num_of_cities >= 500 }
	}
	immediate = { omni_tribal_cycle_start_effect = yes }
}

# The monthly monitor calls this after the 7,300-day rise clock expires.
# No queued 20-year event can survive cancellation and hit a later cycle.
tribal_flavour_events.4 = {
	type = country_event
	hidden = yes
	trigger = {
		has_variable = is_facing_tribal_crisis
		NOT = { has_variable = omni_tribal_rise_clock }
		war = no
		has_civil_war = no
		omni_tribal_cycle_valid_realm_trigger = yes
	}
	immediate = { omni_tribal_cycle_collapse_effect = yes }
}
