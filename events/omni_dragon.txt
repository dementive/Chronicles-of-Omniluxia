# Console encounter for testing: prefer a native horde in the player's lands.
omni_dragon.1 = {
    type = country_event
    hidden = yes
    immediate = {
        save_scope_as = omni_dragon_observer
        if = {
            limit = {
                exists = c:BAR
                c:BAR = { NOT = { any_country_sub_unit = { sub_unit_type = omni_dragon } } }
            }
            c:BAR = {
                random_unit = {
                    limit = { omni_dragon_horde_candidate = yes unit_location = { owner = root } }
                    save_scope_as = omni_dragon_test_horde
                }
                if = {
                    limit = { NOT = { exists = scope:omni_dragon_test_horde } }
                    random_unit = {
                        limit = { omni_dragon_horde_candidate = yes }
                        save_scope_as = omni_dragon_test_horde
                    }
                }
            }
        }
        if = {
            limit = { exists = scope:omni_dragon_test_horde }
            scope:omni_dragon_test_horde = { omni_awaken_wild_dragon = yes }
        }
        # No barbarian horde to wake: raise one on the next rotation continent.
        else_if = {
            limit = { NOT = { omni_wild_dragon_alive = yes } }
            omni_spawn_rotating_wild_dragon = yes
            if = {
                limit = { NOT = { exists = scope:omni_new_wild_dragon } }
                trigger_event = { id = omni_dragon.3 }
            }
        }
        else_if = {
            limit = {
                exists = global_var:omni_dragon_horde
                global_var:omni_dragon_horde = { any_sub_unit = { sub_unit_type = omni_dragon } }
            }
            global_var:omni_dragon_horde.unit_location = { save_scope_as = omni_dragon_sighting }
            trigger_event = { id = omni_dragon.2 }
        }
        else = { trigger_event = { id = omni_dragon.3 } }
    }
}

# Console test: mark a den and start the hunt so the monthly pulse turns the
# next barbarian horde into a Wild Morrigon. Points at a living one if present.
# The game cannot create armies for the barbarians, so hordes come from the den.
omni_dragon.6 = {
    type = country_event
    title = omni_dragon.6.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:omni_dragon_test_sighting }
                desc = omni_dragon.6.desc
            }
            triggered_desc = {
                trigger = { exists = scope:omni_dragon_test_den }
                desc = omni_dragon.6.desc_den
            }
            desc = omni_dragon.6.desc_fail
        }
    }
    picture = Event_dragons_burning_ships
    goto_location = scope:omni_dragon_test_den
    immediate = {
        save_scope_as = omni_dragon_observer
        # Diagnostics: written to logs/debug.log.
        if = { limit = { exists = c:BAR } debug_log = "omni_dragon.6: c:BAR exists" }
        else = { debug_log = "omni_dragon.6: c:BAR does NOT exist" }
        if = { limit = { exists = global_var:omni_dragon_den } debug_log = "omni_dragon.6: den already set" }
        if = {
            limit = { any_province = { omni_dragon_den_candidate = yes } }
            debug_log = "omni_dragon.6: den candidates found"
        }
        else = { debug_log = "omni_dragon.6: NO den candidates found" }
        if = {
            limit = { exists = c:BAR c:BAR = { any_country_sub_unit = { sub_unit_type = omni_dragon } } }
            debug_log = "omni_dragon.6: barbarians already have a dragon"
        }
        if = {
            limit = {
                exists = c:BAR
                c:BAR = { NOT = { any_country_sub_unit = { sub_unit_type = omni_dragon } } }
            }
            if = {
                limit = { NOT = { exists = global_var:omni_dragon_den } }
                random_province = {
                    limit = {
                        omni_dragon_den_candidate = yes
                    }
                    set_global_variable = { name = omni_dragon_den value = this }
                    add_province_modifier = { name = omni_dragon_stirring duration = -1 }
                }
            }
            if = {
                limit = { exists = global_var:omni_dragon_den }
                set_global_variable = omni_dragon_hunt
                global_var:omni_dragon_den = { save_scope_as = omni_dragon_test_den }
            }
        }
        # Point at the living wild dragon, new or already in the field.
        if = {
            limit = {
                exists = global_var:omni_dragon_horde
                global_var:omni_dragon_horde = { any_sub_unit = { sub_unit_type = omni_dragon } }
            }
            global_var:omni_dragon_horde.unit_location = {
                save_scope_as = omni_dragon_test_sighting
                save_scope_as = omni_dragon_test_den
            }
        }
    }
    option = { name = omni_dragon.6.a }
}

# Daily dragon upkeep: track wounds and breach forts the dragon is besieging.
omni_dragon.8 = {
    type = country_event
    hidden = yes
    immediate = {
        omni_dragon_maintain_slot = { N = 1 }
        omni_dragon_maintain_slot = { N = 2 }
        omni_dragon_maintain_slot = { N = 3 }
        omni_dragon_maintain_slot = { N = 4 }
        omni_dragon_maintain_slot = { N = 5 }
        omni_dragon_maintain_slot = { N = 6 }
        omni_dragon_maintain_slot = { N = 7 }
        omni_dragon_maintain_slot = { N = 8 }
        omni_dragon_maintain_slot = { N = 9 }
        omni_dragon_maintain_slot = { N = 10 }
        # The last living cohort found above leads to the dragon army.
        if = {
            limit = { exists = scope:omni_dragon_slot_cohort }
            scope:omni_dragon_slot_cohort.owning_unit = {
                save_scope_as = omni_dragon_rider_unit
                unit_location = {
                    if = {
                        limit = {
                            fort_level >= 1
                            has_siege = yes
                            siege = {
                                has_breach = no
                                siege_controller = { this = root }
                            }
                        }
                        siege = { add_breach = 1 }
                    }
                }
            }
        }
        omni_dragon_update_rider = yes
        if = {
            limit = {
                OR = {
                    exists = var:omni_dragon_cohort_1
                    exists = var:omni_dragon_cohort_2
                    exists = var:omni_dragon_cohort_3
                    exists = var:omni_dragon_cohort_4
                    exists = var:omni_dragon_cohort_5
                    exists = var:omni_dragon_cohort_6
                    exists = var:omni_dragon_cohort_7
                    exists = var:omni_dragon_cohort_8
                    exists = var:omni_dragon_cohort_9
                    exists = var:omni_dragon_cohort_10
                }
            }
            trigger_event = { id = omni_dragon.8 days = 1 }
        }
        else = {
            remove_variable = omni_dragon_tracking
        }
    }
}

# Neighbours of a newly crowned dragon dominion hear the news. Sent by
# omni_try_dragon_settlement, which leaves var:omni_dragon_herald_realm on the
# recipient so the scopes survive the hop between countries.
omni_dragon.12 = {
    type = country_event
    title = omni_dragon.12.t
    desc = omni_dragon.12.desc
    picture = Event_dragons_burning_ships
    left_portrait = current_ruler
    right_portrait = scope:omni_dragon_herald_king
    trigger = {
        exists = var:omni_dragon_herald_realm
        var:omni_dragon_herald_realm = { government = omni_dragon_dominion }
    }
    immediate = {
        var:omni_dragon_herald_realm = {
            save_scope_as = omni_dragon_herald_realm
            current_ruler = { save_scope_as = omni_dragon_herald_king }
            capital_scope = { save_scope_as = omni_dragon_herald_seat }
        }
        remove_variable = omni_dragon_herald_realm
    }
    option = {
        name = omni_dragon.12.a
        add_country_modifier = { name = omni_dragon_vigil duration = 1825 }
        ai_chance = { factor = 50 }
    }
    option = {
        name = omni_dragon.12.b
        trigger = { treasury >= 50 }
        add_treasury = -50
        add_stability = 5
        ai_chance = { factor = 30 }
    }
    option = {
        name = omni_dragon.12.c
        add_country_modifier = { name = omni_dragon_dread duration = 1095 }
        ai_chance = { factor = 20 }
    }
}

# Console only. Stage a complete occupation on the selected player's realm.
omni_dragon.99 = {
    type = country_event
    title = omni_dragon.99.t
    desc = omni_dragon.99.desc
    picture = Event_dragons_burning_ships
    option = {
        name = omni_dragon.99.a
        trigger = {
            exists = capital_scope
            exists = c:REB
            NOT = { exists = global_var:omni_dragon_horde }
            NOT = { government = omni_dragon_dominion }
        }
        capital_scope = { save_scope_as = omni_dragon_test_capital }
        c:REB = {
            omni_choose_dragon_name = { ACTION = omni_create_named_test_dragon }
            scope:omni_test_dragon = {
                every_sub_unit = { add_subunit_strength = 1 }
                set_variable = omni_wild_morrigon
                set_global_variable = { name = omni_dragon_horde value = this }
            }
        }
        if = {
            limit = { exists = global_var:omni_dragon_horde }
            every_owned_province = {
                set_controller = c:REB
                omni_record_dragon_occupation = yes
            }
            trigger_event = { id = omni_dragon.10 days = 1 }
        }
    }
    option = { name = omni_dragon.99.b }
}
