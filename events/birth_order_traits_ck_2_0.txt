# Seed the new monarch's dynasty on accession or country reorganization.
firstborn_traits.1 = {
	hidden = yes
	type = country_event

	trigger = {
		is_monarchy = yes
		exists = current_ruler
	}

	immediate = {
		current_ruler = {
			omni_assign_ruler_birth_order_effect = { INCLUDE_OTHER_FAMILIES = no }
		}
	}
}
